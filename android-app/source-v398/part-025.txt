/* ---- Tile image persistence ----
   BUGFIX (tile pictures vanished after a few hours / never loaded): a tile's picture was only
   ever kept as a plain imgBB URL. The cache above stores that URL, not the picture itself, so
   any slow connection, a CDN hiccup or a link that later stops working left a blank tile. Two
   things were wrong:
     1) the failure fallback was an inline onerror="…innerHTML=${JSON.stringify(svg)}"  -  JSON
        output is full of double quotes, which end a double-quoted HTML attribute early, so
        that handler was a syntax error and NEVER ran: a failed picture left an empty box
        instead of the icon (exactly the blank "Templates" tile);
     2) nothing retried or remembered the picture.
   Now: every tile picture is fetched once, shrunk to a small JPEG/PNG data URL and kept in
   localStorage keyed by its URL, so it keeps showing even if imgBB is slow, blocked or the
   link has expired. The live URL is still tried first (it may be newer), with one retry, and
   the icon fallback is attached with a real listener instead of an inline attribute. */
const TILE_IMG_CACHE_KEY = 'riggy.tileImg.cache.v2'; // bumped: v1 entries were cached before high-quality smoothing was added
const TILE_IMG_MAX_ENTRIES = 12;
const TILE_IMG_MAX_DIM = 320; // tiles are ~80-100 css px square; 320 keeps them sharp on 3x screens
let tileImgCache = {};
try{ tileImgCache = JSON.parse(localStorage.getItem(TILE_IMG_CACHE_KEY) || '{}') || {}; }catch(e){ tileImgCache = {}; }
function saveTileImgCache(){
  try{
    localStorage.setItem(TILE_IMG_CACHE_KEY, JSON.stringify(tileImgCache));
  }catch(e){
    // Storage full: drop the oldest half and try once more, never throw into the UI.
    try{
      const keys = Object.keys(tileImgCache);
      keys.slice(0, Math.ceil(keys.length/2)).forEach(k=>delete tileImgCache[k]);
      localStorage.setItem(TILE_IMG_CACHE_KEY, JSON.stringify(tileImgCache));
    }catch(e2){}
  }
}
// Downloads the picture, shrinks it on a canvas and stores it as a data URL. Best-effort:
// a cross-origin host that doesn't send CORS headers simply can't be snapshotted (canvas
// would be tainted)  -  in that case nothing is stored and the live URL keeps being used.
function cacheTileImage(url){
  if(!/^https?:/i.test(url) || tileImgCache[url] || cacheTileImage._busy[url]) return;
  cacheTileImage._busy[url] = true;
  const probe = new Image();
  probe.crossOrigin = 'anonymous';
  probe.onload = ()=>{
    try{
      const scale = Math.min(1, TILE_IMG_MAX_DIM / Math.max(probe.naturalWidth, probe.naturalHeight));
      const w = Math.max(1, Math.round(probe.naturalWidth * scale));
      const h = Math.max(1, Math.round(probe.naturalHeight * scale));
      const cv = document.createElement('canvas'); cv.width = w; cv.height = h;
      const tctx = cv.getContext('2d');
      tctx.imageSmoothingEnabled = true;
      tctx.imageSmoothingQuality = 'high';
      tctx.drawImage(probe, 0, 0, w, h);
      // PNG keeps transparency for cut-out tile art; fall back to JPEG if it's huge.
      let data = cv.toDataURL('image/png');
      if(data.length > 90000) data = cv.toDataURL('image/jpeg', 0.82);
      if(data.length <= 140000){
        const keys = Object.keys(tileImgCache);
        if(keys.length >= TILE_IMG_MAX_ENTRIES) delete tileImgCache[keys[0]];
        tileImgCache[url] = data;
        saveTileImgCache();
      }
    }catch(e){ /* tainted canvas / quota  -  live URL still works */ }
    delete cacheTileImage._busy[url];
  };
  probe.onerror = ()=>{ delete cacheTileImage._busy[url]; };
  probe.src = url;
}
cacheTileImage._busy = {};

function tileShowFallback(thumb, iconKey){
  if(!thumb) return;
  thumb.classList.remove('tile-photo');
  thumb.innerHTML = TILE_ICONS[iconKey] || TILE_ICONS.spark;
}
// Wires load/error handling onto one tile <img>. Order tried on failure:
//   live URL -> live URL once more (cache-busted) -> locally saved copy -> icon fallback.
function wireTileImage(img, url, iconKey){
  const thumb = img.closest('.home-tile-thumb');
  let stage = 0;
  img.addEventListener('load', ()=>{
    img.classList.add('loaded');
    if(img.src === url || img.dataset.live === '1') cacheTileImage(url);
  });
  img.addEventListener('error', ()=>{
    stage++;
    if(stage === 1 && /^https?:/i.test(url)){
      setTimeout(()=>{ img.dataset.live = '1'; img.src = url + (url.indexOf('?') > -1 ? '&' : '?') + 'r=' + Date.now(); }, 1500);
    } else if(stage <= 2 && tileImgCache[url]){
      stage = 3; img.dataset.live = '0'; img.src = tileImgCache[url];
    } else {
      tileShowFallback(thumb, iconKey);
    }
  });
  img.dataset.live = '1';
  // Paint the saved copy instantly when we have one (no blank flash on a cold/slow network),
  // then let the live URL replace it if it loads.
  if(tileImgCache[url]){
    img.src = tileImgCache[url];
    img.dataset.live = '0';
    const live = new Image();
    live.onload = ()=>{ img.dataset.live = '1'; img.src = url; };
    live.onerror = ()=>{ /* keep the saved copy showing */ };
    if(/^https?:/i.test(url)) live.src = url;
  } else {
    img.src = url;
  }
  // A picture that already finished (or failed) before the listeners existed would otherwise
  // sit invisible  -  settle it explicitly.
  if(img.complete){
    if(img.naturalWidth > 0) img.classList.add('loaded');
    else img.dispatchEvent(new Event('error'));
  }
}

/* ---------------- RENDER: EXAMPLE ANIMATIONS ----------------
   Example clips (GIFs, or videos if the admin supplied one) do NOT autoplay on page load.
   - A clip starts playing only once it has been scrolled into view, and only ONE clip plays
     at a time (the others sit on a still frame).
   - Tapping a clip plays it (and stops whichever clip was playing) or pauses it. There is no
     on-screen play / pause button. A clip the visitor paused by hand stays paused until they
     tap it again or scroll it away and back.
   A GIF cannot be paused, so a paused GIF is shown as a still frame drawn on a canvas and the
   animated <img> is emptied; playing puts the GIF back (it restarts from its first frame). */
const HV = { obs:null, preObs:null, ratios:new Map(), cards:[], current:null, lock:null };
// The clip the visitor paused by tapping it. Only that clip shows the centred pause icon; it is
// cleared as soon as the clip plays again or is scrolled away.
function hvSetLock(card){
  if(HV.lock && HV.lock !== card) HV.lock.classList.remove('hv-user-paused');
  HV.lock = card || null;
  if(card) card.classList.add('hv-user-paused');
}

function hvDrawStill(card, srcImg){
  // Draw srcImg (a loaded <img>) onto the card's still-frame canvas, creating it if needed.
  const w = srcImg.naturalWidth || srcImg.clientWidth, h = srcImg.naturalHeight || srcImg.clientHeight;
  if(!w || !h) return;
  let cv = card.querySelector('canvas.home-gif-freeze');
  if(!cv){
    cv = document.createElement('canvas');
    cv.className = 'home-video-img home-gif-freeze';
    card.querySelector('img.home-gif').after(cv);
  }
  cv.width = w; cv.height = h;
  try{ cv.getContext('2d').drawImage(srcImg, 0, 0, w, h); }catch(_e){}
  cv.classList.remove('hv-hide');
}
function hvLoadStill(card){
  // Loads the GIF once (only when the card is close to the screen) just to paint its first frame.
  if(card.__hvStill) return;
  card.__hvStill = true;
  const img = card.querySelector('img.home-gif');
  if(!img) return;
  const url = img.dataset.gifSrc || '';
  if(!url) return;
  const probe = new Image();
  probe.decoding = 'async';
  probe.onload = ()=>{ if(!card.classList.contains('is-playing')) hvDrawStill(card, probe); };
  probe.src = url;
}
function hvSetPlaying(card, on){
  const vid = card.querySelector('video.home-video-img');
  const img = card.querySelector('img.home-gif');
  if(vid){
    if(on){ const p = vid.play(); if(p && p.catch) p.catch(()=>{}); } else { vid.pause(); }
  } else if(img){
    if(on){
      if(!img.getAttribute('src')){
        img.addEventListener('load', ()=>{
          // hide the still only after the animated frame has really been painted (no flicker)
          requestAnimationFrame(()=>requestAnimationFrame(()=>{
            const cv = card.querySelector('canvas.home-gif-freeze');
            if(cv && card.classList.contains('is-playing')) cv.classList.add('hv-hide');
          }));
        }, {once:true});
        img.src = img.dataset.gifSrc || '';
      }
    } else if(img.getAttribute('src')){
      if(img.complete && img.naturalWidth) hvDrawStill(card, img);   // freeze the frame showing now
      const cv = card.querySelector('canvas.home-gif-freeze'); if(cv) cv.classList.remove('hv-hide');
      img.removeAttribute('src');
    }
  }
  card.classList.toggle('is-playing', on);
  card.classList.toggle('is-paused', !on);
  card.setAttribute('aria-pressed', on ? 'true' : 'false');
}
function hvPlay(card){
  if(HV.current && HV.current !== card) hvSetPlaying(HV.current, false);
  HV.current = card;
  hvSetPlaying(card, true);
}
function hvToggle(card){
  if(card.classList.contains('is-playing')){
    HV.current = null; hvSetLock(card);       // paused by hand: nothing else auto-starts while it stays on screen
    hvSetPlaying(card, false);
  } else {
    hvSetLock(null);
    hvPlay(card);
  }
}
function hvRatio(c){ return HV.ratios.get(c) || 0; }
function hvEvaluate(){
  if(document.hidden){
    if(HV.current){ hvSetPlaying(HV.current, false); HV.current = null; }
    return;
  }
  if(HV.lock && hvRatio(HV.lock) < 0.15) hvSetLock(null);                      // scrolled away: released
  if(HV.current && hvRatio(HV.current) < 0.25){ hvSetPlaying(HV.current, false); HV.current = null; }
  if(HV.current || HV.lock) return;
  if(window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;   // tap-only
  const next = HV.cards.find(c=>hvRatio(c) >= 0.6);
  if(next) hvPlay(next);
}
function hvTeardown(){
  if(HV.obs){ HV.obs.disconnect(); HV.obs = null; }
  if(HV.preObs){ HV.preObs.disconnect(); HV.preObs = null; }
  HV.ratios.clear(); HV.cards = []; HV.current = null; HV.lock = null;
}
function hvInit(wrap){
  hvTeardown();
  HV.cards = Array.from(wrap.querySelectorAll('.home-video-card[data-hv-media]'));
  if(!HV.cards.length) return;
  if(!('IntersectionObserver' in window)){ HV.cards.forEach(hvLoadStill); return; }   // tap-only fallback
  HV.obs = new IntersectionObserver(es=>{
    es.forEach(e=>HV.ratios.set(e.target, e.intersectionRatio));
    hvEvaluate();
  }, {threshold:[0,0.1,0.25,0.4,0.5,0.6,0.75,0.9,1]});
  HV.preObs = new IntersectionObserver(es=>{
    es.forEach(e=>{ if(e.isIntersecting){ hvLoadStill(e.target); HV.preObs && HV.preObs.unobserve(e.target); } });
  }, {rootMargin:'300px 0px'});
  HV.cards.forEach(c=>{ HV.obs.observe(c); HV.preObs.observe(c); });
}

/* ---------------- RENDER: PAYMENT PLANS ---------------- */
/* ================= PLAN PRICING (admin discount + first-purchase offer + promo popup) =====
   A configured promo-popup discount has priority over plan-level and first-purchase discounts
   for its target plan. A popup may target one plan or all paid plans (__all_paid__). If no promo
   applies to a plan, the best applicable regular discount is used as before.
   Returns { finalPrice, originalPrice, off, label } or null if no discount applies. */
function computePlanDiscount(plan){
  const price = Number(plan.price) || 0;
  if(price <= 0) return null;
  function offFor(d){
    if(!d || !d.on) return 0;
    const v = Number(d.value) || 0;
    return d.type === 'amount' ? v : Math.round(price * v / 100);
  }
  const pd = window.activePromoDiscount;
  const promoTargetsPlan = pd && (pd.applyToAll === true || pd.planId === '__all_paid__' || pd.planId === plan.id);
  if(promoTargetsPlan){
    const value = Number(pd.value) || 0;
    const promoOff = pd.type === 'amount' ? value : Math.round(price * value / 100);
    if(promoOff > 0){
      const off = Math.min(promoOff, price); // never discount below ৳0
      return { finalPrice: price - off, originalPrice: price, off, label: pd.label || 'Promo discount', isPromo:true };
    }
  }
  const candidates = [];
  if(plan.discount && plan.discount.on){
    const off = offFor(plan.discount);
    if(off > 0) candidates.push({ off, label: plan.discount.label || 'Discount' });
  }
  if(plan.firstPurchaseOffer && plan.firstPurchaseOffer.on && currentUser && !myHasEverOrdered){
    const off = offFor(plan.firstPurchaseOffer);
    if(off > 0) candidates.push({ off, label: plan.firstPurchaseOffer.label || 'First purchase offer' });
  }
  if(!candidates.length) return null;
  candidates.sort((a,b)=>b.off - a.off);
  const best = candidates[0];
  const off = Math.min(best.off, price); // never discount below ৳0
  return { finalPrice: price - off, originalPrice: price, off, label: best.label };
}

function ethvelleEstimateFace(srcCanvas,joints){
  if(!srcCanvas||!joints||!joints.headTop||!joints.neck) return null;
  const top=joints.headTop, neck=joints.neck;
  const headH=Math.max(0.025,neck.y-top.y);
  const shoulderSpan=Math.abs((joints.shoulderR?.x??(top.x+0.12))-(joints.shoulderL?.x??(top.x-0.12)));
  const headW=Math.max(0.045,Math.min(0.34,shoulderSpan*0.65||headH*0.75));
  const cx=neck.x;
  const face={
    leftEye:{x:Math.max(0.005,cx-headW*0.19),y:Math.min(.99,top.y+headH*.49),w:headW*.18,h:headH*.075},
    rightEye:{x:Math.min(.995,cx+headW*0.19),y:Math.min(.99,top.y+headH*.49),w:headW*.18,h:headH*.075},
    mouth:{x:cx,y:Math.min(.995,top.y+headH*.88),w:headW*.24,h:headH*.095},
    source:'estimated'
  };
  try{
    const ctx=srcCanvas.getContext('2d',{willReadFrequently:true});
    const px=Math.round(Math.max(0,Math.min(srcCanvas.width-1,cx*srcCanvas.width)));
    const py=Math.round(Math.max(0,Math.min(srcCanvas.height-1,(top.y+headH*.66)*srcCanvas.height)));
    const r=Math.max(2,Math.round(Math.min(srcCanvas.width,srcCanvas.height)*.006));
    const d=ctx.getImageData(Math.max(0,px-r),Math.max(0,py-r),Math.min(srcCanvas.width-px+r,r*2+1),Math.min(srcCanvas.height-py+r,r*2+1)).data;
    let R=0,G=0,B=0,n=0;
    for(let i=0;i<d.length;i+=4){

      const lum=.2126*d[i]+.7152*d[i+1]+.0722*d[i+2];
      if(lum<38||lum>246)continue;
      R+=d[i];G+=d[i+1];B+=d[i+2];n++;
    }
    if(n) face.skinColor='#'+[R/n,G/n,B/n].map(v=>Math.round(v).toString(16).padStart(2,'0')).join('');

function ethvelleNormalizeFace(raw,srcCanvas,joints){
  const fallback=ethvelleEstimateFace(srcCanvas,joints);
  if(!raw||typeof raw!=='object') return fallback;
  const out={...(fallback||{}),source:'gemini'};

}

function ethvelleFaceTime(){
  if(Number.isFinite(faceAnimationTimeOverride))return faceAnimationTimeOverride;
  return Number.isFinite(window.currentAbsoluteTimeSec)?window.currentAbsoluteTimeSec:(typeof currentAbsoluteTimeSec==='number'?currentAbsoluteTimeSec:0);
}
function ethvelleMapFacePoint(char,point,bStart,bEnd){
  const mesh=char.skinMesh,face=char.autoRig?.face;
  if(!mesh||!mesh.restBones?.head||!point)return null;
  const rest=mesh.restBones.head,s=bStart('head'),e=bEnd('head');
  const angle=Math.atan2(e.y-s.y,e.x-s.x),len=Math.max(1e-5,Math.hypot(e.x-s.x,e.y-s.y));
  const x=point.x*mesh.srcW,y=point.y*mesh.srcH,dx=x-rest.ax,dy=y-rest.ay;
  const ca=Math.cos(-rest.angle),sa=Math.sin(-rest.angle),lx=dx*ca-dy*sa,ly=dx*sa+dy*ca;
  const scale=Math.max(.92,Math.min(1.08,len/rest.len)),rot=Math.cos(angle),sin=Math.sin(angle),delta=angle-rest.angle;
  return {x:s.x+(lx*scale*Math.cos(angle)-ly*scale*Math.sin(angle)),y:s.y+(lx*scale*Math.sin(angle)+ly*scale*Math.cos(angle)),angle:delta,scale};
}
function drawEthvelleFaceMotion(targetCtx,char,bStart,bEnd){
  const face=char.autoRig?.face,mesh=char.skinMesh;
  if(!face||!mesh||!mesh.restBones?.head)return;
  const t=ethvelleFaceTime();
  const hash=String(char.id||'c').split('').reduce((a,c)=>a+c.charCodeAt(0),0)%37/37;
  const phase=((t+hash*1.7+0.8)%3.7+3.7)%3.7;
  const blink=phase<0.105 || (phase>3.48&&phase<3.58);
  const clipKey=(char.id===activeCharacter()?.id&&activeClip)?activeClip.animKey:(char.clip?.animKey||null);
  const talking=(clipKey==='talk'||clipKey==='talkBlink') || char.faceMotion?.talking===true;
  const skin= /^#[0-9a-f]{6}$/i.test(face.skinColor||'')?face.skinColor:'#dba982';
  function drawFeature(name,mode){
    const f=face[name];if(!f)return;
    if(mode==='eye'&&!blink)return;
    if(mode==='mouth'&&!talking)return;
    const center=ethvelleMapFacePoint(char,f,bStart,bEnd);if(!center)return;
    const w=Math.max(1,Number(f.w||.02)*mesh.srcW*center.scale),h=Math.max(1,Number(f.h||.01)*mesh.srcH*center.scale);
    targetCtx.save();targetCtx.translate(center.x,center.y);targetCtx.rotate(center.angle);
    targetCtx.fillStyle=skin;targetCtx.beginPath();targetCtx.ellipse(0,0,w*.72,h*.9,0,0,Math.PI*2);targetCtx.fill();
    targetCtx.lineCap='round';targetCtx.lineJoin='round';
    if(mode==='eye'){
      if(blink){targetCtx.strokeStyle='#392922';targetCtx.lineWidth=Math.max(1,w*.10);targetCtx.beginPath();targetCtx.moveTo(-w*.48,h*.05);targetCtx.quadraticCurveTo(0,h*.34,w*.48,h*.05);targetCtx.stroke();}
    }else if(mode==='mouth'&&(talking)){
      const open=.10+.5*(.5+.5*Math.sin(t*16.5+hash*3));
      targetCtx.fillStyle='#3e1720';targetCtx.strokeStyle='#3e1720';targetCtx.lineWidth=Math.max(1,w*.055);
      targetCtx.beginPath();targetCtx.ellipse(0,0,w*.47,Math.max(h*.12,h*open),0,0,Math.PI*2);targetCtx.fill();targetCtx.stroke();
      if(open>.34){targetCtx.fillStyle='#e9788a';targetCtx.beginPath();targetCtx.ellipse(0,h*open*.43,w*.22,h*.10,0,0,Math.PI*2);targetCtx.fill();}
    }
    targetCtx.restore();
  }
  drawFeature('leftEye','eye');drawFeature('rightEye','eye');drawFeature('mouth','mouth');
}
