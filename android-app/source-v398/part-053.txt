<script id="ethvelle-v97-crop-commit-guard-js">
(function(){
  'use strict';
  const $=id=>document.getElementById(id);

  // The crop fractions are already part of each image/video/background clip and are
  // consumed by preview + export. Re-apply the crop transform after renderer passes so
  // a later base-transform paint cannot visually undo an applied crop.
  function applyCommittedCrop(){
    try{
      if(window.__ethvelleCropEditingId)return;
      if(typeof window.ethvelleApplyCropPreview==='function')window.ethvelleApplyCropPreview();
    }catch(err){console.warn('Ethvelle crop commit refresh:',err);}
  }
  function wrapCropRender(name){
    const base=window[name];
    if(typeof base!=='function'||base.__evv97CropCommit)return;
    const wrapped=function(){
      const result=base.apply(this,arguments);
      applyCommittedCrop();
      return result;
    };
    wrapped.__evv97CropCommit=true;
    window[name]=wrapped;
  }
  // This final wrapper is deliberately installed after all earlier preview optimizers.
  wrapCropRender('renderOverlayMedia');
  wrapCropRender('applyBackground');
  wrapCropRender('applyBgTransform');

  // Apply crop after the Crop panel closes, then reassert it once more on the following
  // frame. This covers Android WebView's delayed image/video style updates and guarantees
  // that pressing Apply commits the same crop the export path already reads from clip.crop.
  const baseClose=window.closePanel;
  if(typeof baseClose==='function'&&!baseClose.__evv97CropCommit){
    const wrappedClose=function(){
      const cropId=String(window.__ethvelleCropEditingId||'');
      const result=baseClose.apply(this,arguments);
      if(cropId){
        const commit=()=>{
          try{
            const clip=typeof window.ethvelleObjectClipById==='function'?window.ethvelleObjectClipById(cropId):null;
            if(clip&&typeof window.ethvelleApplyCropPreview==='function')window.ethvelleApplyCropPreview();
          }catch(err){console.warn('Ethvelle crop apply:',err);}
        };
        requestAnimationFrame(()=>{commit();requestAnimationFrame(commit);});
      }
      return result;
    };
    wrappedClose.__evv97CropCommit=true;
    window.closePanel=wrappedClose;
  }

  // No MutationObserver is attached to the media layer: observing every video-frame
  // style update would add unnecessary work during playback and timeline interactions.
})();

  function cropFractions(clip){
    let c={left:0,right:0,top:0,bottom:0};
    try{c=typeof window.ethvelleCropFractions==='function'?window.ethvelleCropFractions(clip.crop):(clip.crop||c);}catch(_e){}
    const clamp=v=>Math.max(0,Math.min(.92,Number(v)||0));
    let left=clamp(c.left),right=clamp(c.right),top=clamp(c.top),bottom=clamp(c.bottom);
    if(left+right>.92){const f=.92/(left+right);left*=f;right*=f;}

  function sourceGeometry(clip,element,kind,target){

          clip=typeof backgroundClipForSelection==='function'?backgroundClipForSelection():null;
          el=$('bgImgLayer');
        }
        if(clip&&el){
          const sw=Math.max(1,stage.clientWidth||stageW||1),sh=Math.max(1,stage.clientHeight||stageH||1);
          const iw=Math.max(1,Number(el.videoWidth||el.naturalWidth||clip.sourceWidth)||sw),ih=Math.max(1,Number(el.videoHeight||el.naturalHeight||clip.sourceHeight)||sh);
          const contain=Math.min(sw/iw,sh/ih);
          const fullW=iw*contain,fullH=ih*contain;
          // Shrink the selection box to the cropped region so the white border hugs the visible content.
          let cL=0,cR=0,cT=0,cB2=0;
          try{
            const cropRaw=typeof window.ethvelleCropFractions==='function'?window.ethvelleCropFractions(clip.crop):(clip.crop||{});
            const clampC=v=>Math.max(0,Math.min(.92,Number(v)||0));
            cL=clampC(cropRaw.left);cR=clampC(cropRaw.right);cT=clampC(cropRaw.top);cB2=clampC(cropRaw.bottom);
            if(cL+cR>.92){const f=.92/(cL+cR);cL*=f;cR*=f;}
            if(cT+cB2>.92){const f=.92/(cT+cB2);cT*=f;cB2*=f;}
          }catch(_e){}
          const hasCrop=cL+cR+cT+cB2>.0001;
          if(hasCrop){
            // Cropped visible size
            target.w0=fullW*(1-cL-cR);
            target.h0=fullH*(1-cT-cB2);
            // Shift the centre (cB) to the centre of the cropped region
            const letterX=(sw-fullW)/2,letterY=(sh-fullH)/2;
            const cropCx=letterX+(cL+1-cR)*fullW/2-(sw/2);
            const cropCy=letterY+(cT+1-cB2)*fullH/2-(sh/2);
            target.cB={x:sw/2+cropCx,y:sh/2+cropCy};
          }else{
            target.w0=fullW;target.h0=fullH;
          }
          const tracked=kind==='overlay'&&typeof window.ethvelleGetTrackedOverlayTransform==='function'?window.ethvelleGetTrackedOverlayTransform(clip):(clip.transform||{});
          const keyed=kind==='overlay'&&typeof window.ethvelleObjectKeyframeProperty==='function'&&window.ethvelleObjectKeyframeProperty(clip)==='transform'&&typeof window.ethvelleGetObjectKeyframeValue==='function'
            ?(window.ethvelleGetObjectKeyframeValue(clip,'transform',now)||tracked||clip.transform||{}):
            (kind==='background'&&typeof window.ethvelleGetObjectKeyframeValue==='function'?(window.ethvelleGetObjectKeyframeValue(clip,'transform',now)||clip.transform||{}):(tracked||clip.transform||{}));
          const motion=kind==='overlay'&&typeof window.getEthvelleClipAnimationState==='function'?window.getEthvelleClipAnimationState(clip,now):{x:0,y:0,rotation:0,scale:1};
          target.obj={...(keyed||{}),offsetX:(Number(keyed?.offsetX)||0)+(Number(motion?.x)||0),offsetY:(Number(keyed?.offsetY)||0)+(Number(motion?.y)||0),rotation:(Number(keyed?.rotation)||0)+(Number(motion?.rotation)||0),scale:(Number(keyed?.scale)||1)*(Number(motion?.scale)||1)};
        }
      }catch(err){console.warn('Ethvelle v100 selection bounds:',err);}
      return target;
    };
    wrapped.__evv100NoZoomCrop=true;window.selBoxTarget=wrapped;
  }

})();
