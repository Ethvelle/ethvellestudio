    // clip's `src`, and persistProjectState() writes that string verbatim into the saved
    // project. Blob URLs are only valid for the CURRENT page session  -  the instant the page
    // is closed/refreshed and the project is reopened, that exact blob: address no longer
    // resolves to anything. The clip's data (trimStart/trimEnd/name/thumbs) all survived in
    // localStorage just fine, so the clip still visibly showed up on the timeline after a
    // refresh  -  but its actual picture/video (src) was already dead, so it played back as a
    // blank/broken layer, reading as if "the clip didn't save". FIX: read the file into a
    // self-contained base64 data: URL first (identical technique to the Background fix
    // above), and use THAT as both the thumbnail-generation source and the persisted `src`  - 
    // nothing blob-based is ever stored. Video/audio can still be a large data: URL (same
    // localStorage-quota tradeoff the Background image already accepted  -  see
    // saveProjectsToStorage's own BUGFIX note), but a working clip that's occasionally too
    // big to save is a a strict improvement over a small clip that ALWAYS silently breaks.
    showToast(isVideo ? 'Loading video…' : 'Loading…', {icon: toastIcons.info, duration:1200});
    const reader = new FileReader();
    reader.onload = ()=>{
      const dataUrl = reader.result;
      if(isVideo){
        generateVideoThumbnails(dataUrl, 24, 0, null).then(({duration,width,height})=>{
          addMediaOverlay({ type:'video', src:dataUrl, name:file.name, duration, thumbs:[], sourceWidth:width, sourceHeight:height, appendAfterGroup:uploadAppendGroup });
        });
      } else {
        // Image overlay: single thumbnail tiled across the clip body, like CapCut stickers/images.
        addMediaOverlay({ type:'image', src:dataUrl, name:file.name, duration:3, thumbs:[dataUrl], appendAfterGroup:uploadAppendGroup });
      }
    };
    reader.onerror = ()=>{
      showToast("Couldn't read that file", {icon: toastIcons.info, duration:1800});
    };
    reader.readAsDataURL(file);
    closePanel();
  } else if(uploadTarget==='audio'){
    showToast('Loading audio…', {icon: toastIcons.info, duration:1200});
    // Same base64-instead-of-blob fix as the video/image branch above  -  see that BUGFIX note.
    const reader = new FileReader();
    reader.onload = ()=>{
      const dataUrl = reader.result;
      generateAudioWaveform(dataUrl).then(({peaks, duration})=>{
        addMediaOverlay({ type:'audio', src:dataUrl, name:file.name, duration:duration||3, peaks, appendAfterGroup:uploadAppendGroup });
      });
    };
    reader.onerror = ()=>{
      showToast("Couldn't read that audio file", {icon: toastIcons.info, duration:1800});
    };
    reader.readAsDataURL(file);
    closePanel();
  } else if(uploadTarget==='replace-overlay'){
    // Replace button (see replaceSelectedClip)  -  swaps the SRC/type/thumbnails of the
    // already-selected media clip in place, keeping its id/trimStart/trimEnd/transform
    // exactly where they were on the timeline. Unlike addMediaOverlay this never creates a
    // new clip or changes its position  -  only its underlying picture/video/audio content.
    const ov = overlays.find(o=>o.id===replaceTargetId);
    if(!ov){ closePanel(); return; }
    showToast(isVideo ? 'Loading video…' : isAudio ? 'Loading audio…' : 'Loading…', {icon: toastIcons.info, duration:1200});
    const reader = new FileReader();
    reader.onload = ()=>{
      const dataUrl = reader.result;
      const finish = ({newType, thumbs, peaks, duration, sourceWidth, sourceHeight})=>{
        ov.type = newType;
        ov.src = dataUrl;
        ov.name = file.name;
        ov.sourceDuration = duration;
        if(newType==='video'){ov.sourceWidth=Number(sourceWidth)||0;ov.sourceHeight=Number(sourceHeight)||0;}
        ov.thumbs = newType==='video'?[]:(thumbs||[]);
        ov.peaks = peaks||[];
        // A shape/audio clip has no on-stage transform; an image/video does. Give it one if
        // it's newly gaining a visual and didn't have one before, same shape addMediaOverlay uses.
        if((newType==='video'||newType==='image') && !ov.transform) ov.transform = { scale:1, offsetX:0, offsetY:0, rotation:0 };
        renderOverlayLayers();
        renderOverlayListPanel();
        renderFrame();
        showToast('Replaced', {icon: toastIcons.check, duration:1200});
        persistProjectState();
      };
      if(isVideo){
        generateVideoThumbnails(dataUrl, 24, 0, null).then(({duration,width,height})=> finish({newType:'video', thumbs:[], duration, sourceWidth:width, sourceHeight:height}));
      } else if(isAudio){
        generateAudioWaveform(dataUrl).then(({peaks, duration})=> finish({newType:'audio', peaks, duration: duration||3}));
      } else {
        finish({newType:'image', thumbs:[dataUrl], duration: ov.sourceDuration||3});
      }
    };
    reader.onerror = ()=>{
      showToast("Couldn't read that file", {icon: toastIcons.info, duration:1800});
    };
    reader.readAsDataURL(file);
    closePanel();
    replaceTargetId = null;
  } else {
    showToast('Upload received  -  layer support coming next version');
    closePanel();
  }
  uploadAppendGroup = null;
  e.target.value='';
});

// Unified entry point for adding uploaded media (image/video/audio) into the shared
// Clips or Audio group track. The current playhead decides the insertion point: before/left
// of a clip in its first half, after/right in its second half, or before the next clip in a gap.
// Every item in that track is then packed into one sequential lane.
function addMediaOverlay({type, src, name, duration, thumbs, peaks, appendAfterGroup, deviceMediaKey, deviceMediaInfo, sourceDuration, sourceOffset, sourceWidth, sourceHeight, adjustments, chromaKey, enhanceAmount, filterPreset}){
  const id = 'ov'+Date.now();
  const color = OVERLAY_COLORS[overlays.length % OVERLAY_COLORS.length];
  const clipLen = type==='video' ? Math.max(MIN_CLIP_SEC, Number(duration)||MIN_CLIP_SEC) : Math.max(MIN_CLIP_SEC,Math.min(Number(duration)||3,12)); // video clips keep their full source duration; only stills use the short default
  let startT=0;
  const groupKey = appendAfterGroup || (type==='audio' ? 'audio' : 'clips');
  const groupClips = overlays.filter(o=> (o.type==='audio'?'audio':(o.type==='text'?'text':'clips')) === groupKey);
  // Insert relative to the playhead, like a single-track editor: first half of the
  // clip => before/left of it; second half => after/right of it; a playhead sitting in
  // the gap before a clip => before that clip. Then repack the row sequentially.
  const orderedGroupClips=[...groupClips].sort((a,b)=>(Number(a.trimStart)||0)-(Number(b.trimStart)||0));
  const groupClipLengths=new Map(orderedGroupClips.map(o=>[String(o.id),Math.max(MIN_CLIP_SEC,(Number(o.trimEnd)||0)-(Number(o.trimStart)||0))]));
  let playheadTime=0;
  try{playheadTime=Math.max(0,Number(typeof getCurrentAbsoluteTime==='function'?getCurrentAbsoluteTime():(typeof currentAbsoluteTimeSec!=='undefined'?currentAbsoluteTimeSec:0))||0);}catch(_e){}
  let insertIndex=orderedGroupClips.length;
  for(let i=0;i<orderedGroupClips.length;i++){
    const existing=orderedGroupClips[i];
    const start=Number(existing.trimStart)||0;
    const len=groupClipLengths.get(String(existing.id))||MIN_CLIP_SEC;
    const end=start+len;
    // A playhead exactly at / just before a clip's leading edge is treated as "before it".
    if(playheadTime <= start+0.001){insertIndex=i;break;}
    if(playheadTime < end-0.001){
      insertIndex=playheadTime < start+len/2 ? i : i+1;
      break;
    }
  }
  startT=orderedGroupClips.slice(0,insertIndex).reduce((sum,o)=>sum+(groupClipLengths.get(String(o.id))||MIN_CLIP_SEC),0);
  const newClip = {
    id, name: name || (type==='video'?'Video':type==='audio'?'Audio':'Image'), color,
    type, src, deviceMediaKey: deviceMediaKey || undefined,
    deviceMediaName: deviceMediaInfo&&deviceMediaInfo.name || undefined,
    deviceMediaSize: deviceMediaInfo&&Number(deviceMediaInfo.size)||undefined,
    deviceMediaLastModified: deviceMediaInfo&&Number(deviceMediaInfo.lastModified)||undefined,
    sourceDuration: Number(sourceDuration)>0?Number(sourceDuration):(Number(duration)||clipLen), sourceOffset: Math.max(0,Number(sourceOffset)||0),
    sourceWidth: Number(sourceWidth)>0?Number(sourceWidth):0, sourceHeight: Number(sourceHeight)>0?Number(sourceHeight):0,
    trimStart: startT, trimEnd: startT + clipLen,
    adjustments: adjustments&&typeof adjustments==='object'?{...adjustments}:undefined,
    chromaKey: chromaKey&&typeof chromaKey==='object'?{enabled:!!chromaKey.enabled,color:chromaKey.color||'#00ff00',tolerance:Number(chromaKey.tolerance)||75,softness:Number(chromaKey.softness)||24}:undefined,
    enhanceAmount: Number(enhanceAmount)||0, filterPreset:filterPreset||'original',
    thumbs: thumbs||[], peaks: peaks||[], speed: 1,
    // FEATURE (volume control + audio extraction): every clip that can carry sound (video
    // or audio) now has its own volume multiplier, 0 (silent) to 2 (2x boost), defaulting
    // to 1 (unchanged)  -  see the Volume panel (renderVolumePanel/volumePanelSetVolume,
    // mirrors the existing Speed panel) and its live application in renderOverlayMedia/
    // renderAudioOverlayMedia below. An image/shape clip has nothing to carry volume on, so
    // stays undefined there (same pattern transform already uses for audio).
    volume: (type==='video'||type==='audio') ? 1 : undefined,
    // Own independent size/position/rotation (image/video overlays only  -  audio/shape have
    // no on-stage visual of their own to move). Lets the Transform panel adjust THIS clip's
    // picture on the canvas the same way it already adjusts the stickman  -  see
    // updateTransform's selectedClip-aware BUGFIX note.
    transform: (type==='video'||type==='image') ? { scale:1, offsetX:0, offsetY:0, rotation:0 } : null
  };
  const orderedWithNew=[...orderedGroupClips.slice(0,insertIndex),newClip,...orderedGroupClips.slice(insertIndex)];
  let groupCursor=0;
  orderedWithNew.forEach(item=>{
    const oldStart=Number(item.trimStart)||0;
    const oldKeyframeTimes=Array.isArray(item.ethvelleMediaKeyframes)?item.ethvelleMediaKeyframes.map(k=>Number(k&&k.t)||0):[];
    const len=item===newClip?clipLen:(groupClipLengths.get(String(item.id))||Math.max(MIN_CLIP_SEC,(Number(item.trimEnd)||0)-(Number(item.trimStart)||0)));
    item.trimStart=groupCursor;
    item.trimEnd=groupCursor+len;
    const movedBy=item.trimStart-oldStart;
    if(item!==newClip&&Math.abs(movedBy)>0.000001)shiftTimelineKeyframesOnce(item.ethvelleMediaKeyframes,oldKeyframeTimes,movedBy);
    groupCursor=item.trimEnd;
  });
  startT=Number(newClip.trimStart)||0;
  overlays.push(newClip);
  timelineDuration = Math.max(Number(timelineDuration)||0, groupCursor, startT+clipLen);
  renderOverlayLayers();
  syncTimelineInnerWidth();
  renderRuler();
  onClipSelected(newClip,'add');
  renderOverlayListPanel();
  // Immediately redraw the current playhead frame after the timeline is rearranged, so
  // the preview reflects the selected clip order without requiring a later play/scrub action.
  renderFrame();
  showToast((type==='video'?'Video':type==='audio'?'Audio':'Image')+(appendAfterGroup?' added to layer':' added'), {icon: toastIcons.check, duration:1200});
  persistProjectState();
}

// Background clips use the same midpoint insertion rule as every other timeline row:
// first half inserts before/left; second half inserts after/right; gaps insert before next.
function addBackgroundClip(src){
  const id='bg'+Date.now(),len=Math.max(MIN_CLIP_SEC,Number(BG_DEFAULT_DURATION)||3);
  const ordered=[...backgrounds].sort((a,b)=>(Number(a.trimStart)||0)-(Number(b.trimStart)||0));
  let playhead=0;try{playhead=Math.max(0,Number(typeof getCurrentAbsoluteTime==='function'?getCurrentAbsoluteTime():currentAbsoluteTimeSec)||0);}catch(_e){}
  let at=ordered.length;
  for(let i=0;i<ordered.length;i++){
    const c=ordered[i],st=Number(c.trimStart)||0,l=Math.max(MIN_CLIP_SEC,(Number(c.trimEnd)||st+MIN_CLIP_SEC)-st),en=st+l;
    if(playhead<=st+.001){at=i;break;}
    if(playhead<en-.001){at=playhead<(st+en)/2?i:i+1;break;}
  }
  const clip={id,src,trimStart:0,trimEnd:len,transform:{scale:1,offsetX:0,offsetY:0,rotation:0}};
  const arranged=[...ordered.slice(0,at),clip,...ordered.slice(at)];let cursor=0;
  arranged.forEach(c=>{
    const oldStart=Number(c.trimStart)||0;
    const oldKeyframeTimes=Array.isArray(c.ethvelleMediaKeyframes)?c.ethvelleMediaKeyframes.map(k=>Number(k&&k.t)||0):[];
    const l=c===clip?len:Math.max(MIN_CLIP_SEC,(Number(c.trimEnd)||0)-(Number(c.trimStart)||0));
    c.trimStart=cursor;c.trimEnd=cursor+l;
    const movedBy=c.trimStart-oldStart;
    if(c!==clip&&Math.abs(movedBy)>0.000001)shiftTimelineKeyframesOnce(c.ethvelleMediaKeyframes,oldKeyframeTimes,movedBy);
    cursor=c.trimEnd;
  });
  backgrounds.splice(0,backgrounds.length,...arranged);timelineDuration=Math.max(Number(timelineDuration)||0,cursor);
  renderBackgroundTrack();syncTimelineInnerWidth();renderRuler();
  onClipSelected(clip,'add');
  refreshCurrentBgFromPlayhead();applyBackground();updateSelBox();persistProjectState();return clip;
}

// animationAccess[key] is a list of plan ids an admin checked for this animation (see the
// admin panel's multi-select Animation access rules). Older data saved before multi-plan
// support stored a single plan-id (or name, or 'free'/empty) as a bare string  -  normalized
// here into a one-item array so both shapes read identically.
function getAnimationAccessPlans(key){
  const map=siteContent.animationAccess||{}; const raw=map[key];
  const plans=typeof getPlans==='function'?getPlans():[];
  const free=plans.find(p=>Number(p.price)===0)||{id:'p_free',name:'Free',price:0};
  const ids=Array.isArray(raw)?raw.map(String).filter(Boolean):(raw==null||raw==='')?[]:[String(raw)];
  if(!ids.length)return [free];
  const resolved=ids.map(id=>{
    const normalized=id.toLowerCase();
    return plans.find(p=>String(p.id)===id) ||
      plans.find(p=>String(p.name||'').toLowerCase()===normalized) ||
      ((normalized==='free'||normalized==='p_free')?free:null);
  }).filter(Boolean);
  return resolved.length?resolved:[free];
}
// The badge/lock shown to a visitor is driven by the CHEAPEST plan an admin checked for this
// animation  -  e.g. if both Pro and Studio are checked, a Pro subscriber should already be able
// to use it rather than being told they need Studio.
function getAnimationRequiredPlan(key){
  const options=getAnimationAccessPlans(key);
  return options.slice().sort((a,b)=>Number(a.price||0)-Number(b.price||0))[0];
}
function isAnimationLocked(key){
  const required=getAnimationRequiredPlan(key);
  if(!required||Number(required.price)<=0)return false;
  const current=typeof getMyPlanData==='function'?getMyPlanData():{id:'p_free',name:'Free',price:0};
  // Treat the plans as tiers by price, so a higher tier (or an exact plan match) can use the
  // animation even if its id isn't literally in the checked list  -  the cheapest checked plan
  // sets the floor price a user's current plan must meet.
  const options=getAnimationAccessPlans(key);
  const directMatch=!!current && options.some(p=>String(p.id)===String(current.id));
  if(directMatch) return false;
  return !current || Number(current.price||0) < Number(required.price||0);
}

function ensureAdminFontLoaded(assetOrFamily){

  if(!family)return;
  const safeFamily=family.replace(/\\/g,'\\\\').replace(/"/g,'\\"');
  if(asset?.src){
    const src=String(asset.src);
    const key='ethvelle-font-'+family.toLowerCase().replace(/[^a-z0-9]+/g,'-');
    if(document.getElementById(key))return;
    if(/\.(woff2?|ttf|otf)(?:[?#].*)?$/i.test(src)){

      style.textContent='@font-face{font-family:"'+safeFamily+'";src:url("'+src.replace(/"/g,'%22')+'");font-display:swap;}';
      document.head.appendChild(style);
    }else if(/^https:\/\//i.test(src)){
      const link=document.createElement('link');link.id=key;link.rel='stylesheet';link.href=src;document.head.appendChild(link);
    }
  }else{
    const key='ethvelle-font-'+family.toLowerCase().replace(/[^a-z0-9]+/g,'-');

  }
}

async function downloadPromptImage(button){
  if(!button||button.disabled)return;
  const url=button.dataset.src||'';
  if(!url){showToast('Image is unavailable',{icon:toastIcons.info});return;}
  const kind=button.dataset.kind==='template'?'template':'reference';
  const title=String(button.dataset.title||'Ethvelle').normalize('NFKD').replace(/[\u0300-\u036f]/g,'').replace(/[^a-zA-Z0-9_-]+/g,'-').replace(/^-+|-+$/g,'').slice(0,55)||'Ethvelle';
  const baseName=`${title}-${kind}-image`;
  const originalText=button.querySelector('span')?.textContent||'Download';
  button.disabled=true;
  if(button.querySelector('span'))button.querySelector('span').textContent='Downloading…';
  try{
    const response=await fetch(url,{mode:'cors',credentials:'omit'});
    if(!response.ok)throw new Error('Image request failed');
    const blob=await response.blob();
    if(!blob.size)throw new Error('Empty image');
    const mime=(blob.type||'').split(';')[0].toLowerCase();
    const mimeExt={'image/jpeg':'jpg','image/jpg':'jpg','image/png':'png','image/webp':'webp','image/gif':'gif','image/avif':'avif','image/bmp':'bmp','image/svg+xml':'svg'};
    let ext=mimeExt[mime]||'';
    if(!ext){try{const path=new URL(url,location.href).pathname;const match=path.match(/\.([a-z0-9]{2,5})$/i);ext=match?match[1].toLowerCase():'jpg';}catch(e){ext='jpg';}}
    const objectUrl=URL.createObjectURL(blob);
    const link=document.createElement('a');link.href=objectUrl;link.download=`${baseName}.${ext}`;link.style.display='none';
    document.body.appendChild(link);link.click();link.remove();
    setTimeout(()=>URL.revokeObjectURL(objectUrl),1500);
    showToast('Image downloaded',{icon:toastIcons.check});
  }catch(error){
    // Some image hosts block cross-origin downloads. Open the source directly as a fallback.
    const link=document.createElement('a');link.href=url;link.target='_blank';link.rel='noopener noreferrer';link.download=`${baseName}.jpg`;link.style.display='none';
    document.body.appendChild(link);link.click();link.remove();
    showToast('Opening image  -  use Save image if download does not start',{icon:toastIcons.info});
  }finally{
    button.disabled=false;
    if(button.querySelector('span'))button.querySelector('span').textContent=originalText;
  }
}

// FEATURE (admin-authored Custom Animations, published from the admin panel's new
// "Custom animations" page): merges every admin-published preset into the live animPresets
// object so they show up in the Animate picker exactly like idle/walk/wave/jump do, for
// every user. Only ever ADDS/overwrites admin-authored keys (each one prefixed 'custom_' by
// the admin panel when it creates them, so a custom preset can never collide with or
// shadow a built-in key)  -  never deletes from animPresets, so removing a custom preset in
// the admin panel just means its key stops being re-applied on the next sync; a project
// that already has a clip using a deleted custom preset keeps working until the user picks
// something else (same "orphaned animKey" tolerance evalPreset/animPresets[clip.animKey]
// already has to handle anyway if a key is ever missing).
function applyCustomAnimPresets(){
  // Shape published by the admin panel's Custom Animations page: a flat array of
  // { id, key, name, duration, keyframes }, one per preset  -  key is always 'custom_'-prefixed
  // (see the admin panel's slugifyAnimName) and keyframes are already in radians (the admin
  // panel converts from the degrees an admin types in before publishing).
  const list = siteContent.customAnimPresets;
  if(!Array.isArray(list)) return;
  list.forEach(entry=>{
    if(!entry || !entry.key || !entry.name) return;
    if(!String(entry.key).startsWith('custom_')) return; // defensive: admin panel always sets this prefix
    if(!Array.isArray(entry.keyframes) || !entry.keyframes.length) return; // malformed  -  skip, don't crash the picker
    animPresets[entry.key] = { name:String(entry.name), duration: Number(entry.duration)>0 ? Number(entry.duration) : 1, keyframes: entry.keyframes };
  });
}
function defaultContactInfo(){ return { email:'support@ethvelle.com', whatsapp:'' }; }

// MIGRATION: the built-in Character/Keyframe tiles used to open the ratio picker
// (actionType:'ratio') like Video does. They now have real destinations of their own (the
// Models tab / the new Keyframe info screen  -  see defaultTiles). A site that saved tiles to
// Firestore BEFORE this change (any edit to any tile re-saves the whole array  -  see saveTile
// in the admin panel) would otherwise keep reopening the ratio picker forever, since
// Firestore's stored copy wins over defaultTiles(). Patched here, once per read, rather than
// in Firestore directly, so it self-heals on every device without a manual admin re-save.
function migrateLegacyTiles(tiles){
  if(!Array.isArray(tiles)) return tiles;
  return tiles.map(t=>{
    if(!t || !t.builtin || t.actionType !== 'ratio') return t;
    if(t.id === 't_default_1') return { ...t, actionType:'screen', actionValue:'models' };
    if(t.id === 't_default_4') return { ...t, actionType:'screen', actionValue:'keyframeInfo' };
    return t;
  });
}
function getTiles(){ return migrateLegacyTiles(siteContent.tiles) || defaultTiles(); }

// Resolves a banner/animation/tile tap: either navigate in-app, open the new-project ratio
// picker (the "create" tiles' original behavior), or open an external link.
function runAdminAction(actionType, actionValue){
  if(actionType === 'link' && actionValue){
    window.open(actionValue, '_blank', 'noopener');
  } else if(actionType === 'ratio'){
    openRatioModal();
  } else if(actionValue){
    goTo(actionValue);
  }
}

/* ================= PROMO POPUP =================
   Shows at most one promo popup per app session (not per visit to Home  -  once dismissed or
   acted on, it won't reappear until the page is fully reloaded), picked from whichever active
   popups exist, after its own configured delay. If several are active, the one with the
   shortest delay fires first; the rest are skipped for this session rather than queued, so the
   person is never hit with a second popup right after closing the first.
   window.activePromoDiscount is read by computePlanDiscount() (Payment screen) once a
   "Give a discount" popup is tapped  -  { planId, type, value, label } or null. */
window.activePromoDiscount = null;
const ETHVELLE_PROMO_ATTRIBUTION_KEY = 'ethvelle.promoAttribution.v1';
function setPromoAttribution(pp){
  if(!pp || !pp.id) return;
  try{ sessionStorage.setItem(ETHVELLE_PROMO_ATTRIBUTION_KEY, JSON.stringify({popupId:String(pp.id), clickedAt:Date.now()})); }catch(e){}
}
function getPromoAttribution(){
  try{
    const raw=sessionStorage.getItem(ETHVELLE_PROMO_ATTRIBUTION_KEY);
    if(!raw) return null;
    const data=JSON.parse(raw);
    if(!data || !data.popupId) return null;
    // Attribution is tab/session scoped and expires after four hours to avoid stale cross-journey credit.
    if(data.clickedAt && Date.now()-Number(data.clickedAt)>4*60*60*1000){ sessionStorage.removeItem(ETHVELLE_PROMO_ATTRIBUTION_KEY); return null; }
    return {popupId:String(data.popupId)};
  }catch(e){ return null; }
}
function clearPromoAttribution(){ try{ sessionStorage.removeItem(ETHVELLE_PROMO_ATTRIBUTION_KEY); }catch(e){} }
function withPromoAttribution(details){
  const base=details && typeof details==='object' ? details : {};
  const attribution=getPromoAttribution();
  return attribution ? {...base, popupId:attribution.popupId, source:'promo_popup'} : base;
}
function trackPromoAttributedEvent(kind, details){
  const attribution=getPromoAttribution();
  if(!attribution) return;
  trackEvent(kind, {...(details && typeof details==='object' ? details : {}), popupId:attribution.popupId, source:'promo_popup'});
}
let promoPopupShownThisSession = false;
let promoPopupTimer = null;
let promoPopupCurrent = null;
// When Firestore is available, do not schedule a popup from the stale local cache. Wait for
// the first server-confirmed snapshot, then schedule from the newest published data. This
// prevents the old popup/offer appearing on the first visit after an admin update.
let promoPopupContentServerConfirmed = !(firebaseReady && fbDb);
function promoPopupVisibleOnScreen(pp,screenName){
  const targets=Array.isArray(pp&&pp.showOn)?pp.showOn:(typeof (pp&&pp.showOn)==='string'?[pp.showOn]:['home']);
  return targets.includes('all') || targets.includes(screenName);
}
// Audience scope is independent from the action/discount target. Legacy popups that predate
// audiencePlans remain visible to everyone; newly configured popups can target guests, Free
// users, or one/more paid plan IDs.
function promoPopupVisibleToCurrentUser(pp){
  const targets = Array.isArray(pp && pp.audiencePlans) ? pp.audiencePlans : ['__all__'];
  if(targets.includes('__all__')) return true;
  if(!currentUser) return targets.includes('guest');
  const plans = getPlans();
  const activePlan = plans.find(p=>p.id===myActivePlanId) || plans.find(p=>Number(p.price)===0);
  if(!activePlan || Number(activePlan.price)<=0) return targets.includes('free');
  return targets.includes(activePlan.id);
}
function wirePromoPopups(forceReschedule=false, screenOverride=null){
  if(promoPopupShownThisSession || !promoPopupContentServerConfirmed) return;
  if(promoPopupTimer){
    if(!forceReschedule) return;
    clearTimeout(promoPopupTimer); promoPopupTimer=null;
  }
  const screenName=screenOverride || activeScreenId() || 'home';
  const eligible=getPromoPopups().filter(pp=>pp&&pp.active!==false&&safeImageUrl(pp.image)&&promoPopupVisibleOnScreen(pp,screenName)&&promoPopupVisibleToCurrentUser(pp));
  if(!eligible.length)return;
  const soonest=eligible.slice().sort((a,b)=>(Number(a.delaySeconds)||0)-(Number(b.delaySeconds)||0))[0];
  promoPopupTimer=setTimeout(()=>{
    promoPopupTimer=null;
    if(promoPopupShownThisSession)return;
    const currentScreen=activeScreenId();
    const latest=getPromoPopups().filter(pp=>pp&&pp.active!==false&&safeImageUrl(pp.image)&&promoPopupVisibleOnScreen(pp,currentScreen)&&promoPopupVisibleToCurrentUser(pp))
      .slice().sort((a,b)=>(Number(a.delaySeconds)||0)-(Number(b.delaySeconds)||0))[0];
    if(latest) showPromoPopup(latest);
    else wirePromoPopups(true,currentScreen); // audience/screen may have changed while the delay was running
  },Math.max(0,Number(soonest.delaySeconds)||0)*1000);
}
function showPromoPopup(pp){
  promoPopupCurrent = pp;
  promoPopupShownThisSession = true;
  trackEvent('promo_popup_shown', { popupId:pp && pp.id, screen:trackingActiveScreen() });
  document.getElementById('promoPopupImg').src = safeImageUrl(pp.image);
  document.getElementById('promoPopupBackdrop').classList.add('active');
}
function closePromoPopup(){
  document.getElementById('promoPopupBackdrop').classList.remove('active');
}
function handlePromoPopupTap(){
  const pp = promoPopupCurrent;
  closePromoPopup();
  if(!pp) return;
  setPromoAttribution(pp);
  trackEvent('promo_popup_click', { popupId:pp.id, screen:trackingActiveScreen(), source:'promo_popup' });
  if(pp.actionType === 'discount' && pp.discountPlanId){
    // A promo can target one paid plan or every paid plan (__all_paid__). The promo is
    // intentionally kept for the rest of this page session so the prices stay consistent
    // while the user browses the Payment screen and proceeds to checkout.
    // Older saved popups predate discountScope; treat them as all-plan promos so a
    // previously configured home popup no longer discounts only its legacy target plan.
    const applyToAll = pp.discountPlanId === '__all_paid__' || pp.discountScope !== 'specific';
    const selectionPlanId = pp.discountPlanId !== '__all_paid__' ? pp.discountPlanId : '';
    window.activePromoDiscount = {
      planId: applyToAll ? '__all_paid__' : pp.discountPlanId,
      applyToAll,
      type: pp.discountType || 'percent',
      value: Number(pp.discountValue) || 0,
      label: 'Promo discount'
    };
    _plansSig = null;
    goTo('payment');
    // Navigation alone does not redraw the plans; rebuild them now so the promo prices are
    // visible immediately instead of waiting for an unrelated Firestore snapshot.
    renderPaymentPlans();
    if(selectionPlanId){
      const card = document.querySelector('#screen-payment .plan-card[data-plan="'+selectionPlanId+'"]');
      if(card) selectPlan(card);
    }
  } else {
    runAdminAction(pp.actionType, pp.actionValue);
  }
}

/* ---------------- SAFE HELPERS ---------------- */
// Only ever put http(s) URLs or inline image data URLs into an <img src>.
function safeImageUrl(u){
  return (typeof u === 'string' && /^(https?:\/\/|data:image\/)/i.test(u)) ? u : '';
}
// Re-rendering replaces the whole strip, which snaps the swipe position back to slide 1 and
// makes every picture flash. Firestore snapshots fire more often than the content actually
// changes, so each renderer remembers what it last drew and skips identical data.
let _bannerSig = null, _animSig = null, _plansSig = null, _tutorialsSig = null;

// Banner / example-card taps go through ONE delegated listener reading data-* attributes.
// (The old inline onclick="…runAdminAction('screen', "home")" put JSON's double quotes inside a
// double-quoted attribute, which cut the attribute short  -  so taps never fired.)
function wireContentTaps(){
  ['homeBannerScroll','homeVideoScroll','homeTileRow','tutorialGrid'].forEach(id=>{
    const el = document.getElementById(id);
    if(!el || el.dataset.tapsWired) return;
    el.dataset.tapsWired = '1';
    el.addEventListener('click', e=>{
      const hit = e.target.closest('[data-atype]');
      if(!hit || !el.contains(hit)) return;
      runAdminAction(hit.dataset.atype, hit.dataset.avalue || '');
    });
  });
}

// Image load / error handling for banner <img>s. A failed load is retried once (a CDN that is
// briefly slow shouldn't leave a blank banner) before showing the fallback panel.
function bannerImgLoaded(img){
  const box = img.closest('.home-banner');
  if(box){ box.classList.remove('failed'); box.classList.add('loaded'); }
}
function bannerImgFailed(img){
  const box = img.closest('.home-banner');
  const url = img.getAttribute('src') || '';
  if(!img.dataset.retried && /^https?:/i.test(url)){
    img.dataset.retried = '1';
    setTimeout(()=>{ img.src = url + (url.indexOf('?') > -1 ? '&' : '?') + 'r=' + Date.now(); }, 2500);
    return;
  }
  if(box){ box.classList.remove('loaded'); box.classList.add('failed'); }
}

    if(img){

            <span>${escapeHtml(actionText)}</span>

          </button>` : '';
      return `<div class="home-banner-slide">
        <div class="home-banner has-image">
          <img src="${escapeHtml(img)}" alt="${escapeHtml(b.title||'Banner')}" decoding="async"
               onload="bannerImgLoaded(this)" onerror="bannerImgFailed(this)">
          <div class="home-banner-fail"><svg viewBox="0 0 24 24" fill="none" stroke-width="1.8" stroke-linecap="round" stroke-linejoin="round"><rect x="3" y="3" width="18" height="18" rx="2"/><circle cx="9" cy="9" r="2"/><path d="M21 15l-5-5L5 21"/></svg></div>
