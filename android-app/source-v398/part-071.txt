      const cx=w/2,half=cx/SP;
      const lo=Math.max(Math.ceil(min),Math.floor(pos-half)-1);
      const hi=Math.min(Math.floor(max),Math.ceil(pos+half)+1);
      for(let v=lo;v<=hi;v++){
        const x=cx+(v-pos)*SP;
        if(x<-2||x>w+2)continue;
        if(Math.abs(x-cx)<1.6)continue; // the blue marker takes this tick's place
        const edge=Math.min(x,w-x);
        const fade=Math.max(0,Math.min(1,(edge-SP*.4)/(SP*2.6)));
        if(fade<=0)continue;
        const major=(v%MAJOR===0);
        ctx.fillStyle=major?`rgba(255,255,255,${(.93*fade).toFixed(3)})`:`rgba(255,255,255,${(.40*fade).toFixed(3)})`;
        rr(x-.8,BASE_Y-TICK_H,1.6,TICK_H,.8);
      }
      ctx.fillStyle=BLUE;
      rr(cx-1.3,IND_TOP,2.6,IND_H,1.3);
    }

    // Set the ruler position and push whole-number changes into the real slider.
    function apply(v,emit){
      pos=clamp(v);
      const iv=Math.round(pos);
      const input=inputEl();
      if(emit&&input&&Number(input.value)!==iv){
        input.value=iv;
        try{window.updateTransform();}catch(err){console.warn('Ethvelle transform:',err);}
        try{input.dispatchEvent(new Event('input',{bubbles:true}));}catch(_e){}
        lastEmit=iv;
      }
      canvas.setAttribute('aria-valuenow',String(iv));
      draw();
    }
    function finishChange(){
      const input=inputEl();
      if(input&&lastEmit!==null){try{input.dispatchEvent(new Event('change',{bubbles:true}));}catch(_e){}}
      lastEmit=null;
    }

    function stopAnim(){if(anim){cancelAnimationFrame(anim);anim=0;}}
    function settle(){
      stopAnim();
      const from=pos,to=Math.round(pos);
      if(Math.abs(from-to)<.001){apply(to,true);finishChange();return;}
      const t0=performance.now(),dur=130;
      const step=now=>{
        const k=Math.min(1,(now-t0)/dur),e=1-Math.pow(1-k,3);
        apply(from+(to-from)*e,true);
        if(k<1)anim=requestAnimationFrame(step);else{anim=0;apply(to,true);finishChange();}
      };
      anim=requestAnimationFrame(step);
    }
    function fling(vel){ // vel in units per ms
      stopAnim();
      let last=performance.now();
      const [min,max]=range();
      const step=now=>{
        const dt=Math.min(34,now-last);last=now;
        pos+=vel*dt;
        vel*=Math.exp(-dt/240);
        if(pos<=min||pos>=max){apply(pos<=min?min:max,true);anim=0;finishChange();return;}
        apply(pos,true);
        if(Math.abs(vel)<.0035){anim=0;settle();return;}
        anim=requestAnimationFrame(step);
      };
      anim=requestAnimationFrame(step);
    }

    canvas.addEventListener('pointerdown',e=>{
      if(drag||(e.pointerType==='mouse'&&e.button!==0))return;
      stopAnim();
      drag={id:e.pointerId,x0:e.clientX,p0:pos,s:[{t:e.timeStamp,x:e.clientX}]};
      lastEmit=null;
      try{canvas.setPointerCapture(e.pointerId);}catch(_e){}
      e.preventDefault();e.stopPropagation();
    },{passive:false});
    canvas.addEventListener('pointermove',e=>{
      if(!drag||drag.id!==e.pointerId)return;
      drag.s.push({t:e.timeStamp,x:e.clientX});
      if(drag.s.length>8)drag.s.shift();
      // Content follows the finger: drag left -> ticks slide left -> value goes up.
      pending=drag.p0-(e.clientX-drag.x0)/SP;
      if(!raf)raf=requestAnimationFrame(()=>{raf=0;if(pending!==null){apply(pending,true);pending=null;}});
      if(e.cancelable)e.preventDefault();e.stopPropagation();
    },{passive:false});
    const end=e=>{
      if(!drag||drag.id!==e.pointerId)return;
      if(raf){cancelAnimationFrame(raf);raf=0;}
      if(pending!==null){apply(pending,true);pending=null;}
      const s=drag.s,last=s[s.length-1];
      let vel=0;
      for(let i=s.length-2;i>=0;i--){ if(last.t-s[i].t>=70||i===0){const dt=Math.max(1,last.t-s[i].t);vel=-(last.x-s[i].x)/dt/SP;break;} }
      if(e.type==='pointercancel'||e.timeStamp-last.t>90)vel=0; // paused before lifting: no fling
      drag=null;
      try{canvas.releasePointerCapture(e.pointerId);}catch(_e){}
      if(Math.abs(vel)>.012)fling(Math.max(-2.2,Math.min(2.2,vel)));else settle();
      e.stopPropagation();
    };
    canvas.addEventListener('pointerup',end,{passive:false});
    canvas.addEventListener('pointercancel',end,{passive:false});
    canvas.addEventListener('keydown',e=>{
      const step=e.shiftKey?MAJOR:1;
      if(e.key==='ArrowLeft'||e.key==='ArrowDown'){apply(Math.round(pos)-step,true);finishChange();e.preventDefault();}
      else if(e.key==='ArrowRight'||e.key==='ArrowUp'){apply(Math.round(pos)+step,true);finishChange();e.preventDefault();}
    });

    const inst={
      key:def.key,canvas,resize,draw,
      busy:()=>!!drag||!!anim,
      // Follow the real slider (opening the panel, playhead moved, reset, keyframes...). No emit.
      refresh(){
        if(drag||anim)return;
        const i=inputEl();if(!i)return;
        const [min,max]=range();
        pos=Math.max(min,Math.min(max,Number(i.value)||0));
        canvas.setAttribute('aria-valuemin',String(min));canvas.setAttribute('aria-valuemax',String(max));
        canvas.setAttribute('aria-valuenow',String(Math.round(pos)));
        canvas.setAttribute('aria-label',def.label);
        draw();
      }
    };
    if(typeof ResizeObserver==='function'){new ResizeObserver(()=>resize()).observe(canvas);}
    return inst;
  }

  /* ---------- panel wiring ---------- */
  function refreshAll(){instances.forEach(r=>{try{r.refresh();}catch(_e){}});}

  function setTab(root,name){
    root.querySelectorAll('.evcx-tab').forEach(b=>{
      const on=b.dataset.evcxTab===name;b.classList.toggle('active',on);b.setAttribute('aria-selected',on?'true':'false');
    });
    root.querySelectorAll('.evcx-pane').forEach(p=>p.classList.toggle('active',p.dataset.evcxPane===name));
    // Panes that were hidden have no width yet: measure once they are shown.
    requestAnimationFrame(()=>{instances.forEach(r=>{r.resize();r.refresh();});});
  }

  function initRoot(root){
    if(!root||root.dataset.evcxReady==='1')return;
    root.dataset.evcxReady='1';
    instances.clear();
    root.querySelectorAll('canvas[data-evcx-ruler]').forEach(c=>{const r=makeRuler(c);if(r)instances.set(r.key,r);});
    root.addEventListener('click',e=>{
      const tab=e.target.closest&&e.target.closest('[data-evcx-tab]');
      if(tab){setTab(root,tab.dataset.evcxTab);return;}
      const act=e.target.closest&&e.target.closest('[data-evcx-act]');
      if(!act)return;
      if(act.dataset.evcxAct==='reset'){
        try{window.resetTransform();}catch(err){console.warn('Ethvelle reset:',err);}
        refreshAll();
      }else if(act.dataset.evcxAct==='ok'){
        try{window.closePanel();}catch(_e){}
      }
    });
    const titleEl=$('panelTitle'),t=$('evcxTitle');
    if(t&&titleEl&&titleEl.textContent.trim())t.textContent=titleEl.textContent.trim();
    // Measure after layout, sync with the real slider values, then once more when the sheet has settled.
    requestAnimationFrame(()=>{
      try{if(typeof window.syncTransformSliders==='function')window.syncTransformSliders();}catch(_e){}
      instances.forEach(r=>{r.resize();r.refresh();});
      setTimeout(()=>{instances.forEach(r=>{r.resize();r.refresh();});},320);
    });
  }

  function watchPanel(){
    const panel=$('toolPanel'),bodyEl=$('panelBody');
    if(!panel||!bodyEl)return;
    const sync=()=>{
      const root=bodyEl.querySelector('.evcx-root');
      panel.classList.toggle('evcx-on',!!root);
      if(root)initRoot(root);else instances.clear();
    };
    new MutationObserver(sync).observe(bodyEl,{childList:true});
    sync();
  }

  // Keep the rulers in step with every place the app re-syncs the sliders.
  const baseSync=window.syncTransformSliders;
  if(typeof baseSync==='function'&&!baseSync.__evcx){
    const wrapped=function(){const r=baseSync.apply(this,arguments);try{refreshAll();}catch(_e){}return r;};
    wrapped.__evcx=true;window.syncTransformSliders=wrapped;
  }

  // The Transform panel now opens with the CapCut layout (always starts on Position, like CapCut).
  if(typeof panelContent!=='undefined'&&panelContent&&panelContent.transform){
    panelContent.transform.body=body();
  }

  if(document.readyState==='loading')document.addEventListener('DOMContentLoaded',watchPanel,{once:true});
  else watchPanel();

  window.addEventListener('resize',()=>instances.forEach(r=>r.resize()),{passive:true});
  window.__evcxTransformApi={refresh:refreshAll};
})();
</script>
