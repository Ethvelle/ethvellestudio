(function(){
  'use strict';
  /* IDs are the identity key used by the layer renderer, selectors and save model.
     A repeated ID would render the same logical clip twice (often as an extra lane) while
     both visual copies still resolve to the first matching object. Keep the first model
     only; intentionally duplicated media still works because real duplicates have new IDs. */
  const previousRender=window.renderOverlayLayers;
  if(typeof previousRender==='function'&&!previousRender.__evv126UniqueIds){
    const wrapped=function(){
      if(Array.isArray(overlays)){
        const seen=new Set(),kept=[];
        for(const clip of overlays){
          if(!clip||clip.id==null){kept.push(clip);continue;}
          const id=String(clip.id);
          if(seen.has(id))continue;
          seen.add(id);kept.push(clip);
        }
        if(kept.length!==overlays.length){overlays.length=0;for(const clip of kept)overlays.push(clip);}
      }
      return previousRender.apply(this,arguments);
    };
    wrapped.__evv126UniqueIds=true;window.renderOverlayLayers=wrapped;
  }
})();
