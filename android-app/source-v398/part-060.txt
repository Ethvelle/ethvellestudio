<script id="ethvelle-v146-held-preview-full-scroll">
(function(){
  'use strict';
  if(window.__ethvelleV146HeldPreviewFullScrollInstalled)return;
  window.__ethvelleV146HeldPreviewFullScrollInstalled=true;

  let gesture=null;
  const ROOT='#ethvelleCharacterPropsBackdrop';
  const STAGE='.evpa-prop-stage-wrap';

  function getPreviewSheet(e){
    const back=document.querySelector(ROOT+'.active');
    if(!back)return null;
    const stage=back.querySelector(STAGE);
    const sheet=back.querySelector('.evpa-props-sheet');
    if(!stage||!sheet||!(e.target===stage||stage.contains(e.target)))return null;
    return {stage,sheet};
  }

  // Intercept the preview gesture before the canvas's held-object drag listener.
  // A vertical gesture scrolls the complete sheet; a horizontal gesture is left
  // to the existing object-positioning handler.
  document.addEventListener('pointerdown',function(e){
    if(e.pointerType==='mouse'&&e.button!==0)return;
    const target=getPreviewSheet(e);
    if(!target)return;
    gesture={pointerId:e.pointerId,startX:e.clientX,startY:e.clientY,lastX:e.clientX,
      lastY:e.clientY,mode:'pending',sheet:target.sheet};
  },{capture:true,passive:true});

  document.addEventListener('pointermove',function(e){
    const g=gesture;
    if(!g||g.pointerId!==e.pointerId)return;
    const totalX=e.clientX-g.startX;
    const totalY=e.clientY-(g.startY===undefined?e.clientY:g.startY);
    if(g.mode==='pending'&&Math.max(Math.abs(totalX),Math.abs(totalY))>6){
      if(Math.abs(totalY)>=Math.abs(totalX)*0.9)g.mode='scroll';
      else if(Math.abs(totalX)>Math.abs(totalY)*0.9)g.mode='object';
    }
    if(g.mode==='scroll'){
      const max=Math.max(0,g.sheet.scrollHeight-g.sheet.clientHeight);
      if(max>0){
        // Incremental scrolling avoids the short travel/clamping seen with the old
        // fixed startScroll calculation and continues correctly across long swipes.
        const delta=g.lastY-e.clientY;
        g.sheet.scrollTop=Math.max(0,Math.min(max,g.sheet.scrollTop+delta));
        if(e.cancelable)e.preventDefault();
        e.stopPropagation();
        if(e.stopImmediatePropagation)e.stopImmediatePropagation();
      }
    }
    g.lastX=e.clientX;
    g.lastY=e.clientY;
  },{capture:true,passive:false});

  function finish(e){
    if(gesture&&(!e||gesture.pointerId===e.pointerId))gesture=null;
  }
  document.addEventListener('pointerup',finish,true);
  document.addEventListener('pointercancel',finish,true);
  document.addEventListener('lostpointercapture',finish,true);
})();
</script>
