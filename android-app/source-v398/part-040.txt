  /* Runs a capture plan. fg=true holds the decoder for the whole run (import); fg=false takes the lock per frame and yields to playback. */
  async function runPlan(set,src,times,opt){
    const h=opt.h,q=opt.q,total=times.length;let done=0,okCount=0;const retry=[];
    const queue=times.filter(ms=>!set.urls.has(ms));const skipped=total-queue.length;done=skipped;
    const step=async ms=>{
      if(!opt.fg)await idleWait();
      const blob=opt.fg?await grab(src,ms/1000,h,q):await withLock(()=>grab(src,ms/1000,h,q));
      if(blob){if(addFrame(set,ms,blob)){await persistFrame(set,ms,blob);okCount++;}}else return false;
      return true;
    };
    let failStreak=0;
    for(let i=0;i<queue.length;i++){
      if(opt.deadline&&Date.now()>opt.deadline)break;
      if(opt.shouldStop&&opt.shouldStop())break;
      const ok=await step(queue[i]);
      if(!ok){retry.push(queue[i]);failStreak++;}else failStreak=0;
      done++;
      try{if(opt.onProgress)opt.onProgress(Math.min(100,done/Math.max(1,total)*100));}catch(_e){}
      if(opt.onBatch&&okCount&&okCount%25===0)opt.onBatch();
      if(failStreak>=6&&!opt.fg){opt.aborted=true;break;}
      await delay(opt.fg?0:40);
    }
    if(!opt.aborted&&retry.length&&!(opt.deadline&&Date.now()>opt.deadline)){
      for(const ms of retry){if(opt.shouldStop&&opt.shouldStop())break;await step(ms);}
    }
    return okCount;
  }
