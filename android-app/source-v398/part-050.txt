  function timelineEntryElement(row,entry){

      const end=Number(entry.model&&entry.model.trimEnd)||start;

    // already selected still covers the playhead, it stays selected for the whole slide  - 
    // the border never jumps to the other clip. Only when the selected clip no longer covers
    // the playhead does the pointer's lane pick a new one.
    const keepRef=currentSelectedRef();

      const keepEntry=entries.find(entry=>String(entry.id)===keepRef&&(entry.characterIndex==null||keepOwner==null||String(entry.characterIndex)===String(keepOwner)));
      if(keepEntry)return {id:keepEntry.id,characterIndex:keepEntry.characterIndex};

    // pointer's lane only to choose between those data-matched candidates; time membership

    });

<script id="ethvelle-v82-viewport-fit-js">
(function(){
  'use strict';
  const screen=document.getElementById('screen-editor');
  const wrap=screen&&screen.querySelector('.canvas-wrap');
  const stage=document.getElementById('canvasStage');
  if(!screen||!wrap||!stage)return;

  let fitRaf=0;
  function activeRatio(){
    try{
      const value=(typeof currentProject!=='undefined'&&currentProject&&currentProject.ratio)||'16:9';
      const match=String(value).match(/^\s*(\d+(?:\.\d+)?)\s*:\s*(\d+(?:\.\d+)?)\s*$/);
      return match&&Number(match[1])>0&&Number(match[2])>0 ? `${Number(match[1])}:${Number(match[2])}` : '16:9';
    }catch(_e){return '16:9';}
  }
  function refit(){
    fitRaf=0;
    if(!screen.classList.contains('active'))return;
    const ratio=activeRatio();
    const parts=ratio.split(':').map(Number);
    const rw=parts[0],rh=parts[1];
    if(!(rw>0&&rh>0))return;

    // Set the non-shrinking flex behaviour before measuring/re-sizing. setupCanvasStage
    // computes the largest frame that fits both available dimensions while preserving ratio.
    stage.style.flex='0 0 auto';
    stage.style.aspectRatio=rw+' / '+rh;
    if(typeof setupCanvasStage==='function')setupCanvasStage(ratio);

    // Wait for the WebView to commit the final post-rotation layout before repainting.
    requestAnimationFrame(()=>{
      if(!screen.classList.contains('active'))return;
      // renderFrame already invokes drawStick, so avoid drawing the stage twice on rotation.
      try{if(typeof renderFrame==='function')renderFrame();}catch(_e){}
      try{if(typeof updateSelBox==='function')updateSelBox();}catch(_e){}
    });
  }
  function scheduleRefit(){
    if(fitRaf)return;
    // Two frames let Android/WebView finish applying the new visual viewport dimensions.
    fitRaf=requestAnimationFrame(()=>{
      fitRaf=requestAnimationFrame(refit);
    });
  }

  // The observer catches the actual editor preview-area resize, including address-bar and
  // orientation changes that do not always arrive as a reliable window.resize event.
  if(typeof ResizeObserver==='function'){
    const observer=new ResizeObserver(scheduleRefit);
    observer.observe(wrap);
  }
  window.addEventListener('resize',scheduleRefit,{passive:true});
  window.addEventListener('orientationchange',scheduleRefit,{passive:true});
  if(window.visualViewport)window.visualViewport.addEventListener('resize',scheduleRefit,{passive:true});

  // Re-fit after the editor becomes visible; the initial observer notification can occur
  // while its parent screen is display:none.
  const editorObserver=new MutationObserver(()=>{
    if(screen.classList.contains('active'))scheduleRefit();
  });
  editorObserver.observe(screen,{attributes:true,attributeFilter:['class']});
  scheduleRefit();
})();

  const selectedId=()=>{try{return typeof selectedClip!=='undefined'?selectedClip:null;}catch(_e){return null;}};
  const objectClip=id=>{
    if(id==null)return null;
    if(typeof window.ethvelleObjectClipById==='function')return window.ethvelleObjectClipById(id);
    const key=String(id);
    const bgList=typeof backgrounds!=='undefined'?backgrounds:(window.backgrounds||[]);
    const ovList=typeof overlays!=='undefined'?overlays:(window.overlays||[]);
    return (Array.isArray(bgList)?bgList.find(x=>String(x.id)===key):null)
      ||(Array.isArray(ovList)?ovList.find(x=>String(x.id)===key):null);
  };
  const nowTime=()=>typeof window.getCurrentAbsoluteTime==='function'?Number(window.getCurrentAbsoluteTime()):Number(window.currentAbsoluteTimeSec)||0;
  const currentTransform=(clip,t)=>{
    if(!clip)return {scale:1,offsetX:0,offsetY:0,rotation:0};
    const animated=typeof window.ethvelleGetObjectKeyframeValue==='function'?window.ethvelleGetObjectKeyframeValue(clip,'transform',t):null;
    return cloneTransform(animated||clip.transform||{});
  };
  // Resolve an EXISTING transform key only when the playhead is on its diamond.
  // Editing elsewhere never creates a keyframe implicitly.
  function transformKeyAtPlayhead(clip,t){
    const keys=Array.isArray(clip&&clip.ethvelleMediaKeyframes)?clip.ethvelleMediaKeyframes:[];
    const tolerance=Math.max(0.001,typeof window.ethvelleObjectKeyframeContactWindow==='function'?Number(window.ethvelleObjectKeyframeContactWindow()):0.05);
    const chosen=window.__ethvelleSelectedObjectKeyframe;
    if(chosen&&String(chosen.id)===String(clip&&clip.id)&&Number.isInteger(Number(chosen.index))){
      const candidate=keys[Number(chosen.index)];
      if(candidate&&candidate.property==='transform'&&Math.abs(Number(candidate.t)-Number(t))<=tolerance)return candidate;
    }
    return keys.reduce((best,item)=>item&&item.property==='transform'&&Number.isFinite(Number(item.t))&&Math.abs(Number(item.t)-Number(t))<=tolerance&&(!best||Math.abs(Number(item.t)-Number(t))<Math.abs(Number(best.t)-Number(t)))?item:best,null);
  }
  function editableTransform(clip,t){
    if(!clip)return null;
    if(!clip.transform)clip.transform={scale:1,offsetX:0,offsetY:0,rotation:0};
    const property=typeof window.ethvelleObjectKeyframeProperty==='function'?window.ethvelleObjectKeyframeProperty(clip):'transform';
    if(property!=='transform')return clip.transform;
    const key=transformKeyAtPlayhead(clip,t);
    if(key){
      const idx=(clip.ethvelleMediaKeyframes||[]).indexOf(key);
      if(idx>=0)window.__ethvelleSelectedObjectKeyframe={id:String(clip.id),index:idx,source:'playhead'};
      if(!key.value||typeof key.value!=='object')key.value=cloneTransform(clip.transform);
      return key.value;
    }
    return clip.transform;
  }
  // Apply an edit without inserting a new marker. On a keyframe, only that existing key is
  // changed. Between keyframes, apply the requested transform delta to the entire authored
  // path and base value, preserving the animation curve and the original key count.
  function applyTransformEdit(clip,t,desired){
    if(!clip)return null;
    if(!clip.transform)clip.transform={scale:1,offsetX:0,offsetY:0,rotation:0};
    const wanted=cloneTransform(desired),key=transformKeyAtPlayhead(clip,t);
    if(key){
      key.value=wanted;
      const idx=(clip.ethvelleMediaKeyframes||[]).indexOf(key);
      if(idx>=0)window.__ethvelleSelectedObjectKeyframe={id:String(clip.id),index:idx,source:'playhead'};
      return key.value;
    }
    const keys=Array.isArray(clip.ethvelleMediaKeyframes)?clip.ethvelleMediaKeyframes:[];
    const transformKeys=keys.filter(k=>k&&k.property==='transform'&&k.value&&typeof k.value==='object');
    if(transformKeys.length){
      const before=currentTransform(clip,t);
      const ratio=Math.max(.0001,wanted.scale)/Math.max(.0001,before.scale);
      const dx=wanted.offsetX-before.offsetX,dy=wanted.offsetY-before.offsetY;
      const dr=((wanted.rotation-before.rotation+540)%360)-180;
      const shift=value=>{
        const v=cloneTransform(value);
        v.scale=Math.max(.0001,v.scale*ratio);v.offsetX+=dx;v.offsetY+=dy;v.rotation+=dr;
        return v;
      };
      transformKeys.forEach(k=>{k.value=shift(k.value);});
      clip.transform=shift(clip.transform);
      if(clip.ethvelleMediaKeyframeBase&&clip.ethvelleMediaKeyframeBase.transform&&typeof clip.ethvelleMediaKeyframeBase.transform==='object'){
        clip.ethvelleMediaKeyframeBase.transform=shift(clip.ethvelleMediaKeyframeBase.transform);
      }
      return currentTransform(clip,t);
    }
    clip.transform=wanted;
    return clip.transform;
  }
