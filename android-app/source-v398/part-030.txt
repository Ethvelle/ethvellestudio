function sheetDrawDressBody(g,img,d,rot,k,a,S,B,BA,sh){
  try{
    const iw=img.naturalWidth,ih=img.naturalHeight,kx=k*d.sx,ky=k*d.sy;
    if(!(iw>0&&ih>0)||!(Math.abs(kx)>1e-6)||!(Math.abs(ky)>1e-6)||!S||!B)return false;
    const Br={};Object.keys(B).forEach(n=>{const b=B[n];Br[n]=Object.assign({},b,{angle:(BA&&Number.isFinite(BA[n]))?BA[n]:b.angle});});
    const R=sheetSolveRig(Br,BA);
    if(!R.pelvis||!R.neck)return false;
    const rotR=Math.atan2(R.neck.e.y-R.pelvis.s.y,R.neck.e.x-R.pelvis.s.x)-d.ra+d.rot;
    const loc=(P,o,r)=>{const dx=P.x-o.x,dy=P.y-o.y,c=Math.cos(r),s=Math.sin(r);return {x:(dx*c+dy*s)/kx+d.pvx,y:(-dx*s+dy*c)/ky+d.pvy};};
    const legs=['L','R'].map(sd=>{
      const T1=S['thigh'+sd],T2=S['shin'+sd],T3=S['foot'+sd],R1=R['thigh'+sd],R2=R['shin'+sd],R3=R['foot'+sd];
      if(!T1||!T2||!R1||!R2)return null;
      const pt=(Sx,Rx,end)=>({c:loc(end?Sx.e:Sx.s,a.s,rot),r:loc(end?Rx.e:Rx.s,R.pelvis.s,rotR)});
      const H=pt(T1,R1,false),K=pt(T2,R2,false),A=T3&&R3?pt(T3,R3,false):pt(T2,R2,true);
      // v373: half-width of the leg art, in the garment's own pixels, so the dress can always cover the leg.
      const hwOf=key=>{const pp=sh&&sh.parts&&sh.parts[key],dd=sh&&sh.draw&&sh.draw[key];return pp&&dd&&Number(pp.w)>0&&Math.abs(d.sx)>1e-6?.5*pp.w*Math.abs(dd.sx)/Math.abs(d.sx):0;};
      const hwT=hwOf('leg'+sd+'_upper'),hwS=hwOf('leg'+sd+'_lower')||hwT*.8;
      return {Hy:H.r.y,Ky:K.r.y,Ay:A.r.y,Hx:H.r.x,Kx:K.r.x,Ax:A.r.x,hwT,hwS,dK:{x:K.c.x-K.r.x,y:K.c.y-K.r.y},dA:{x:A.c.x-A.r.x,y:A.c.y-A.r.y}};
    });
    if(!legs[0]||!legs[1])return false;
    if(Math.max(Math.hypot(legs[0].dK.x,legs[0].dK.y),Math.hypot(legs[0].dA.x,legs[0].dA.y),Math.hypot(legs[1].dK.x,legs[1].dK.y),Math.hypot(legs[1].dA.x,legs[1].dA.y))<.6)return false;
    const lf=legs[0].Ax<=legs[1].Ax?legs[0]:legs[1],rt=lf===legs[0]?legs[1]:legs[0];
    const yTop=Math.max(0,Math.min(ih-3,(lf.Hy+rt.Hy)/2));
    // v373 EVEN STRETCH + ALWAYS COVER THE LEGS.
    // Old version: each leg's pull was blended over the narrow gap between the two ankles, so all of the stretching was
    // squeezed into one small area, and nothing checked that the cloth still covered the legs.
    // Now every row of the garment is stretched LINEARLY across the garment's whole width: the left edge follows the left
    // leg, the right edge follows the right leg, and everything in between is spread evenly (the same on the left,
    // the middle and the right). The vertical pull is spread across the width the same way. Then two guards keep the
    // legs inside the cloth: the side edges are never allowed to cut inside a leg, and the hem is never allowed to
    // rise above a foot (both only act when the cloth would otherwise lose a margin it had at rest).
    const clamp01=t=>t<0?0:t>1?1:t,lerp=(p,q,t)=>p+(q-p)*t;
    const ease=clamp01;   // linear along each leg bone = exactly how the leg itself moves, so the cloth can never lag behind it
    const dispOf=(L,y)=>{
      if(y<=L.Hy)return {x:0,y:0};
      if(y<=L.Ky){const t=ease((y-L.Hy)/Math.max(1e-3,L.Ky-L.Hy));return {x:L.dK.x*t,y:L.dK.y*t};}
      if(y<=L.Ay){const t=ease((y-L.Ky)/Math.max(1e-3,L.Ay-L.Ky));return {x:lerp(L.dK.x,L.dA.x,t),y:lerp(L.dK.y,L.dA.y,t)};}
      return L.dA;
    };
    const restX=(L,y)=>y<=L.Ky?lerp(L.Hx,L.Kx,clamp01((y-L.Hy)/Math.max(1e-3,L.Ky-L.Hy))):lerp(L.Kx,L.Ax,clamp01((y-L.Ky)/Math.max(1e-3,L.Ay-L.Ky)));
    const halfW=(L,y)=>y<=L.Ky?lerp(L.hwT,L.hwS,clamp01((y-L.Hy)/Math.max(1e-3,L.Ky-L.Hy))):L.hwS;
    // where the opaque cloth starts and ends on every row (read once from the picture, then cached on it)
    if(!img.__dressExt||img.__dressExt.iw!==iw||img.__dressExt.ih!==ih){
      let L=null,R=null,sc=1;
      try{
        sc=Math.min(1,256/iw);const cw=Math.max(2,Math.round(iw*sc)),ch=Math.max(2,Math.round(ih*sc));
        const cv=document.createElement('canvas');cv.width=cw;cv.height=ch;
        const cg=cv.getContext('2d',{willReadFrequently:true});cg.drawImage(img,0,0,cw,ch);
        const px=cg.getImageData(0,0,cw,ch).data;L=new Float32Array(ch).fill(-1);R=new Float32Array(ch).fill(-1);
        for(let y=0;y<ch;y++){let a0=-1,a1=-1;for(let x=0;x<cw;x++){if(px[(y*cw+x)*4+3]>24){if(a0<0)a0=x;a1=x;}}if(a0>=0){L[y]=a0/sc;R[y]=(a1+1)/sc;}}
        let last=-1;for(let y=0;y<ch;y++){if(L[y]>=0)last=y;else if(last>=0){L[y]=L[last];R[y]=R[last];}}
        let nxt=-1;for(let y=ch-1;y>=0;y--){if(L[y]>=0)nxt=y;else if(nxt>=0){L[y]=L[nxt];R[y]=R[nxt];}}
        for(let y=0;y<ch;y++)if(L[y]<0){L[y]=0;R[y]=iw;}
      }catch(e){L=null;R=null;sc=1;}
      img.__dressExt={iw,ih,L,R,sc};
    }
    const EX=img.__dressExt;
    const extAt=(arr,sy,dflt)=>{if(!arr)return dflt;const f=Math.max(0,Math.min(arr.length-1,sy*EX.sc)),i0=Math.floor(f),i1=Math.min(arr.length-1,i0+1);return lerp(arr[i0],arr[i1],f-i0);};
    const edgesAt=sy=>{let eL=extAt(EX.L,sy,0),eR=extAt(EX.R,sy,iw);if(!(eR-eL>iw*.05)){eL=0;eR=iw;}return {eL,eR};};
    const pad=iw*.012,padY=ih*.01;
    // hem guard: at the bottom row, the cloth above each foot must move down at least as far as that foot does
    const bot=edgesAt(ih-1),extraBot=(()=>{let need=0;[lf,rt].forEach(L=>{
      const u=clamp01((restX(L,ih)-bot.eL)/Math.max(1,bot.eR-bot.eL)),hem=lerp(lf.dA.y,rt.dA.y,u),m=Math.max(0,ih-L.Ay);
      need=Math.max(need,L.dA.y-hem-Math.max(0,m-padY));});return need;})();
    const NX=14,NY=18,rows=[];
    for(let j=0;j<=NY;j++){
      const sy=yTop+(ih-yTop)*j/NY,row=[],dl=dispOf(lf,sy),dr=dispOf(rt,sy),E=edgesAt(sy),W=E.eR-E.eL;
      // side guard: how much spare cloth each leg has beyond it at rest (never less than a small pad is demanded)
      const slackL=L=>Math.max(0,(restX(L,sy)-halfW(L,sy)-E.eL)-pad),slackR=L=>Math.max(0,(E.eR-(restX(L,sy)+halfW(L,sy)))-pad);
      let dLe=Math.min(dl.x,dr.x+slackL(rt)),dRe=Math.max(dr.x,dl.x-slackR(lf));
      dRe=Math.max(dRe,dLe-.7*W);                                   // never fold the row over itself
      const wy=Math.pow(clamp01((sy-yTop)/Math.max(1e-3,ih-yTop)),1.5),ex=Math.max(0,extraBot)*wy;
      for(let i=0;i<=NX;i++){
        const sx=iw*i/NX,u=clamp01((sx-E.eL)/W);
        row.push({sx,sy,x:sx+lerp(dLe,dRe,u),y:sy+lerp(dl.y,dr.y,u)+ex});
      }
      rows.push(row);
    }
    for(let j=1;j<=NY;j++)for(let i=0;i<=NX;i++){const p=rows[j-1][i],q=rows[j][i],mn=(q.sy-p.sy)*.25;if(q.y<p.y+mn)q.y=p.y+mn;}
    for(let j=0;j<=NY;j++)for(let i=1;i<=NX;i++){const p=rows[j][i-1],q=rows[j][i],mn=(q.sx-p.sx)*.25;if(q.x<p.x+mn)q.x=p.x+mn;}
    const ov=.9/Math.max(1e-6,Math.min(Math.abs(kx),Math.abs(ky)));
    g.drawImage(img,0,0,iw,Math.min(ih,Math.ceil(yTop)+1),-d.pvx,-d.pvy,iw,Math.min(ih,Math.ceil(yTop)+1));
    const tri=(s0,s1,s2)=>{
      const den=(s1.sx-s0.sx)*(s2.sy-s0.sy)-(s2.sx-s0.sx)*(s1.sy-s0.sy);if(Math.abs(den)<1e-8)return;
      const d0={x:s0.x-d.pvx,y:s0.y-d.pvy},d1={x:s1.x-d.pvx,y:s1.y-d.pvy},d2={x:s2.x-d.pvx,y:s2.y-d.pvy};
      const A1=((d1.x-d0.x)*(s2.sy-s0.sy)-(d2.x-d0.x)*(s1.sy-s0.sy))/den,B1=((d1.y-d0.y)*(s2.sy-s0.sy)-(d2.y-d0.y)*(s1.sy-s0.sy))/den;
      const C1=((d2.x-d0.x)*(s1.sx-s0.sx)-(d1.x-d0.x)*(s2.sx-s0.sx))/den,D1=((d2.y-d0.y)*(s1.sx-s0.sx)-(d1.y-d0.y)*(s2.sx-s0.sx))/den;
      const E1=d0.x-(A1*s0.sx+C1*s0.sy),F1=d0.y-(B1*s0.sx+D1*s0.sy);
      const ccx=(d0.x+d1.x+d2.x)/3,ccy=(d0.y+d1.y+d2.y)/3,out=p=>{const vx=p.x-ccx,vy=p.y-ccy,l=Math.hypot(vx,vy)||1;return {x:p.x+vx/l*ov,y:p.y+vy/l*ov};};
      const p0=out(d0),p1=out(d1),p2=out(d2);
      const bx=Math.max(0,Math.floor(Math.min(s0.sx,s1.sx,s2.sx))-1),by=Math.max(0,Math.floor(Math.min(s0.sy,s1.sy,s2.sy))-1);
      const bx1=Math.min(iw,Math.ceil(Math.max(s0.sx,s1.sx,s2.sx))+1),by1=Math.min(ih,Math.ceil(Math.max(s0.sy,s1.sy,s2.sy))+1);
      if(bx1<=bx||by1<=by)return;
      g.save();g.beginPath();g.moveTo(p0.x,p0.y);g.lineTo(p1.x,p1.y);g.lineTo(p2.x,p2.y);g.closePath();g.clip();
      g.transform(A1,B1,C1,D1,E1,F1);g.drawImage(img,bx,by,bx1-bx,by1-by,bx,by,bx1-bx,by1-by);g.restore();
    };
    for(let j=0;j<NY;j++)for(let i=0;i<NX;i++){
      const p00=rows[j][i],p10=rows[j][i+1],p01=rows[j+1][i],p11=rows[j+1][i+1];
      tri(p00,p10,p01);tri(p10,p11,p01);
    }
    return true;
  }catch(err){return false;}
}

function drawSheetCharacter(g, char){
  const sh = char.sheet;
  if(!sh || !sh.parts || !sh.draw || !Array.isArray(sh.order)) return;
  if(Number(sh.v||0)<7){
    if(sh.edit && sh.edit.st) normalizeLegacySheetMouthLayout(sh.parts,sh.draw,sh.edit.st);
    sh.v=7;
  }
  const sheetRig = sheetRigForCharacter(char), B = sheetRig.bones, S = sheetSolveRig(B, sheetRig.baseAngles);
  let L = 0; ['pelvis','spine_lower','spine_upper','chest','neck'].forEach(n=>{ if(B[n]) L += B[n].length; });
  const k = L/Math.max(1e-6, sh.refLen);
  if(!g.__sheetImg) g.__sheetImg = {};
  sheetPreloadImages(sh);
  const getImg = sheetImgFor;
  const parentXform = {};
  const modelType=String(char.modelType||(sh&&sh.modelType)||'humanoid');
  const isAnimalFaceKit = modelType==='animal' && SHEET_FACE_KIT_NESTED.some(key=>sh.parts[key]);
  const isHumanoidFaceKit = (modelType==='humanoid' || isMythicModelType(modelType) || isWingedHumanoidType(modelType)) && SHEET_HUMANOID_FACE_KIT_KEYS.some(key=>!!sh.parts[key]);
  const humanoidHasClosedEyes = isHumanoidFaceKit && !!sh.parts.eye_closed_L && !!sh.parts.eye_closed_R;
  const isBirdFaceKit = modelType==='bird' && SHEET_BIRD_FACE_KIT_KEYS.every(key=>!!sh.parts[key]);
  const hasSeparateFaceKit=isAnimalFaceKit||isHumanoidFaceKit||isBirdFaceKit;
  const faceKitDone = {};
  const faceKitExpr = char.expr || {};
  const faceKitEyesClosed = faceKitExpr.eyeState==='closed' || (faceKitExpr.eyeState!=='open' && (faceKitExpr.blink||faceKitExpr.ambientBlink));
  // Normalize z-order at render time too, so older saved characters receive the same
  // arm-depth and tail-over-body fixes without requiring a re-import.
  const renderOrder=sheetDepthOrderForView(sh.order,modelType,sh.view||char.view||'front');
  let faceKitHead = null;
  if(hasSeparateFaceKit){
    const hd=sh.draw&&sh.draw.head,hbn=SHEET_BONES.head,ha=hbn&&S[hbn[0]],hb=hbn&&S[hbn[1]];
    if(hd&&ha&&hb){
      const hAng=Math.atan2(hb.e.y-ha.s.y,hb.e.x-ha.s.x);
      faceKitHead={x:ha.s.x,y:ha.s.y,rot:hAng-(Number(hd.ra)||0)+(Number(hd.rot)||0),sx:k*(Number(hd.sx)||1),sy:k*(Number(hd.sy)||1)};
    }
  }
  const drawFaceKitPart = key=>{
    if(faceKitDone[key]) return; faceKitDone[key]=true;
    const px=faceKitHead||parentXform.head;if(!px)return;
    const sourceKey=(key==='beak_closed')?sheetActiveKey('beak_closed',faceKitExpr,sh.parts):key;
    const part=sh.parts[sourceKey],rawNest=part&&sheetNestInfo(sh,sourceKey);
    const closedBeakNest=sourceKey==='beak_open'?sheetNestInfo(sh,'beak_closed'):null;
    // Open/closed artwork is a direct swap: preserve the saved open-art scale and pivot, but
    // inherit the closed beak's head-local position/rotation so animation cannot make it jump.
    const nest=rawNest&&closedBeakNest?{...rawNest,nestDx:closedBeakNest.nestDx,nestDy:closedBeakNest.nestDy,nestRot:closedBeakNest.nestRot}:rawNest;
    if(!part||!nest||!part.src)return;
    if(sourceKey==='mouth_inner'&&!sheetInnerMouthShown(sh,faceKitExpr,!!sheetRuntimeHeadVariant(sh,faceKitExpr)))return;
    if((sourceKey==='eye_L'||sourceKey==='eye_R'||sourceKey==='bird_eye')&&faceKitEyesClosed&&(isAnimalFaceKit||humanoidHasClosedEyes))return;
    if(sourceKey==='bird_eye'&&faceKitEyesClosed&&isBirdFaceKit&&sh.parts.bird_eye_closed)return;
    if((sourceKey==='eye_closed_L'||sourceKey==='eye_closed_R'||sourceKey==='bird_eye_closed')&&!faceKitEyesClosed)return;
    const img=getImg(part.src);
    if(!img.complete||!img.naturalWidth){if(!img.__hook){img.__hook=true;img.addEventListener('load',()=>{if(typeof drawStick==='function')drawStick();});}return;}
    if(sourceKey==='mouth_inner'){
      const clipped=sheetAnimalMouthInnerLayer(sh);if(!clipped)return;
      g.save();g.translate(px.x,px.y);g.rotate(px.rot);g.scale(px.sx,px.sy);
      g.drawImage(clipped.canvas,-clipped.pivotX,-clipped.pivotY,sh.parts.head.w,sh.parts.head.h);g.restore();return;
    }
    let moodRot=0;
    if(sourceKey==='bird_brow'){
      // A side-profile bird has one visible brow: use a clear up/down tilt, not the tiny
      // mirrored two-brow angles used by humanoids. This keeps each expression readable.
      const emo=faceKitExpr.emo||'';
      if(emo==='angry')moodRot=.30;       // tip slopes down into an angry brow
      else if(emo==='sad')moodRot=-.24;   // inner brow lifts for a worried/sad look
      else if(emo==='laugh')moodRot=.13;
      else if(emo==='shock')moodRot=-.10;
    }else if(sourceKey==='brow_L'||sourceKey==='brow_R'){
      const left=sourceKey==='brow_L',emo=faceKitExpr.emo||'';
      if(emo==='angry')moodRot=(left?1:-1)*.18;
      else if(emo==='sad')moodRot=(left?-1:1)*.11;
      else if(emo==='laugh')moodRot=(left?1:-1)*.07;
      else if(emo==='shock')moodRot=(left?-1:1)*.04;
    }
    const pivotX=Number.isFinite(Number(nest.nestPivotX))?Number(nest.nestPivotX):Number((sh.draw[sourceKey]||{}).pvx??part.dpx)||0;
    const pivotY=Number.isFinite(Number(nest.nestPivotY))?Number(nest.nestPivotY):Number((sh.draw[sourceKey]||{}).pvy??part.dpy)||0;
    g.save();g.translate(px.x,px.y);g.rotate(px.rot);
    g.translate((Number(nest.nestDx)||0)*px.sx,(Number(nest.nestDy)||0)*px.sy);
    g.rotate((Number(nest.nestRot)||0)+moodRot);
    let sx=k*(Number(nest.nestSx)||1),sy=k*(Number(nest.nestSy)||1);
    // Humanoid kits WITHOUT closed-eye parts (legacy 17-part) have no separate closed-eyelid art. Keep their open eyes visible
    // rather than flattening them to 4.5% height during blink/auto-blink. Old 6-part bird kits
    // still use the legacy squash; new 7-part bird kits swap to their dedicated eyelid asset.
    const shouldSquash=faceKitEyesClosed && isBirdFaceKit && sourceKey==='bird_eye' && !sh.parts.bird_eye_closed;
    if(shouldSquash){g.translate(0,(Number(part.h)||img.naturalHeight)*.49);g.scale(1,.045);g.translate(0,-(Number(part.h)||img.naturalHeight)*.49);}
    g.scale(sx,sy);g.drawImage(img,-pivotX,-pivotY,part.w,part.h);g.restore();
  };
  renderOrder.forEach(key=>{
    if(hasSeparateFaceKit&&key==='mouth_inner')return;
    if((isAnimalFaceKit||isHumanoidFaceKit)&&key==='head')drawFaceKitPart('mouth_inner');
    if(isAnimalFaceKit&&SHEET_FACE_KIT_NESTED.includes(key)){drawFaceKitPart(key);return;}
    if(isHumanoidFaceKit&&['eye_L','eye_R','eye_closed_L','eye_closed_R','brow_L','brow_R'].includes(key))return;
    if(isBirdFaceKit&&SHEET_BIRD_FACE_NESTED.includes(key)){drawFaceKitPart(key);return;}
    if(SHEET_ALT_OF[key]||SHEET_FACE_NESTED.includes(key)||(isAnimalFaceKit&&SHEET_FACE_KIT_NESTED.includes(key))||(isHumanoidFaceKit&&SHEET_HUMANOID_FACE_KIT_KEYS.includes(key))||(isBirdFaceKit&&SHEET_BIRD_FACE_NESTED.includes(key)))return;
    if(isAnimalFaceKit&&(key==='eye_L'||key==='eye_R')&&faceKitEyesClosed)return;
    if((key==='eye_L'||key==='eye_R')&&char.expr&&(char.expr.blink||char.expr.ambientBlink||char.expr.eyeState==='closed')&&!hasSeparateFaceKit)return;
    if(key==='mouth_inner'&&!sheetInnerMouthShown(sh,char.expr,!!sheetRuntimeHeadVariant(sh,char.expr)))return;
    const extraHead=key==='head'?sheetRuntimeHeadVariant(sh,char.expr):null;
    const extraHand=key==='handL'||key==='handR'?sheetRuntimeHandVariant(sh,char.expr,key==='handL'?'L':'R'):null;
    const dynamic=extraHead||extraHand;
    const useKey=dynamic?key:sheetActiveKey(key,char.expr,sh.parts);
    const d=sh.draw[useKey],part=dynamic?{src:dynamic.src,w:sh.parts[key].w,h:sh.parts[key].h,dpx:sh.parts[key].dpx,dpy:sh.parts[key].dpy}:sh.parts[useKey];
    let sb=SHEET_BONES[useKey];
    if(useKey==='legL_lower'&&sh.parts.shoeL)sb=['shinL','shinL'];
    if(useKey==='legR_lower'&&sh.parts.shoeR)sb=['shinR','shinR'];
    const a=sb&&S[sb[0]],b=sb&&S[sb[1]];if(!d||!part||!a||!b)return;
    const img=getImg(part.src);
    if(!img.complete||!img.naturalWidth){if(!img.__hook){img.__hook=true;img.addEventListener('load',()=>{if(typeof drawStick==='function')drawStick();});}return;}
    const ang=Math.atan2(b.e.y-a.s.y,b.e.x-a.s.x),rot=ang-d.ra+d.rot;
    if(key==='head')parentXform.head={x:a.s.x,y:a.s.y,rot,sx:k*d.sx,sy:k*d.sy};
    const mirror=extraHand&&key==='handL'&&extraHand.side!=='L'?-1:1;
    g.save();g.translate(a.s.x,a.s.y);g.rotate(rot);g.scale(k*d.sx*mirror,k*d.sy);
    if(dynamic)g.drawImage(img,-d.pvx,-d.pvy,sh.parts[key].w,sh.parts[key].h);
    else if(key==='torso'&&sh.dress&&sheetDrawDressBody(g,img,d,rot,k,a,S,B,sheetRig.baseAngles,sh)){/* drawn as a stretching dress */}
    else g.drawImage(img,-d.pvx,-d.pvy);
    g.restore();
  });
  // Dedicated face kits are painted after the base head; mouth fill is inserted just before that head.
  if(isAnimalFaceKit){SHEET_FACE_KIT_NESTED.forEach(drawFaceKitPart);}
  if(isHumanoidFaceKit){['eye_closed_L','eye_closed_R','eye_L','eye_R','brow_L','brow_R'].forEach(drawFaceKitPart);}
  if(isBirdFaceKit){['bird_eye_closed','bird_eye','bird_brow','beak_closed'].forEach(drawFaceKitPart);}
  // Keep legacy full-head blink eye windows available for older Animal kits only.
  if(isAnimalFaceKit&&char.expr&&(char.expr.blink||char.expr.ambientBlink||char.expr.eyeState==='closed')&&(char.expr.emo||char.expr.customExpressionId)&&parentXform.head){
    const blinkPart=sh.parts.head_blink||sheetLatestHeadPart(sh,'blink-head');
    const headKey=sheetActiveKey('head',{...char.expr,blink:false,ambientBlink:false},sh.parts);
    const headPart=sh.parts[headKey]||sh.parts.head,headDraw=sh.draw[headKey]||sh.draw.head;
    if(blinkPart&&blinkPart.src&&headPart&&headDraw){
      const img=getImg(blinkPart.src);
      if(!img.complete||!img.naturalWidth){if(!img.__hook){img.__hook=true;img.addEventListener('load',()=>{if(typeof drawStick==='function')drawStick();});}}
      else{
        const bw=Number(blinkPart.w)||img.naturalWidth,bh=Number(blinkPart.h)||img.naturalHeight;

// Ambient eye-blinking: independent of whichever animation preset is playing (or none at all), each
// custom character with a head_blink part occasionally swaps to it for a brief moment. Characters
// without that part are untouched forever -- the Math.random() check never even fires for them.
function edScheduleBlinks(){
  const tick = ()=>{
    characters.forEach(c=>{
      // Idle blinks use their own flag (expr.ambientBlink), so they never fight the Talk + Blink clip (expr.blink).
      if(c.type === 'custom' && c.sheet && c.sheet.parts && (c.sheet.parts.head_blink || (c.sheet.parts.eye_L && (c.sheet.parts.eye_closed_L || sheetLatestHeadPart(c.sheet,'blink-head'))) || (c.sheet.parts.eye_L && c.sheet.parts.eye_R && c.sheet.parts.mouth_inner) || c.sheet.parts.bird_eye || sheetLatestHeadPart(c.sheet,'blink-head')) && !(c.expr && c.expr.disableAmbientBlink)){
        if(!c.expr) c.expr = sheetExprFresh();
        if(!c.expr.ambientBlink && Math.random() < 0.55){
          c.expr.ambientBlink = true; drawStick();
          setTimeout(()=>{ c.expr.ambientBlink = false; drawStick(); }, 110+Math.random()*80);
        }
      }
    });
    setTimeout(tick, 2200+Math.random()*2600);
  };
  setTimeout(tick, 1500+Math.random()*1500);
}
function updateModelAssetFromChar(c){
  if(!c || !c.sheet || !c.assetId) return;
  let a = modelAssets.find(x=>String(x.id) === String(c.assetId));
  // A character may originate from an admin-published model. Promote that asset to the local
  // editable library before saving, so a character-side edit is not silently discarded.
  if(!a && Array.isArray(publishedModelAssets)){
    const published = publishedModelAssets.find(x=>String(x.id) === String(c.assetId));
    if(published){ a=JSON.parse(JSON.stringify(published)); delete a.origin; modelAssets.push(a); }
  }
  if(!a) return;
  a.name = c.name || a.name;
  a.updatedAt = Date.now();
  a.sheet = JSON.parse(JSON.stringify(c.sheet));
  a.rigOverride = undefined; a.parts = {};
  a.thumb = c.thumb || c.sheet.thumb;
  a.previewThumb = undefined; // the model was edited; the cached full-body preview is now stale
  a.modelType = c.modelType || (c.sheet && c.sheet.modelType) || a.modelType || 'humanoid';
  saveModelAssetsToStorage();
}

/* ================= LADY WALK (v369) =================

function ethvelleIs21Pose(p){
  return !!p && ['pelvis','spine_lower','spine_upper','chest','neck','clavicleL','clavicleR','upperArmL','upperArmR','forearmL','forearmR','thighL','thighR','shinL','shinR','footL','footR'].some(k=>Object.prototype.hasOwnProperty.call(p,k));

function ethvelleDiscrete21(preset,tFrac,field){
  const kfs=preset&&preset.keyframes;if(!Array.isArray(kfs))return undefined;
  let val;
  for(let i=0;i<kfs.length;i++)if(kfs[i].t<=tFrac&&kfs[i][field]!==undefined)val=kfs[i][field];

}
function ethvelleComplete21Pose(preset,tFrac,pose){
  const p=ethvelleNormalizeLegacyPoseTo21(pose||{});
  if(preset&&preset._ethvelleAnimKey)p.__ethvelleAnimKey=preset._ethvelleAnimKey;

  if(fist!==undefined)p.gestureFist=fist?1:0;
  else p.gestureFist=Number(p.gestureFist)||0;

  const mouth=ethvelleDiscrete21(preset,tFrac,'mouth');
  const blink=ethvelleDiscrete21(preset,tFrac,'blink');
  p.__mouth=mouth===undefined?'closed':mouth;
  p.__blink=blink===undefined?false:!!blink;
  return p;
}

// Updated presets authored for a parent-relative 21-bone rig also need a compatibility pose
// for the original stickman and saved legacy rigs. The sheet renderer uses the full pose below.
function ethvelle21PoseToLegacy(p){
  const out={}, skeleton=new Set(['pelvis','spine_lower','spine_upper','chest','neck','head','clavicleL','clavicleR','upperArmL','upperArmR','forearmL','forearmR','handL','handR','thighL','thighR','shinL','shinR','footL','footR','hipY']);
  Object.keys(p||{}).forEach(k=>{if(!skeleton.has(k)&&!k.startsWith('__ethvelle'))out[k]=p[k];});
  const spine=(Number(p.spine_lower)||0)+(Number(p.spine_upper)||0)+(Number(p.chest)||0);
  out.spine=spine;
  out.head=spine+(Number(p.neck)||0)+(Number(p.head)||0);
  ['L','R'].forEach(S=>{
    const arm=(Number(p['clavicle'+S])||0)+(Number(p['upperArm'+S])||0)+spine;
    out['arm'+S]=arm;
    out['arm'+S+'_lo']=arm+(Number(p['forearm'+S])||0);
    out['hand'+S]=out['arm'+S+'_lo']+(Number(p['hand'+S])||0);
    const leg=(Number(p.pelvis)||0)+(Number(p['thigh'+S])||0);
    out['leg'+S]=leg;
    out['leg'+S+'_lo']=leg+(Number(p['shin'+S])||0)+(Number(p['foot'+S])||0);
  });
  if(p.hipY!==undefined)out.hipY=p.hipY;
  return out;
}

function ethvelleModernPoseForClipAtTime(tSecAbsolute,clipOverride,segmentsOverride){
  const useClip=clipOverride!==undefined?clipOverride:activeClip;
  const useSegments=segmentsOverride!==undefined?segmentsOverride:splitStickmanSegments;
  if(!useClip)return null;
  const hit=stickmanClipAt(tSecAbsolute,useClip,useSegments);if(!hit)return null;
  const clip=hit.clip,preset=animPresets[clip.animKey];
  if(!preset||preset._ethvelleRigVersion!==21)return null;
  const speed=clip.speed||1, localT=Math.max(0,(tSecAbsolute-clip.trimStart)*speed);
  const loopT=(localT%preset.duration)/preset.duration;
  let pose=ethvelleComplete21Pose(preset,loopT,ethvelleOriginalEvalPreset21(preset,loopT));
  if(Array.isArray(clip.keyframes)&&clip.keyframes.length){
    const kfs=[...clip.keyframes].sort((a,b)=>a.t-b.t), WINDOW=.36;
    const holdAndBlend=(kf,base)=>{
      const dist=Math.abs(tSecAbsolute-kf.t);if(dist>=WINDOW)return null;
      const weight=.5*(1+Math.cos(Math.PI*(dist/WINDOW)));
      const kp=ethvelleNormalizeLegacyPoseTo21(kf.pose||{}), blended={};
      const keys=new Set([...Object.keys(base),...Object.keys(kp)]);
      keys.forEach(k=>{
        if(k.startsWith('__'))return;
        const bv=base[k]??0;
        blended[k]=lerp(bv,kp[k]??bv,weight);
      });
      return {...base,...blended};
    };
    if(tSecAbsolute<=kfs[0].t)pose=holdAndBlend(kfs[0],pose)||pose;
    else if(tSecAbsolute>=kfs[kfs.length-1].t)pose=holdAndBlend(kfs[kfs.length-1],pose)||pose;
    else{
      let i=0;while(i<kfs.length-1&&tSecAbsolute>kfs[i+1].t)i++;
      const a=kfs[i],b=kfs[i+1],near=Math.abs(tSecAbsolute-a.t)<=Math.abs(tSecAbsolute-b.t)?a:b,far=near===a?b:a;
      pose=holdAndBlend(near,pose)||holdAndBlend(far,pose)||pose;
    }
    // Preserve preset-level discrete expression fields through the user keyframe blend.
    const mouth=ethvelleDiscrete21(preset,loopT,'mouth'),blink=ethvelleDiscrete21(preset,loopT,'blink'),fist=ethvelleDiscrete21(preset,loopT,'fist');
    pose.__mouth=mouth===undefined?'closed':mouth;pose.__blink=blink===undefined?false:!!blink;
    pose.gestureFist=fist===undefined?(Number(pose.gestureFist)||0):(fist?1:0);
  }
  return pose;
}

/* v326: knee bend (degrees, backward) for the LEFT-bone leg over one walk loop (0..1).
   The right-bone leg uses the same curve shifted by half a loop. Peak = foot in the air (leg lifting
   and swinging forward), small bend at toe-off, almost straight when the foot is planted. */
const ETHVELLE_WALK_KNEE_KEYS=[[0,12],[0.12,26],[0.25,44],[0.36,46],[0.46,20],[0.5,6],[0.75,3],[1,12]];
function ethvelleWalkKneeFlexDeg(t){
  t=((Number(t)||0)%1+1)%1;
  const k=ETHVELLE_WALK_KNEE_KEYS;
  for(let i=0;i<k.length-1;i++){
    if(t>=k[i][0]&&t<=k[i+1][0]){
      const u=(t-k[i][0])/((k[i+1][0]-k[i][0])||1),e=.5-.5*Math.cos(Math.PI*u);
      return k[i][1]+(k[i+1][1]-k[i][1])*e;
    }
  }
  return k[k.length-1][1];
}
/* v327: hip joint. The two thighs used to pivot too far apart / too low, so a gap opened under the
   pelvis triangle when the legs swung wide. Pull both hip joints a little toward the centre and up so
   the thigh tops stay overlapped with the pelvis piece. The pull grows a bit with the thigh swing.
   Tune with the three numbers below (bigger = closer / higher). */
const ETHVELLE_HIP_IN_REST=0.12, ETHVELLE_HIP_IN_SWING=0.18, ETHVELLE_HIP_UP_REST=0.008, ETHVELLE_HIP_UP_SWING=0.012;
function ethvelleTightenHips(rig,char,p){
  if(!rig||!rig.bones)return;
  const type=(char&&(char.modelType||(char.sheet&&char.sheet.modelType)))||'humanoid';
  if(type!=='humanoid')return;
  const ov=(char&&char.sheet&&char.sheet.rigOverride)||{};
  const scale=stageH*(Number(ov.figFrac)||.5), deg=Math.PI/180;
  ['L','R'].forEach(S=>{
    const b=rig.bones['thigh'+S]; if(!b||!b.off)return;
    const sw=p?Math.min(1,Math.abs(Number(p['thigh'+S])||0)/(24*deg)):0;
    b.off={x:b.off.x*(1-(ETHVELLE_HIP_IN_REST+ETHVELLE_HIP_IN_SWING*sw)),
           y:b.off.y-(ETHVELLE_HIP_UP_REST+ETHVELLE_HIP_UP_SWING*sw)*scale};
  });
}
