  // denoise/contrast/sharpen/vibrance passes on the GPU; Canvas2D is the compatibility path.
  function createEnhancePipeline(sourceWidth,sourceHeight,outWidth,outHeight,pipeOpts){
    const sw=Math.max(2,Math.round(sourceWidth||2)),sh=Math.max(2,Math.round(sourceHeight||2));
    const ow=Math.max(2,Math.round(outWidth||sw)),oh=Math.max(2,Math.round(outHeight||sh));
    const stage=document.createElement('canvas');stage.width=sw;stage.height=sh;
    const stageCtx=stage.getContext('2d',{alpha:false,desynchronized:true});
    if(!stageCtx)throw new Error('Frame staging is unavailable in this browser.');
    stageCtx.imageSmoothingEnabled=true;stageCtx.imageSmoothingQuality='high';
    const vertexSource='attribute vec2 a_position; varying vec2 v_uv; void main(){v_uv=(a_position+1.0)*0.5;gl_Position=vec4(a_position,0.0,1.0);}';
    const fragmentSource=`
      precision highp float;
      uniform sampler2D u_image;
      uniform vec2 u_sourceSize;
      uniform vec2 u_outputSize;
      uniform float u_amount;
      varying vec2 v_uv;
      float luma(vec3 c){return dot(c,vec3(0.2126,0.7152,0.0722));}
      float cubic(float v){v=abs(v);if(v<=1.0)return 1.5*v*v*v-2.5*v*v+1.0;if(v<2.0)return -0.5*v*v*v+2.5*v*v-4.0*v+2.0;return 0.0;}
      vec4 bicubicSample(vec2 uv){
#ifdef EV_UNITY
        /* v288: output is the same size as the source, so every output pixel centre lands exactly on a source
           texel centre and the 16-tap bicubic collapses to that single texel. Identical result, 1 fetch. */
        return texture2D(u_image,uv);
#else
#ifdef EV_LITE
        /* v289 lite: one hardware-bilinear tap for the upscale. The enhance passes below (denoise / contrast / unsharp) still run. */
        return texture2D(u_image,uv);
#else
#ifdef EV_FAST
        /* v288: exact Catmull-Rom (same weights as the legacy loop below) using 9 hardware-bilinear fetches
           instead of 16 nearest fetches. */
        vec2 sp=uv*u_sourceSize;
        vec2 t1=floor(sp-0.5)+0.5;
        vec2 f=sp-t1;
        vec2 w0=f*(-0.5+f*(1.0-0.5*f));
        vec2 w1=1.0+f*f*(-2.5+1.5*f);
        vec2 w2=f*(0.5+f*(2.0-1.5*f));
        vec2 w3=f*f*(-0.5+0.5*f);
        vec2 w12=w1+w2;
        vec2 o12=w2/max(w12,vec2(0.0001));
        vec2 inv=1.0/u_sourceSize;
        vec2 p0=(t1-1.0)*inv;
        vec2 p3=(t1+2.0)*inv;
        vec2 p12=(t1+o12)*inv;
        vec4 c=texture2D(u_image,vec2(p0.x,p0.y))*(w0.x*w0.y);
        c+=texture2D(u_image,vec2(p12.x,p0.y))*(w12.x*w0.y);
        c+=texture2D(u_image,vec2(p3.x,p0.y))*(w3.x*w0.y);
        c+=texture2D(u_image,vec2(p0.x,p12.y))*(w0.x*w12.y);
        c+=texture2D(u_image,vec2(p12.x,p12.y))*(w12.x*w12.y);
        c+=texture2D(u_image,vec2(p3.x,p12.y))*(w3.x*w12.y);
        c+=texture2D(u_image,vec2(p0.x,p3.y))*(w0.x*w3.y);
        c+=texture2D(u_image,vec2(p12.x,p3.y))*(w12.x*w3.y);
        c+=texture2D(u_image,vec2(p3.x,p3.y))*(w3.x*w3.y);
        return c;
#else
        vec2 pixel=uv*u_sourceSize-0.5;vec2 base=floor(pixel);vec2 f=fract(pixel);
        vec4 sum=vec4(0.0);float total=0.0;
        for(int y=-1;y<=2;y++){for(int x=-1;x<=2;x++){
          float weight=cubic(float(x)-f.x)*cubic(float(y)-f.y);
          vec2 at=(base+vec2(float(x),float(y))+0.5)/u_sourceSize;
          sum+=texture2D(u_image,clamp(at,vec2(0.0),vec2(1.0)))*weight;total+=weight;
        }}
        return sum/max(total,0.0001);
#endif
#endif
#endif
      }
      void main(){
        vec4 raw=bicubicSample(v_uv);vec3 original=raw.rgb;
        float a=clamp(u_amount,0.0,1.0);
        if(a<0.0001){gl_FragColor=vec4(original,raw.a);return;}
        vec2 t=1.0/u_sourceSize;
        vec3 n1=texture2D(u_image,clamp(v_uv+vec2(t.x,0.0),vec2(0.0),vec2(1.0))).rgb;
        vec3 n2=texture2D(u_image,clamp(v_uv-vec2(t.x,0.0),vec2(0.0),vec2(1.0))).rgb;
        vec3 n3=texture2D(u_image,clamp(v_uv+vec2(0.0,t.y),vec2(0.0),vec2(1.0))).rgb;
        vec3 n4=texture2D(u_image,clamp(v_uv-vec2(0.0,t.y),vec2(0.0),vec2(1.0))).rgb;
        float y=luma(original);
        float w1=exp(-abs(y-luma(n1))*16.0)*0.25;
        float w2=exp(-abs(y-luma(n2))*16.0)*0.25;
        float w3=exp(-abs(y-luma(n3))*16.0)*0.25;
        float w4=exp(-abs(y-luma(n4))*16.0)*0.25;
        vec3 local=(n1*w1+n2*w2+n3*w3+n4*w4)/max(w1+w2+w3+w4,0.0001);
        vec3 denoised=mix(original,local,a*0.18);
        float localY=luma((n1+n2+n3+n4)*0.25);
        float adaptive=clamp(abs(y-localY)*3.0,0.0,1.0);
        vec3 contrasted=(denoised-0.5)*(1.0+a*(0.20+0.15*adaptive))+0.5;
        float shadow=pow(max(0.0,1.0-y),2.0);
        contrasted+=vec3(shadow*a*0.105);
        vec3 sharpened=contrasted+(denoised-local)*a*0.58;
        float gray=luma(sharpened);
        float saturation=max(max(sharpened.r,sharpened.g),sharpened.b)-min(min(sharpened.r,sharpened.g),sharpened.b);
        sharpened+=(sharpened-vec3(gray))*a*0.27*(1.0-clamp(saturation,0.0,1.0));
        // A second, lighter post-upscale sharpen keeps fine edges legible after interpolation.
        sharpened+=(original-local)*a*0.12;
        gl_FragColor=vec4(clamp(sharpened,0.0,1.0),raw.a);
      }`;

  /* v287 BEFORE/AFTER SYNC FIX. The two halves used to be filled by two separate reads of the live video
     (the AFTER half through a full-resolution WebGL pass that was also rebuilt every time playback started or
     stopped), so AFTER could trail BEFORE by 0.1-0.5 s. Now:
       1. ONE snapshot of the current video frame is taken per update;
       2. BEFORE and AFTER are both drawn from that same snapshot in the same synchronous block, so they can never
          show different moments;
       3. updates are driven by the video's own frame callback (requestVideoFrameCallback) plus seeked/loadeddata,
          so both halves advance on exactly the frames the video presents;
       4. the preview pipeline works on the small snapshot and is NOT rebuilt on play/pause, so it is cheap and
          never stalls the start of playback. */
  function syncCanvasCompare(force){
    force=(force===true);
    const root=$('ethvelleCanvasCompareRoot');if(!root)return;
    const target=getEnhanceTarget();if(!target)return;
    const before=root.querySelector('.evcc-before .evcc-media'),after=root.querySelector('.evcc-after .evcc-media');if(!before||!after)return;
    const source=referenceMediaForCompare(target);if(!source)return;
    const isVideo=source.tagName==='VIDEO';
    const sw=Number(source.videoWidth||source.naturalWidth||source.width)||0,sh=Number(source.videoHeight||source.naturalHeight||source.height)||0;
    if(!sw||!sh||(isVideo&&source.readyState<2))return;
    const live=!!(isPlaying||window.__ethvelleTimelineScrubbing);
    const now=performance.now();
    if(live&&!force){
      // While playing, the video frame callback drives the updates; only fall back to the render loop when it is unavailable.
      if(isVideo&&compareRvfcEl===source&&typeof source.requestVideoFrameCallback==='function')return;
      if(compareLastPlaybackSyncAt&&now-compareLastPlaybackSyncAt<33)return;
    }
    const factor=Math.min(1,COMPARE_PREVIEW_LONG_SIDE/Math.max(sw,sh)),pw=Math.max(2,Math.round(sw*factor)),ph=Math.max(2,Math.round(sh*factor));
    const amount=Math.max(0,Math.min(100,Number($('evltEnhanceAmount')?.value??target.model.enhanceAmount??0)||0));
    const filter=typeof window.ethvelleAdjustmentFilter==='function'?window.ethvelleAdjustmentFilter(target.model.adjustments,0):'none';
    const key=`${source.currentSrc||source.src||''}|${pw}x${ph}|${amount}|${filter}`,t=isVideo?Number(source.currentTime)||0:0;
    if(!live&&!force&&source===compareLastSource&&key===compareLastKey&&t===compareLastTime)return;
    if(!isVideo&&!force&&source===compareLastSource&&key===compareLastKey)return;
    const started=performance.now();
    // 1) ONE snapshot of the current frame - both halves are built from this and nothing else.
    if(!compareSnap)compareSnap=document.createElement('canvas');
    if(compareSnap.width!==pw||compareSnap.height!==ph){compareSnap.width=pw;compareSnap.height=ph;compareSnapCtx=null;}
    if(!compareSnapCtx){compareSnapCtx=compareSnap.getContext('2d',{alpha:false});if(!compareSnapCtx)return;compareSnapCtx.imageSmoothingEnabled=true;compareSnapCtx.imageSmoothingQuality='high';}
    try{compareSnapCtx.drawImage(source,0,0,pw,ph);}catch(_e){return;}
    // 2) Preview pipeline sized to the snapshot (rebuilt only if the preview size itself changes).
    const pkey=pw+'x'+ph;
    if(comparePipelineKey!==pkey||!comparePipeline){
      try{if(comparePipeline)comparePipeline.dispose();comparePipeline=window.ethvelleCreateEnhancePipeline(pw,ph,pw,ph);comparePipelineKey=pkey;}
      catch(err){console.warn('Ethvelle: comparison processor unavailable',err);comparePipeline=null;comparePipelineKey='';}
    }
    let processed=null;
    if(comparePipeline){try{processed=comparePipeline.render(compareSnap,filter,amount);}catch(err){console.warn('Ethvelle: comparison frame failed',err);}}
    // 3) Paint BEFORE and AFTER back-to-back in the same synchronous block, from the same snapshot.
    if(before.width!==pw||before.height!==ph){before.width=pw;before.height=ph;}
    if(after.width!==pw||after.height!==ph){after.width=pw;after.height=ph;}
    const beforeCtx=before.getContext('2d'),afterCtx=after.getContext('2d');
    if(beforeCtx){beforeCtx.save();beforeCtx.filter=filter;beforeCtx.imageSmoothingEnabled=true;beforeCtx.imageSmoothingQuality='high';beforeCtx.clearRect(0,0,pw,ph);beforeCtx.drawImage(compareSnap,0,0,pw,ph);beforeCtx.restore();}
    if(afterCtx&&processed){afterCtx.clearRect(0,0,pw,ph);afterCtx.imageSmoothingEnabled=true;afterCtx.imageSmoothingQuality='high';afterCtx.drawImage(processed,0,0,pw,ph);}
    if(!live)setCanvasComparePosition(canvasComparePercent);
    compareLastSource=source;compareLastKey=key;compareLastTime=t;compareLastPlaybackSyncAt=performance.now();compareLastCost=compareLastPlaybackSyncAt-started;
  }
