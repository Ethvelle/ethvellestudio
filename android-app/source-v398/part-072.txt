  function mountRuler(canvas,def){
    const ctx=canvas.getContext('2d');
    let w=0,dpr=1,pos=0,drag=null,raf=0,pending=null,anim=0,lastEmit=null;
    const clampV=v=>Math.max(def.min,Math.min(def.max,v));
    function rr(x,y,ww,hh,r){
      ctx.beginPath();
      if(ctx.roundRect){ctx.roundRect(x,y,ww,hh,r);}
      else{ctx.moveTo(x+r,y);ctx.arcTo(x+ww,y,x+ww,y+hh,r);ctx.arcTo(x+ww,y+hh,x,y+hh,r);ctx.arcTo(x,y+hh,x,y,r);ctx.arcTo(x,y,x+ww,y,r);ctx.closePath();}
      ctx.fill();
    }
    function resize(){
      const cw=canvas.clientWidth|0;if(!cw)return false;
      dpr=Math.min(3,window.devicePixelRatio||1);
      if(cw!==w||canvas.width!==Math.round(cw*dpr)){w=cw;canvas.width=Math.round(w*dpr);canvas.height=Math.round(ROW_H*dpr);}
      draw();return true;
    }
    function draw(){
      if(!w)return;
      ctx.setTransform(dpr,0,0,dpr,0,0);ctx.clearRect(0,0,w,ROW_H);
      const cx=w/2,half=cx/SP;
      const lo=Math.max(Math.ceil(def.min),Math.floor(pos-half)-1);
      const hi=Math.min(Math.floor(def.max),Math.ceil(pos+half)+1);
      for(let v=lo;v<=hi;v++){
        const x=cx+(v-pos)*SP;
        if(x<-2||x>w+2)continue;
        if(Math.abs(x-cx)<1.6)continue;
        const edge=Math.min(x,w-x);
        const fade=Math.max(0,Math.min(1,(edge-SP*.4)/(SP*2.6)));
        if(fade<=0)continue;
        const major=(v%MAJOR===0);
        ctx.fillStyle=major?`rgba(255,255,255,${(.93*fade).toFixed(3)})`:`rgba(255,255,255,${(.40*fade).toFixed(3)})`;
        rr(x-.8,BASE_Y-TICK_H,1.6,TICK_H,.8);
      }
      ctx.fillStyle=BLUE;rr(cx-1.3,IND_TOP,2.6,IND_H,1.3);
    }
    function apply(v,emit){
      pos=clampV(v);const iv=Math.round(pos);
      if(emit&&lastEmit!==iv){lastEmit=iv;try{def.input(iv);}catch(err){console.warn('Ethvelle ruler:',err);}}
      canvas.setAttribute('aria-valuenow',String(iv));draw();
    }
    let dirty=false;
    function finish(){if(lastEmit!==null){try{def.commit(Math.round(pos));}catch(err){console.warn('Ethvelle ruler:',err);}}lastEmit=null;}
    function stopAnim(){if(anim){cancelAnimationFrame(anim);anim=0;}}
    function settle(){
      stopAnim();const from=pos,to=Math.round(pos);
      if(Math.abs(from-to)<.001){apply(to,true);finish();return;}
      const t0=performance.now(),dur=130;
      const step=now=>{const k=Math.min(1,(now-t0)/dur),e=1-Math.pow(1-k,3);apply(from+(to-from)*e,true);if(k<1)anim=requestAnimationFrame(step);else{anim=0;apply(to,true);finish();}};
      anim=requestAnimationFrame(step);
    }
    function fling(vel){
      stopAnim();let last=performance.now();
      const step=now=>{
        const dt=Math.min(34,now-last);last=now;pos+=vel*dt;vel*=Math.exp(-dt/240);
        if(pos<=def.min||pos>=def.max){apply(pos<=def.min?def.min:def.max,true);anim=0;finish();return;}
        apply(pos,true);
        if(Math.abs(vel)<.0035){anim=0;settle();return;}
        anim=requestAnimationFrame(step);
      };
      anim=requestAnimationFrame(step);
    }
    canvas.addEventListener('pointerdown',e=>{
      if(drag||(e.pointerType==='mouse'&&e.button!==0))return;
      stopAnim();drag={id:e.pointerId,x0:e.clientX,p0:pos,s:[{t:e.timeStamp,x:e.clientX}]};lastEmit=null;
      try{canvas.setPointerCapture(e.pointerId);}catch(_e){}
      e.stopPropagation();
    },{passive:true});
    canvas.addEventListener('pointermove',e=>{
      if(!drag||drag.id!==e.pointerId)return;
      drag.s.push({t:e.timeStamp,x:e.clientX});if(drag.s.length>8)drag.s.shift();
      pending=drag.p0-(e.clientX-drag.x0)/SP;
      if(!raf)raf=requestAnimationFrame(()=>{raf=0;if(pending!==null){apply(pending,true);pending=null;}});
      if(e.cancelable)e.preventDefault();e.stopPropagation();
    },{passive:false});
    const end=e=>{
      if(!drag||drag.id!==e.pointerId)return;
      if(raf){cancelAnimationFrame(raf);raf=0;}
      if(pending!==null){apply(pending,true);pending=null;}
      const s=drag.s,last=s[s.length-1];let vel=0;
      for(let i=s.length-2;i>=0;i--){if(last.t-s[i].t>=70||i===0){const dt=Math.max(1,last.t-s[i].t);vel=-(last.x-s[i].x)/dt/SP;break;}}
      if(e.type==='pointercancel'||e.timeStamp-last.t>90)vel=0;
      drag=null;try{canvas.releasePointerCapture(e.pointerId);}catch(_e){}
      if(Math.abs(vel)>.012)fling(Math.max(-2.2,Math.min(2.2,vel)));else settle();
      e.stopPropagation();
    };
    canvas.addEventListener('pointerup',end);
    canvas.addEventListener('pointercancel',end);
    canvas.addEventListener('keydown',e=>{
      const step=e.shiftKey?MAJOR:1;
      if(e.key==='ArrowLeft'||e.key==='ArrowDown'){apply(Math.round(pos)-step,true);finish();e.preventDefault();}
      else if(e.key==='ArrowRight'||e.key==='ArrowUp'){apply(Math.round(pos)+step,true);finish();e.preventDefault();}
    });
    pos=clampV(Number(def.value)||0);
    canvas.setAttribute('role','slider');canvas.setAttribute('tabindex','0');canvas.setAttribute('aria-label',def.label||'');
    canvas.setAttribute('aria-valuemin',String(def.min));canvas.setAttribute('aria-valuemax',String(def.max));canvas.setAttribute('aria-valuenow',String(Math.round(pos)));
    if(typeof ResizeObserver==='function'){new ResizeObserver(()=>resize()).observe(canvas);}
    requestAnimationFrame(()=>{resize();setTimeout(resize,320);});
    return {resize,draw};
  }

  let S = null;                                          // the stroke being painted
  const bp = new Map(); let pinch = null;                // pointers on the canvas / two-finger pan-zoom state
  const pos = e=>{ const r = ED.cv.getBoundingClientRect(); return { x:e.clientX-r.left, y:e.clientY-r.top }; };

  function buildTargets(){
    const W = edWorld(), list = B.scope === 'sel' ? (ED.sel ? [ED.sel] : []) : ED.order, seen = {}, out = [];
    list.forEach(k=>{
      if(!W[k] || !ED.st[k] || !edKeyVisible(k) || edIsLocked(k)) return;
      const ak = ED.assign[k], art = ED.arts[ak]; if(!art || !art.canvas || seen[ak]) return; seen[ak] = 1;
      const w = W[k], st = ED.st[k];
      out.push({ k, ak, art, st, w, c:Math.cos(w.rot), s:Math.sin(w.rot), sx:w.sx, sy:w.sy, ctx:null, cur:null, orig:null, changed:false, x0:1e9, y0:1e9, x1:-1, y1:-1 });
    });
    return out;
  }
  function load(T){
    if(T.cur) return;
    T.ctx = T.art.canvas.getContext('2d');
    T.cur = T.ctx.getImageData(0, 0, T.art.w, T.art.h);
    T.orig = new Uint8ClampedArray(T.cur.data);          // pixels as they were when the stroke began - every dab is computed from these, so overlapping dabs never compound
  }
  function dab(T, fx, fy, r){
    const dx = fx-T.w.x, dy = fy-T.w.y;
    const lx = T.st.pvx + (dx*T.c + dy*T.s)/T.sx, ly = T.st.pvy + (-dx*T.s + dy*T.c)/T.sy;   // brush centre in the part's own pixels
    const rx = r/Math.abs(T.sx), ry = r/Math.abs(T.sy), aw = T.art.w, ah = T.art.h;
    const y0 = Math.max(0, Math.floor(ly-ry)), y1 = Math.min(ah-1, Math.ceil(ly+ry));
    const bx0 = Math.max(0, Math.floor(lx-rx)), bx1 = Math.min(aw-1, Math.ceil(lx+rx));
    if(y1 < y0 || bx1 < bx0) return false;
    load(T);
    const d = T.cur.data, o = T.orig, all = B.mode === 'all', t = darkLimit();
    let hit = false, ex0 = 1e9, ex1 = -1, ey0 = 1e9, ey1 = -1;
    for(let y = y0; y <= y1; y++){
      const ty = (y+.5-ly)/ry, q = 1-ty*ty; if(q < 0) continue;
      const half = rx*Math.sqrt(q), xs = Math.max(0, Math.ceil(lx-half-.5)), xe = Math.min(aw-1, Math.floor(lx+half-.5));
      for(let x = xs; x <= xe; x++){
        const i = (y*aw+x)*4, a = o[i+3]; if(!a) continue;
        let rem = 1;
        if(!all){ const m = Math.max(o[i], o[i+1], o[i+2]); rem = m <= t ? 1 : m >= t+FEATHER ? 0 : 1-(m-t)/FEATHER; }
        if(rem <= 0) continue;
        let na = Math.round(a*(1-rem)); if(na < 10) na = 0;
        if(na < d[i+3]){ d[i+3] = na; hit = true; if(x<ex0)ex0=x; if(x>ex1)ex1=x; if(y<ey0)ey0=y; if(y>ey1)ey1=y; }
      }
    }
    if(!hit) return false;
    T.ctx.putImageData(T.cur, 0, 0, ex0, ey0, ex1-ex0+1, ey1-ey0+1);
    T.changed = true; T.x0 = Math.min(T.x0, ex0); T.y0 = Math.min(T.y0, ey0); T.x1 = Math.max(T.x1, ex1); T.y1 = Math.max(T.y1, ey1);
    return true;
  }
  function addPoint(p){
    if(!S) return;
    const r = (B.size/2)/ED.cam.z, sp = Math.max(1, B.size*0.2), pts = [];
    if(!S.last) pts.push(p);
    else { const dx = p.x-S.last.x, dy = p.y-S.last.y, n = Math.max(1, Math.ceil(Math.hypot(dx,dy)/sp)); for(let i = 1; i <= n; i++) pts.push({ x:S.last.x+dx*i/n, y:S.last.y+dy*i/n }); }
    try{
      pts.forEach(q=>{ const fx = (q.x-ED.cam.x)/ED.cam.z, fy = (q.y-ED.cam.y)/ED.cam.z; S.targets.forEach(T=>dab(T, fx, fy, r)); });
    }catch(err){
      console.error('Ethvelle brush:', err); cancelStroke();
      try{ showToast("This part's image can't be edited here", { duration:2200 }); }catch(_e){}
      return;
    }
    S.last = p; edDraw();
  }
  function startStroke(){
    const targets = buildTargets();
    if(!targets.length){
      if(B.scope === 'sel'){ try{ showToast('Select a part first, or switch to All parts', { duration:1800 }); }catch(_e){} }
      S = { targets:[], last:null }; return;
    }
    S = { targets, last:null };
  }
  function cancelStroke(){
    if(!S) return;
    S.targets.forEach(T=>{ if(T.changed && T.ctx) T.ctx.putImageData(new ImageData(new Uint8ClampedArray(T.orig), T.art.w, T.art.h), 0, 0); });
    S = null; edDraw();
  }
  function refreshArt(art, x, y, img){
    art.src = null; art.thumb = null;                    // forces Done to re-encode this part from the cleaned canvas
    if(art.alpha) for(let row = 0; row < img.height; row++) for(let col = 0; col < img.width; col++) art.alpha[(y+row)*art.w + x+col] = img.data[(row*img.width+col)*4+3] > 30 ? 1 : 0;
  }
  function commit(){
    const st = S; S = null; if(!st) return;
    const ts = st.targets.filter(T=>T.changed); if(!ts.length) return;
