    if(!node||!t)return;
    const enabled=!!(t.textBackgroundEnabled||t.textHighlightEnabled);
    node.classList.toggle('ev-text-highlight-plate',enabled);
    if(!enabled){['--ev-text-highlight-fill','--ev-text-highlight-padding','--ev-text-highlight-inset','--ev-text-highlight-radius'].forEach(k=>node.style.removeProperty(k));return;}
    const col=normalizeHex(colorOverride||t.textBackgroundColor||t.textHighlightColor||'#182A66');
    const op=t.textBackgroundOpacity==null?(t.textHighlightOpacity==null?78:t.textHighlightOpacity):t.textBackgroundOpacity;
    const pad=Math.max(0,Number(t.textBackgroundPadding==null?(t.textHighlightPadding==null?8:t.textHighlightPadding):t.textBackgroundPadding));
    const radius=Math.max(0,Number(t.textBackgroundRadius==null?(t.textHighlightRadius==null?8:t.textHighlightRadius):t.textBackgroundRadius));
    const h=col.slice(1),n=[0,2,4].map(i=>parseInt(h.slice(i,i+2),16));
    node.style.setProperty('--ev-text-highlight-fill',`rgba(${n[0]},${n[1]},${n[2]},${Math.max(0,Math.min(100,Number(op==null?78:op)))/100})`);
    node.style.setProperty('--ev-text-highlight-padding',pad+'px');node.style.setProperty('--ev-text-highlight-inset',(-pad)+'px');node.style.setProperty('--ev-text-highlight-radius',radius+'px');
    if(window.ethvelleSyncTextBackplateShape)window.ethvelleSyncTextBackplateShape(node,t);
  }
  let colorSaveTimer=null;
  function persistNow(){
    if(colorSaveTimer!==null){clearTimeout(colorSaveTimer);colorSaveTimer=null;}
    try{if(typeof persistProjectState==='function')persistProjectState({skipHistory:true});}catch(_e){try{if(typeof persistProjectState==='function')persistProjectState();}catch(__){}}
  }
  function persistOne(){
    if(colorSaveTimer!==null)clearTimeout(colorSaveTimer);
    colorSaveTimer=setTimeout(()=>{colorSaveTimer=null;try{if(typeof persistProjectState==='function')persistProjectState({skipHistory:true});}catch(_e){try{if(typeof persistProjectState==='function')persistProjectState();}catch(__){}}},100);
  }
  window.addEventListener('pagehide',persistNow,{passive:true});
  // Swatch changes are applied to the already-rendered node instead of rerendering the whole stage.
  // This prevents slow Highlight/Stroke palette taps and keeps the swatch row at its exact scroll offset.
  window.ethvellePicsartSetColor=function(key,color){
    const value=normalizeHex(color),t=typeof window.ethvellePicsartGetTarget==='function'?window.ethvellePicsartGetTarget():null;
    if(!t){if(typeof priorSetColor==='function')return priorSetColor.apply(this,arguments);return;}
    const composer=$('ethvelleTextComposer')?.classList.contains('active');
    if(composer&&typeof priorSetColor==='function'){return priorSetColor.apply(this,arguments);}
    t[key]=value;
    if(key==='textColor'){
      t.textGradient='';t.textTexture='';t.textTextureSize='100% 100%';
      if(Number(t.textOpacity)===0)t.textOpacity=100;
    }
    const activeTab=$('evpaTabs')?.querySelector('.evpa-tab.active')?.dataset.tab||'';
    if(key==='textBackgroundColor'&&(activeTab==='highlight'||activeTab==='background'))t.textHighlightColor=value;
    const node=getTextNode(t),glyph=node?.querySelector(':scope > .evpa-text-effect-layer');
    if(node){
      if(key==='textColor'){
        const fill=rgba(value,t.textOpacity);
        node.style.setProperty('color',fill,'important');
        node.style.setProperty('background-image','none','important');
        node.style.setProperty('background-clip','border-box','important');
        node.style.setProperty('-webkit-background-clip','border-box','important');
        if(glyph){
          glyph.style.setProperty('color',fill,'important');
          glyph.style.setProperty('-webkit-text-fill-color',fill,'important');
          glyph.style.setProperty('background-image','none','important');
          glyph.style.setProperty('background-clip','border-box','important');
          glyph.style.setProperty('-webkit-background-clip','border-box','important');
          glyph.style.setProperty('background-size','auto','important');
          glyph.style.setProperty('opacity','1','important');
          glyph.querySelectorAll('span').forEach(span=>{
            span.style.setProperty('color',fill,'important');
            span.style.setProperty('-webkit-text-fill-color',fill,'important');
            span.style.setProperty('background-image','none','important');
            span.style.removeProperty('background-clip');
            span.style.removeProperty('-webkit-background-clip');
            span.style.removeProperty('background-size');
            span.style.removeProperty('background-position');
            span.style.removeProperty('background-repeat');
          });
        }
      }else if(key==='strokeColor'){
        const width=Math.max(0,Number(t.strokeWidth)||0),stroke=width>0?`${width}px ${rgba(value,t.strokeOpacity)}`:'0 transparent';
        if(glyph){glyph.style.webkitTextStroke=stroke;glyph.style.paintOrder='stroke fill';}
        node.style.webkitTextStroke='0 transparent';node.style.paintOrder='normal';
      }else if(key==='shadowColor'){
        const shadow=t.shadowEnabled?`${(Number(t.shadowX)||0)}px ${Math.max(4,Number(t.shadowY)||6)}px ${Math.max(0,Number(t.shadowBlur)||0)}px ${rgba(value,t.shadowOpacity)}`:'none';
        if(glyph)window.ethvelleApplyTextShadow(glyph,shadow,!!(String(t.textGradient||'').trim()||String(t.textTexture||'').trim()));else node.style.textShadow=shadow;
        node.style.textShadow='none';
      }else if(key==='textBackgroundColor'){
        plateStyle(node,t,value);
      }
    }
    const palette=$('evpaPalette');if(palette)palette.querySelectorAll('.evpa-swatch[data-color]').forEach(b=>b.classList.toggle('active',normalizeHex(b.dataset.color)===value));
    persistOne();
  };
  // Refresh glyph-only shadow/stroke when a model redraw changes the numeric controls.
  function syncGlyphPaint(){
    let list=[];try{list=typeof overlays!=='undefined'?overlays:(window.overlays||[]);}catch(_e){list=window.overlays||[];}
    list.filter(o=>o&&o.type==='text').forEach(t=>{
      const node=getTextNode(t);if(!node)return;const glyph=node.querySelector(':scope > .evpa-text-effect-layer');if(!glyph)return;
      const strokeWidth=Math.max(0,Number(t.strokeWidth)||0);
      glyph.style.webkitTextStroke=strokeWidth>0?`${strokeWidth}px ${rgba(t.strokeColor||'#000000',t.strokeOpacity)}`:'0 transparent';
      glyph.style.paintOrder='stroke fill';
      window.ethvelleApplyTextShadow(glyph,t.shadowEnabled?`${(Number(t.shadowX)||0)}px ${Math.max(4,Number(t.shadowY)||6)}px ${Math.max(0,Number(t.shadowBlur)||0)}px ${rgba(t.shadowColor||'#000000',t.shadowOpacity)}`:'none',!!(String(t.textGradient||'').trim()||String(t.textTexture||'').trim()));
      node.style.webkitTextStroke='0 transparent';node.style.paintOrder='normal';node.style.textShadow='none';
      if(t.textBackgroundEnabled||t.textHighlightEnabled)plateStyle(node,t);
    });
  }
  ['renderFrame','renderOverlayLayers'].forEach(name=>{
    const original=window[name];if(typeof original!=='function'||original.__evvelleV14Paint)return;
    const wrapped=function(){const result=original.apply(this,arguments);syncGlyphPaint();return result;};
    wrapped.__evvelleV14Paint=true;window[name]=wrapped;
  });
  // Keep section palette visibility accurate on both startup and tab changes, without rebuilding it.
  document.addEventListener('click',e=>{const tab=e.target?.closest?.('#evpaTabs .evpa-tab');if(!tab)return;requestAnimationFrame(()=>{const dock=$('ethvellePicsartTextDock');if(!dock)return;const font=tab.dataset.tab==='font';if(font)dock.classList.remove('evpa-no-palette');});},true);
  if(document.readyState==='loading')document.addEventListener('DOMContentLoaded',syncGlyphPaint,{once:true});else syncGlyphPaint();
})();

<script id="ethvelle-v19-contextual-toolbar">
(function(){
  'use strict';
  const byId=id=>document.getElementById(id);

  function selectedId(){
    try{return String(typeof selectedClip!=='undefined'&&selectedClip!=null?selectedClip:(window.selectedClip??''));}
    catch(_e){return String(window.selectedClip??'');}
  }
  function selectedOverlay(){
    const id=selectedId();
    try{
      const list=typeof overlays!=='undefined'?overlays:(window.overlays||[]);
      return (list||[]).find(item=>item&&String(item.id)===id)||null;
    }catch(_e){return null;}
  }
  function setTextEditVisibility(){
    const edit=byId('clipTextEditTool');
    if(!edit)return;
    const clip=selectedOverlay();
    const isText=!!clip&&clip.type==='text';
    // This is the final toolbar sync layer, after legacy toolbar scripts have run.
    // Only a selected text clip is allowed to expose the pencil/text-edit control.
    edit.style.display=isText?'flex':'none';
    edit.setAttribute('aria-hidden',isText?'false':'true');
  }
  function setContextualAddAction(){
    const add=byId('clipAddMediaTool');
    if(!add||add.dataset.evV19ContextAdd==='1')return;
    add.dataset.evV19ContextAdd='1';
    // Replaces the old inline handler rather than adding a second click listener, so a text
    // selection cannot accidentally launch the device media picker as well.
    add.onclick=function(event){
      if(event){event.preventDefault();event.stopPropagation();}
      const clip=selectedOverlay();
      if(clip&&clip.type==='text'){
        if(typeof window.openEthvelleTextComposer==='function'){
          window.openEthvelleTextComposer('__ethvelle_new_text_layer__');
        }
        return false;
      }
      const isAudio=!!clip&&clip.type==='audio';
      if(typeof window.triggerUpload==='function')window.triggerUpload(isAudio?'audio':'overlay',isAudio?'audio':'clips');
      return false;
    };
  }
  function refresh(){setContextualAddAction();setTextEditVisibility();}

  const oldSync=window.syncBottomToolsMode;
  if(typeof oldSync==='function'&&!oldSync.__evV19ContextToolbar){
    const wrapped=function(){const result=oldSync.apply(this,arguments);refresh();return result;};
    wrapped.__evV19ContextToolbar=true;window.syncBottomToolsMode=wrapped;
  }
  const oldSelect=window.selectClipForChrome;
  if(typeof oldSelect==='function'&&!oldSelect.__evV19ContextToolbar){
    const wrapped=function(){const result=oldSelect.apply(this,arguments);refresh();return result;};
    wrapped.__evV19ContextToolbar=true;window.selectClipForChrome=wrapped;
  }
  const oldDeselect=window.deselectAllClips;
  if(typeof oldDeselect==='function'&&!oldDeselect.__evV19ContextToolbar){

  }
