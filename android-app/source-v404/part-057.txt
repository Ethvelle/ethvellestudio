  function parseRgb(css){

      const copy=document.createElement('canvas');copy.width=frame.width;copy.height=frame.height;
      const ctx=copy.getContext('2d',{willReadFrequently:true});if(!ctx)throw new Error('Pixel reader is unavailable');
      ctx.drawImage(frame,0,0);
      ctx.getImageData(0,0,1,1); // throws if any source tainted the canvas
      return {canvas:copy,ctx};
    };

    try{snap=build();}
    catch(err){
      // A remote image/video without CORS blocks pixel reads. Retry without those sources
      // so the picker still samples the clips and layers it is allowed to read.
      try{snap=build({safe:true});}
      finally{if(typeof window.renderFrame==='function')try{window.renderFrame();}catch(_e){}}

    const data=snap.ctx.getImageData(px,py,1,1).data;

    if(aa<1){
      const bg=parseRgb(getComputedStyle(stage).backgroundColor),ba=bg[3];

      rr=rr*aa+br*(1-aa);gg=gg*aa+bgc*(1-aa);bb=bb*aa+bbg*(1-aa);

  // Text and Audio rows. The scroll delta is folded into the drag delta, keeping the
  // dragged clip directly under the finger while the timeline content moves underneath.
  let timelineEdgeAutoScrollRaf = 0;

  function timelineDragDeltaPixels(dragState, clientX){
    if(!dragState)return 0;

    const startX=Number(dragState.startX)||0;
    const startScroll=Number.isFinite(Number(dragState.startScrollLeft))?Number(dragState.startScrollLeft):(scroller?Number(scroller.scrollLeft)||0:0);
    const currentScroll=scroller?Number(scroller.scrollLeft)||0:startScroll;
    return (Number(clientX)-startX)+(currentScroll-startScroll);
  }

  function ensureTimelineRoomForDraggedClip(clip){

        scroller.scrollLeft=after;
        try{if(typeof state.update==='function')state.update();}catch(err){console.warn('Ethvelle timeline auto-scroll update failed',err);}
      }
    }
    if(timelineEdgeAutoScrollState===state)timelineEdgeAutoScrollRaf=requestAnimationFrame(runTimelineEdgeAutoScroll);
  }

  }
