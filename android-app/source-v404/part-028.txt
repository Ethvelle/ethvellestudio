// Down-samples a part's alpha silhouette onto a small fixed grid so two parts of slightly different
// pixel sizes can still be compared shape-for-shape.
function edSiloGrid(art, gw, gh){
  const g = new Float32Array(gw*gh);
  for(let gy=0; gy<gh; gy++) for(let gx=0; gx<gw; gx++){
    const x0 = Math.floor(gx*art.w/gw), x1 = Math.max(x0+1, Math.floor((gx+1)*art.w/gw));
    const y0 = Math.floor(gy*art.h/gh), y1 = Math.max(y0+1, Math.floor((gy+1)*art.h/gh));
    let s = 0, n = 0;
    for(let y=y0; y<y1; y++) for(let x=x0; x<x1; x++){ s += art.alpha[y*art.w+x]; n++; }
    g[gy*gw+gx] = n ? s/n : 0;
  }
  return g;
}
// True when the RIGHT-side piece looks like the same drawing as the LEFT-side piece just pasted
// again (same orientation) rather than a genuine mirror image of it -- the classic "both hands drawn
// with the thumb on the same side" mistake that makes one arm look anatomically backwards. Compares
// silhouettes only (not colour/shading), both as-is and with the right side flipped; if flipping would
// clearly line the two shapes up better, the art was never mirrored to begin with.
function edLooksUnmirrored(artL, artR){
  const gw = 16, gh = 28, gL = edSiloGrid(artL, gw, gh), gR = edSiloGrid(artR, gw, gh), gRf = new Float32Array(gw*gh);
  for(let y=0; y<gh; y++) for(let x=0; x<gw; x++) gRf[y*gw+x] = gR[y*gw+(gw-1-x)];
  const diff = (a,b)=>{ let s = 0; for(let i=0;i<gw*gh;i++) s += Math.abs(a[i]-b[i]); return s; };
  const direct = diff(gL, gR), flipped = diff(gL, gRf);
  // Only act when the pair is clearly a near-duplicate in the first place, and mirroring it would
  // unambiguously line the shapes up better -- avoids "fixing" two legitimately different poses.
  return direct < gw*gh*0.06 && direct < flipped*0.7;
}

    if(!R || !T || !H || !st || !st.head || view !== 'front' || modelType !== 'humanoid') return st;
    if(T.dress || !T.bodyIncludesNeckAndWaist || (opts && opts.beast)) return st;
    const P = R.parts, rT = P.torso, rH = P.head;
    if(!rT || !rH) return st;
    const sz = a=>Math.sqrt(Math.max(1,a.w)*Math.max(1,a.h));
    const mk = a=>({ ax:0, ay:0, pvx:a.dpx, pvy:a.dpy, s:1, wx:1, hy:1, fx:1, fy:1, rot:0 });
    const ensure = k=>{ if(!A[k]) return null; if(!st[k]) st[k] = mk(A[k]); return st[k]; };
    // 1) base-body parts: only the scale comes from the sheet (their positions are measured from each model's own artwork)
    ['torso','armL_upper','armL_lower','handL','armR_upper','armR_lower','handR','legL_upper','legL_lower','shoeL','legR_upper','legR_lower','shoeR'].forEach(k=>{
      if(st[k] && P[k] && Number.isFinite(P[k].s)) st[k].s = P[k].s;
    });
    // 2) expression heads: same neck offset as the sheet, measured from the base head
    const kTx = T.w/rT.w, kTy = T.h/rT.h;
    ['head_sad','head_angry','head_shock','head_laugh'].forEach(k=>{
      const r = P[k], s0 = ensure(k); if(!r || !s0) return;
      s0.ax = st.head.ax + (r.ax - rH.ax)*kTx; s0.ay = st.head.ay + (r.ay - rH.ay)*kTy;
      s0.rot = r.rot; s0.fx = r.fx; s0.fy = r.fy; s0.wx = r.wx; s0.hy = r.hy;
    });
    // 3) face kit nested in the head: position as a fraction of the head artwork, size as a fraction of the head's drawn size
    const headRefSize = sz(rH)*rH.s, headNewSize = sz(H)*st.head.s;
    ['mouth_inner','eye_L','eye_R','eye_closed_L','eye_closed_R','brow_L','brow_R'].forEach(k=>{
      const r = P[k], s0 = ensure(k); if(!r || !s0) return;
      s0.ax = r.ax*(H.w/rH.w); s0.ay = r.ay*(H.h/rH.h);
      s0.s = (r.s*sz(r)/headRefSize)*headNewSize/sz(A[k]);
      s0.rot = r.rot; s0.fx = r.fx; s0.fy = r.fy; s0.wx = r.wx; s0.hy = r.hy;
    });
    // 4) hand gestures nested in the forearm: position as a fraction of the forearm artwork, size as a fraction of the plain hand
    ['L','R'].forEach(S=>{
      const fk = 'arm'+S+'_lower', hk = 'hand'+S, rF = P[fk], rHa = P[hk], F = A[fk], Ha = A[hk];
      if(!rF || !rHa || !F || !Ha || !st[hk]) return;
      const handRefSize = sz(rHa)*rHa.s, handNewSize = sz(Ha)*st[hk].s;
      ['point','thumb','palm','hold'].forEach(g=>{
        const k = hk+'_'+g, r = P[k], s0 = ensure(k); if(!r || !s0) return;
        s0.ax = r.ax*(F.w/rF.w); s0.ay = r.ay*(F.h/rF.h);
        s0.s = (r.s*sz(r)/handRefSize)*handNewSize/sz(A[k]);
        s0.rot = r.rot; s0.fx = r.fx; s0.fy = r.fy; s0.wx = r.wx; s0.hy = r.hy;
      });
    });
  }catch(e){ console.warn('Ethvelle: default position sheet could not be applied', e); }
  return st;
}

/* ---------- 2) editor state + geometry ---------- */
/* EDIT STANCE: every model opens in the editor with the same T-pose shown in the joint-overlay reference
   (arms straight out to the sides, legs a little apart). The stance is an offset over each limb's own rotation
   (st.rot stays 0 = "no change"). When Done is pressed, edCompile() keeps this stance as the saved default rig pose
   so the model lines up with the visible joints in the main video editor as well. Values are degrees; L parts are on the viewer's left. */
const ED_STANCE = { armL_upper:90, armR_upper:-90, legL_upper:8, legR_upper:-8 };
// The T-pose display stance is for front-facing sheets only. In side view every limb is shown hanging straight down.
function edStanceOf(k){ return ED.view === 'side' ? 0 : (ED_STANCE[k]||0); }
const ED = { view:'front', facing:1, modelType:'humanoid', open:false, arts:{}, assign:{}, st:{}, order:[], sel:'torso', cam:{z:1,x:0,y:0}, undo:[], redo:[], bones:true, lock:false, mode:'size', kind:'new', editIdx:null, name:'Model', init:null, cv:null, g:null, dpr:1, ready:false, extraVariants:{headParts:[],expressions:[],gestures:[]}, pendingVariantType:null };
const edArt = k=>ED.arts[ED.assign[k]];
// World (figure-space) placement of every part: { x, y (pivot), rot (rad, incl. parents), sx, sy (signed scale) }.
function edWorld(rest){
  const W = {};
  const go = k=>{
    if(W[k] !== undefined) return W[k];
    const st = ED.st[k];
    if(!st) return W[k] = null;   // this model doesn't have this (optional) part -- skip it, don't crash
    const p = SHEET_PARENT[k], sx = st.s*st.wx*st.fx, sy = st.s*st.hy*st.fy;
    let x, y, rot = (st.rot + (rest ? 0 : edStanceOf(k)))*Math.PI/180;
    if(!p){ x = st.ax; y = st.ay; }
    else {
      const P = go(p); if(!P) return W[k] = null;
      const ps = ED.st[p], c = Math.cos(P.rot), s = Math.sin(P.rot), vx = (st.ax-ps.pvx)*P.sx, vy = (st.ay-ps.pvy)*P.sy;
      x = P.x + vx*c - vy*s; y = P.y + vx*s + vy*c; rot += P.rot;
    }
    return W[k] = { x, y, rot, sx, sy };
  };
  SHEET_KEYS.forEach(go);
  return W;
}
function edL2W(W, k, lx, ly){ const w = W[k], st = ED.st[k], vx = (lx-st.pvx)*w.sx, vy = (ly-st.pvy)*w.sy, c = Math.cos(w.rot), s = Math.sin(w.rot); return { x:w.x+vx*c-vy*s, y:w.y+vx*s+vy*c }; }
// v370 MIRROR FIX: Flip H / Flip V must mirror only the ARTWORK of a part, never its bones or joints.
// (a) A bone's far end is measured with the part's opening flip, so a user flip never swings the bone line / rig joint.
let edInitStCache = { src:null, st:{} };
function edInitFlip(k){
  if(edInitStCache.src !== ED.init){ edInitStCache = { src:ED.init, st:{} }; try{ edInitStCache.st = (typeof ED.init === 'string' ? JSON.parse(ED.init) : ED.init).st || {}; }catch(e){} }
  const s = edInitStCache.st[k];
  return { fx: s && s.fx < 0 ? -1 : 1, fy: s && s.fy < 0 ? -1 : 1 };
}
function edL2Wbone(W, k, lx, ly){
  const w = W[k]; if(!w || /^tail/.test(k)) return edL2W(W, k, lx, ly);
  const f = edInitFlip(k), o = {}; o[k] = { ...w, sx:Math.abs(w.sx)*f.fx, sy:Math.abs(w.sy)*f.fy };
  return edL2W(o, k, lx, ly);
}
// (b) Face artwork nested in the head (eyes, brows, mouths) is part of the picture, so it still flips with the head.
function edIsFaceNestedKey(c){
  return edIsFaceOverlayKey(c) || c==='mouth_closed' || c==='mouth_open' || c==='mouth_hold'
    || (typeof SHEET_FACE_NESTED!=='undefined' && SHEET_FACE_NESTED.includes(c))
    || (typeof SHEET_FACE_KIT_NESTED!=='undefined' && SHEET_FACE_KIT_NESTED.includes(c))
    || (typeof SHEET_BIRD_FACE_NESTED!=='undefined' && SHEET_BIRD_FACE_NESTED.includes(c));
}
// (c) Runs mutate() (which changes how part k is flipped) and then puts every child joint back on the exact
// world position it had, so the skeleton is untouched.
function edKeepJoints(k, mutate){
  const W0 = edWorld(), kids = Object.keys(ED.st).filter(c=>SHEET_PARENT[c]===k && ED.st[c] && W0[c] && !edIsFaceNestedKey(c));
  mutate();
  kids.forEach(c=>edSetPos(c, W0[c].x, W0[c].y));
}
function edW2L(W, k, px, py){ const w = W[k], st = ED.st[k], dx = px-w.x, dy = py-w.y, c = Math.cos(w.rot), s = Math.sin(w.rot); return { x:st.pvx + (dx*c+dy*s)/w.sx, y:st.pvy + (-dx*s+dy*c)/w.sy }; }
// Moves part k so its pivot lands on world point (x,y) by rewriting its attach point (children follow automatically).
function edSetPos(k, x, y){
  const st = ED.st[k], p = SHEET_PARENT[k];
  if(!p){ st.ax = x; st.ay = y; return; }
  const W = edWorld(), P = W[p], ps = ED.st[p], c = Math.cos(P.rot), s = Math.sin(P.rot), dx = x-P.x, dy = y-P.y;
  st.ax = ps.pvx + (dx*c+dy*s)/P.sx; st.ay = ps.pvy + (-dx*s+dy*c)/P.sy;
}
// LOCK MODE helper: the bone that ends at joint `k` belongs to k's parent part. Rotates + stretches that parent part so the
// joint lands on world point (tx,ty); k and everything below it are carried along by the hierarchy.
function edBoneDrive(k, tx, ty){
  if(edIsLocked(SHEET_PARENT[k]) || edIsLocked(k)) return;   // v328: never rotate/stretch a locked part
  const p = SHEET_PARENT[k], ps = ED.st[p], cs = ED.st[k], stance = edStanceOf(p), P = edWorld()[p];
  const dx = cs.ax-ps.pvx, dy = cs.ay-ps.pvy, Tx = tx-P.x, Ty = ty-P.y, sxv = dx*P.sx;
  let sy = P.sy;
  if(Math.abs(dy) > 1){
    const d2 = Tx*Tx+Ty*Ty-sxv*sxv;
    if(d2 > 0){ ps.hy = Math.max(0.2, Math.min(4, Math.sqrt(d2)/(Math.abs(dy)*ps.s))); sy = ps.s*ps.hy*ps.fy; }
  }
  const world = Math.atan2(Ty,Tx) - Math.atan2(dy*sy, sxv), inherited = P.rot - (ps.rot+stance)*Math.PI/180;
  let r = (world-inherited)*180/Math.PI - stance; ps.rot = ((r+180)%360+360)%360-180;
}
// The tail bone runs from the tail root (the pivot dot) to the far end of the tail piece. Its tip is a real,
// draggable joint: dragging it swings the tail around the root and stretches/shrinks it to follow the finger.
// v344: animals/birds have a sideways tail (tip at the left-middle edge); a beast man has an upright tail (tip at the bottom).
function edHasTail(){ return isMythicModelType(ED.modelType) || ED.modelType==='animal' || ED.modelType==='bird' || (ED.modelType==='humanoid' && !!ED.beast && !!edArt('tail')); }
function edTailTipLocal(a){ return isMythicModelType(ED.modelType) ? {x:(Number.isFinite(a.botCx)?a.botCx:a.w*.5),y:a.h*.96} : ED.modelType==='humanoid' ? { x:(Number.isFinite(a.botCx)?a.botCx:a.w/2), y:a.h*0.96 } : { x:0, y:a.h*0.5 }; }
function edTailTip(W){
  if(!edHasTail()) return null;
  if(isMythicModelType(ED.modelType)){if(!W||!W.tail_tip||!ED.st.tail_tip||!edArt('tail_tip')||!edKeyVisible('tail_tip'))return null;const tl=edTailTipLocal(edArt('tail_tip'));return edL2W(W,'tail_tip',tl.x,tl.y);}
  if(!W || !W.tail || !ED.st.tail || !edArt('tail') || !edKeyVisible('tail')) return null;
  const tl=edTailTipLocal(edArt('tail'));
  return edL2W(W,'tail',tl.x,tl.y);
}
function edBoneSegs(W){

    }else if(W.tail&&st.tail&&edArt('tail')){const tl=edTailTipLocal(edArt('tail'));add(pv('tail'),far('tail',tl.x,tl.y));}
    if(isWingedHumanoidType(ED.modelType)){

      if(W.wingR&&st.wingR&&edArt('wingR'))add(pv('wingR'),far('wingR',edArt('wingR').w,st.wingR.pvy));
    }
    ['L','R'].forEach(S=>{
      const up='arm'+S+'_upper',lo='arm'+S+'_lower',hand='hand'+S,th='leg'+S+'_upper',sh='leg'+S+'_lower',shoe='shoe'+S;
      add(chest,pv(up));
      if(W[lo]){add(pv(up),pv(lo));if(W[hand]){add(pv(lo),pv(hand));add(pv(hand),far(hand,st[hand].pvx,edArt(hand).h));}else add(pv(lo),far(lo,st[lo].pvx,edArt(lo).h));}
      else if(W[up])add(pv(up),far(up,st[up].pvx,edArt(up).h));
      if(W[sh]){add(pv(th),pv(sh));if(W[shoe]){add(pv(sh),pv(shoe));add(pv(shoe),far(shoe,st[shoe].pvx,edArt(shoe).h));}else add(pv(sh),far(sh,st[sh].pvx,edArt(sh).h));}
      else if(W[th])add(pv(th),far(th,st[th].pvx,edArt(th).h));
    });
  }
  return segs;
}
// Selection handles of part k, in SCREEN pixels: rot = bottom-right corner (scale + rotate), len = stretch length, wid = stretch width.
function edHandles(k, W){
  const a = edArt(k), st = ED.st[k], Z = ED.cam.z, S = p=>({ x:ED.cam.x+p.x*Z, y:ED.cam.y+p.y*Z }), L = (x,y)=>S(edL2W(W,k,x,y));
  // handles sit on the side / end FARTHEST from the joint (stable ratios) and are pushed a little outside the box (never stacked on each other)
  const farY = st.pvy < a.h/2 ? a.h : 0, sideX = st.pvx <= a.w/2 ? a.w : 0, ctr = L(a.w/2, a.h/2);
  const out = (p, d)=>{ const dx = p.x-ctr.x, dy = p.y-ctr.y, l = Math.hypot(dx,dy) || 1; return { x:p.x+dx/l*d, y:p.y+dy/l*d }; };
  return { corners:[L(0,0),L(a.w,0),L(a.w,a.h),L(0,a.h)], rot:out(L(sideX,farY),15), len:out(L(st.pvx,farY),19), wid:out(L(sideX,a.h/2),19) };
}
// True when key k should actually be shown/tappable right now. An alt expression texture (head_blink,
// mouth_open, handL_fist, handR_fist) is invisible unless it's the one currently selected for editing;
// its base hides right back while that's happening, so the two never draw stacked on each other.
const ED_FACE_OVERLAY_KEYS = new Set(['mouth_inner','eye_L','eye_R','eye_closed_L','eye_closed_R','brow_L','brow_R']);
function edIsFaceOverlayKey(k){return ED_FACE_OVERLAY_KEYS.has(String(k||''));}
function edKeyVisible(k){
  // Facial overlays never replace the base head. Selecting an eye, lid, brow or inner mouth
  // must leave the complete head visible under the selected feature's white bounds.
  if(k==='head' && ED.sel && edIsFaceOverlayKey(ED.sel))return true;
  if(SHEET_ALT_OF[k]) return k === ED.sel;
  const alts = SHEET_ALTS[k]; if(alts && alts.includes(ED.sel) && !(k==='head'&&edIsFaceOverlayKey(ED.sel))) return false;
  return true;
}
// Legacy editor oval-joint blend helpers removed; do not repaint over imported part artwork.
function edPick(p, W){
  const edTipHit = (radius)=>{
    const tipKey=isMythicModelType(ED.modelType)?'tail_tip':'tail';
    if(!ED.bones || edIsLocked(tipKey)) return null;
    const t = edTailTip(W); if(!t) return null;
    const q = { x:ED.cam.x+t.x*ED.cam.z, y:ED.cam.y+t.y*ED.cam.z };
    return Math.hypot(p.x-q.x, p.y-q.y) < (radius||20) ? { type:'tailtip', key:tipKey } : null;
  };
  if(ED.lock && ED.bones){
    const tipLocked = edTipHit(); if(tipLocked) return tipLocked;
    // Mouth crops are nested face artwork, not bones: keep their pixels draggable even while
    // every structural joint is locked to its connected artwork. Their pivot dots are hidden
    // in edDraw(), so exclude them from bone-point hit-testing as well.
    const jointKeys = ED.order.filter(k=>W[k] && edKeyVisible(k) && !edIsLocked(k) && !edIsFaceOverlayKey(k) && k!=='mouth_closed' && k!=='mouth_open' && k!=='mouth_hold').sort((a,b)=>(a===ED.sel?-1:0)-(b===ED.sel?-1:0));
    for(const k of jointKeys){
      const q={x:ED.cam.x+W[k].x*ED.cam.z,y:ED.cam.y+W[k].y*ED.cam.z};
      if(Math.hypot(p.x-q.x,p.y-q.y)<20) return {type:'pivot',key:k};
    }
    const fx=(p.x-ED.cam.x)/ED.cam.z, fy=(p.y-ED.cam.y)/ED.cam.z;
    for(let i=ED.order.length-1;i>=0;i--){
      const k=ED.order[i];
      if((k!=='mouth_closed' && k!=='mouth_open' && k!=='mouth_hold') || !W[k] || !edKeyVisible(k) || edIsLocked(k)) continue;
      const a=edArt(k), l=edW2L(W,k,fx,fy), lx=Math.round(l.x), ly=Math.round(l.y);
      for(const [ox,oy] of [[0,0],[2,0],[-2,0],[0,2],[0,-2]]){
        const X=lx+ox,Y=ly+oy;
        if(X>=0 && Y>=0 && X<a.w && Y<a.h && a.alpha[Y*a.w+X]) return {type:'move',key:k};
      }
    }
    // v348: locked mode must still let a plain tap select a body part. Only joint dots
    // are draggable while locked; selecting the artwork itself never starts a transform.
    for(let i=ED.order.length-1;i>=0;i--){
      const k=ED.order[i];
      if(!W[k] || !edKeyVisible(k) || edIsLocked(k)) continue;
      const a=edArt(k), l=edW2L(W,k,fx,fy), lx=Math.round(l.x), ly=Math.round(l.y);
      for(const [ox,oy] of [[0,0],[2,0],[-2,0],[0,2],[0,-2]]){
        const X=lx+ox,Y=ly+oy;
        if(X>=0 && Y>=0 && X<a.w && Y<a.h && a.alpha[Y*a.w+X]) return {type:'select',key:k};
      }
    }
    return {type:'locked',key:ED.sel};
  }
  const tipExact = edTipHit(11); if(tipExact) return tipExact;   // the width knob of a selected tail sits right next to the tip dot
  if(ED.sel){ const h = edHandles(ED.sel, W); for(const n of ['rot','len','wid']) if(Math.hypot(p.x-h[n].x, p.y-h[n].y) < 20) return { type:n, key:ED.sel }; }
  const tipFree = edTipHit(); if(tipFree) return tipFree;
  if(ED.bones){
    const keys = ED.order.filter(k=>W[k] && edKeyVisible(k) && !edIsLocked(k) && !edIsFaceOverlayKey(k)).sort((a,b)=>(a===ED.sel?-1:0)-(b===ED.sel?-1:0));
    for(const k of keys){ const q = { x:ED.cam.x+W[k].x*ED.cam.z, y:ED.cam.y+W[k].y*ED.cam.z }; if(Math.hypot(p.x-q.x, p.y-q.y) < 17) return { type:'pivot', key:k }; }
  }
  const fx = (p.x-ED.cam.x)/ED.cam.z, fy = (p.y-ED.cam.y)/ED.cam.z;
  const opaqueAt = k=>{
    const a = edArt(k), l = edW2L(W, k, fx, fy), lx = Math.round(l.x), ly = Math.round(l.y);
    for(const [ox,oy] of [[0,0],[2,0],[-2,0],[0,2],[0,-2]]){
      const X = lx+ox, Y = ly+oy; if(X >= 0 && Y >= 0 && X < a.w && Y < a.h && a.alpha[Y*a.w+X]) return true;
    }
    return false;
  };
  let hitKey = null;
  for(let i = ED.order.length-1; i >= 0; i--){
    const k = ED.order[i]; if(!W[k] || !edKeyVisible(k) || edIsLocked(k)) continue;
    if(opaqueAt(k)){ hitKey = k; break; }
  }
  // v332 FIX (small part grabbed through its own box selects the big part underneath):
  // a thin/small selected part (eyebrow, eye, ...) is mostly transparent inside its selection
  // border, so a drag that starts inside that border missed the part's pixels and fell through
  // to the large part below it (usually the Head), which then got selected. If the touch is
  // inside (or right on) the SELECTED part's drawn border, and what the pixel test found is
  // only a part layered BELOW it (or nothing) that is bigger than the selected part, keep
  // the selected part and move it. A part layered ABOVE the selected one with real pixels
  // under the finger still wins, so tapping an eyebrow while the head is selected works.
  const selK = ED.sel;
  if(selK && hitKey !== selK && W[selK] && edKeyVisible(selK) && !edIsLocked(selK) && edArt(selK) && ED.st[selK]){
    const selIdx = ED.order.indexOf(selK), hitIdx = hitKey == null ? -1 : ED.order.indexOf(hitKey);
    if(hitKey == null || hitIdx < selIdx){
      const sq = edHandles(selK, W).corners;
      const quadArea = q=>{ let s = 0; for(let i = 0; i < 4; i++){ const a = q[i], b = q[(i+1)%4]; s += a.x*b.y - b.x*a.y; } return Math.abs(s)/2; };
      const inQuad = (q, pt)=>{ let pos = false, neg = false; for(let i = 0; i < 4; i++){ const a = q[i], b = q[(i+1)%4], c = (b.x-a.x)*(pt.y-a.y) - (b.y-a.y)*(pt.x-a.x); if(c > 0) pos = true; else if(c < 0) neg = true; } return !(pos && neg); };
      const nearEdge = (q, pt, r)=>{ for(let i = 0; i < 4; i++){ const a = q[i], b = q[(i+1)%4], dx = b.x-a.x, dy = b.y-a.y, l2 = dx*dx+dy*dy || 1, t = Math.max(0, Math.min(1, ((pt.x-a.x)*dx+(pt.y-a.y)*dy)/l2)); if(Math.hypot(pt.x-(a.x+t*dx), pt.y-(a.y+t*dy)) <= r) return true; } return false; };
      if(inQuad(sq, p) || nearEdge(sq, p, 8)){
        let selWins = hitKey == null;
        if(!selWins){ try{ selWins = quadArea(sq) < quadArea(edHandles(hitKey, W).corners); }catch(_e){ selWins = false; } }
        if(selWins) return { type:'move', key:selK };
      }
    }
  }
  if(hitKey != null) return { type:'move', key:hitKey };
  return { type:'pan' };
}

/* ---------- editor drawing ---------- */
function edDraw(){
  if(!ED.open || !ED.g) return;
  const g = ED.g, cv = ED.cv, dpr = ED.dpr, Z = ED.cam.z, W = edWorld();
  g.setTransform(1,0,0,1,0,0); g.clearRect(0,0,cv.width,cv.height);
  g.setTransform(dpr*Z,0,0,dpr*Z,dpr*ED.cam.x,dpr*ED.cam.y); g.imageSmoothingQuality = 'high';
  ED.order.forEach(k=>{
    if(!W[k] || !edKeyVisible(k)) return;
    const w = W[k], st = ED.st[k]; g.save(); g.translate(w.x,w.y); g.rotate(w.rot); g.scale(w.sx,w.sy); g.drawImage(edArt(k).canvas, -st.pvx, -st.pvy);
    const preview=ED.previewVariant;
    if(preview&&preview.targetKey===k&&preview.src){
      const cache=window.__ethvelleModelLabVariantPreviewCache||(window.__ethvelleModelLabVariantPreviewCache=new Map());
      let previewImage=cache.get(preview.src);
      if(!previewImage){previewImage=new Image();previewImage.onload=()=>{if(ED.previewVariant&&ED.previewVariant.src===preview.src)edDraw();};previewImage.onerror=()=>{};cache.set(preview.src,previewImage);previewImage.src=preview.src;}
      if(previewImage.complete&&previewImage.naturalWidth){const art=edArt(k);g.drawImage(previewImage,-st.pvx,-st.pvy,art.w,art.h);}
    }
    g.restore();
  });
  g.setTransform(dpr,0,0,dpr,0,0);
  const S = p=>({ x:ED.cam.x+p.x*Z, y:ED.cam.y+p.y*Z });
  // Joint-bridge painting intentionally removed: it generated oval/crescent artifacts over source art.
  if(ED.bones){
    const segs = edBoneSegs(W); g.lineCap = 'round';
    [['rgba(255,255,255,.95)',ED.lock?7:6],[ED.lock?'#12B886':'#2F5CE8',ED.lock?4:3]].forEach(([c,lw])=>{ g.strokeStyle = c; g.lineWidth = lw; g.beginPath(); segs.forEach(([a,b])=>{ const A = S(a), B = S(b); g.moveTo(A.x,A.y); g.lineTo(B.x,B.y); }); g.stroke(); });
  }
  if(ED.sel){
    const h = edHandles(ED.sel, W), c = h.corners;
    g.beginPath(); g.moveTo(c[0].x,c[0].y); for(let i=1;i<4;i++) g.lineTo(c[i].x,c[i].y); g.closePath();
    g.lineJoin = 'round'; g.strokeStyle = 'rgba(0,0,0,.5)'; g.lineWidth = 4; g.stroke(); g.strokeStyle = '#fff'; g.lineWidth = 2; g.stroke();
    // Part selection stays visible in locked mode; manipulation knobs remain unlocked-only.
    if(!ED.lock){
      const rot = W[ED.sel].rot;
      const knob = (p, ang, glyph)=>{
        g.save(); g.translate(p.x,p.y); g.rotate(ang);
        g.beginPath(); g.arc(0,0,11,0,Math.PI*2); g.fillStyle = '#fff'; g.shadowColor = 'rgba(0,0,0,.45)'; g.shadowBlur = 6; g.fill(); g.shadowBlur = 0;
        g.strokeStyle = '#2F5CE8'; g.fillStyle = '#2F5CE8'; g.lineWidth = 2; g.lineCap = 'round'; glyph(); g.restore();
      };
      const arrows = ()=>{ g.beginPath(); g.moveTo(0,-5); g.lineTo(0,5); g.stroke(); [[-1,-5],[1,5]].forEach(([d,y])=>{ g.beginPath(); g.moveTo(-3,y-d*3); g.lineTo(0,y); g.lineTo(3,y-d*3); g.stroke(); }); };
      knob(h.wid, rot + Math.PI/2, arrows);
      knob(h.len, rot, arrows);
      knob(h.rot, 0, ()=>{ g.beginPath(); g.arc(0,0,5,Math.PI*0.15,Math.PI*1.55); g.stroke(); g.beginPath(); g.moveTo(-1,-6.5); g.lineTo(2.5,-4.2); g.lineTo(-1.5,-2.4); g.closePath(); g.fill(); });
    }
  }
  if(ED.bones) ED.order.forEach(k=>{
    if(!W[k] || !edKeyVisible(k)) return;
    // Facial overlays are positioned inside the head artwork, not skeleton joints. Their
    // pivot dots intercepted taps intended for an eye/brow and made the head jump/disappear.
    if(edIsFaceOverlayKey(k) || k==='mouth_closed' || k==='mouth_open' || k==='mouth_hold') return;
    const p = S(W[k]), on = k === ED.sel;
    g.beginPath(); g.arc(p.x,p.y,on?9:7,0,Math.PI*2); g.fillStyle = '#fff'; g.fill();
    g.beginPath(); g.arc(p.x,p.y,on?6.5:4.5,0,Math.PI*2); g.fillStyle = ED.lock ? (on ? '#12B886' : '#3fd6a6') : (on ? '#2F5CE8' : '#5b7cf0'); g.fill();
  });
  if(ED.bones){
    const tailTip = edTailTip(W);
    if(tailTip){
      const p = S(tailTip), on = ED.sel === (isMythicModelType(ED.modelType)?'tail_tip':'tail');
      g.beginPath(); g.arc(p.x,p.y,on?9:7,0,Math.PI*2); g.fillStyle = '#fff'; g.fill();
      g.beginPath(); g.arc(p.x,p.y,on?6.5:4.5,0,Math.PI*2); g.fillStyle = ED.lock ? (on ? '#12B886' : '#3fd6a6') : (on ? '#2F5CE8' : '#5b7cf0'); g.fill();
    }
  }
}

function edFit(){
  const W = edWorld(); let x0 = 1e9, y0 = 1e9, x1 = -1e9, y1 = -1e9;
  ED.order.forEach(k=>{ const a = edArt(k); if(!a) return; [[0,0],[a.w,0],[a.w,a.h],[0,a.h]].forEach(([lx,ly])=>{ const p = edL2W(W,k,lx,ly); x0 = Math.min(x0,p.x); x1 = Math.max(x1,p.x); y0 = Math.min(y0,p.y); y1 = Math.max(y1,p.y); }); });
  const bw = Math.max(1,x1-x0), bh = Math.max(1,y1-y0), z = Math.min((ED.cw-36)/bw, (ED.ch-36)/bh);
  ED.cam.z = Math.max(0.05, z); ED.cam.x = ED.cw/2 - (x0+x1)/2*ED.cam.z; ED.cam.y = ED.ch/2 - (y0+y1)/2*ED.cam.z;
  edSyncZoom(); edDraw();
}

// Facial reactions, mouths and each side's hand gestures are interchangeable variants. Paired
// arms/legs share size and rotation edits without destroying their different anatomical positions.
function edSimilarKeys(k){
  // v333: every part family now has a rule, for humanoid, animal and bird models alike.
  // Head variants (incl. the open-eye head), mouths (incl. mouth_hold), hands (incl. holding hands).
  if(ED_SIM.head.test(k)) return SHEET_KEYS.filter(x=>ED_SIM.head.test(x));
  if(ED_SIM.mouth.test(k)) return ['mouth_closed','mouth_open','mouth_hold'];
  // v318: open eyes (eye_L/eye_R) and closed eyes (eye_closed_L/eye_closed_R) are left/right pairs.
  const em=k.match(/^(eye|eye_closed)_([LR])$/);
  if(em) return [em[1]+'_L',em[1]+'_R'];
  // v333: eyebrows are a left/right pair (humanoid + animal face kits).
  if(/^brow_[LR]$/.test(k)) return ['brow_L','brow_R'];
  // Hand transforms are a paired character-level edit, not side-specific: include both base hands
  // and every matching gesture on both sides. Per-part defaults still preserve each gesture's own
  // wrist pivot and the left/right model's original mirror direction.
  if(ED_SIM.hand.test(k)) return SHEET_KEYS.filter(x=>ED_SIM.hand.test(x));
  const lm=k.match(/^(arm|leg)([LR])_(upper|lower)$/);
  if(lm){ const other=lm[2]==='L'?'R':'L'; return [k,lm[1]+other+'_'+lm[3]]; }
  // v333: shoes are a left/right pair.
  if(/^shoe[LR]$/.test(k)) return ['shoeL','shoeR'];
  // v333: bird flight wings are a left/right pair; bird eye and beak states are interchangeable variants.
  if(/^wing[LR]_fly$/.test(k)) return ['wingL_fly','wingR_fly'];
  if(/^bird_eye(?:_closed)?$/.test(k)) return ['bird_eye','bird_eye_closed'];
  if(/^beak_(?:closed|open)$/.test(k)) return ['beak_closed','beak_open'];
  return [k];
}
// v333: shared matchers so "which parts are similar" and "how to transfer the edit" never disagree.
const ED_SIM={
  // v371: head variants include the emotion heads (sad / angry / shocked / laughing) - they were missing, so "Apply similar" found no matches.
  head:/^head(?:_(?:blink|open|sad|angry|shock|laugh))?$/,
  mouth:/^mouth_(?:closed|open|hold)$/,
  hand:/^hand[LR](?:_(?:fist|point|thumb|palm|hold))?$/,
  birdVariant:/^(?:bird_eye(?:_closed)?|beak_(?:closed|open))$/
};
function edApplyToSimilarParts(){
  const k=ED.sel;
  if(!k || !ED.st[k]){ showToast('Select a part first',{duration:1300}); return; }
  let init;
  try{ init = typeof ED.init === 'string' ? JSON.parse(ED.init).st : ED.init.st; }
  catch(e){ showToast('Could not read the starting layout',{duration:1500}); return; }
  const headGroup=ED_SIM.head.test(k);
  const keys=edSimilarKeys(k);
  const group=keys.filter(x=>x!==k && !edIsLocked(x) && ED.st[x] && init[x] && ED.arts[ED.assign[x]]);
  if(!group.length){ showToast(keys.length<=1 ? 'This part has no matching pair to copy to' : 'No other matching parts are present in this sheet',{duration:1500}); return; }
  const src=ED.st[k], src0=init[k], srcArt=edArt(k);
  if(!src0 || !srcArt){ showToast('Could not read the selected part',{duration:1500}); return; }
  const ratio=(v,b)=>Math.abs(b)>1e-6?v/b:1;
  const variants=headGroup || ED_SIM.mouth.test(k) || ED_SIM.hand.test(k) || ED_SIM.birdVariant.test(k);
  let applied=0;
  edPush();
  group.forEach(q=>{
    const dst=ED.st[q], d0=init[q], art=edArt(q);
    if(!dst || !d0 || !art) return;
    const prevDst={...dst};
    // Transfer edit DELTAS from the selected piece, relative to its opening state. This preserves
    // the destination's own detected proportions while copying every intentional transform edit.
    dst.s=Math.max(.3,Math.min(3,d0.s*ratio(src.s,src0.s)));
    dst.wx=Math.max(.2,Math.min(4,d0.wx*ratio(src.wx,src0.wx)));
    dst.hy=Math.max(.2,Math.min(4,d0.hy*ratio(src.hy,src0.hy)));
    dst.rot=d0.rot+(src.rot-src0.rot);
    dst.fx=src.fx===src0.fx ? d0.fx : -d0.fx;
    dst.fy=src.fy===src0.fy ? d0.fy : -d0.fy;
    if(variants){
      const srcParent=SHEET_PARENT[k], dstParent=SHEET_PARENT[q];
      const pa=srcParent&&edArt(srcParent), pb=dstParent&&edArt(dstParent);
      const sx=pa&&pb ? pb.w/Math.max(1,pa.w) : 1;
      const sy=pa&&pb ? pb.h/Math.max(1,pa.h) : 1;
      const sp0=srcParent&&init[srcParent], dp0=dstParent&&init[dstParent];
      const sign=v=>v ? Math.sign(v.s*v.wx*v.fx)||1 : 1;
      const flipLocalX=ED.view!=='side' && sign(sp0)!==sign(dp0) ? -1 : 1;
      dst.ax=d0.ax+(src.ax-src0.ax)*sx*flipLocalX;
      dst.ay=d0.ay+(src.ay-src0.ay)*sy;
    } else {
      const mirror=ED.view!=='side';
      dst.ax=d0.ax+(mirror?-(src.ax-src0.ax):(src.ax-src0.ax));
      dst.ay=d0.ay+(src.ay-src0.ay);
    }
    // Keep each part's own pivot but copy only the selected part's pivot CHANGE (relative to its opening state).
    // FIX: this used to be measured from the art's default pivot (dpx/dpy) while ax/ay were measured from the
    // opening layout, so any model with an already-moved pivot made the matching part jump away / seem to vanish.
    dst.pvx=d0.pvx+(src.pvx-src0.pvx)*(art.w/Math.max(1,srcArt.w));
    dst.pvy=d0.pvy+(src.pvy-src0.pvy)*(art.h/Math.max(1,srcArt.h));
    // Safety net: never let a bad number hide a part. If anything went non-finite, put this part back as it was.
    if(!['s','wx','hy','rot','fx','fy','ax','ay','pvx','pvy'].every(f=>Number.isFinite(dst[f]))){ Object.assign(dst, prevDst); return; }
    applied++;
  });
  if(!applied){ ED.undo.pop(); showToast('No matching parts could be updated',{duration:1500}); return; }
  edAfterChange(true);
  showToast(headGroup ? 'Applied head edits to '+applied+' head variants' : 'Applied edits to '+applied+' matching parts',{duration:1800});
}

// NEAR/FAR limb pairs in a side-view sheet: two pieces with the same silhouette (the far one a touch darker,
// see SHEET_PROMPT_SIDE's "NO MIRRORING IN SIDE VIEW" rule). When the model turns around, the near and far
// limbs trade places, so the far leg becomes the near one and vice versa.
const SHEET_FLIP_PAIRS = [['armL_upper','armR_upper'], ['armL_lower','armR_lower'], ['handL','handR'], ['legL_upper','legR_upper'], ['legL_lower','legR_lower'], ['shoeL','shoeR'], ['wingL','wingR'],

// Turns the WHOLE side-view model around so it faces the other way. Every piece turns, limbs and hands included:
// its art is reflected (fx), its joint is mirrored across the body's centre line and its angle is negated, so a foot
// or hand that pointed forward now points the new way. The near and far limbs first trade places together with their
// own state (art, size, stretch, rotation and joint edits), so nothing the person tuned is lost. Parents go first so
// children are placed against their new parent transforms. Side view only.
// ED.facing flips as well: the animation presets read it, so arms and legs swing toward the way the model now faces.
function edFlipModel(){
  if(ED.view !== 'side') return;
  edPush();
  SHEET_FLIP_PAIRS.forEach(([l,r])=>{
    if(!ED.arts[ED.assign[l]] || !ED.arts[ED.assign[r]]) return;   // basic sheets may not have the fist pair
    const t = ED.assign[l]; ED.assign[l] = ED.assign[r]; ED.assign[r] = t;
    const s = ED.st[l]; ED.st[l] = ED.st[r]; ED.st[r] = s;
  });
  const W0 = edWorld(true), cx = W0.torso.x, done = {};
  const go = k => {
    if(done[k] || !ED.st[k] || !W0[k]) return; done[k] = true;
    const p = SHEET_PARENT[k]; if(p) go(p);
    const st = ED.st[k], old = W0[k];
    st.fx *= -1;                                   // reflect this piece's art (limbs and hands included)
    if(p){
      const P = edWorld(true)[p];
      st.rot = (-old.rot - P.rot)*180/Math.PI;
      edSetPos(k, 2*cx - old.x, old.y);
    } else st.rot = -old.rot*180/Math.PI;
  };
  SHEET_KEYS.forEach(go);
  ED.facing = -(ED.facing || 1);
  edSyncVals(); edDraw();
  const flipBtn=document.getElementById('edFlipModelBtn');
  if(flipBtn){ flipBtn.setAttribute('aria-pressed', ED.facing < 0 ? 'true' : 'false'); flipBtn.title='Turn the entire side-profile rig 180 degrees. Near/far limbs swap, and every piece stays editable'; }
  showToast(ED.facing < 0 ? 'Model turned to face left' : 'Model turned to face right', { duration:1500 });
}

function edPosSheetNum(v){ return Number.isFinite(v) ? Math.round(v*100000)/100000 : v; }
function edBuildPositionSheet(){
  let init={}; try{ init = typeof ED.init === 'string' ? JSON.parse(ED.init).st : (ED.init && ED.init.st) || {}; }catch(e){}
  const clean=o=>{ const r={}; Object.keys(o||{}).forEach(f=>{ const v=o[f]; if(typeof v==='number') r[f]=edPosSheetNum(v); else if(typeof v==='boolean'||typeof v==='string') r[f]=v; }); return r; };
  const parts={};
  Object.keys(ED.st||{}).forEach(k=>{
    const a=ED.arts && ED.arts[(ED.assign&&ED.assign[k])||k];
    parts[k]={ label:(typeof sheetLabel==='function'?sheetLabel(k):k), art:a?{ w:a.w, h:a.h, dpx:edPosSheetNum(a.dpx), dpy:edPosSheetNum(a.dpy) }:null, position:clean(ED.st[k]), startLayout:clean(init[k]), locked:!!(ED.lockedParts&&ED.lockedParts[k]) };

}

// Only joint attachment coordinates and bone rotations are restored. Artwork scale, width, length,
// flips and the rest of the part edits remain untouched. The "default" is the pose/layout this
// specific model had when this editing session opened.
// v379: parts whose edited position/rotation must survive Done on a plain humanoid (head + both legs: thigh, shin, shoe).
function edKeepOnDone(k){ return k==='head' || /^(leg[LR]_(upper|lower)|shoe[LR])$/.test(String(k||'')); }
function edSkeletonAtDefault(keepHead){
  if(!ED.init || !ED.st) return false;
  let base; try{ base=JSON.parse(ED.init).st||{}; }catch(e){ return false; }
  for(const k of Object.keys(ED.st)){
    if(k==='tail' && ED.modelType==='humanoid') continue;   // v344: the beast-man tail keeps the rest pose chosen in the editor
    if(keepHead && edKeepOnDone(k)) continue;   // v378/v379: head and leg positions edited by the user are saved, never reset on Done
    // Mouth crops are independent face overlays nested inside the head, not skeleton joints.
    // Their ax/ay/rot are part of the model's saved facial layout and must not block the
    // skeleton-default check after a user deliberately repositions them.
    if(k==='mouth_closed'||k==='mouth_open'||k==='mouth_hold'||(typeof SHEET_FACE_NESTED!=='undefined'&&SHEET_FACE_NESTED.includes(k))||(typeof SHEET_FACE_KIT_NESTED!=='undefined'&&SHEET_FACE_KIT_NESTED.includes(k))||(typeof SHEET_HUMANOID_FACE_KIT_KEYS!=='undefined'&&SHEET_HUMANOID_FACE_KIT_KEYS.includes(k))||(typeof SHEET_BIRD_FACE_NESTED!=='undefined'&&SHEET_BIRD_FACE_NESTED.includes(k))||(typeof SHEET_ALT_OF!=='undefined'&&SHEET_ALT_OF[k])) continue;
    const cur=ED.st[k], ref=base[k]; if(!cur || !ref) continue;
    for(const prop of ['ax','ay','rot']){
      if(Number.isFinite(ref[prop]) && (!Number.isFinite(cur[prop]) || Math.abs(cur[prop]-ref[prop])>1e-4)) return false;
    }
  }
  return true;
}
function edResetSkeletonToDefault(silent, keepHead){
  if(!ED.open || !ED.init || !ED.st) return false;
  let base; try{ base=JSON.parse(ED.init).st||{}; }catch(e){ return false; }
  const keys=Object.keys(ED.st).filter(k=>{
    if(k==='tail' && ED.modelType==='humanoid') return false;   // v344: never reset the beast-man tail
    if(keepHead && edKeepOnDone(k)) return false;   // v378/v379: keep the edited head and leg positions (Done)
    // Reset only structural joint attachments. Keep mouth overlay transforms as edited by
    // the user; they are nested in the head and are not part of the skeleton pose.
    if(k==='mouth_closed'||k==='mouth_open'||k==='mouth_hold'||(typeof SHEET_FACE_NESTED!=='undefined'&&SHEET_FACE_NESTED.includes(k))||(typeof SHEET_FACE_KIT_NESTED!=='undefined'&&SHEET_FACE_KIT_NESTED.includes(k))||(typeof SHEET_HUMANOID_FACE_KIT_KEYS!=='undefined'&&SHEET_HUMANOID_FACE_KIT_KEYS.includes(k))||(typeof SHEET_BIRD_FACE_NESTED!=='undefined'&&SHEET_BIRD_FACE_NESTED.includes(k))||(typeof SHEET_ALT_OF!=='undefined'&&SHEET_ALT_OF[k])) return false;
    const cur=ED.st[k], ref=base[k];
    return !!(cur && ref && ['ax','ay','rot'].some(prop=>Number.isFinite(ref[prop]) && (!Number.isFinite(cur[prop]) || Math.abs(cur[prop]-ref[prop])>1e-4)));
  });
  if(keys.length){
    edPush();
    keys.forEach(k=>['ax','ay','rot'].forEach(prop=>{
      const cur=ED.st[k], ref=base[k];
      if(cur && ref && Number.isFinite(ref[prop])) cur[prop]=ref[prop];
    }));
    if(!ED.st[ED.sel]) ED.sel=ED.st.torso?'torso':Object.keys(ED.st)[0]||null;
    edAfterChange(true);
  }
  const ok=edSkeletonAtDefault(keepHead);
  if(!silent) showToast(ok ? (keys.length?'Skeleton reset to its default position':'Skeleton is already in its default position') : 'Could not reset this skeleton', {duration:1600});
  return ok;
}

// Bone <-> body-part lock. OFF: a joint dot moves on its own (the art stays where it is, so you can line the bones up with the art).
// ON: the body part is locked to its bone, so dragging a joint moves the part with it; an elbow / knee dot swings and stretches
