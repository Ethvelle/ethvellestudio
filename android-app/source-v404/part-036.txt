<script id="ethvelle-v20-selection-toolbar-guards">
(function(){
  'use strict';
  const $=id=>document.getElementById(id);
  let rafPending=false;
  function currentId(){
    try{return String(typeof selectedClip!=='undefined'&&selectedClip!=null?selectedClip:(window.selectedClip??''));}
    catch(_e){return String(window.selectedClip??'');}
  }
  function currentOverlay(id){
    try{
      const list=typeof overlays!=='undefined'?overlays:(window.overlays||[]);
      return (list||[]).find(o=>o&&String(o.id)===String(id))||null;
    }catch(_e){return null;}
  }
  function textIsReallySelected(){
    const id=currentId();
    const selectedOverlayEl=document.querySelector('#timelineInner .overlay-clip.selected[data-overlay-id]');
    if(selectedOverlayEl){
      return selectedOverlayEl.dataset.group==='text' && String(selectedOverlayEl.dataset.overlayId)===id;
    }
    // If another kind of timeline clip is visibly selected, never trust an older selectedClip id.
    if(document.querySelector('#timelineInner .anim-clip.selected, #timelineInner .track-clip.selected'))return false;
    const overlay=currentOverlay(id);
    return !!overlay&&overlay.type==='text';
  }
  function syncPencil(){
    rafPending=false;
    const button=$('clipTextEditTool');
    const isText=textIsReallySelected();
    document.body.classList.toggle('ev-v20-text-layer-selected',isText);
    if(button){
      button.style.setProperty('display',isText?'flex':'none','important');
      button.setAttribute('aria-hidden',isText?'false':'true');
      button.tabIndex=isText?0:-1;
    }
  }
  function scheduleSync(){if(rafPending)return;rafPending=true;requestAnimationFrame(()=>requestAnimationFrame(syncPencil));}
  // The original editor has multiple toolbar wrappers. Wrap the final versions so the pencil
  // rule is reapplied after each legacy toolbar update rather than racing it.
  ['syncBottomToolsMode','selectClipForChrome','deselectAllClips'].forEach(name=>{
    const old=window[name];
    if(typeof old!=='function'||old.__evV20PencilGuard)return;
    const wrapped=function(){const result=old.apply(this,arguments);scheduleSync();return result;};
    wrapped.__evV20PencilGuard=true;window[name]=wrapped;
  });
  ['click','pointerup','touchend'].forEach(type=>document.addEventListener(type,e=>{
    if(e.target&&e.target.closest&&e.target.closest('#timelineInner,#overlayRows,#overlayMediaLayer,#bottomToolsClip'))scheduleSync();
  },true));
  const timeline=$('timelineInner');
  if(timeline&&typeof MutationObserver!=='undefined'){
    const observer=new MutationObserver(records=>{
      if(records.some(r=>r.type==='childList'||(r.type==='attributes'&&r.attributeName==='class')))scheduleSync();
    });
    observer.observe(timeline,{subtree:true,childList:true,attributes:true,attributeFilter:['class']});
  }
  // Keep the colored progress rail synchronized without drawing an outline around the rail.
  function updateRangeProgress(input){
    if(!input||input.type!=='range')return;
    const min=Number(input.min||0),max=Number(input.max||100),value=Number(input.value||0);
    const percent=max>min?Math.max(0,Math.min(100,(value-min)/(max-min)*100)):0;
    input.style.setProperty('--ev-range-progress',percent+'%');
  }
  function initRanges(root=document){root.querySelectorAll('#ethvellePicsartTextDock .evpa-range input[type="range"]').forEach(updateRangeProgress);}
  document.addEventListener('input',e=>{if(e.target&&e.target.matches&&e.target.matches('#ethvellePicsartTextDock .evpa-range input[type="range"]'))updateRangeProgress(e.target);},true);
  const dock=$('ethvellePicsartTextDock');
  if(dock&&typeof MutationObserver!=='undefined')new MutationObserver(()=>initRanges(dock)).observe(dock,{subtree:true,childList:true});
  if(document.readyState==='loading')document.addEventListener('DOMContentLoaded',()=>{initRanges();scheduleSync();},{once:true});
  else {initRanges();scheduleSync();}
})();

<script id="ethvelle-text-polish-v21-js">
(function(){
  'use strict';
  const $=id=>document.getElementById(id);
  const rangeSelector='#ethvellePicsartTextDock input[type="range"],#ethvelleTextComposer input[type="range"],#ethvelleTextStylePanel input[type="range"],#ethvelleColorPicker input[type="range"],#evpaCustomEffectEditor input[type="range"],#textPanelBody .text-range-line input[type="range"],#evltEnhanceAmount,#chromaTolerance,#chromaSoftness,#ethvelleChromaColorPicker #evckLight';
  function updateProgress(input){
    if(!input||input.type!=='range')return;
    const min=Number(input.min===''?0:(input.min||0)),max=Number(input.max===''?100:(input.max||100)),value=Number(input.value||0);
    const pct=max>min?Math.max(0,Math.min(100,(value-min)/(max-min)*100)):0;
    const nextProgress=pct+'%';if(input.style.getPropertyValue('--ev-range-progress')!==nextProgress)input.style.setProperty('--ev-range-progress',nextProgress);
  }
  function initProgress(root=document){try{root.querySelectorAll(rangeSelector).forEach(updateProgress);}catch(_){}}
  document.addEventListener('input',e=>{if(e.target&&e.target.matches&&e.target.matches(rangeSelector))updateProgress(e.target);},true);
  document.addEventListener('change',e=>{if(e.target&&e.target.matches&&e.target.matches(rangeSelector))updateProgress(e.target);},true);

  function currentTarget(){try{return typeof window.ethvellePicsartGetTarget==='function'?window.ethvellePicsartGetTarget():null;}catch(_){return null;}}
  function currentColorKey(){try{return typeof window.ethvellePicsartGetColorTarget==='function'?window.ethvellePicsartGetColorTarget():'textColor';}catch(_){return 'textColor';}}
  function norm(v){return String(v||'').trim().toUpperCase();}
  function syncColorMarkers(){
    const t=currentTarget();if(!t)return;
    const key=currentColorKey();const color=norm(t[key]||t.textColor||'#FFFFFF');
    const palette=$('evpaPalette');if(palette){/* v222-fix: one owner for palette markers.     const composer=$('evtcColorStrip');if(composer)composer.querySelectorAll('.evtc-color-swatch[data-color]').forEach(b=>b.classList.toggle('active',norm(b.dataset.color)===norm(t.textColor||'#FFFFFF')));
  }
  function syncEffectMarkers(){
    const controls=$('evpaControls'),tabs=$('evpaTabs'),t=currentTarget();if(!controls||!tabs||!t)return;
    const active=tabs.querySelector('.evpa-tab.active')?.dataset.tab;
    if(active!=='gradient'&&active!=='texture')return;
    const current=String(active==='gradient'?(t.textGradient||''):(t.textTexture||''));
    controls.querySelectorAll('.evpa-gradient-chip[data-evpa-effect][data-evpa-fill]').forEach(b=>{
      let fill='';try{fill=decodeURIComponent(b.dataset.evpaFill||'');}catch(_){fill=b.dataset.evpaFill||'';}
      b.classList.toggle('selected',b.dataset.evpaEffect===active&&fill===current);
    });
    // Saved image-texture chips can be created by the separate image-texture module.
    if(active==='texture'){
      let saved=[];try{saved=JSON.parse(localStorage.getItem('ethvelleCustomTextTexturesV1')||'[]');}catch(_){saved=[];}

        const item=saved[Number(b.dataset.imageTextureIndex)];b.classList.toggle('selected',!!item&&String(item.css||'')===current);
      });
    }
  }
  function syncAll(){initProgress();syncColorMarkers();syncEffectMarkers();}

  const observedRoots=new WeakSet();
  function observeRoot(root,callback){
    if(!root||observedRoots.has(root)||typeof MutationObserver==='undefined')return;
    observedRoots.add(root);
    new MutationObserver(callback).observe(root,{childList:true,subtree:true,attributes:true,attributeFilter:['class']});
  }
  function bindObservers(){

      const wrapped=function(){const result=originalSetColor.apply(this,arguments);syncColorMarkers();return result;};wrapped.__evV21ColorSync=true;window.ethvellePicsartSetColor=wrapped;
    }
    const originalComposerColor=window.setEthvelleComposerColor;

  }

    if(textPanel&&!textPanel.dataset.evV21Observed){
      textPanel.dataset.evV21Observed='1';observeRoot(textPanel,()=>initProgress(textPanel));initProgress(textPanel);
    }
  });
  if(document.body)bodyObserver.observe(document.body,{childList:true,subtree:true});
