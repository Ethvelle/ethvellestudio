    ctx.putImageData(a,0,0);
  }catch(err){ console.warn('Ethvelle: face sharpen skipped',err); }
}
// Name kept for existing callers; it now builds the face-only preview.
function generateModelGridFullBodyPreview(asset){
  const faceSrc = modelFaceSourceFor(asset);
  if(!faceSrc) return Promise.resolve('');
  const faceSig = modelFaceSig(asset);
  if(modelFacePreviewCache.has(faceSrc) && (modelFacePreviewSigs.get(faceSrc)||'') === faceSig) return Promise.resolve(modelFacePreviewCache.get(faceSrc));
  if(modelFacePreviewPending.has(faceSrc)) return modelFacePreviewPending.get(faceSrc);
  const task = (modelFaceQueue = modelFaceQueue.then(()=>new Promise(r=>setTimeout(r,0))).then(()=>new Promise(resolve=>{
    const img = new Image();
    const finish = (img)=>{
      try{
        const iw = img.naturalWidth || img.width, ih = img.naturalHeight || img.height;
        if(!iw || !ih){ resolve(''); return; }
        // Crop to the visible (non-transparent) pixels so the face fills the card.
        let sx=0, sy=0, sw=iw, sh=ih;
        if(iw*ih <= 9000000){
          const probe = document.createElement('canvas'); probe.width=iw; probe.height=ih;
          const pc = probe.getContext('2d',{willReadFrequently:true});
          pc.drawImage(img,0,0);
          const px = pc.getImageData(0,0,iw,ih).data;
          let minX=iw, minY=ih, maxX=-1, maxY=-1;
          for(let y=0;y<ih;y++){
            const row=y*iw*4;
            for(let x=0;x<iw;x++){
              if(px[row+x*4+3]>12){ if(x<minX)minX=x; if(x>maxX)maxX=x; if(y<minY)minY=y; if(y>maxY)maxY=y; }
            }
          }
          if(maxX>=minX && maxY>=minY){
            const pad = Math.max(2,Math.round(Math.max(maxX-minX+1,maxY-minY+1)*.03));
            minX=Math.max(0,minX-pad); minY=Math.max(0,minY-pad);
            maxX=Math.min(iw-1,maxX+pad); maxY=Math.min(ih-1,maxY+pad);
            sx=minX; sy=minY; sw=maxX-minX+1; sh=maxY-minY+1;
          }
        }
        // Square 2K output. The face is fitted to ~96% of the tile and ENLARGED when the artwork is smaller,
        // so every card gets a full-resolution tile instead of a small bitmap the browser has to stretch.
        const SIZE = MODEL_FACE_PREVIEW_PX, longest = Math.max(sw,sh);
        const scale = (SIZE*.96)/longest;
        const tw = Math.max(1,Math.round(sw*scale)), th = Math.max(1,Math.round(sh*scale));
        const crop = document.createElement('canvas'); crop.width = sw; crop.height = sh;
        crop.getContext('2d').drawImage(img,sx,sy,sw,sh,0,0,sw,sh);
        const big = modelFaceResizeSteps(crop,tw,th);
        const out = document.createElement('canvas'); out.width = SIZE; out.height = SIZE;
        const oc = out.getContext('2d'); oc.imageSmoothingEnabled = true; oc.imageSmoothingQuality = 'high';
        oc.drawImage(big,Math.round((SIZE-tw)/2),Math.round((SIZE-th)/2));
        if(scale > 1.25) modelFaceUnsharp(out, Math.min(4,Math.max(1.2,scale*.4)), Math.min(1.1,.45+scale*.08));
        else modelFaceUnsharp(out, 1, .35);
        if(out.toBlob) out.toBlob(blob=>resolve(blob ? URL.createObjectURL(blob) : out.toDataURL('image/png')),'image/png');
        else resolve(out.toDataURL('image/png'));
      }catch(err){
        console.warn('Ethvelle: could not build face model preview',err);
        resolve('');
      }
    };
    img.onload = ()=>{ modelFaceComposite(asset, img).then(base=>finish(base || img), ()=>finish(img)); };
    img.onerror = ()=>resolve('');
    img.src = faceSrc;
  }))).then(url=>{
    modelFacePreviewPending.delete(faceSrc);
    if(url){ modelFacePreviewCache.set(faceSrc,url); modelFacePreviewSigs.set(faceSrc,faceSig); modelFacePreviewFailed.delete(faceSrc); } else modelFacePreviewFailed.add(faceSrc);
    return url;
  });
  modelFacePreviewPending.set(faceSrc,task);
  return task;
}

// v389: admin models arrive as a light index (one tiny head-only thumbnail); their artwork lives in a separate package.
// So that they get the same sharp face tile as the user's own models (head + open eyes + eyebrows), the package is
// loaded quietly (one at a time, cached in IndexedDB by evFetchModelSheet) and the tile is rendered from it.
let evFaceFetchChain = Promise.resolve();
function evFaceTileFor(asset){
  if(!asset) return Promise.resolve('');
  if(asset.sheet && asset.sheet.parts) return modelFaceSourceFor(asset) ? generateModelGridFullBodyPreview(asset) : Promise.resolve('');
  if(asset.origin !== 'admin' || !asset.hasPackage || typeof evFetchModelSheet !== 'function') return Promise.resolve('');
  const run = evFaceFetchChain.then(()=>evFetchModelSheet(asset)).then(sheet=>{
    asset.sheet = sheet;
    try{ const live = Array.isArray(publishedModelAssets) ? publishedModelAssets.find(x=>String(x.id)===String(asset.id)) : null; if(live && live !== asset) live.sheet = sheet; }catch(_e){}
    return modelFaceSourceFor(asset) ? generateModelGridFullBodyPreview(asset) : '';
  }).catch(()=>'');
  evFaceFetchChain = run.then(()=>{}, ()=>{});
  return run;
}

// Tapping a saved model from the Projects list (as opposed to from inside an already-open
// project's "Add character" sheet) has nothing to add the character TO yet, so it starts a
// brand new project first, opens the editor, then drops the model in as the starting
// character  -  landing the person straight on the canvas with their model already there.
async function useModelAssetFromProjects(id){
  const asset = getAvailableModelAssets().find(a=>String(a.id)===String(id));
  if(!asset) return;
  if(evModelIsLocked(asset)){ evModelUpgrade(asset); return; }
  if(!(asset.sheet&&asset.sheet.parts) && asset.origin==='admin' && asset.hasPackage){
    if(!(await evEnsureModelSheet(asset))) return;
  }
  const plan = getMyPlanData();
  if(!plan.unlimitedProjects && plan.maxProjects != null && projects.length >= plan.maxProjects){
    showToast('Your plan allows up to ' + plan.maxProjects + ' projects  -  upgrade for more, or delete one first', {icon: toastIcons.info, duration:3000});
    return;
  }
  const proj = {
    id: Date.now(),
    name: asset.name || ('Untitled ' + (projects.length+1)),
    ratio: '9:16',
    created: new Date().toISOString(),
    duration: 2,
    savedState: null
  };
  projects.unshift(proj);
  saveProjectsToStorage();
  trackEvent('project');
  renderProjects();
  openEditor(proj);
  setTimeout(()=>addCharacterFromModelAsset(asset), 0);
}

/* ================= MULTI-SELECT (long-press a project to start) =================
   Long-press any project card (~450ms) to enter select mode. In select mode a tap toggles the
   card, the top bar shows the count + Select all + Delete, and Delete opens a confirm sheet
   that deletes everything selected in one go. Cancel (or deselecting the last card) leaves
   select mode. One delegated pointer handler on the grid handles both tap and long-press, so
   it keeps working after every renderProjects() rebuild. */
let selectMode = false;
const selectedProjectIds = new Set();
const LONG_PRESS_MS = 450;
const MOVE_TOLERANCE = 10; // px of finger drift that cancels a long-press (so scrolling doesn't select)

function updateSelectUI(){
  const screen = document.getElementById('screen-projects');
  screen.classList.toggle('select-mode', selectMode);
  const n = selectedProjectIds.size;
  const title = document.getElementById('selectCount');
  if(title) title.textContent = 'Select projects';

function enterSelectMode(firstId){
  selectMode = true;
  selectedProjectIds.clear();
  if(firstId != null) selectedProjectIds.add(firstId);
  if(navigator.vibrate){ try{ navigator.vibrate(18); }catch(e){} }
  renderProjects();
  updateSelectUI();
}
function exitSelectMode(){
  selectMode = false;
  selectedProjectIds.clear();
  renderProjects();
  updateSelectUI();
}
function toggleProjectSelected(id){
  if(selectedProjectIds.has(id)) selectedProjectIds.delete(id); else selectedProjectIds.add(id);

  const card = document.querySelector('.project-card[data-pid="'+id+'"]');
  if(card) card.classList.toggle('selected', selectedProjectIds.has(id));
  if(selectedProjectIds.size === 0){ exitSelectMode(); return; }
  updateSelectUI();
}

  else projects.forEach(p=>selectedProjectIds.add(p.id));

(function wireProjectGesture(){
  const grid = document.getElementById('projectGrid');
  let timer = null, startX = 0, startY = 0, pressedId = null, longFired = false;

  function clear(){ clearTimeout(timer); timer = null; }
  grid.addEventListener('pointerdown', e=>{
    if(e.target.closest('.project-more')) return;
    const card = e.target.closest('.project-card');
    if(!card) return;
    pressedId = Number(card.dataset.pid);
    startX = e.clientX; startY = e.clientY; longFired = false;
    clear();
    if(!selectMode){
      timer = setTimeout(()=>{
        longFired = true;
        enterSelectMode(pressedId);
      }, LONG_PRESS_MS);
    }
  });
  grid.addEventListener('pointermove', e=>{
    if(timer && (Math.abs(e.clientX-startX) > MOVE_TOLERANCE || Math.abs(e.clientY-startY) > MOVE_TOLERANCE)) clear();
  });
  ['pointerup','pointercancel','pointerleave'].forEach(t=>grid.addEventListener(t, clear));
  // click fires after pointerup: a long-press must NOT also count as a tap/open
  grid.addEventListener('click', e=>{
    if(e.target.closest('.project-more')) return;
    const card = e.target.closest('.project-card');
    if(!card) return;
    if(longFired){ longFired = false; return; }
    const id = Number(card.dataset.pid);
    if(selectMode) toggleProjectSelected(id);
    else openEditorById(id);
  });
  // stop the browser's long-press context menu / image-save menu on cards
  grid.addEventListener('contextmenu', e=>{ if(e.target.closest('.project-card')) e.preventDefault(); });
})();

function confirmPlan(plan){
  const card = document.querySelector('#screen-payment .plan-card[data-plan="'+plan+'"]');
  if(card && !card.classList.contains('selected')) selectPlan(card);
  let planData = getPlans().find(p=>p.id===plan);
  if(!planData){
    // Rare fallback: confirmPlan fired before renderPaymentPlans() replaced the placeholder
    // cards with the real Firestore-backed ones (e.g. a very early tap). Read price straight
    // off the card's own rendered price text so a paid plan never gets mistaken for free.
    const priceText = card ? card.querySelector('.plan-price') : null;
    const priceNum = priceText ? parseInt((priceText.textContent||'').replace(/[^\d]/g,''), 10) || 0 : 0;
    const descText = card ? card.querySelector('.plan-desc') : null;
    planData = { id:plan, name:(card?card.dataset.name:plan) || plan, price:priceNum, desc:(descText?descText.textContent:'') };
  }
  // Track each explicit plan confirmation click, including the final plan choice for this
  // session. The Pro-specific counter is driven by the actual plan configured in Firestore.
  const planEventDetail={ planId:planData.id, planName:planData.name, planPrice:planData.price };
  trackEvent('plan_cta_click', withPromoAttribution(planEventDetail));
  trackPromoAttributedEvent('promo_plan_cta_click', planEventDetail);
  trackEvent('plan_confirm_click', withPromoAttribution(planEventDetail));
  trackPromoAttributedEvent('promo_plan_confirm_click', planEventDetail);
  const chosenName = String(planData.name || '').trim().toLowerCase();
  if(String(planData.id).toLowerCase()==='pro' || chosenName==='pro'){
    trackEvent('upgrade_to_pro_click', withPromoAttribution(planEventDetail));
    trackPromoAttributedEvent('promo_upgrade_to_pro_click', planEventDetail);
  }
  if(Number(planData.price) === 0){
    showToast((planData.name||'Free') + ' plan selected', {icon: toastIcons.check, duration:1600});
    return;
  }
  requireLoginThen(()=>{
    // Apply whatever discount (admin/first-purchase/promo) is currently active for this plan
    //  -  same computation renderPaymentPlans() used to show the struck-through price  -  so
    // checkout always charges exactly what the card displayed. originalPrice/discountLabel are
    // carried through so the checkout screen and the submitted order record both show it.
    const disc = computePlanDiscount(planData);
    const checkoutData = disc
      ? { ...planData, price: disc.finalPrice, originalPrice: disc.originalPrice, discountLabel: disc.label, isPromoDiscount: !!disc.isPromo }
      : { ...planData, originalPrice: null, discountLabel: null, isPromoDiscount:false };
    openCheckout(checkoutData);
  });
}

function onCheckoutProofPicked(e){
  const files = Array.from(e.target.files || []);
  e.target.value = '';
  if(!files.length) return;
  document.getElementById('checkoutError').style.display = 'none';
  let skippedNonImage = false, hitLimit = false;
  files.forEach(file=>{
    if(!/^image\//i.test(file.type)){ skippedNonImage = true; return; }
    const dup = checkoutProofs.some(p=> p.file.name === file.name && p.file.size === file.size && p.file.lastModified === file.lastModified);
    if(dup) return;
    if(checkoutProofs.length >= CHECKOUT_MAX_PROOFS){ hitLimit = true; return; }
    checkoutProofs.push({ file, url: URL.createObjectURL(file) });
    const proofDetail={ proofIndex:checkoutProofs.length, planId:checkoutPlan && checkoutPlan.id, planName:checkoutPlan && checkoutPlan.name };
    trackEvent('payment_proof_selected', withPromoAttribution(proofDetail));
    trackPromoAttributedEvent('promo_proof_image_selected', proofDetail);
  });
  renderCheckoutProofs();
  if(hitLimit) showCheckoutError('সর্বোচ্চ ' + toBnDigits(CHECKOUT_MAX_PROOFS) + 'টি ছবি আপলোড করা যাবে');
  else if(skippedNonImage) showCheckoutError('শুধু ছবির ফাইল বাছাই করুন');
}
function toBnDigits(n){
  return String(n).replace(/\d/g, d=>'\u09E6\u09E7\u09E8\u09E9\u09EA\u09EB\u09EC\u09ED\u09EE\u09EF'[d]);
}

// Compress payment proof screenshots locally so they fit safely inside a Firestore order.
// The image data is saved in proofImage / proofImages; no third-party image host is used.
function checkoutUploadProof(file, idx){
  return new Promise((resolve)=>{
    const url=URL.createObjectURL(file),img=new Image();
    img.onload=()=>{
      try{
        let scale=Math.min(1,960/Math.max(img.naturalWidth||1,img.naturalHeight||1));
        let canvas=document.createElement('canvas'),data='';
        for(let attempt=0;attempt<8;attempt++){
          canvas.width=Math.max(1,Math.round((img.naturalWidth||1)*scale));
          canvas.height=Math.max(1,Math.round((img.naturalHeight||1)*scale));
          const ctx=canvas.getContext('2d');ctx.imageSmoothingEnabled=true;ctx.imageSmoothingQuality='high';ctx.fillStyle='#fff';ctx.fillRect(0,0,canvas.width,canvas.height);ctx.drawImage(img,0,0,canvas.width,canvas.height);
          const quality=Math.max(.42,.70-attempt*.045);data=canvas.toDataURL('image/jpeg',quality);
          if(data.length<=125000)break;
          scale*=.84;
        }
        URL.revokeObjectURL(url);resolve(data&&data.length<=150000?data:'');
      }catch(err){URL.revokeObjectURL(url);resolve('');}
    };
    img.onerror=()=>{URL.revokeObjectURL(url);resolve('');};
    img.src=url;
  });
}

/* ================= HOME BANNER: swipe <-> dots sync =================
   Refactored into a named, re-callable function (was a run-once IIFE) so it can be re-wired
   after renderHomeBanners() replaces the slide/dot markup with admin-configured banners  - 
   otherwise the dots would keep listening to detached elements from the very first render. */
let _bannerDotsObserver = null, _bannerScrollHandler = null, _bannerAutoTimer = null;
const BANNER_AUTOPLAY_MS = 5000;
function wireBannerDots(){
  const scroller = document.getElementById('homeBannerScroll');
  const dotsWrap = document.getElementById('homeBannerDots');
  if(!scroller || !dotsWrap) return;
  const dots = Array.from(dotsWrap.children);
  const banners = Array.from(scroller.querySelectorAll('.home-banner-slide'));
  function setActive(i){ dots.forEach((d,k)=>d.classList.toggle('active', k===i)); }
  function currentIndex(){
    // pick the banner whose centre is closest to the scroller's centre
    const mid = scroller.scrollLeft + scroller.clientWidth/2;
    let best = 0, bestDist = Infinity;
    banners.forEach((b,i)=>{
      const c = b.offsetLeft + b.offsetWidth/2;
      const d = Math.abs(c - mid);
      if(d < bestDist){ bestDist = d; best = i; }
    });
    return best;
  }
  function goToSlide(i){
    const b = banners[i]; if(!b) return;
    scroller.scrollTo({ left: b.offsetLeft, behavior:'smooth' });
  }
  let raf = null;
  // The scroller element lives for the whole session while its slides/dots are re-created,
  // so drop the previous handler first  -  otherwise every re-render stacked another listener.
  if(_bannerScrollHandler) scroller.removeEventListener('scroll', _bannerScrollHandler);
  _bannerScrollHandler = ()=>{
    if(raf) return;
    raf = requestAnimationFrame(()=>{ raf = null; setActive(currentIndex()); });
  };
  scroller.addEventListener('scroll', _bannerScrollHandler, {passive:true});
  // tap a dot to jump to that banner
  dots.forEach((d,i)=>d.addEventListener('click', ()=>{ goToSlide(i); restartBannerAutoplay(); }));
  // The home screen is display:none until goTo('home') runs, so layout numbers are 0 at load.
  // Re-sync whenever it becomes visible / resizes so dots start on the right banner.
  if(_bannerDotsObserver) _bannerDotsObserver.disconnect();
  if(window.ResizeObserver){
    _bannerDotsObserver = new ResizeObserver(()=>setActive(currentIndex()));
    _bannerDotsObserver.observe(scroller);
  }

  /* ---- Auto-rotate: advance to the next banner every 5s ----
     A manual swipe/tap restarts the 5s timer so the banner doesn't jump away right after the
     person interacts with it. Paused while the tab/page is hidden so it doesn't burn cycles
     or jump slides while the person isn't looking. */
  function stopBannerAutoplay(){ if(_bannerAutoTimer){ clearInterval(_bannerAutoTimer); _bannerAutoTimer = null; } }
  function startBannerAutoplay(){
    stopBannerAutoplay();
    if(banners.length < 2) return;
    if(document.hidden) return;
    _bannerAutoTimer = setInterval(()=>{
      const next = (currentIndex() + 1) % banners.length;
      goToSlide(next);
    }, BANNER_AUTOPLAY_MS);
  }
  window.restartBannerAutoplay = function(){ startBannerAutoplay(); };
  // Manual swipe should also reset the timer so autoplay doesn't fight the person's gesture.
  scroller.addEventListener('touchstart', ()=>stopBannerAutoplay(), {passive:true});
  scroller.addEventListener('touchend', ()=>startBannerAutoplay(), {passive:true});
  document.removeEventListener('visibilitychange', _bannerVisibilityHandler);
  _bannerVisibilityHandler = ()=>{ if(document.hidden) stopBannerAutoplay(); else startBannerAutoplay(); };
  document.addEventListener('visibilitychange', _bannerVisibilityHandler);
  startBannerAutoplay();
}

wireBannerDots();

let _ethvelleAddCharPointerDownRow = null;
let _ethvelleAddCharPointerDownOrigin = 'other';
let _ethvelleAddCharPointerDownAt = 0;
let _ethvelleAddCharPointerDownId = null;
function _ethvelleResetAddCharPointer(){
  _ethvelleAddCharPointerDownRow = null;
  _ethvelleAddCharPointerDownOrigin = 'other';
  _ethvelleAddCharPointerDownAt = 0;
  _ethvelleAddCharPointerDownId = null;
}
// Record the original touch BEFORE timeline/clip handlers run. Use window capture rather
// than a timeline/document listener: later capture handlers can stop propagation before
// document is reached, which allowed a character-row tap to be mistaken for an Add-row tap.
window.addEventListener('pointerdown', e=>{
  const target=e.target;
  _ethvelleAddCharPointerDownRow = target && target.closest ? target.closest('.char-add-row') : null;
  _ethvelleAddCharPointerDownOrigin = _ethvelleAddCharPointerDownRow ? 'add' :
    (target && target.closest && target.closest('#characterRows .char-row') ? 'character' : 'other');
  _ethvelleAddCharPointerDownAt = Date.now();
  _ethvelleAddCharPointerDownId = e.pointerId;
}, true);
window.addEventListener('pointercancel', e=>{
  if(_ethvelleAddCharPointerDownId===e.pointerId)_ethvelleResetAddCharPointer();
}, true);
// Safety net for mobile browsers that re-hit-test the synthesized click after the row
// selection has repainted the timeline. A click landing on Add Character is accepted ONLY
// when its pointerdown also began on that exact Add Character row.
window.addEventListener('click', e=>{
  const addRow=e.target && e.target.closest ? e.target.closest('.char-add-row') : null;
  if(!addRow)return;
  const valid=_ethvelleAddCharPointerDownOrigin==='add' &&
    _ethvelleAddCharPointerDownRow===addRow && Date.now()-_ethvelleAddCharPointerDownAt<1200;
  if(!valid){
    e.preventDefault();
    e.stopPropagation();
    if(e.stopImmediatePropagation)e.stopImmediatePropagation();
    _ethvelleResetAddCharPointer();
  }
}, true);
