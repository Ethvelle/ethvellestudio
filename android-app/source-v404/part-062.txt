  function clipRate(ov){

  function fadeAmount(ov,t){
    let cfg={};try{cfg=typeof getEthvelleAudioSettings==='function'?getEthvelleAudioSettings(ov)||{}:{};}catch(_e){}
    const duration=Math.max(.001,Number(ov.trimEnd)-Number(ov.trimStart));
    let fi=clamp(cfg.fadeIn,0,8),fo=clamp(cfg.fadeOut,0,8);
    if(fi+fo>duration){const k=duration/(fi+fo);fi*=k;fo*=k;}
    let amount=1;
    if(fi>0)amount=Math.min(amount,clamp((t-Number(ov.trimStart))/fi,0,1));
    if(fo>0)amount=Math.min(amount,clamp((Number(ov.trimEnd)-t)/fo,0,1));
    return amount;
  }

  function ensureAnchor(ctx){
    // One shared timeline->AudioContext mapping per play run. Every clip start/stop time is
    // derived from this single mapping, so the end of clip A and the start of clip B land on
    // the exact same sample instead of each being computed from a jittery per-frame reading.
    if(!anchor||anchor.run!==runSerial){
      const tl=(isPlaying&&typeof getCurrentAbsoluteTime==='function')?getCurrentAbsoluteTime():(Number(currentAbsoluteTimeSec)||0);
      anchor={ctx:ctx.currentTime,tl,lead:0,run:runSerial};
      try{if(window.__evPlaybackDiagnostics)window.__evPlaybackDiagnostics.recordAnchor();}catch(_e){}
    }
    return anchor;
  }
  function sameSourceContinuation(a,b){
    if(!a||!b||a===b||a.type!==b.type||a.reversePlayback||b.reversePlayback)return false;
    if(String(a.src||'')!==String(b.src||''))return false;
    const ra=clipRate(a),rb=clipRate(b);
    if(Math.abs(ra-rb)>.0005)return false;
    if(Math.abs((Number(a.trimEnd)||0)-(Number(b.trimStart)||0))>.006)return false;
    const endSrc=(Number(a.sourceOffset)||0)+((Number(a.trimEnd)||0)-(Number(a.trimStart)||0))*ra;
    if(Math.abs(endSrc-(Number(b.sourceOffset)||0))>.02)return false;
    if(Math.abs(clipVolume(a,Number(a.trimEnd)||0)-clipVolume(b,Number(b.trimStart)||0))>.01)return false;
    return true;
  }

    return true;

  // The clip that continues the same waveform right before/after `ov` (a plain split). A cut
  // between two such halves must NOT be faded, otherwise a tiny dip would be audible.
  function continuationNeighbor(ov,dir){
    if(!Array.isArray(overlays))return null;
    for(let i=0;i<overlays.length;i++){
      const o=overlays[i];
      if(!o||o===ov||o.type!==ov.type)continue;
      if(dir<0){
        if(sameSourceContinuation(o,ov)&&audibleHere(o,Math.max(0,(Number(o.trimEnd)||0)-.001)))return o;
      }else if(sameSourceContinuation(ov,o)&&audibleHere(o,(Number(o.trimStart)||0)+.001))return o;
    }
    return null;
  }
  function stopScheduledOne(id){
    const entry=scheduled.get(String(id));if(!entry)return;
    scheduled.delete(String(id));
    let soft=false;
    try{
      const ctx=audioContext();
      // A source that is sounding right now is released with an ~8ms ramp, not chopped, so
      // pause / seek / edit never leaves a click at the stop point.
      if(ctx&&entry.env&&entry.when<=ctx.currentTime+.001){
        const t=ctx.currentTime,g=entry.env.gain;
        g.cancelScheduledValues(t);g.setValueAtTime(g.value,t);g.linearRampToValueAtTime(0,t+.008);
        entry.source.onended=null;entry.source.stop(t+.012);
        setTimeout(()=>{try{entry.source.disconnect();}catch(_e){}try{entry.env.disconnect();}catch(_e){}},80);
        soft=true;
      }
    }catch(_e){soft=false;}
    if(!soft){
      try{entry.source.onended=null;entry.source.stop();}catch(_e){}
      try{entry.source.disconnect();}catch(_e){}
      try{if(entry.env)entry.env.disconnect();}catch(_e){}
    }
    if(entry.route==='video-graph'){
      const nodes=videoAudioNodes[String(id)];
      if(nodes&&nodes.source===entry.source)nodes.source=null;
    }
  }

  // Internal decoder rebuffer: freeze the AudioContext and canonical timeline together.
  // Unlike a user pause, this deliberately does NOT stop, disconnect, or recreate any
  // scheduled AudioBufferSourceNode. AudioContext suspension preserves each source's sample.
  let internalRebufferHold=null;
  async function suspendForRebuffer(){
    if(internalRebufferHold)return internalRebufferHold;
    const ctx=audioContext();
    const startTime=(isPlaying&&typeof getCurrentAbsoluteTime==='function')
      ?getCurrentAbsoluteTime():(Number(currentAbsoluteTimeSec)||0);
    const hold={ctx,run:runSerial,time:startTime,suspendFailed:false};

    // Stop the performance-clock loop immediately, without entering the public pause wrapper
    // (the v165 public pause wrapper intentionally calls stopScheduledAll()).
    if(playbackRaf!==null){try{cancelAnimationFrame(playbackRaf);}catch(_e){}playbackRaf=null;}
    setCurrentAbsoluteTime(startTime);
    isPlaying=false;
    playbackStartedAtMs=0;
    playbackStartTimeSec=startTime;
    internalRebufferHold=hold;

    // Suspend first, then read currentTime after the suspend promise resolves. The context may
    // finish its current render quantum before stopping; this calculation advances the held
    // timeline to the exact audio sample at which the source nodes were frozen.
    if(ctx&&ctx.state==='running'){
      try{const pending=ctx.suspend();if(pending&&typeof pending.then==='function')await pending;}
      catch(err){hold.suspendFailed=true;if(window.__evVideoDebug)console.warn('[evAudio] rebuffer suspend:',err);}
    }
    // A real user pause/seek/edit may invalidate the run while suspend() is pending.
    // In that case the public action owns the clock and this internal hold must not rewind it.
    if(internalRebufferHold!==hold||runSerial!==hold.run)return hold;
    let frozen=startTime;
    if(ctx&&anchor&&anchor.run===hold.run&&Number.isFinite(ctx.currentTime)){
      frozen=anchor.tl+(ctx.currentTime-anchor.ctx-anchor.lead);
    }
    frozen=Math.max(0,Number(frozen)||0);
    setCurrentAbsoluteTime(frozen);
    playbackStartTimeSec=frozen;
    hold.time=frozen;
    try{if(typeof renderFrame==='function')renderFrame();}catch(_e){}
    return hold;
  }
  async function resumeAfterRebuffer(){
    const hold=internalRebufferHold;
    const ctx=(hold&&hold.ctx)||audioContext();
    // Resume the same suspended context: previously scheduled sources continue at their
    // exact frozen sample. Do not call startPlayback(), which is a new-run rescheduler.
    if(ctx&&ctx.state==='suspended'){
      const pending=ctx.resume();if(pending&&typeof pending.then==='function')await pending;
    }
    let t=Math.max(0,Number(currentAbsoluteTimeSec)||(hold&&hold.time)||0);
