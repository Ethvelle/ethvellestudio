(function(){
  'use strict';
  var LOW_CREDITS=100; /* available credits below this (including 0) = compact capsule */
  function syncCreditCapsule(){
    var button=document.getElementById('homeAccountSummary');
    var balance=document.getElementById('homeAccountBalance');
    if(!button||!balance)return;
    var value=balance.querySelector('.ev-v394-available-value');
    var compact=false;
    if(value){
      var n=parseInt(String(value.textContent).replace(/[^0-9]/g,''),10);
      compact=(isNaN(n)?0:n)<LOW_CREDITS;
    }
    button.classList.toggle('ev-v399-credit-compact',compact);
  }
  var balanceEl=document.getElementById('homeAccountBalance');
  if(balanceEl&&typeof MutationObserver==='function'){
    new MutationObserver(syncCreditCapsule).observe(balanceEl,{childList:true,subtree:true,characterData:true});
  }
  syncCreditCapsule();
})();
</script>
