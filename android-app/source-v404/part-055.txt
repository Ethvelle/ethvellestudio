  const duplicateControls=new Set([
    'clipMediaAccessTool','evcaFrameCaptureTool-clip','evv72SaveFrame-clip','evv72AddFrame-clip'
  ]);
  let lastSignature='';

  function removeMirrors(){
    keyRow.querySelectorAll('.evv-kf-mirror').forEach(el=>el.remove());
    lastSignature='';
  }
  function sourceTools(){
    return Array.from(clipRow.children).filter(el=>{
      if(!el.classList?.contains('tool-item'))return false;
      if(duplicateControls.has(el.id))return false;
      if(el.id==='clipDeleteCharacterTool')return false; // No whole-character Delete in keyframe mode.
      const label=(el.querySelector('span:last-child')?.textContent||'').trim().toLowerCase();
      if(label==='delete')return false;                 // No normal selected-layer/clip Delete in keyframe mode.
      if(label==='done')return false;
      return getComputedStyle(el).display!=='none';
    });
  }
  function buildMirrors(){
    const selected=typeof selectedKfIdx!=='undefined'&&selectedKfIdx!==null;
    const ref=typeof selectedClip==='undefined'?'':String(selectedClip||'');
    const isCharacterKeyframe=selected&&(ref==='main'||/^seg\d+$/.test(ref));
    if(!isCharacterKeyframe){removeMirrors();return;}

    // Recompute the normal character-row visibility using the app's existing sync logic,
    // then immediately restore the selected-keyframe row before the browser paints.
    if(typeof oldSync==='function'){
      const savedKfIndex=selectedKfIdx;
      selectedKfIdx=null;
      try{oldSync.apply(window,[]);}finally{selectedKfIdx=savedKfIndex;}
      const defaultRow=document.getElementById('bottomToolsDefault');
      [defaultRow,clipRow].forEach(row=>row?.style.setProperty('display','none','important'));
      keyRow.style.setProperty('display','flex','important');
    }

    const sources=sourceTools();
    // Keep the existing Add model action available while the two Delete actions are hidden.
    const addCharacterTool=document.querySelector('#bottomToolsDefault [data-tool="model"]');
    if(addCharacterTool)sources.unshift(addCharacterTool);
    const signature=sources.map(el=>(el.id||'')+':'+(el.textContent||'').trim()).join('|');
    if(signature===lastSignature&&keyRow.querySelectorAll('.evv-kf-mirror').length===sources.length)return;
    removeMirrors();
    sources.forEach(source=>{
      const proxy=source.cloneNode(true);
      proxy.removeAttribute('id');
      proxy.classList.add('evv-kf-mirror');
      proxy.setAttribute('role','button');
      proxy.setAttribute('tabindex','0');
      proxy.setAttribute('aria-label',(source.getAttribute('aria-label')||(source.textContent||'').trim()||'Character tool').trim());
      Array.from(proxy.attributes||[]).forEach(attr=>{if(/^on/i.test(attr.name))proxy.removeAttribute(attr.name);});
      proxy.querySelectorAll('[id]').forEach(node=>node.removeAttribute('id'));
      proxy.querySelectorAll('*').forEach(node=>{
        Array.from(node.attributes||[]).forEach(attr=>{if(/^on/i.test(attr.name))node.removeAttribute(attr.name);});
      });
      proxy.style.setProperty('display','flex','important');
      const activate=e=>{
        if(e){e.preventDefault();e.stopPropagation();}
        source.click();
      };
      proxy.addEventListener('click',activate);
      proxy.addEventListener('keydown',e=>{if(e.key==='Enter'||e.key===' '){activate(e);}});
      keyRow.appendChild(proxy);
    });
    lastSignature=signature;
  }

  const oldSync=window.syncBottomToolsMode;
  if(typeof oldSync==='function'&&!oldSync.__ethvelleV110KeyframeTools){
    const wrapped=function(){
      const result=oldSync.apply(this,arguments);
      buildMirrors();
      return result;
    };
    wrapped.__ethvelleV110KeyframeTools=true;
    window.syncBottomToolsMode=wrapped;
  }
  // This is an initialization-only fallback; later keyframe selections flow through the wrapper above.
  buildMirrors();
})();
