      if(e.isPrimary===false||(e.pointerType==='mouse'&&e.button!==0))return;
      // Geometry is stable for the duration of one swipe. Read the scroll range once here,
      // instead of asking layout for scrollWidth/clientWidth on every RAF flush and glide tick.
      state.maxScroll=Math.max(0,row.scrollWidth-row.clientWidth);
      if(state.maxScroll<=2)return;
      stopGlide();
      state.id=e.pointerId;state.mode='pending';state.x=state.lastX=e.clientX;state.y=e.clientY;
      state.lastAt=clock();state.startScroll=row.scrollLeft;state.targetScroll=row.scrollLeft;state.velocity=0;
    },{capture:true,passive:true});
    row.addEventListener('pointermove',e=>{
      if(state.id!==e.pointerId)return;
      const dx=e.clientX-state.x,dy=e.clientY-state.y;
      if(state.mode==='pending'){
        if(Math.max(Math.abs(dx),Math.abs(dy))<6)return;
        if(Math.abs(dy)>Math.abs(dx)*1.12){
          // Keep the pointer identity until pointerup so the generated click from a vertical
          // swipe that starts on a tool cannot activate that tool. Clearing state.id here
          // made pointerup invisible to this controller and let the synthetic click through.
          state.mode='cancelled';
          state.suppressUntil=clock()+500;
          return;
        }
        if(Math.abs(dx)<6)return;
        state.mode='drag';
        state.suppressUntil=clock()+260;
        try{row.setPointerCapture(e.pointerId);}catch(_e){}
      }
      if(state.mode!=='drag')return;
      const t=clock(),dt=Math.max(1,t-state.lastAt),deltaX=e.clientX-state.lastX;
      const instant=-deltaX/dt;
      state.velocity=clamp(state.velocity*.32+instant*.68,-2.4,2.4);
      state.lastX=e.clientX;state.lastAt=t;
      state.targetScroll=state.startScroll-dx;
      if(!state.moveRaf)state.moveRaf=requestAnimationFrame(flush);
      if(e.cancelable)e.preventDefault();
      e.stopPropagation();
    },{passive:false});
    row.addEventListener('pointerup',e=>{
      if(state.id!==e.pointerId)return;
      const dragged=state.mode==='drag';
      if(state.moveRaf){cancelAnimationFrame(state.moveRaf);state.moveRaf=0;flush();}
      state.id=null;state.mode='idle';
      if(dragged){state.suppressUntil=clock()+260;try{row.releasePointerCapture(e.pointerId);}catch(_e){}glide();}
    },{passive:true});
    row.addEventListener('pointercancel',e=>{
      if(state.id!==e.pointerId)return;
      state.id=null;state.mode='idle';state.velocity=0;state.suppressUntil=0;
      if(state.moveRaf){cancelAnimationFrame(state.moveRaf);state.moveRaf=0;}
    },{passive:true});
    row.addEventListener('click',e=>{
      if(!state.suppressUntil)return;
      if(clock()>state.suppressUntil){state.suppressUntil=0;return;}
      state.suppressUntil=0;e.preventDefault();e.stopPropagation();e.stopImmediatePropagation();
    },true);

      if(Math.abs(e.clientX-(r.left+r.width/2))>13||Math.abs(e.clientY-(r.top+r.height/2))>Math.min(24,r.height/2+1)){
        if(e.cancelable)e.preventDefault();e.stopPropagation();e.stopImmediatePropagation();
      }
    }
  },{capture:true,passive:false});

  /* 3. Finish clip-row scrolling with gentle inertia after the direct RAF handoff. */
  let clipGlideRaf=0,clipGlideGeneration=0;
  function stopClipGlide(){clipGlideGeneration++;if(clipGlideRaf){cancelAnimationFrame(clipGlideRaf);clipGlideRaf=0;}}
  function startClipGlide(){
    const generation=++clipGlideGeneration;
    const motion=window.__ethvelleV110ClipScrollMotion;
    if(!motion||!motion.scroller||!motion.scroller.isConnected)return;
    motion.active=false;
    let velocity=clamp(Number(motion.velocity)||0,-2.4,2.4);
    const scroller=motion.scroller;
    // Flush the last finger position first; the original helper may have one RAF queued.
    requestAnimationFrame(()=>{
      if(generation!==clipGlideGeneration)return;
      if(Number.isFinite(motion.targetTop))scroller.scrollTop=clamp(motion.targetTop,0,Math.max(0,scroller.scrollHeight-scroller.clientHeight));
      if(Math.abs(velocity)<0.07)return;
      let previous=clock();
      const tick=()=>{
        const t=clock(),dt=Math.min(32,Math.max(1,t-previous));previous=t;
        const max=Math.max(0,scroller.scrollHeight-scroller.clientHeight);
        const next=clamp(scroller.scrollTop+velocity*dt,0,max);
        scroller.scrollTop=next;
        velocity*=Math.pow(0.93,dt/16.667);
        if(next<=0||next>=max||Math.abs(velocity)<0.035){clipGlideRaf=0;return;}
        clipGlideRaf=requestAnimationFrame(tick);
      };
      clipGlideRaf=requestAnimationFrame(tick);
    });
  }
  document.addEventListener('pointerdown',e=>{
    if(e.target?.closest?.('#screen-editor #timelineScroll')){
      stopClipGlide();
      const m=window.__ethvelleV110ClipScrollMotion;if(m)m.active=false;
    }
  },{capture:true,passive:true});
  document.addEventListener('pointerup',()=>{
    const m=window.__ethvelleV110ClipScrollMotion;
    if(m&&m.active)startClipGlide();
  },{capture:true,passive:true});
  document.addEventListener('pointercancel',()=>{
    const m=window.__ethvelleV110ClipScrollMotion;if(m)m.active=false;
    stopClipGlide();
  },{capture:true,passive:true});

  /* 4. Selection-outline X: block the legacy pointer/click delete handlers, then deselect. */
  const closeButton=$('sbDelete');
  if(closeButton){
    closeButton.title='Deselect layer';closeButton.setAttribute('aria-label','Deselect layer');
    closeButton.setAttribute('role','button');closeButton.tabIndex=0;
    closeButton.addEventListener('pointerdown',e=>{e.stopPropagation();e.stopImmediatePropagation();},true);
    const deselect=e=>{
      e.preventDefault();e.stopPropagation();e.stopImmediatePropagation();
      if(typeof window.deselectAllClips==='function')window.deselectAllClips();
      else if(typeof deselectAllClips==='function')deselectAllClips();
    };
    closeButton.addEventListener('click',deselect,true);
    closeButton.addEventListener('keydown',e=>{if(e.key==='Enter'||e.key===' '){deselect(e);}},true);
  }
})();
