<script id="ethvelle-v96-direct-crop-and-performance-js">
(function(){
  'use strict';
  const $=id=>document.getElementById(id);
  const clamp=(v,min,max)=>Math.max(min,Math.min(max,Number(v)||0));

  /* ------------------------------------------------------------------
     1. Manual crop: use the actual transformed crop-root quad to map
        screen movement into the crop frame's local coordinate system.
        This includes canvas zoom, parent rotation and media rotation.
     ------------------------------------------------------------------ */
  let cropDrag=null;
  function getCropContext(){
    const id=String(window.__ethvelleCropEditingId||((typeof selectedClip!=='undefined'&&selectedClip!=null)?selectedClip:''));
    if(!id)return null;
    let clip=null;
    try{clip=typeof window.ethvelleObjectClipById==='function'?window.ethvelleObjectClipById(id):null;}catch(_e){}
    if(!clip)return null;
    const ovs=(typeof overlays!=='undefined'&&Array.isArray(overlays))?overlays:[];
    const bgs=(typeof backgrounds!=='undefined'&&Array.isArray(backgrounds))?backgrounds:[];
    let el=null,kind='';
    if(ovs.includes(clip)){
      kind='overlay';
      el=(typeof overlayMediaEls!=='undefined'&&overlayMediaEls)?overlayMediaEls[id]:null;
      if(!el){const node=document.querySelector(`.overlay-media-item[data-overlay-id="${id.replace(/\\/g,'\\\\').replace(/"/g,'\\"')}"]`);el=node?.querySelector('img,video,canvas')||node;}
    }else if(bgs.includes(clip)){
      kind='background';el=$('bgImgLayer');
    }
    if(!kind)return null;
    const stage=$('canvasStage');if(!stage)return null;
    const sw=Math.max(1,stage.clientWidth),sh=Math.max(1,stage.clientHeight);
    const iw=Math.max(1,el&&(el.videoWidth||el.naturalWidth||el.width)||Number(clip.sourceWidth)||sw);
    const ih=Math.max(1,el&&(el.videoHeight||el.naturalHeight||el.height)||Number(clip.sourceHeight)||sh);
    const scale=Math.min(sw/iw,sh/ih),w=iw*scale,h=ih*scale;
    return {id,clip,kind,el,stage,sw,sh,iw,ih,m:{x:(sw-w)/2,y:(sh-h)/2,w,h,aspect:iw/ih}};
  }
  function cropValues(clip){
    const raw=typeof window.ethvelleCropFractions==='function'?window.ethvelleCropFractions(clip.crop):(clip.crop||{});
    return {left:clamp(raw.left,0,.92),right:clamp(raw.right,0,.92),top:clamp(raw.top,0,.92),bottom:clamp(raw.bottom,0,.92)};
  }
  function normalizeCrop(c){
    let left=clamp(c.left,0,.92),right=clamp(c.right,0,.92),top=clamp(c.top,0,.92),bottom=clamp(c.bottom,0,.92);
    if(left+right>.92){const f=.92/(left+right);left*=f;right*=f;}
    if(top+bottom>.92){const f=.92/(top+bottom);top*=f;bottom*=f;}
    return {left,right,top,bottom};
  }
  function localToClientInverse(root,stage){
    // Preferred path: getBoxQuads includes every transform in the ancestor chain and the
    // cropRoot's own translate/rotate/scale. Its basis vectors map local pixels to screen pixels.
    try{
      if(typeof root.getBoxQuads==='function'){
        const q=root.getBoxQuads({box:'border'})[0]||root.getBoxQuads()[0];
        const w=Math.max(1,root.offsetWidth),h=Math.max(1,root.offsetHeight);
        if(q&&q.p1&&q.p2&&q.p4){
          const a=(q.p2.x-q.p1.x)/w,b=(q.p2.y-q.p1.y)/w;
          const c=(q.p4.x-q.p1.x)/h,d=(q.p4.y-q.p1.y)/h;
          const det=a*d-b*c;
          if(Math.abs(det)>.000001)return {a:d/det,b:-b/det,c:-c/det,d:a/det};
        }
      }
    }catch(_e){}
    // Fallback: compose the 2D transforms of the crop root and every ancestor. Translation
    // and transform-origin do not affect deltas; the linear matrix does. This also handles
    // inline-canvas zoom/rotation in browsers where getBoxQuads is unavailable.
    try{
      const M=window.DOMMatrixReadOnly||window.DOMMatrix;
      let a=1,b=0,c=0,d=1;
      for(let node=root;node&&node.nodeType===1;node=node.parentElement){
        const cs=getComputedStyle(node),raw=cs.transform;
        let na=1,nb=0,nc=0,nd=1;
        if(M&&raw&&raw!=='none'){
          const matrix=new M(raw);na=matrix.a;nb=matrix.b;nc=matrix.c;nd=matrix.d;
        }
        const zoom=parseFloat(cs.zoom);const z=Number.isFinite(zoom)&&zoom>0?zoom:1;
        na*=z;nb*=z;nc*=z;nd*=z;
        const nextA=na*a+nc*b,nextB=nb*a+nd*b,nextC=na*c+nc*d,nextD=nb*c+nd*d;
        a=nextA;b=nextB;c=nextC;d=nextD;
      }
      const det=a*d-b*c;
      if(Math.abs(det)>.000001)return {a:d/det,b:-b/det,c:-c/det,d:a/det};
      return {a:1,b:0,c:0,d:1};
    }catch(_e){return {a:1,b:0,c:0,d:1};}
  }
  function paintManualCrop(ctx,c){

  function cropPointerInStage(e,root,stage,inverse){
    // Convert the contact point to the crop-root's local stage coordinates. getBoxQuads
    // gives the true transformed top-left; the centre-based fallback remains correct for
    // the root's 50% transform-origin when getBoxQuads is unavailable in a WebView.

        if(q&&q.p1){

          return {x:inverse.a*dx+inverse.c*dy,y:inverse.b*dx+inverse.d*dy};

  function startManualCrop(e){
    const root=$('evV89CropRoot'),frame=$('evV89CropFrame');
    if(!root||!frame||getComputedStyle(root).display==='none')return false;
    if(e.pointerType==='mouse'&&e.button!==0)return false;

      else if(nearT&&nearR)handle='ne';
      else if(nearB&&nearL)handle='sw';
      else if(nearB&&nearR)handle='se';
      else if(nearL)handle='w';
      else if(nearR)handle='e';
      else if(nearT)handle='n';
      else if(nearB)handle='s';

      return true;

    const c={...d.startCrop},h=d.handle;
    const w0=1-d.startCrop.left-d.startCrop.right,hh0=1-d.startCrop.top-d.startCrop.bottom;
    const min=.04;
    if(h==='move'){
      c.left=clamp(d.startCrop.left+dx,0,1-w0);c.right=1-w0-c.left;
      c.top=clamp(d.startCrop.top+dy,0,1-hh0);c.bottom=1-hh0-c.top;
    }else if(d.ratio==='free'){
      if(h.includes('w'))c.left=clamp(d.startCrop.left+dx,0,1-d.startCrop.right-min);
      if(h.includes('e'))c.right=clamp(d.startCrop.right-dx,0,1-d.startCrop.left-min);
      if(h.includes('n'))c.top=clamp(d.startCrop.top+dy,0,1-d.startCrop.bottom-min);
      if(h.includes('s'))c.bottom=clamp(d.startCrop.bottom-dy,0,1-d.startCrop.top-min);
    }else{
      const ratioParts=String(d.ratio).split(':').map(Number);
      const wanted=(ratioParts[0]/ratioParts[1])/ctx.m.aspect;
      if(!Number.isFinite(wanted)||wanted<=0){cropDrag=null;return false;}
      if(h==='e'||h==='w'){
        let w=clamp(w0+(h==='e'?dx:-dx),min,1),height=clamp(w/wanted,min,1);w=height*wanted;
        if(h==='e')c.right=1-d.startCrop.left-w;else c.left=1-d.startCrop.right-w;
        c.top=(1-height)/2;c.bottom=1-height-c.top;
      }else if(h==='n'||h==='s'){
        let height=clamp(hh0+(h==='s'?dy:-dy),min,1),w=clamp(height*wanted,min,1);height=w/wanted;
        if(h==='s')c.bottom=1-d.startCrop.top-height;else c.top=1-d.startCrop.bottom-height;
        c.left=(1-w)/2;c.right=1-w-c.left;
      }else{
        const dirX=h.includes('w')?-1:1,dirY=h.includes('n')?-1:1;
        const wCandidate=clamp(w0+dirX*dx,min,1),hCandidate=clamp(hh0+dirY*dy,min,1);
        let w=Math.abs(wCandidate-w0)/Math.max(min,w0)>=Math.abs(hCandidate-hh0)/Math.max(min,hh0)?wCandidate:hCandidate*wanted;
        let height=w/wanted;
        if(height>1){height=1;w=height*wanted;}if(w>1){w=1;height=w/wanted;}
        if(dirX<0)c.left=1-d.startCrop.right-w;else c.right=1-d.startCrop.left-w;
        if(dirY<0)c.top=1-d.startCrop.bottom-height;else c.bottom=1-d.startCrop.top-height;
      }
    }
    const next=normalizeCrop(c);
    ctx.clip.crop=next;
    paintManualCrop(ctx,next);
    // Update readouts only; do not rebuild the media renderer or persist on every pointer frame.
    ['left','top','right','bottom'].forEach(k=>{const v=$('evcr-'+k+'-value');if(v)v.textContent=Math.round(next[k]*100)+'%';const input=$('evcr-'+k);if(input)input.value=Math.round(next[k]*100);});
    e.preventDefault();e.stopPropagation();if(e.stopImmediatePropagation)e.stopImmediatePropagation();
    return true;
  }
  function finishManualCrop(e){
    if(!cropDrag||e.pointerId!==undefined&&e.pointerId!==cropDrag.pointerId)return false;
    const d=cropDrag;cropDrag=null;
    try{if(e.pointerId!==undefined&&d.capture?.hasPointerCapture?.(e.pointerId))d.capture.releasePointerCapture(e.pointerId);}catch(_e){}
    try{if(typeof window.persistProjectState==='function')window.persistProjectState({skipHistory:true});}catch(_e){}
    e?.preventDefault?.();e?.stopPropagation?.();if(e?.stopImmediatePropagation)e.stopImmediatePropagation();
    return true;
  }
  // Window capture runs before the old document-level crop handler and before canvas drag handlers.
  window.addEventListener('pointerdown',e=>{if(startManualCrop(e))return;},{capture:true,passive:false});
  window.addEventListener('pointermove',e=>{if(cropDrag)updateManualCrop(e);},{capture:true,passive:false});
  window.addEventListener('pointerup',e=>{if(cropDrag)finishManualCrop(e);},{capture:true,passive:false});
  window.addEventListener('pointercancel',e=>{if(cropDrag)finishManualCrop(e);},{capture:true,passive:false});

  /* ------------------------------------------------------------------
     2. Native bottom tool-strip movement. The final CSS enables native
        horizontal overflow scrolling and Android momentum without competing

  /*
     The bottom tool rows now use native horizontal overflow scrolling. This avoids

  /* ------------------------------------------------------------------
     3. Coalesce selection-box rendering to one paint per display frame.
        Drag values still update on every input event; only DOM painting
        is coalesced. Flush before pointer-up saves the final transform.
     ------------------------------------------------------------------ */
  const baseSelBoxApply=window.selBoxApply;
  let pendingSelTarget=null,selPaintRaf=0,selPaintThis=null,lastDragSliderSync=0;
  const syncTransformBase=window.syncTransformSliders;
  if(typeof syncTransformBase==='function'&&!syncTransformBase.__evv95DragThrottle){
    const throttled=function(){
      if(window.__ethvelleLayerDragPainting){
        const now=performance.now();if(now-lastDragSliderSync<90)return;
        lastDragSliderSync=now;
      }
      return syncTransformBase.apply(this,arguments);
    };
    throttled.__evv95DragThrottle=true;window.syncTransformSliders=throttled;
  }
  function flushSelPaint(){
    if(selPaintRaf){cancelAnimationFrame(selPaintRaf);selPaintRaf=0;}
    if(!pendingSelTarget||typeof baseSelBoxApply!=='function')return;
    const target=pendingSelTarget,context=selPaintThis;pendingSelTarget=null;selPaintThis=null;
    window.__ethvelleLayerDragPainting=true;
    try{baseSelBoxApply.call(context,target);}catch(err){console.warn('Ethvelle drag paint:',err);}
    finally{window.__ethvelleLayerDragPainting=false;}
  }
  if(typeof baseSelBoxApply==='function'){
    const coalesced=function(tg){pendingSelTarget=tg;selPaintThis=this;if(!selPaintRaf)selPaintRaf=requestAnimationFrame(flushSelPaint);};
    coalesced.__evv95Raf=true;window.selBoxApply=coalesced;
    document.addEventListener('pointerup',flushSelPaint,true);
    document.addEventListener('pointercancel',flushSelPaint,true);
  }

  /* Transform sliders previously called the full media renderer on every input event.
     Keep the keyframe/model writes from the existing updateTransform(), but draw only the
     selected media element during that short slider gesture. */
  const baseRenderOverlay=window.renderOverlayMedia;
  if(typeof baseRenderOverlay==='function'&&!baseRenderOverlay.__evv95FastTransform){
    const fastRender=function(time){
      if(window.__ethvelleFastTransformInput){
        try{
          const id=String(typeof selectedClip!=='undefined'&&selectedClip!=null?selectedClip:'');
          const clip=(typeof overlays!=='undefined'&&Array.isArray(overlays))?overlays.find(o=>String(o.id)===id):null;
          if(clip&&(clip.type==='image'||clip.type==='video')&&typeof window.applyOverlayTransformPreviewFast==='function'){
            window.applyOverlayTransformPreviewFast({obj:clip.transform||{scale:1,offsetX:0,offsetY:0,rotation:0}});
            if(typeof window.updateSelBox==='function')window.updateSelBox();
            return;
          }
        }catch(err){console.warn('Ethvelle fast transform render:',err);}
      }
      return baseRenderOverlay.apply(this,arguments);
    };
    fastRender.__evv95FastTransform=true;window.renderOverlayMedia=fastRender;
  }
  const baseUpdateTransform=window.updateTransform;
  if(typeof baseUpdateTransform==='function'&&!baseUpdateTransform.__evv95FastTransform){
    const fastUpdate=function(){
      window.__ethvelleFastTransformInput=true;
      try{return baseUpdateTransform.apply(this,arguments);}
      finally{window.__ethvelleFastTransformInput=false;}
    };
    fastUpdate.__evv95FastTransform=true;window.updateTransform=fastUpdate;
  }

  /* Character-joint drawing is also painted at most once per display frame while a pointer
     is held on the canvas. Pose calculations themselves remain immediate. */
  const baseDrawStick=window.drawStick;
  if(typeof baseDrawStick==='function'&&!baseDrawStick.__evv95Raf){
    let canvasPointerActive=false,pendingDraw=false,drawRaf=0,drawingNow=false,drawArgs=[];
    function flushDraw(){
      if(drawRaf){cancelAnimationFrame(drawRaf);drawRaf=0;}
      if(!pendingDraw)return;
      pendingDraw=false;drawingNow=true;
      try{baseDrawStick.apply(window,drawArgs);}catch(err){console.warn('Ethvelle canvas drag paint:',err);}
      finally{drawingNow=false;drawArgs=[];}
    }
    const wrappedDraw=function(){
      if((!canvasPointerActive&&!window.__ethvelleFastTransformInput)||drawingNow)return baseDrawStick.apply(this,arguments);
      pendingDraw=true;drawArgs=Array.from(arguments);
      if(!drawRaf)drawRaf=requestAnimationFrame(flushDraw);
    };
    wrappedDraw.__evv95Raf=true;window.drawStick=wrappedDraw;
    window.addEventListener('pointerdown',e=>{if(e.target?.closest?.('#stickCanvas'))canvasPointerActive=true;},{capture:true,passive:true});
    const endCanvas=e=>{if(!canvasPointerActive)return;canvasPointerActive=false;flushDraw();};
    document.addEventListener('pointerup',endCanvas,true);
    document.addEventListener('pointercancel',endCanvas,true);
    window.addEventListener('blur',()=>{canvasPointerActive=false;flushDraw();});
  }
})();
