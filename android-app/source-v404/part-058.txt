<script id="ethvelle-v133-complete-history-js">
(function(){
  'use strict';
  if(window.__ethvelleV133HistoryInstalled)return;
  window.__ethvelleV133HistoryInstalled=true;

  const editor=document.getElementById('screen-editor');
  const activePointers=new Set();
  const originalPushHistory=window.pushHistory;
  let transactionOpen=false;
  let commitTimer=null;
  // Invalidates delayed commits when Undo/Redo or project reset is requested. Without this,
  // a pending pointer/input timer could run just after Undo and accidentally erase redo.
  let transactionEpoch=0;

  function isEditorEvent(e){
    if(!editor||!e||!e.target)return false;
    if(!editor.contains(e.target))return false;
    // Undo and redo must not record themselves as new edits.
    if(e.target.closest&&e.target.closest('#undoBtn,#redoBtn'))return false;
    return true;
  }
  function clearCommitTimer(){
    if(commitTimer!==null){clearTimeout(commitTimer);commitTimer=null;}
  }
  function beginTransaction(){
    clearCommitTimer();
    if(transactionOpen)return;
    try{
      // Project opening already seeds history with resetHistory(). Only seed here as a
      // safety net for a newly-created editor; avoid serializing large media projects on
      // every ordinary tap/drag start when the previous action is already committed.
      if(typeof pushHistory==='function'&&!restoringHistory&&typeof historyIndex!=='undefined'&&historyIndex<0)pushHistory();
    }catch(err){console.warn('Ethvelle history baseline:',err);}
    transactionOpen=true;
  }
  function commitTransaction(delay){
    if(!transactionOpen)return;
    clearCommitTimer();
    const epoch=transactionEpoch;
    commitTimer=setTimeout(()=>{
      commitTimer=null;
      if(epoch!==transactionEpoch)return;
      // Pointer capture is occasionally cancelled by Android WebView while a panel is
      // rebuilding. Retry briefly rather than leaving the transaction permanently open.
      if(activePointers.size){commitTransaction(90);return;}
      try{
        // Bypass the transaction-aware wrapper below for this one final snapshot.
        if(typeof originalPushHistory==='function'&&!restoringHistory)originalPushHistory();
      }catch(err){console.warn('Ethvelle history commit:',err);}
      transactionOpen=false;
    },Math.max(0,Number(delay)||0));
  }
  function flushPendingTransaction(){
    clearCommitTimer();
    const hadPending=transactionOpen;
    activePointers.clear();
    transactionOpen=false;

  function v140VideoPreferred(id,time){
    // v279: every active forward-playing video layer is audible, not just the topmost one, so two
    // overlapping copies of a clip are heard together (doubled) instead of one being auto-muted.
    if(!Array.isArray(overlays))return false;
    const t=Number.isFinite(Number(time))?Number(time):Number(currentAbsoluteTimeSec)||0;
    const ov=overlays.find(o=>o&&o.type==='video'&&String(o.id)===String(id));
    return !!(ov&&!ov.reversePlayback&&t>=Number(ov.trimStart||0)&&t<Number(ov.trimEnd||0));
  }
  function v140RampVideoAudible(el,ov){
    if(!el||el.tagName!=='VIDEO')return false;
    const id=String((ov&&ov.id)||el.dataset.evV72OverlayId||'');
    const live=v140LiveOverlay(id,'video');
    if(!live)return false;
    const t=Number(currentAbsoluteTimeSec)||0;
    const active=t>=Number(live.trimStart||0)&&t<Number(live.trimEnd||0);
    if(el.dataset.evV140Priming==='1'&&!active){v140SetVideoGain(el,0,0);return true;}
    const aligned=isPlaying&&!live.reversePlayback&&active&&el.isConnected&&el.readyState>=2&&!el.seeking&&
      el.dataset.evPlayingEventSeen==='1'&&!(evVideoBufferStall&&String(evVideoBufferStall.id)===id);
    // Keep audio attached to a live video element during ordinary mobile decoder drift.
    // The visual compositor already gates presentation separately; pausing the whole video
    // here left playback visibly running after the browser's mute flag had been restored.
    if(!aligned){v140StopVideo(el,true);return false;}
    if(!v140VideoPreferred(id,t)){
      v140SetVideoGain(el,0,0);return false;
    }
    v140MuteOtherVideos(el);
    if(el.__evGainRampRaf)return true;
    const rawTarget=el.dataset.evTargetVolume==null?(live.volume==null?1:Number(live.volume)):Number(el.dataset.evTargetVolume);
    const target=Math.max(0,Math.min(1,Number.isFinite(rawTarget)?rawTarget:1));
    const nodes=v140EnsureVideoAudioNodes(el,id);
    if(el.dataset.evGainRampDone==='1'){
      v140SetVideoGain(el,target,0);return true;
    }
    const start=performance.now();
    if(nodes)v140SetVideoGain(el,0,0);
    else{try{el.volume=0;el.muted=false;el.removeAttribute('muted');}catch(_e){}}
    el.dataset.evGainRampActive='1';
    const step=()=>{
      el.__evGainRampRaf=0;
      const now=Number(currentAbsoluteTimeSec)||0;
      const current=v140LiveOverlay(id,'video');
      if(!current||!isPlaying||now<Number(current.trimStart||0)||now>=Number(current.trimEnd||0)||
         (evVideoBufferStall&&String(evVideoBufferStall.id)===id)||el.seeking||!el.isConnected){v140StopVideo(el,true);return;}
      const amount=Math.min(1,Math.max(0,(performance.now()-start)/110));
      v140SetVideoGain(el,target*amount,0);
      if(amount<1)el.__evGainRampRaf=requestAnimationFrame(step);
      else{delete el.dataset.evGainRampActive;el.dataset.evGainRampDone='1';v140SetVideoGain(el,target,0);}
    };
    el.__evGainRampRaf=requestAnimationFrame(step);
    return true;
  }
  function v140SafePlayVideo(el,ov){
    if(!el||el.tagName!=='VIDEO')return Promise.resolve(false);
    const id=String((ov&&ov.id)||el.dataset.evV72OverlayId||'');
    const live=v140LiveOverlay(id,'video');
    if(!live)return Promise.resolve(false);
    const now=Number(currentAbsoluteTimeSec)||0;
    const active=now>=Number(live.trimStart||0)&&now<Number(live.trimEnd||0);
    if(el.dataset.evV140Priming==='1'&&!active){v140SetVideoGain(el,0,0);return Promise.resolve(false);}
    if(live.reversePlayback||!active||(evVideoBufferStall&&String(evVideoBufferStall.id)===id)){
      v140StopVideo(el,true);return Promise.resolve(false);
    }
    if(el.seeking||el.readyState<2)return Promise.resolve(!el.paused);
    const expected=evOverlayVideoExpectedLocalTime(live,now,el);
    const drift=Math.abs((Number(el.currentTime)||0)-expected);
    const seekAt=Number(el.dataset.evPlaySeekAt)||0;
    const justSeeked=seekAt>0&&performance.now()-seekAt<1500;
    const tolerance=(!el.paused||justSeeked)?EV_VIDEO_ALIGN_TOL_PLAYING:EV_VIDEO_ALIGN_TOL_START;
    if(drift>tolerance){
      v140StopVideo(el,true);
      if(!el.seeking&&el.readyState>=1){
        const latency=Math.max(0,Math.min(1200,Number(el.dataset.evSeekLatencyMs)||0));
        const speed=Math.max(.05,Number(live.speed)||1);
        const maxT=Math.max(.05,(Number(live.sourceDuration)||Number(el.duration)||expected+.05)-.02);
        const target=Math.max(0,Math.min(maxT,expected+(latency/1000)*speed));
        el.dataset.evPreciseSyncTarget=String(target);el.dataset.evSeekRequestAt=String(performance.now());
        try{el.currentTime=target;}catch(_e){delete el.dataset.evPreciseSyncTarget;}
      }
      return Promise.resolve(false);
    }
    delete el.dataset.evPreciseSyncTarget;
    v140EnsureVideoAudioNodes(el,id);v140InstrumentPlay(el,'video',id);
    try{el.muted=false;el.defaultMuted=false;el.removeAttribute('muted');el.volume=1;}catch(_e){}
    if(!el.paused){if(isPlaying&&el.dataset.evPlayingEventSeen==='1')v140RampVideoAudible(el,live);return Promise.resolve(true);}
    if(el.dataset.evV140PlayPending==='1')return el.__evV140PrimePromise||Promise.resolve(false);
    el.dataset.evV140PlayPending='1';
    let nativePromise;
    try{nativePromise=el.play();}catch(err){delete el.dataset.evV140PlayPending;v140ReportPlayError(err,'video',id);return Promise.resolve(false);}
    const attempt=Promise.resolve(nativePromise).then(()=>{
      delete el.dataset.evV140PlayPending;
      const current=v140LiveOverlay(id,'video'),t=Number(currentAbsoluteTimeSec)||0;
      if(current&&isPlaying&&t>=Number(current.trimStart||0)&&t<Number(current.trimEnd||0))v140RampVideoAudible(el,current);
      else v140StopVideo(el,true);
      return true;
    },err=>{delete el.dataset.evV140PlayPending;v140ReportPlayError(err,'video',id);return false;});
    el.__evV140PrimePromise=attempt;
    return attempt;
  }
