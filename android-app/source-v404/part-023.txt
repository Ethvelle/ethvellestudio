    const w=220, h=Math.max(1,Math.round(w*(stageH/stageW)));
    const cover=document.createElement('canvas');cover.width=w;cover.height=h;
    const cx=cover.getContext('2d');
    if(!cx)return false;
    cx.fillStyle='#FFFFFF';cx.fillRect(0,0,w,h);
    const coverCharacters=ethvelleCloneCharactersForFirstFrameCover();
    ethvelleDrawCoverCharacters(cx,coverCharacters,w,h);
    const data=cover.toDataURL('image/jpeg',0.76);
    if(!data || data.length<100)return false;
    project.thumb=data;
    const idx=projects.findIndex(p=>p.id===project.id);
    if(idx>-1)projects[idx]=project;
    saveProjectsToStorage();
    if(document.getElementById('screen-projects')?.classList.contains('active'))renderProjects();
    return true;
  }catch(err){
    console.warn('Ethvelle: first-frame cover capture failed',err);
    return false;
  }
}
function queueProjectFirstFrameCover(delay){
  if(!currentProject || !stageW || !stageH) return;
  const pid=currentProject.id;
  clearTimeout(projectCoverTimer);
  projectCoverTimer=setTimeout(()=>{
    if(!currentProject || currentProject.id!==pid || !document.getElementById('screen-editor')?.classList.contains('active')) return;
    if(isPlaying){ queueProjectFirstFrameCover(1800); return; }
    const project=currentProject;
    captureProjectFirstFrameCoverSync(project);
    generateProjectThumbnail().then(thumb=>{
      // Don't let a slow background-image load from an old project overwrite another project's cover.
      if(!thumb || !currentProject || currentProject.id!==pid || !document.getElementById('screen-editor')?.classList.contains('active')) return;
      saveProjectThumbnailResult(project,thumb);
    });
  },Math.max(0,Number(delay)||0));
}
function saveEditorStateOnLeave(){
  if(!currentProject || !document.getElementById('screen-editor')?.classList.contains('active')) return;
  const now=Date.now();
  if(lastLifecycleCoverProjectId===currentProject.id && now-lastLifecycleCoverAt<700) return;
  lastLifecycleCoverAt=now; lastLifecycleCoverProjectId=currentProject.id;
  clearTimeout(projectCoverTimer);
  try{ persistProjectState({skipHistory:true}); }catch(e){ console.warn('Ethvelle: leave-save state error',e); }
  if(isPlaying) pausePlaybackAtCurrentTime();
  const project=currentProject, pid=currentProject.id;
  captureProjectFirstFrameCoverSync(project);
  generateProjectThumbnail().then(thumb=>{
    if(!thumb || !currentProject || currentProject.id!==pid) return;
    saveProjectThumbnailResult(project,thumb);
  });
}
document.addEventListener('visibilitychange',()=>{if(document.visibilityState==='hidden')saveEditorStateOnLeave();});
window.addEventListener('pagehide',saveEditorStateOnLeave);

function saveTextOverlay(){
  const text=String(document.getElementById('textContentInput')?.value||'').trim();
  if(!text){showToast('Write some text first', {icon:toastIcons.info,duration:1500});return;}
  const family=document.getElementById('textFontFamily')?.value || 'Hind Siliguri';
  ensureAdminFontLoaded(family);
  const size=Math.max(16,Math.min(96,Number(document.getElementById('textFontSize')?.value)||36));
  const color=document.getElementById('textFontColor')?.value || '#ffffff';
  let clip=selectedTextOverlay();
  if(clip){
    clip.textContent=text; clip.name=text.replace(/\s+/g,' ').slice(0,22); clip.fontFamily=family; clip.fontSize=size; clip.textColor=color; clip.fontWeight=700;
  }else{
    const len=5;
    const startT=(typeof window.ethvellePlanTextInsertStart==='function'?window.ethvellePlanTextInsertStart(len):Math.max(0,Number(currentAbsoluteTimeSec)||0));
    clip={id:'ov'+Date.now()+Math.random().toString(36).slice(2,5),name:text.replace(/\s+/g,' ').slice(0,22)||'Text',color:'#454B73',type:'text',textContent:text,fontFamily:family,fontSize:size,textColor:color,fontWeight:700,sourceDuration:len,trimStart:startT,trimEnd:startT+len,thumbs:[],peaks:[],speed:1,transform:{scale:1,offsetX:0,offsetY:0,rotation:0}};
    overlays.push(clip);
    timelineDuration=Math.max(timelineDuration,clip.trimEnd);
  }
  renderOverlayLayers(); syncTimelineInnerWidth(); renderRuler(); selectClipForChrome(clip.id); renderFrame(); closePanel(); persistProjectState();
  showToast(clip.type==='text'?'Text saved':'Text added', {icon:toastIcons.check,duration:1300});
}

function hexToRgb(hex){
  let h=String(hex||'#00ff00').replace('#','').trim();
  if(h.length===3) h=h.split('').map(ch=>ch+ch).join('');
  if(!/^[0-9a-f]{6}$/i.test(h)) h='00ff00';
  return {r:parseInt(h.slice(0,2),16),g:parseInt(h.slice(2,4),16),b:parseInt(h.slice(4,6),16)};
}
function paintChromaKeyFrame(source,canvas,cfg,maxSide){
  const sw=source?.videoWidth||source?.naturalWidth||source?.width||0;
  const sh=source?.videoHeight||source?.naturalHeight||source?.height||0;
  if(!sw||!sh) return false;
  const ratio=Math.min(1,(Number(maxSide)>0?Number(maxSide):Math.max(sw,sh))/Math.max(sw,sh));
  const w=Math.max(1,Math.round(sw*ratio)),h=Math.max(1,Math.round(sh*ratio));
  if(canvas.width!==w) canvas.width=w; if(canvas.height!==h) canvas.height=h;
  const ctx=canvas.getContext('2d',{willReadFrequently:true});
  if(!ctx) return false;
  try{
    ctx.clearRect(0,0,w,h); ctx.drawImage(source,0,0,w,h);
    const frame=ctx.getImageData(0,0,w,h),d=frame.data,key=hexToRgb(cfg?.color);
    const tolerance=Math.max(0,Number(cfg?.tolerance)||75),softness=Math.max(0,Number(cfg?.softness)||0),edge=Math.max(1,softness);
    for(let i=0;i<d.length;i+=4){
      const dr=d[i]-key.r,dg=d[i+1]-key.g,db=d[i+2]-key.b;
      const distance=Math.sqrt(dr*dr+dg*dg+db*db);
      if(distance<=tolerance) d[i+3]=0;
      else if(softness>0 && distance<tolerance+softness) d[i+3]=Math.round(d[i+3]*Math.max(0,Math.min(1,(distance-tolerance)/edge)));
    }
    ctx.putImageData(frame,0,0); return true;
  }catch(err){
    console.warn('Ethvelle: chroma-key pixel processing failed (the video may be cross-origin)',err);
    return false;
  }
}

function animPresetFitsCharacter(preset,character){

function previewAnim(key){

// The toggle switch is the ONLY thing that actually commits a preset to the selected clip
// now (see the .anim-toggle CSS note). Tapping an already-on toggle turns the animation back
// off (the clip reverts to no preset / the character's rest pose) rather than being a one-way
// switch, since a real on/off control should go both ways.
function toggleApplyAnim(key){
  const preset = animPresets[key];
  const character=activeCharacter();
  if(!preset || !character) return;
  const activeType=String(character.modelType||(character.sheet&&character.sheet.modelType)||'humanoid');
  if(!animPresetFitsCharacter(preset,character)){if(typeof showToast==='function')showToast('Select a matching character type for this animation.',{icon:toastIcons.info,duration:1800});return;}
  cancelAnimationPreview(false);
  pausePlayback();

  // Only an explicitly selected main/split animation clip is a toggle target. If the user
  // has selected a background, overlay, or character row, tapping Add must create a new
  // animation clip instead of silently toggling currentAnimKey without changing the clip.
  let target = null;
  if(selectedClip === 'main' && activeClip){
    target = {ref:'main', clip:activeClip};
  } else if(typeof selectedClip === 'string' && selectedClip.startsWith('seg')){
    const idx = parseInt(selectedClip.slice(3),10);
    if(Number.isInteger(idx) && splitStickmanSegments[idx]) target = {ref:'seg'+idx, clip:splitStickmanSegments[idx]};
  } else if(selectedClip == null && activeClip){
    // No explicit selection means the main animation clip is the default target. This avoids
    // an apparent flaky toggle where the list UI showed the main animation as selected but
    // the click instead created a separate clip because selectedClip was null.
    target = {ref:'main', clip:activeClip};
  }
  const currentlyApplied = !!(target && target.clip.animKey === key);
  if(!currentlyApplied && isAnimationLocked(key)){
    openAnimationUpgrade(getAnimationRequiredPlan(key)?.id);
    drawStick();
    renderAnimList();
    return;
  }

  if(currentlyApplied){
    if(Object.prototype.hasOwnProperty.call(target.clip,'__evFaceActionSnapshot')){
      const before=target.clip.__evFaceActionSnapshot;
      if(before===null)delete target.clip.faceActions;else target.clip.faceActions=JSON.parse(JSON.stringify(before));
      delete target.clip.__evFaceActionSnapshot;
    }
    target.clip.animKey = null;
    if(target.ref === 'main') currentAnimKey = null;
    renderTimelineClip();
    renderAnimList();
    // renderFrame restores the authored rest pose when a clip has no active preset, so
    // repeatedly toggling an animation on/off cannot leave the character stuck mid-motion.
    renderFrame();
    showToast(preset.name + ' turned off', {duration:1200});
    persistProjectState();
    return;
  }

  try{const req=getAnimationRequiredPlan(key);trackEvent(Number(req&&req.price)>0?'animation_paid_used':'animation_free_used',{itemLabel:preset.name});}catch(_e){}
  if(target){
    target.clip.animKey = key;
    if(target.ref === 'main') currentAnimKey = key;
    renderTimelineClip();
    renderAnimList();
    seekAbsoluteTimelineSeconds(target.clip.trimStart);
  } else if(activeClip){
    appendAnimationClip(key);
    renderAnimList();
  } else {
    setActiveAnimation(key);
    renderAnimList();
  }
  const committedFaceClip=target?target.clip:activeClip;
  if(preset.faceActions&&committedFaceClip){
    if(!Object.prototype.hasOwnProperty.call(committedFaceClip,'__evFaceActionSnapshot'))committedFaceClip.__evFaceActionSnapshot=committedFaceClip.faceActions?JSON.parse(JSON.stringify(committedFaceClip.faceActions)):null;
    committedFaceClip.faceActions={expression:'none',mouth:'preset',blink:'auto',eyes:'auto',leftHand:'none',rightHand:'none',...preset.faceActions};
  }
  showToast(preset.name + (target ? ' applied' : ' added'), {icon:toastIcons.check,duration:1400});
  // Start at the chosen clip's first frame to make the newly committed animation visible
  // immediately. The row itself only previews; this real button is the commit action.
  renderFrame();
  startPlayback();
  persistProjectState();
}

/* ---------------- Real media generation: video thumbnails + audio waveforms ----------------
   CapCut shows the actual content of a clip in its track (filmstrip of real frames for
   video, a rendered waveform for audio) rather than a flat color block. We generate both
   client-side from the uploaded file so what's shown is genuine, not a placeholder. */

// Captures `count` evenly-spaced frames from a <video> as JPEG data URLs using an offscreen
// canvas. Resolves once all frames are captured (or on failure, resolves with []).
function generateVideoThumbnails(url, count){
  // Kept as a compatibility name for older upload handlers. Video frames are no longer
  // batch-extracted here; the timeline's single lazy decoder owns all video thumbnails.
  return new Promise(resolve=>{
    const video=document.createElement('video');
    let settled=false,timer=0;
    video.muted=true;video.playsInline=true;video.preload='metadata';
    video.setAttribute('playsinline','');
    if(/^https?:/i.test(String(url||'')))video.crossOrigin='anonymous';
    const finish=()=>{
      if(settled)return;settled=true;clearTimeout(timer);
      const result={frames:[],duration:Number(video.duration)||1,width:Number(video.videoWidth)||0,height:Number(video.videoHeight)||0};
      try{video.pause();video.removeAttribute('src');video.load();}catch(_e){}
      resolve(result);
    };
    video.addEventListener('loadedmetadata',finish,{once:true});
    video.addEventListener('error',finish,{once:true});
    timer=setTimeout(finish,15000);
    try{video.src=String(url||'');video.load();}catch(_e){finish();}
  });
}

// Decodes an audio file with Web Audio API and returns { peaks: Float32Array-like downsampled
// min/max pairs, duration } so we can draw a real waveform without keeping the whole PCM buffer.
function generateAudioWaveform(url){
  return fetch(url)
    .then(r=>r.arrayBuffer())
    .then(buf=>{
      const AC = window.AudioContext || window.webkitAudioContext;
      const actx = new AC();
      return actx.decodeAudioData(buf).then(audioBuffer=>{
        const raw = audioBuffer.getChannelData(0);
        const samples = 200; // resolution of the drawn waveform, independent of clip length
        const blockSize = Math.max(1, Math.floor(raw.length/samples));
        const peaks = [];
        for(let i=0;i<samples;i++){
          const start = i*blockSize;
          let min=1, max=-1;
          for(let j=0;j<blockSize && (start+j)<raw.length;j++){
            const v = raw[start+j];
            if(v<min) min=v;
            if(v>max) max=v;
          }
          peaks.push([min,max]);
        }
        const duration = audioBuffer.duration;
        actx.close();
        return { peaks, duration };
      });
    })
    .catch(()=>({ peaks: [], duration: 1 }));
}

  const file = e.target.files[0];
  if(!file) return;
  const isVideo = file.type.startsWith('video/');
  const isAudio = file.type.startsWith('audio/');

  // Standard Android/WebView picker path: persist overlay media as an IndexedDB Blob with a

  const pickedTarget = String(uploadTarget||'');
  const pickedName = String(file.name||'');
  const pickedIsVideo = isVideo || /\.(mp4|m4v|f4v|mov|webm|mkv|avi|3gp|3gpp|3g2|mpeg|mpg|mpe|m2v|ogv|ts|mts|m2ts|wmv|flv)$/i.test(pickedName);
  const pickedIsAudio = isAudio || /\.(mp3|m4a|aac|wav|ogg|opus|flac|aiff?)$/i.test(pickedName);
  const pickedIsImage = /^image\//i.test(file.type||'') || /\.(png|apng|jpe?g|jfif|webp|gif|avif|bmp|dib|heic|heif|tif|tiff|svg|ico|jxl)$/i.test(pickedName);
  if(['overlay','media','audio','replace-overlay'].includes(pickedTarget) &&
     typeof window.__ethvelleStoreMediaFile==='function' && (pickedIsVideo||pickedIsAudio||pickedIsImage)){
    const pickedFile = file;
    const pickedAppendGroup = uploadAppendGroup || null;
    const pickedReplaceId = replaceTargetId;
    e.preventDefault();
    e.stopImmediatePropagation(); // do not let the later legacy video listener import this twice
    closePanel();
    uploadAppendGroup = null;
    replaceTargetId = null;
    e.target.value='';
    (async()=>{
      let mediaKey=null, mediaUrl=null, sessionOnly=false;
      try{
        try{mediaKey = await window.__ethvelleStoreMediaFile(pickedFile);}catch(storageError){
          console.warn('Ethvelle durable media storage unavailable; using a safe import fallback:',storageError);
          if(pickedFile.size<=3*1024*1024){
            mediaUrl=await new Promise((resolve,reject)=>{const reader=new FileReader();reader.onload=()=>resolve(String(reader.result||''));reader.onerror=()=>reject(reader.error||new Error('Could not read the selected media.'));reader.readAsDataURL(pickedFile);});
          }else{mediaUrl=URL.createObjectURL(pickedFile);sessionOnly=true;}
        }
        if(!mediaUrl)mediaUrl = URL.createObjectURL(pickedFile);
        const type = pickedTarget==='audio' ? 'audio' : pickedIsVideo ? 'video' : pickedIsAudio ? 'audio' : 'image';
        let duration = type==='video' ? 12 : 3;
        let thumbs = [], peaks = [], sourceWidth=0, sourceHeight=0;
        if(type==='video'){
          showToast('Preparing video clip…',{icon:toastIcons.info,duration:1200});
          const result = await generateVideoThumbnails(mediaUrl,24,0,null); // metadata only; frame work is lazy
          thumbs = [];
          duration = Math.max(MIN_CLIP_SEC,Number(result&&result.duration)||12);
          sourceWidth=Number(result&&result.width)||0;sourceHeight=Number(result&&result.height)||0;
        }else if(type==='audio'){
          showToast('Preparing audio clip…',{icon:toastIcons.info,duration:1200});
          const result = await generateAudioWaveform(mediaUrl);
          peaks = Array.isArray(result&&result.peaks) ? result.peaks : [];
          duration = Math.max(MIN_CLIP_SEC,Number(result&&result.duration)||3);
        }

        if(pickedTarget==='replace-overlay'){
          const target = overlays.find(o=>String(o.id)===String(pickedReplaceId));
          if(!target) throw new Error('The selected overlay is no longer available.');
          target.type = type;
          target.src = mediaUrl;
          target.deviceMediaKey = mediaKey;
          target.deviceMediaName = pickedFile.name || '';
          target.deviceMediaSize = Number(pickedFile.size)||undefined;
          target.deviceMediaLastModified = Number(pickedFile.lastModified)||undefined;
          target.name = pickedFile.name || (type==='video'?'Video':type==='audio'?'Audio':'Image');
          target.sourceDuration = duration;
          target.sourceOffset = 0;
          target.sourceWidth = sourceWidth;target.sourceHeight = sourceHeight;
          target.thumbs = [];
          target.peaks = peaks;
          target.speed = Number(target.speed)||1;
          target.volume = (type==='video'||type==='audio') ? (target.volume==null?1:target.volume) : undefined;
          if(type==='video'||type==='image') target.transform = target.transform || {scale:1,offsetX:0,offsetY:0,rotation:0};
          else target.transform = null;
          renderOverlayLayers();
          renderOverlayListPanel();
          renderFrame();
          persistProjectState();
        }else{
          addMediaOverlay({type,src:mediaUrl,name:pickedFile.name,duration,thumbs,peaks,sourceWidth,sourceHeight,appendAfterGroup:pickedAppendGroup,deviceMediaKey:mediaKey,deviceMediaInfo:{name:pickedFile.name,size:pickedFile.size,lastModified:pickedFile.lastModified}});
        }
        // Do not leave a project as a lightweight localStorage index until its full, latest
        // IndexedDB snapshot - including the new clip record and stable media key - is committed.
        if(typeof window.__ethvelleDeviceSaveNow==='function') await window.__ethvelleDeviceSaveNow();
        if(sessionOnly){
          showToast('Clip added for this session. Device storage is unavailable, so export a backup before leaving.',{icon:toastIcons.info,duration:3600});
        }else if(!mediaKey){
          showToast('Clip added. This browser saved it in the project data; keep a backup of important work.',{icon:toastIcons.check,duration:2600});
        }else{
          if(pickedTarget==='replace-overlay')showToast('Overlay media replaced and saved',{icon:toastIcons.check,duration:1500});
        }
      }catch(err){
        // Once an overlay references this key, keep both the Blob record and its live object URL
        // even if the full project snapshot write fails. Deleting it here would guarantee that
        // the just-added clip breaks as soon as the user refreshes.
        const alreadyAttached = !!(typeof overlays!=='undefined' && Array.isArray(overlays) && overlays.some(o=>o&&((mediaKey&&String(o.deviceMediaKey||'')===String(mediaKey))||(mediaUrl&&String(o.src||'')===String(mediaUrl)))));
        if(alreadyAttached){
          try{if(typeof persistProjectState==='function')persistProjectState({skipHistory:true});}catch(_e){}
          try{if(typeof window.__ethvelleDeviceSaveNow==='function')await window.__ethvelleDeviceSaveNow();}catch(saveErr){console.error('Ethvelle project snapshot retry:',saveErr);}
          showToast(mediaKey?'Clip is still in the timeline. Its device save did not finish; export a backup before leaving.':'Clip is still in the timeline for this session. Export a backup before leaving because device storage is unavailable.',{icon:toastIcons.info,duration:3800});
        }else{
          if(mediaUrl){try{URL.revokeObjectURL(mediaUrl);}catch(_e){}}
          if(mediaKey&&typeof window.__ethvelleDeleteStoredMediaFile==='function'){
            try{await window.__ethvelleDeleteStoredMediaFile(mediaKey);}catch(_e){}
          }
          showToast(err&&err.message ? err.message : 'Could not save this overlay. Please try again.',{icon:toastIcons.info,duration:2600});
        }
        console.error('Ethvelle persistent overlay import:',err);
      }
    })();
    return;
  }

  if(uploadTarget==='background'){
    // FEATURE (multiple sequential backgrounds): picking an image via the "Add background"
    // badge/empty-state now APPENDS a brand new background clip after the last one already
    // on the row (same "append after existing content" behavior addMediaOverlay already
    // gives Clips/Audio), instead of overwriting the single project-wide background in
    // place. Reads the file into a self-contained base64 data: URL first  -  never a
    // `url(blob:...)` object-URL  -  for the same reason documented on the overlay/media
    // branch below: blob URLs go stale the moment the page is closed/reopened, which used
    // to show up as a broken-image glyph on reload and in exported video.
    // UPDATED: close the Background panel the instant a file is chosen, BEFORE the async
    // FileReader finishes  -  previously it only closed inside onload, so the panel stayed
    // open (visible behind the freshly-added clip) for the whole read/decode. Now the panel
    // is gone immediately and the new 5s background simply appears on the timeline.
    closePanel();
    const reader = new FileReader();
    reader.onload = ()=>{
      addBackgroundClip(`url(${reader.result})`);
      showToast('Background added  -  drag to move, pull the corner to resize', {duration:2200});
    };
    reader.onerror = ()=>{
      showToast("Couldn't read that image", {icon: toastIcons.info, duration:1800});
    };
    reader.readAsDataURL(file);
  } else if(uploadTarget==='replace-background'){
    // Swaps ONLY the selected background clip's own image in place  -  keeps its
    // trimStart/trimEnd/transform untouched, mirroring how 'replace-overlay' below swaps a
    // Clips-row clip's content without moving or resizing it.
    const target = backgrounds.find(b=>b.id===replaceTargetId);
    if(!target){ closePanel(); return; }
    const reader = new FileReader();
    reader.onload = ()=>{
      target.src = `url(${reader.result})`;
      replaceTargetId = null;
      refreshCurrentBgFromPlayhead();
      applyBackground();
      renderBackgroundTrack();
      showToast('Background replaced', {icon: toastIcons.check, duration:1600});
