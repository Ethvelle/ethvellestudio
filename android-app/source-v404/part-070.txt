<script id="ethvelle-v304-caret-retype-js">
(function(){
  'use strict';
  const $=id=>document.getElementById(id);
  let caretEl=null,rafId=0,lastKey='',moInput=null,moHost=null,moHostEl=null,globalsWired=false;

  const composerOn=()=>{const r=$('ethvelleTextComposer');return !!(r&&r.classList.contains('active'));};

  function ensureCaretEl(input){
    if(!caretEl){caretEl=document.createElement('div');caretEl.id='evtcBlueCaret';caretEl.setAttribute('aria-hidden','true');}
    if(caretEl.parentNode!==input.parentNode)input.parentNode.appendChild(caretEl);
    return caretEl;
  }
  function hideCaret(input){
    if(caretEl&&caretEl.style.display!=='none')caretEl.style.display='none';
    if(input&&input.classList.contains('ev-v304-own-caret'))input.classList.remove('ev-v304-own-caret');
  }

  /* 1) The textarea only supplies typing, selection and the on-screen keyboard; the styled preview paints the
        text. Give the textarea the preview's own letter-spacing / line-height / text-transform so both lay the
        text out identically (it used to keep its CSS defaults, so any Spacing / line-height change, or even
        just a big font, pushed the caret away from the text). */
  function syncTypography(input,glyph,host){
    const ls=glyph.style.letterSpacing;
    const want={
      'letter-spacing':(ls&&ls!=='normal')?ls:'0px',
      'line-height':glyph.style.lineHeight||'1.16',
      'text-transform':(host&&host.style.textTransform)||'none'
    };
    for(const k in want){
      if(input.style.getPropertyValue(k)!==want[k]||input.style.getPropertyPriority(k)!=='important')input.style.setProperty(k,want[k],'important');
    }
  }

  /* 2) Caret position measured from the preview's real glyphs, so it stays beside the visible text whatever
        font, size, stroke, spacing, bend or perspective is applied. Returns null when it cannot be measured
        (then the native caret, also blue, is used). */
  /* v305: the outline (stroke), its 3D extrusion and the font size make the visible text wider than its glyph box,
        so a caret placed exactly on the glyph edge touched / hid under the outline. Measure how far the outline
        reaches on screen and keep the caret a small, size-based distance outside of it. */
  function edgePad(host,glyph){
    const gs=getComputedStyle(glyph),fs=parseFloat(gs.fontSize)||40;
    let k=1;                                                 // on-screen scale of the preview (zoom / transform)
    try{if(glyph.offsetWidth>0){const bw=glyph.getBoundingClientRect().width/glyph.offsetWidth;if(isFinite(bw)&&bw>0.05)k=bw;}}catch(_e){}
    let out=0;
    const clone=host.querySelector(':scope > .evpa-stroke-clone');
    if(clone){
      const cs=getComputedStyle(clone);
      const w=(parseFloat(cs.webkitTextStrokeWidth)||0)/2;   // the stroke is painted at 2x width, half of it outside the glyph
      const vis=parseFloat(cs.opacity);
      if(w>0.01&&!(vis===0)){
        out=w;
        if(typeof window.ethvelleStrokeExtrude==='function'&&cs.filter&&cs.filter!=='none'){
          try{out+=window.ethvelleStrokeExtrude(w,1,cs.webkitTextStrokeColor).depth*0.32;}catch(_e){}
        }
      }
    }
    const gap=Math.max(3,Math.min(12,fs*0.07));            // breathing room that grows with the font size
    return (out+gap)*k;
  }

  function caretFromGlyph(input,glyph){
    const val=input.value,idx=input.selectionStart;
    if(!val||idx==null)return null;
    const segs=[];let total=0;
    const walker=document.createTreeWalker(glyph,NodeFilter.SHOW_TEXT);
    for(let n=walker.nextNode();n;n=walker.nextNode()){if(n.data.length){segs.push({n:n,s:total});total+=n.data.length;}}
    if(total!==val.length)return null;                      // preview has not caught up with the text yet
    const rectOf=(a,b)=>{
      for(let i=0;i<segs.length;i++){
        const sg=segs[i];
        if(a>=sg.s&&a<sg.s+sg.n.data.length){
          const rg=document.createRange();
          rg.setStart(sg.n,a-sg.s);rg.setEnd(sg.n,Math.min(b,sg.s+sg.n.data.length)-sg.s);
          const rs=rg.getClientRects();
          return rs.length?rs[rs.length-1]:null;
        }
      }
      return null;
    };
    const pack=(x,top,h,e)=>(isFinite(x)&&isFinite(top)&&isFinite(h)&&h>0)?{x:x,top:top,h:h,e:e||0}:null;   // e: +1 = after the last glyph of a line, -1 = before the first glyph, 0 = between glyphs
    const gs=getComputedStyle(glyph),rtl=gs.direction==='rtl';
    const fs=parseFloat(gs.fontSize)||40;
    const lhPx=parseFloat(gs.lineHeight)||fs*1.16;
    let r;
    if(idx>0){
      if(val.charAt(idx-1)==='\n'){                         // caret sits at the start of a new line
        if(idx<val.length&&val.charAt(idx)!=='\n'){
          r=rectOf(idx,idx+1);
          return r?pack(rtl?r.right:r.left,r.top,r.height,-1):null;
        }
        let k=idx-1,nl=0;while(k>=0&&val.charAt(k)==='\n'){nl++;k--;}
        if(k<0)return null;
        r=rectOf(k,k+1);if(!r)return null;                  // empty line: same column as the line above, one line lower
        const gr=glyph.getBoundingClientRect(),al=getComputedStyle(input).textAlign;
        const l=gr.left+(parseFloat(gs.paddingLeft)||0),rr=gr.right-(parseFloat(gs.paddingRight)||0);
        const x=al==='center'?(l+rr)/2:(al==='right'?rr:l);
        return pack(x,r.top+nl*lhPx,r.height);
      }
      let a=idx-1;
      const lo=val.charCodeAt(idx-1);
      if(lo>=0xDC00&&lo<=0xDFFF&&idx>=2){const hi=val.charCodeAt(idx-2);if(hi>=0xD800&&hi<=0xDBFF)a=idx-2;}
      r=rectOf(a,idx);
      return r?pack(rtl?r.left:r.right,r.top,r.height,(idx>=val.length||val.charAt(idx)==='\n')?1:0):null;
    }
    r=rectOf(0,1);                                          // caret before the first character
    return r?pack(rtl?r.right:r.left,r.top,r.height,-1):null;
  }

  function update(){
    rafId=0;
    const input=$('evtcStageText');
    if(!input)return;
    if(!composerOn()){hideCaret(input);return;}
    const host=$('evtcStagePreview');
    const glyph=(host&&host.style.display!=='none')?host.querySelector(':scope > .evpa-text-effect-layer'):null;
    let pos=null,pad=0,rtlDir=1;
    if(glyph){
      syncTypography(input,glyph,host);
      if(document.activeElement===input&&input.selectionStart===input.selectionEnd){
        try{pos=caretFromGlyph(input,glyph);if(pos&&pos.e){pad=edgePad(host,glyph);rtlDir=getComputedStyle(glyph).direction==='rtl'?-1:1;}}catch(_e){pos=null;}
      }
    }
    if(!pos){hideCaret(input);return;}
    const c=ensureCaretEl(input),h=Math.max(8,pos.h);
    pos.x+=pos.e*rtlDir*pad;
    c.style.left=(pos.x-1)+'px';c.style.top=pos.top+'px';c.style.height=h+'px';
    if(c.style.display!=='block')c.style.display='block';
    const key=Math.round(pos.x*10)+':'+Math.round(pos.top*10)+':'+Math.round(h*10);
    if(key!==lastKey){lastKey=key;c.style.animation='none';void c.offsetWidth;c.style.animation='';}   // solid right after it moves
    if(!input.classList.contains('ev-v304-own-caret'))input.classList.add('ev-v304-own-caret');
  }
  function schedule(){if(!rafId)rafId=requestAnimationFrame(update);}

  function wire(){
    const input=$('evtcStageText');
    if(input&&!input.__ev304){
      input.__ev304=true;
      ['input','keydown','keyup','click','focus','blur','select','pointerup','touchend','compositionupdate','compositionend'].forEach(ev=>input.addEventListener(ev,schedule,{passive:true}));
      if(typeof MutationObserver==='function'){moInput=new MutationObserver(schedule);moInput.observe(input,{attributes:true,attributeFilter:['style','class']});}
    }
    const host=$('evtcStagePreview');
    if(host&&host!==moHostEl&&typeof MutationObserver==='function'){

    }
    if(!globalsWired){
      globalsWired=true;
      document.addEventListener('selectionchange',schedule,{passive:true});
      window.addEventListener('resize',schedule,{passive:true});
      if(window.visualViewport){window.visualViewport.addEventListener('resize',schedule,{passive:true});window.visualViewport.addEventListener('scroll',schedule,{passive:true});}
      if(document.fonts&&document.fonts.addEventListener)document.fonts.addEventListener('loadingdone',schedule);
      /* Safety net: re-measure a few times a second while the writing screen is open. */
      setInterval(function(){if(composerOn()){wire();schedule();}},160);
    }
  }

  /* ---- Retype: reopen the writing screen for the selected text layer from the text-editing bottom toolbar ---- */
  const RETYPE_ICON='<svg viewBox="0 0 24 24" aria-hidden="true"><path d="M12 20h9"/><path d="M16.5 3.5a2.1 2.1 0 0 1 3 3L8 18l-4 1 1-4Z"/></svg>';

  }
  wrapAfter('openEthvelleTextComposer',function(){wire();schedule();});

  wrapAfter('openEthvelleTextStylePanel',ensureRetypeTab);
  wrapAfter('ethvellePicsartSetTool',ensureRetypeTab);

  if(document.readyState==='loading')document.addEventListener('DOMContentLoaded',function(){ensureRetypeTab();wire();},{once:true});
  window.addEventListener('load',function(){ensureRetypeTab();},{once:true});
