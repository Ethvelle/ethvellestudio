    return el.classList.contains('evpa-color-wheel')||el.classList.contains('evpa-transparent')||

  function restoreSwatchChildren(swatches){
    if(!swatches)return;

    const grid=swatches.querySelector(':scope > .evpa-color-grid, :scope > .evpa-font-color-row');

  function layoutColorTab(swatches){
    if(swatches.querySelector(':scope > .evpa-color-utilities')&&swatches.querySelector(':scope > .evpa-color-grid')){
      const u=swatches.querySelector(':scope > .evpa-color-utilities');
      const g=swatches.querySelector(':scope > .evpa-color-grid');
      if(u&&g&&u.nextElementSibling!==g){swatches.insertBefore(u,g);}

    if(!kids.length)return;

    const grid=document.createElement('div');grid.className='evpa-color-grid';grid.setAttribute('aria-label','Color swatches');
    kids.forEach(el=>{if(el.classList.contains('evpa-swatch')&&el.hasAttribute('data-color'))grid.appendChild(el);else utilities.appendChild(el);});

  function layoutFontTab(swatches){

    const fontMode=!dock.classList.contains('evpa-in-composer')&&tab==='font';
    const bottomMode=!dock.classList.contains('evpa-in-composer')&&!dock.classList.contains('evpa-no-palette')&&['highlight','stroke','shadow'].includes(tab);
    dock.classList.toggle('evpa-color-grid-mode',colorMode);

    if(!swatches)return;
    if(colorMode){layoutColorTab(swatches);}
    else if(fontMode){layoutFontTab(swatches);}

    else {restoreSwatchChildren(swatches);}

  /* Thumb-only ranges: no overlay layers. Reject track taps; allow thumb drags only. */
  function geometry(input){
    const r=input.getBoundingClientRect();
    let thumbW=0;
    try{thumbW=parseFloat(getComputedStyle(input,'::-webkit-slider-thumb').width)||0;}catch(_e){}
    if(!thumbW)try{thumbW=parseFloat(getComputedStyle(input,'::-moz-range-thumb').width)||0;}catch(_e){}
    const half=Math.max(7,Math.min(16,(thumbW||18)/2));
    const cs=getComputedStyle(input);
    const padLeft=parseFloat(cs.paddingLeft)||0,padRight=parseFloat(cs.paddingRight)||0;
    const trackLeft=r.left+padLeft+half;
    const trackRight=Math.max(trackLeft+1,r.right-padRight-half);
    const min=Number.isFinite(parseFloat(input.min))?parseFloat(input.min):0;
    const max=Number.isFinite(parseFloat(input.max))?parseFloat(input.max):100;
    const value=Number.isFinite(parseFloat(input.value))?parseFloat(input.value):min;
    let frac=max===min?0:clamp((value-min)/(max-min),0,1);
    const rtl=cs.direction==='rtl';
    if(rtl)frac=1-frac;
    return {r,trackLeft,trackRight,min,max,step:input.step==='any'?0:(Number(input.step)||1),thumbX:trackLeft+frac*(trackRight-trackLeft),thumbY:r.top+r.height/2,half,rtl};
  }
  function nearThumb(input,x,y,g){
    g=g||geometry(input);
    // Generous finger hit-area around the visible circular thumb, never on the rest of the rail.
    const hitX=Math.max(9,Math.min(12,g.half+2));
    const hitY=Math.max(10,Math.min(13,g.r.height/2+3));
    return Math.abs(x-g.thumbX)<=hitX && Math.abs(y-g.thumbY)<=hitY;
  }
  function dispatch(input,type){input.dispatchEvent(new Event(type,{bubbles:true}));}
  function setFromX(input,clientX,g){
    g=g||geometry(input);
    let frac=clamp((clientX-g.trackLeft)/(g.trackRight-g.trackLeft),0,1);
    if(g.rtl)frac=1-frac;
    let value=g.min+frac*(g.max-g.min);
    if(g.step>0)value=g.min+Math.round((value-g.min)/g.step)*g.step;
    value=clamp(value,g.min,g.max);
    const decimals=(String(input.step||'1').split('.')[1]||'').length;
    const next=String(Number(value.toFixed(Math.min(7,decimals||0))));
    if(input.value!==next){input.value=next;dispatch(input,'input');}
  }
  function isEditorRange(el){return !!(el&&el.type==='range'&&el.closest&&el.closest(RANGE_ROOTS));}
  let drag=null,dragRaf=0;
  function stopPointer(e){
    if(e.cancelable)e.preventDefault();
    e.stopPropagation();
    if(e.stopImmediatePropagation)e.stopImmediatePropagation();
  }
  function paintPendingDrag(){
    dragRaf=0;
    const d=drag;
    if(!d||!d.moved)return;
    setFromX(d.input,d.lastX-d.pointerOffset,d.geometry);
  }
  function flushDrag(cancelled){
    const d=drag;if(!d)return;
    if(dragRaf){cancelAnimationFrame(dragRaf);dragRaf=0;}
    drag=null;
    if(cancelled){
      if(d.input.value!==d.startValue){d.input.value=d.startValue;dispatch(d.input,'input');dispatch(d.input,'change');}
    }else if(d.moved){
      setFromX(d.input,d.lastX-d.pointerOffset,d.geometry);
      dispatch(d.input,'change');
    }
    try{d.input.releasePointerCapture(d.pointerId);}catch(_e){}
  }
  function onPointerDown(e){
    const input=e.target;
    if(!isEditorRange(input)||input.disabled||input.matches(':disabled'))return;
    if(e.pointerType==='mouse'&&e.button!==0)return;
    const g=geometry(input);
    if(!nearThumb(input,e.clientX,e.clientY,g)){
      stopPointer(e);return;
    }
    drag={pointerId:e.pointerId,input,geometry:g,startX:e.clientX,startY:e.clientY,lastX:e.clientX,
      pointerOffset:e.clientX-g.thumbX,startValue:String(input.value),moved:false,downEvent:e,startedInline:false};
    try{input.setPointerCapture(e.pointerId);}catch(_e){}
    stopPointer(e);
  }
  function onPointerMove(e){
    const d=drag;if(!d||d.pointerId!==e.pointerId)return;
    stopPointer(e);
    d.lastX=e.clientX;
    if(!d.moved&&Math.hypot(e.clientX-d.startX,e.clientY-d.startY)<2)return;
    d.moved=true;
    // Preserve the legacy Position-slider start hook, but run it only once a real drag begins.
    if(!d.startedInline){
      d.startedInline=true;
      try{if(typeof d.input.onpointerdown==='function')d.input.onpointerdown.call(d.input,d.downEvent);}catch(err){console.warn('Slider start hook:',err);}
    }
    if(!dragRaf)dragRaf=requestAnimationFrame(paintPendingDrag);
  }
  function onPointerUp(e,cancelled){
    if(!drag||drag.pointerId!==e.pointerId)return;
    stopPointer(e);
    if(Number.isFinite(e.clientX))drag.lastX=e.clientX;
    flushDrag(!!cancelled);
  }
