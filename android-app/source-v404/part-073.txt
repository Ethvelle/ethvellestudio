    if(!targets.length){
      if(B.scope === 'sel'){ try{ showToast('Select a part first, or switch to All parts', { duration:1800 }); }catch(_e){} }
      S = { targets:[], last:null }; return;
    }
    S = { targets, last:null };
  }
  function cancelStroke(){
    if(!S) return;
    S.targets.forEach(T=>{ if(T.changed && T.ctx) T.ctx.putImageData(new ImageData(new Uint8ClampedArray(T.orig), T.art.w, T.art.h), 0, 0); });
    S = null; edDraw();
  }
  function refreshArt(art, x, y, img){
    art.src = null; art.thumb = null;                    // forces Done to re-encode this part from the cleaned canvas
    if(art.alpha) for(let row = 0; row < img.height; row++) for(let col = 0; col < img.width; col++) art.alpha[(y+row)*art.w + x+col] = img.data[(row*img.width+col)*4+3] > 30 ? 1 : 0;
  }
  function commit(){
    const st = S; S = null; if(!st) return;
    const ts = st.targets.filter(T=>T.changed); if(!ts.length) return;
    edPush();                                            // snapshot of the state BEFORE this stroke (tagged with the brush versions below)
    ts.forEach(T=>{
      const x = T.x0, y = T.y0, w = T.x1-T.x0+1, h = T.y1-T.y0+1, aw = T.art.w;
      const before = T.ctx.createImageData(w, h), after = T.ctx.getImageData(x, y, w, h);
      for(let row = 0; row < h; row++){ const so = ((y+row)*aw+x)*4; before.data.set(T.orig.subarray(so, so+w*4), row*w*4); }
      const v = (ED.pxVer[T.ak]||0)+1;
      (ED.pxPatch[T.ak] = ED.pxPatch[T.ak] || {})[v] = { x, y, w, h, before, after };
      ED.pxVer[T.ak] = v;
      refreshArt(T.art, x, y, after);
    });
    edAfterChange(true);
  }

  /* ---------- undo / redo: keep brush versions in step with the editor's own history ---------- */
  const tag = (j, ver)=>{ try{ return j.slice(0, -1) + ',"px":' + JSON.stringify(ver || {}) + '}'; }catch(_e){ return j; } };
  const readPx = j=>{ try{ return JSON.parse(j).px || null; }catch(_e){ return null; } };
  function applyPx(target){
    if(!target) return;
    const keys = new Set(Object.keys(target).concat(Object.keys(ED.pxVer)));
    keys.forEach(ak=>{
      const art = ED.arts[ak]; if(!art || !art.canvas) return;
      const ctx = art.canvas.getContext('2d'), patches = ED.pxPatch[ak] || {}, t = target[ak] || 0; let c = ED.pxVer[ak] || 0;
      while(c > t){ const p = patches[c]; if(p){ ctx.putImageData(p.before, p.x, p.y); refreshArt(art, p.x, p.y, p.before); } c--; }
      while(c < t){ const p = patches[c+1]; if(p){ ctx.putImageData(p.after, p.x, p.y); refreshArt(art, p.x, p.y, p.after); } c++; }
      if(c) ED.pxVer[ak] = c; else delete ED.pxVer[ak];
    });
  }
  const oPush = window.edPush, oUndo = window.edUndo, oRedo = window.edRedo;
  window.edPush = function(){
    const r = oPush.apply(this, arguments);
    const i = ED.undo.length-1; if(i >= 0) ED.undo[i] = tag(ED.undo[i], ED.pxVer);
    return r;
  };

    const cur = Object.assign({}, ED.pxVer), target = readPx(ED.undo[ED.undo.length-1]);
    const r = oUndo.apply(this, arguments);

    if(target){ applyPx(target); edAfterChange(true); }
    return r;
  };
  window.edRedo = function(){
    if(!ED.redo.length) return oRedo.apply(this, arguments);
    const cur = Object.assign({}, ED.pxVer), target = readPx(ED.redo[ED.redo.length-1]);
    const r = oRedo.apply(this, arguments);
    if(ED.undo.length) ED.undo[ED.undo.length-1] = tag(ED.undo[ED.undo.length-1], cur);
    if(target){ applyPx(target); edAfterChange(true); }
    return r;
  };

  /* ---------- canvas gestures (only while the brush is on) ---------- */
  function bDown(e){
    try{ ED.cv.setPointerCapture(e.pointerId); }catch(_e){}
    const p = pos(e); bp.set(e.pointerId, p); B.pt = p; B.show = true;
    if(e.pointerType === 'mouse' && (e.button === 1 || e.button === 2)){ cancelStroke(); pinch = { type:'pan', p0:p, vx:ED.cam.x, vy:ED.cam.y }; return; }
    if(bp.size >= 2){
      cancelStroke(); const [a, b] = [...bp.values()];
      pinch = { type:'pinch', d:Math.hypot(a.x-b.x, a.y-b.y)||1, z:ED.cam.z, c:{ x:(a.x+b.x)/2, y:(a.y+b.y)/2 }, vx:ED.cam.x, vy:ED.cam.y }; return;
    }
    if(pinch) return;
    startStroke(); addPoint(p);
  }
  function bMove(e){
    const p = pos(e);
    if(!bp.has(e.pointerId)){ if(e.pointerType === 'mouse'){ B.pt = p; B.show = true; edDraw(); } return; }
    bp.set(e.pointerId, p); B.pt = p;
    if(pinch){
      if(pinch.type === 'pan'){ ED.cam.x = pinch.vx+p.x-pinch.p0.x; ED.cam.y = pinch.vy+p.y-pinch.p0.y; edDraw(); return; }
      if(bp.size < 2) return;
      const [a, b] = [...bp.values()], d = Math.hypot(a.x-b.x, a.y-b.y), c = { x:(a.x+b.x)/2, y:(a.y+b.y)/2 };
      const z = Math.max(0.08, Math.min(8, pinch.z*d/pinch.d)), k = z/pinch.z;
      ED.cam.z = z; ED.cam.x = c.x-(pinch.c.x-pinch.vx)*k; ED.cam.y = c.y-(pinch.c.y-pinch.vy)*k; edSyncZoom(); edDraw(); return;
    }
    if(S) addPoint(p);
  }
  function bUp(e){
    bp.delete(e.pointerId);
    if(!bp.size){ commit(); pinch = null; if(e.pointerType !== 'mouse') B.show = false; edDraw(); }
  }
  function hookCanvas(){
    const cv = ED.cv; if(!cv || cv.__v374) return; cv.__v374 = true;
    const oD = cv.onpointerdown, oM = cv.onpointermove, oU = cv.onpointerup;
    cv.onpointerdown = function(e){ return B.on ? bDown(e) : (oD && oD.call(this, e)); };
    cv.onpointermove = function(e){ return (B.on || bp.size) ? bMove(e) : (oM && oM.call(this, e)); };
    cv.onpointerup = cv.onpointercancel = function(e){ return (B.on || bp.has(e.pointerId)) ? bUp(e) : (oU && oU.call(this, e)); };
    cv.addEventListener('pointerleave', ()=>{ if(B.on && !bp.size){ B.show = false; edDraw(); } });
  }

  /* ---------- brush ring + lifecycle hooks ---------- */
  const oDraw = window.edDraw;
  window.edDraw = function(){
    const r = oDraw.apply(this, arguments);
    if(B.on && B.pt && B.show && ED.g){
      const g = ED.g; g.save(); g.setTransform(ED.dpr, 0, 0, ED.dpr, 0, 0);
      g.beginPath(); g.arc(B.pt.x, B.pt.y, B.size/2, 0, Math.PI*2);
      g.lineWidth = 2.5; g.strokeStyle = 'rgba(0,0,0,.6)'; g.stroke(); g.lineWidth = 1.25; g.strokeStyle = '#fff'; g.stroke(); g.restore();
    }
    return r;
  };
  const oOpen = window.edOpen;
  window.edOpen = function(){
    if(B.on) setBrush(false);
    const r = oOpen.apply(this, arguments);
    ED.pxVer = {}; ED.pxPatch = {}; S = null; bp.clear(); pinch = null;
    hookCanvas(); syncPanel();
    return r;
  };
  const oClose = window.edClose;
  window.edClose = function(){ if(B.on) setBrush(false); return oClose.apply(this, arguments); };
  if(ED.cv) hookCanvas();
  syncPanel();
})();
</script>

     one-time top-up wallet field, so do not invent a second balance). */
  function creditFigures(){
    let available=0,purchased=0;
    try{available=Math.max(0,Number(typeof myCreditsBalance!=='undefined'?myCreditsBalance:0)||0);}catch(_e){}
    try{
      const owned=(typeof currentUser!=='undefined'&&currentUser&&typeof getMyOwnedPlansForBilling==='function')?getMyOwnedPlansForBilling():[];
      purchased=typeof getMyCombinedCreditAllowance==='function'
        ?Math.max(0,Number(getMyCombinedCreditAllowance(owned))||0):0;
    }catch(_e){}
    return {available,purchased};
  }
  function paintHomeCreditPair(){
    const plan=$('homeAccountPlan'),balance=$('homeAccountBalance'),button=$('homeAccountSummary');
    if(plan)plan.style.setProperty('display','none','important');
    if(!balance||!button)return;
    const signedIn=!!(typeof currentUser!=='undefined'&&currentUser);
    if(!signedIn){
      balance.classList.remove('ev-v394-credit-loading');
      balance.textContent='Sign in for credits';
      button.setAttribute('aria-label','Sign in to view available and purchased credits');
      return;
    }
    if(window.__ethvelleCreditsLoading){
      balance.classList.add('ev-v394-credit-loading');
      balance.textContent='Loading credits';
      button.setAttribute('aria-label','Loading available and purchased credits');
      return;
    }
    balance.classList.remove('ev-v394-credit-loading');
    const values=creditFigures();
    balance.innerHTML='<span class="ev-v394-credit-pair">'
      +'<b class="ev-v394-available-value">'+values.available.toLocaleString()+'</b>'
      +'<span class="ev-v394-credit-slash" aria-hidden="true">/</span>'

    fitNumbers();
  }
