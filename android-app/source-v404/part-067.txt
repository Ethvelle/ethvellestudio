<script id="ethvelle-v255-anim-clip-name-fit-js">
(function(){
  'use strict';
  const BASE_FS=10.5;      /* normal one-line label size used everywhere else */
  const STACK_MAX_FS=9.5;  /* biggest size allowed for the two-line stacked label */
  const STACK_MIN_FS=4.5;  /* below this, words are replaced by initials */
  const MIN_FS=3.2;        /* absolute floor  -  keeps a sliver of the name visible */
  const SELECTOR='#timelineInner .anim-clip .anim-clip-label';
  const FALLBACK_NAME='Animation';

  let ctx=null,family='';
  const widths=new Map();
  function setFamily(label){
    try{const f=getComputedStyle(label).fontFamily;if(f&&f!==family){family=f;widths.clear();}}catch(_e){}
  }
  /* Width of `text` at BASE_FS; scaled linearly for other sizes. */
  function measure(text){
    let w=widths.get(text);
    if(w!==undefined)return w;
    if(!ctx){const c=document.createElement('canvas');ctx=c.getContext&&c.getContext('2d');}
    if(ctx){ctx.font='700 '+BASE_FS+'px '+(family||'sans-serif');w=ctx.measureText(text).width;}
    else w=text.length*BASE_FS*0.58;
    widths.set(text,w);
    return w;
  }
  const clamp=(v,a,b)=>Math.max(a,Math.min(b,v));

  /* Pure decision logic  -  given the name and the clip's pixel width, choose how to draw it. */
  function plan(full,clipW){
    const gutter=clamp(clipW*0.2,1.5,26);        /* keeps clear of the trim grips on wide clips */
    const avail=Math.max(2,clipW-gutter*2);
    const words=full.split(/\s+/).filter(Boolean);
    const wFull=measure(full)+1;

    if(wFull<=avail)return{mode:'single',fs:BASE_FS,lines:[full],avail};

    /* Stage 2: still room for a recognisable chunk -> cut with "..." */
    if(avail>=Math.max(wFull*0.62,28))return{mode:'ellipsis',fs:BASE_FS,lines:[full],avail};

    if(words.length>=2){
      /* Stage 3: first word above, last word below, font shrinks to fit */
      const a=words[0],b=words[words.length-1];
      const wMax=Math.max(measure(a),measure(b))+1;
      const fs=Math.min(STACK_MAX_FS,BASE_FS*avail/wMax);
      if(fs>=STACK_MIN_FS)return{mode:'stack',fs,lines:[a,b],avail};
      /* Stage 4: initials, still stacked */
      const ia=Array.from(a)[0]||'',ib=Array.from(b)[0]||'';
      const wI=Math.max(measure(ia),measure(ib))+1;
      return{mode:'stack-tiny',fs:clamp(BASE_FS*avail/wI,MIN_FS,9),lines:[ia,ib],avail};
    }
    /* One-word name: shrink the font to fit, then fall back to its first letter */
    const fs=BASE_FS*avail/wFull;
    if(fs>=STACK_MIN_FS)return{mode:'shrink',fs,lines:[full],avail};
    const i=Array.from(full)[0]||'';
    return{mode:'single-tiny',fs:clamp(BASE_FS*avail/(measure(i)+1),MIN_FS,9),lines:[i],avail};
  }

  function apply(label){
    const clip=label.closest('.anim-clip');if(!clip)return;
    const w=clip.offsetWidth||parseFloat(clip.style.width)||0;
    if(w<=0)return;
    setFamily(label);

    /* Work out the real (full) name, even if this label was already rewritten by us. */
    const mine=label.dataset.ev255Shown;
    let full;
    if(mine!==undefined&&label.textContent===mine)full=label.dataset.ev255Full||'';
    else full=(label.textContent||'').replace(/\s+/g,' ').trim();
    if(!full)full=FALLBACK_NAME;                    /* never blank */

    const p=plan(full,w);
    const sig=p.mode+'|'+p.fs.toFixed(1)+'|'+Math.round(p.avail)+'|'+p.lines.join('/')+'|'+full;
    if(label.dataset.ev255Sig===sig)return;

    label.textContent='';
    if(p.mode==='stack'||p.mode==='stack-tiny'){
      p.lines.forEach(t=>{const s=document.createElement('span');s.className='ev255-line';s.textContent=t;label.appendChild(s);});
    }else label.textContent=p.lines[0];

    label.classList.toggle('ev255-stack',p.mode==='stack'||p.mode==='stack-tiny');
    label.classList.toggle('ev255-tiny',p.mode==='stack-tiny'||p.mode==='single-tiny');
    label.style.setProperty('--ev255-max',p.avail.toFixed(1)+'px');
    label.style.setProperty('--ev255-fs',p.fs.toFixed(2)+'px');
    label.dataset.ev255Mode=p.mode;
    label.dataset.ev255Full=full;
