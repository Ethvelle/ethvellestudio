  function previewClipAnimationWindow(ov, category){
    // This preview belongs to an image/video/text clip, not to the character rig. A
    // media-only project therefore uses the same clock and compositor without a character.
    if(!ov || !isVisualModel(ov)) return;

    const start=Math.max(0,Number(ov.trimStart)||0);
    const end=Number.isFinite(Number(ov.trimEnd))?Number(ov.trimEnd):start;
    const len=Math.max(0,end-start);
    if(len<0.05) return;

    let from=start, to=end;
    if(category==='in'){
      // In: preview exactly from the clip's first frame for the chosen In duration.
      const span=Math.min(len,clamp(c.inDuration==null?0.45:c.inDuration,0.1,2.5));
      from=start;
      to=start+span;
    }else if(category==='out'){
      // Out: preview the final part of the clip. Example: a 5s clip / 1s Out = 4s to 5s.
      // v362: play the TAIL of the clip once: ~1s of normal playback, then the Out animation to the last frame.
      const span=Math.min(len,clamp(c.outDuration==null?0.45:c.outDuration,0.1,2.5));
      const lead=Math.min(1.0,Math.max(0,len-span));
      from=end-span-lead;
      to=end;
    }else{
      const cycle=clamp(c.overallDuration==null?1.2:c.overallDuration,0.25,4);
      to=Math.min(end,start+Math.max(0.65,cycle));
    }

    to=Math.max(from+0.04,Math.min(end,to));

    // Keep the shared timeline long enough even in projects with no character at all.
    if(typeof timelineDuration!=='undefined') timelineDuration=Math.max(Number(timelineDuration)||0,end);

    if(isPlaying) pausePlayback();
    if(clipAnimationPreviewRaf) cancelAnimationFrame(clipAnimationPreviewRaf);
    clipAnimationPreviewRaf=0;
    const token=++clipAnimationPreviewToken;

    seekAbsoluteTimelineSeconds(from);
    isPlaying=true;
    playbackStartedAtMs=performance.now();
    playbackStartTimeSec=from;
    const icon=document.getElementById('playIcon');
    if(icon) icon.innerHTML='<path d="M6 5h4v14H6zM14 5h4v14h-4z"/>';

      if(token!==clipAnimationPreviewToken || !isPlaying) return;

      if(t>=to){
