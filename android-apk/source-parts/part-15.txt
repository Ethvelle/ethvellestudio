<script id="ethvelle-v221-pitch-safe-speed-js">
/* Ethvelle v221: pitch-safe clip speed.
   Speeding up or slowing down a clip used to play its audio buffer with a faster/slower
   playbackRate, which also raises/lowers the pitch. This module time-stretches the decoded audio
   (WSOLA: waveform-similarity overlap-add) so the speed changes but the pitch stays the same.
   Live playback and export both use it. Results are cached, computed in small async slices so
   the UI never freezes, and stale jobs are dropped when the speed changes again. */
(function(){
  'use strict';
  if(window.ethvellePitchSafeStretch)return;
  const MAX_ENTRIES=5,FULL_BUFFER_MAX_SEC=150,PAD_SEC=4,MAX_RANGE_SEC=900;
  const cache=[];
  const nextSlice=()=>new Promise(r=>setTimeout(r,0));

  /* Pure WSOLA core: channels = Float32Array[], returns Float32Array[] of length round(inLen/speed). */
  async function wsola(chans,sr,speed,shouldAbort){
    const nch=chans.length,inLen=chans[0].length,D=4;
    let N=Math.round(sr*0.046/(2*D))*(2*D);N=Math.max(256,N);
    const Hs=N/2,Ha=Hs*speed,tol=Math.max(D*8,Math.round(sr*0.012/D)*D);
    const outLen=Math.max(1,Math.round(inLen/speed));
    const out=chans.map(()=>new Float32Array(outLen+N+4));
    const decLen=Math.floor(inLen/D);
    const dec=new Float32Array(decLen+N/D+8);
    for(let i=0;i<decLen;i++){let s=0;const b=i*D;for(let c=0;c<nch;c++){const ch=chans[c];s+=ch[b]+ch[b+1]+ch[b+2]+ch[b+3];}dec[i]=s/(4*nch);}
    const win=new Float32Array(N);for(let i=0;i<N;i++)win[i]=0.5-0.5*Math.cos(2*Math.PI*i/N);
    const L=Hs/D,frames=Math.ceil(outLen/Hs)+1,ch0=chans[0];
    let prev=0,lastYield=(typeof performance!=='undefined'?performance.now():Date.now());
    for(let k=0;k<frames;k++){
      let pos=0;
      if(k>0){
        const target=Math.round(k*Ha);
        pos=Math.min(target,Math.max(0,inLen-1));
        const refStart=prev+Hs,r0=Math.floor(refStart/D);
        if(r0+L<=decLen){
          let lo=Math.max(0,target-tol),hi=Math.min(target+tol,inLen-1);
          lo=Math.ceil(lo/D)*D;
          let best=-Infinity,bestP=pos;
          for(let p=lo;p<=hi;p+=D){
            const q=p/D;if(q+L>decLen)break;
            let dot=0,en=0;
            for(let j=0;j<L;j++){const a=dec[q+j];dot+=a*dec[r0+j];en+=a*a;}
            const score=dot/Math.sqrt(en+1e-6);
            if(score>best){best=score;bestP=p;}
          }
          /* fine alignment at full resolution (channel 0, +-D samples) */
          let fineBest=-Infinity,fineP=bestP;
          if(refStart+Hs<=inLen){
            for(let p=Math.max(0,bestP-D);p<=Math.min(hi,bestP+D);p++){
              if(p+Hs>inLen)break;
              let dot=0,en=0;
              for(let j=0;j<Hs;j+=2){const a=ch0[p+j];dot+=a*ch0[refStart+j];en+=a*a;}
              const score=dot/Math.sqrt(en+1e-6);
              if(score>fineBest){fineBest=score;fineP=p;}
            }
          }
          pos=fineP;
        }
      }
      const base=k*Hs;
      for(let c=0;c<nch;c++){
        const ch=chans[c],o=out[c];
        for(let i=0;i<N;i++){
          const idx=pos+i;if(idx>=inLen)break;
          const w=(k===0&&i<Hs)?1:win[i];
          o[base+i]+=ch[idx]*w;
        }
      }
      prev=pos;
      if((k&15)===15){
        const now=(typeof performance!=='undefined'?performance.now():Date.now());
        if(now-lastYield>14){await nextSlice();lastYield=(typeof performance!=='undefined'?performance.now():Date.now());if(shouldAbort&&shouldAbort())return null;}
      }
    }
    return out.map(o=>o.subarray(0,outLen));
  }

  function makeBuffer(nch,len,sr){
    try{return new AudioBuffer({numberOfChannels:nch,length:len,sampleRate:sr});}
    catch(_e){const C=window.OfflineAudioContext||window.webkitOfflineAudioContext;return new C(nch,Math.max(1,len),sr).createBuffer(nch,len,sr);}
  }
  function liveWanted(entry){
    try{
      if(typeof overlays==='undefined'||!Array.isArray(overlays))return true;
      return overlays.some(o=>o&&(o.type==='video'||o.type==='audio')&&Math.abs((Number(o.speed)||1)-entry.speed)<.0005);
    }catch(_e){return true;}
  }
  async function runJob(entry,transient){
    const buf=entry.src,sr=buf.sampleRate;
    const s0=Math.max(0,Math.floor(entry.start*sr)),s1=Math.min(buf.length,Math.ceil(entry.end*sr));
    if(s1-s0<64){entry.failed=true;return null;}
    const chans=[];for(let c=0;c<buf.numberOfChannels;c++)chans.push(buf.getChannelData(c).subarray(s0,s1));
    const res=await wsola(chans,sr,entry.speed,transient?null:()=>entry.dead||!liveWanted(entry));
    if(!res){entry.dead=true;return null;}
    const out=makeBuffer(res.length,res[0].length,sr);
    for(let c=0;c<res.length;c++){if(out.copyToChannel)out.copyToChannel(res[c],c,0);else out.getChannelData(c).set(res[c]);}
    entry.out=out;entry.originSec=s0/sr;entry.start=s0/sr;entry.end=s1/sr;
    return entry;
  }
  let queue=Promise.resolve();
  function enqueue(entry,transient){
    const job=()=>runJob(entry,transient).catch(err=>{entry.failed=true;try{console.warn('Ethvelle pitch-safe stretch failed:',err);}catch(_e){}return null;});
    if(transient)return job();
    entry.promise=queue.then(()=>{if(entry.dead||!liveWanted(entry)){entry.dead=true;return null;}return job();});
    queue=entry.promise.catch(()=>null);
    return entry.promise;
  }
  function evict(){
    cache.sort((a,b)=>b.last-a.last);
    while(cache.length>MAX_ENTRIES){const i=cache.findIndex((e,idx)=>idx>=MAX_ENTRIES-1&&(e.out||e.dead||e.failed));if(i<0)break;cache.splice(i,1);}
    for(let i=cache.length-1;i>=0;i--){if(cache[i].dead)cache.splice(i,1);}
  }
  function plan(buf,startSec,endSec){
    const total=buf.duration;
    if(total<=FULL_BUFFER_MAX_SEC)return {start:0,end:total};
    return {start:Math.max(0,startSec-PAD_SEC),end:Math.min(total,endSec+PAD_SEC)};
  }
  function find(buf,speed,a,b){
    return cache.find(e=>!e.dead&&!e.failed&&e.src===buf&&Math.abs(e.speed-speed)<.0005&&e.start<=a+.002&&e.end>=b-.002);
  }
  function toResult(e){return {buffer:e.out,originSec:e.originSec};}
  /* Synchronous lookup for the live scheduler: result object when ready, null while computing,
     false when this range cannot be stretched (caller then uses the legacy playbackRate path). */
  function getReady(buf,speed,startSec,endSec){
    if(!buf||!(speed>0))return false;
    const a=Math.max(0,startSec),b=Math.min(buf.duration,Math.max(a+.01,endSec));
    let e=find(buf,speed,a,b);
    if(e){e.last=Date.now();return e.out?toResult(e):null;}
    if(b-a>MAX_RANGE_SEC)return false;
    const p=plan(buf,a,b);
    e={src:buf,speed,start:p.start,end:p.end,out:null,promise:null,last:Date.now(),dead:false,failed:false};
    cache.push(e);evict();enqueue(e,false);
    return null;
  }
  /* Async lookup for export (never cached, never aborted). */
  async function request(buf,speed,startSec,endSec){
    if(!buf||!(speed>0))return null;
    const a=Math.max(0,startSec),b=Math.min(buf.duration,Math.max(a+.01,endSec));
    if(b-a<=0)return null;
    const e={src:buf,speed,start:a,end:b,out:null,last:Date.now(),dead:false,failed:false};
    const r=await enqueue(e,true);
    return r&&r.out?toResult(r):null;
  }
  /* Debounced: pre-compute the stretch for every clip whose speed is not 1x, so playback starts instantly. */
  let prewarmTimer=0;
  function prewarm(resolveBuffer){
    clearTimeout(prewarmTimer);
    prewarmTimer=setTimeout(()=>{
      try{
        if(typeof overlays==='undefined'||!Array.isArray(overlays))return;

          if(!ov||!ov.src||ov.reversePlayback||!['video','audio'].includes(ov.type))return;
          const speed=Number(ov.speed)||1;if(Math.abs(speed-1)<.001)return;
          const buf=resolveBuffer(ov);if(!buf)return;
          const off=Math.max(0,Number(ov.sourceOffset)||0);
          const len=Math.max(0,(Number(ov.trimEnd)||0)-(Number(ov.trimStart)||0))*speed;
          getReady(buf,speed,off,off+len+.1);
        });
      }catch(_e){}
    },350);
  }
  window.ethvellePitchSafeStretch={getReady,request,prewarm,_wsola:wsola};
})();

<script id="ethvelle-v221-media-perspective-js">
/* Ethvelle v221: Perspective for normal media clips (video / image / background), same idea as the text editor's Perspective tool.
   Preview: a CSS "perspective() rotateX() rotateY()" suffix is appended to the clip's transform (innermost, so it tilts the clip in place).
   Export: canvas has no 3D transforms, so the clip is drawn as a small triangle mesh projected with the same maths. */
(function(){
  'use strict';
  const LIMIT=55,DIST=900;
  const SUFFIX_RE=/\s*perspective\(\d+(?:\.\d+)?px\)\s*rotateX\([^)]*\)\s*rotateY\([^)]*\)\s*$/;
  const num=v=>{v=Number(v);return Number.isFinite(v)?Math.max(-LIMIT,Math.min(LIMIT,v)):0;};
  window.ethvellePerspectiveSuffix=function(clip){
    const x=num(clip&&clip.mediaPerspectiveX),y=num(clip&&clip.mediaPerspectiveY);
    return (x||y)?` perspective(${DIST}px) rotateX(${y}deg) rotateY(${x}deg)`:'';
  };
  window.ethvelleStripPerspective=function(value){return String(value||'').replace(SUFFIX_RE,'');};
  window.ethvelleMediaPerspectiveDistance=DIST;
  function project(x,y,a,b,d){
    const cb=Math.cos(b),sb=Math.sin(b),ca=Math.cos(a),sa=Math.sin(a);
    const x1=x*cb,z1=-x*sb;
    const y2=y*ca-z1*sa,z2=y*sa+z1*ca;
    const f=d/(d-z2);
    return [x1*f,y2*f];
  }
  /* Draw img's source rect (sx,sy,sw,sh) into dest rect (dx,dy,dw,dh) tilted around (cx,cy); px/py are degrees like the CSS version. */
  window.ethvelleDrawPerspectiveImage=function(ctx,img,sx,sy,sw,sh,dx,dy,dw,dh,px,py,cx,cy,dist){
    const a=num(py)*Math.PI/180,b=num(px)*Math.PI/180,d=Math.max(50,Number(dist)||DIST);
    const N=12,M=12,V=[];
    for(let j=0;j<=M;j++){
      for(let i=0;i<=N;i++){
        const u=i/N,v=j/M;
        const p=project(dx+u*dw-cx,dy+v*dh-cy,a,b,d);
        V.push({X:cx+p[0],Y:cy+p[1],S:sx+u*sw,T:sy+v*sh});
      }
    }
    const at=(i,j)=>V[j*(N+1)+i];
    function tri(p0,p1,p2){
      const det=(p1.S-p0.S)*(p2.T-p0.T)-(p2.S-p0.S)*(p1.T-p0.T);
      if(!det)return;
      const A=((p1.X-p0.X)*(p2.T-p0.T)-(p2.X-p0.X)*(p1.T-p0.T))/det;
      const B=((p1.Y-p0.Y)*(p2.T-p0.T)-(p2.Y-p0.Y)*(p1.T-p0.T))/det;
      const C=((p2.X-p0.X)*(p1.S-p0.S)-(p1.X-p0.X)*(p2.S-p0.S))/det;
      const D=((p2.Y-p0.Y)*(p1.S-p0.S)-(p1.Y-p0.Y)*(p2.S-p0.S))/det;
      const E=p0.X-A*p0.S-C*p0.T,F=p0.Y-B*p0.S-D*p0.T;
      const gx=(p0.X+p1.X+p2.X)/3,gy=(p0.Y+p1.Y+p2.Y)/3;
      const grow=pt=>{const vx=pt.X-gx,vy=pt.Y-gy,l=Math.hypot(vx,vy)||1;return [pt.X+vx/l*.7,pt.Y+vy/l*.7];};
      const g0=grow(p0),g1=grow(p1),g2=grow(p2);
      const minS=Math.min(p0.S,p1.S,p2.S),maxS=Math.max(p0.S,p1.S,p2.S),minT=Math.min(p0.T,p1.T,p2.T),maxT=Math.max(p0.T,p1.T,p2.T);
      const bx=Math.max(0,Math.floor(minS)),by=Math.max(0,Math.floor(minT));
      const bw=Math.max(1,Math.ceil(maxS)-bx+1),bh=Math.max(1,Math.ceil(maxT)-by+1);
      ctx.save();
      ctx.beginPath();ctx.moveTo(g0[0],g0[1]);ctx.lineTo(g1[0],g1[1]);ctx.lineTo(g2[0],g2[1]);ctx.closePath();ctx.clip();
      ctx.transform(A,B,C,D,E,F);
      try{ctx.drawImage(img,bx,by,bw,bh,bx,by,bw,bh);}catch(_e){}
      ctx.restore();
    }
    for(let j=0;j<M;j++){
      for(let i=0;i<N;i++){
        const p00=at(i,j),p10=at(i+1,j),p01=at(i,j+1),p11=at(i+1,j+1);
        tri(p00,p10,p01);tri(p10,p11,p01);
      }
    }
  };
})();

      for(let i=0;i<overlays.length;i++){
        const ov=overlays[i];if(!isMedia(ov))continue;

  // A split whose second half continues straight on from the first half reuses the same decoder
  // (see transferContinuousSplitVideos), so it needs no separate warm-up.
  function continuesDecoder(prev,next){
    if(!prev||!next||prev.type!=='video'||next.type!=='video'||String(prev.src||'')!==String(next.src||''))return false;
    const ps=Math.max(.05,num(prev.speed,1)),ns=Math.max(.05,num(next.speed,1));
    const srcEnd=num(prev.sourceOffset)+(num(prev.trimEnd)-num(prev.trimStart))*ps;
    return Math.abs(num(prev.trimEnd)-num(next.trimStart))<=.025&&Math.abs(ps-ns)<=.001&&Math.abs(srcEnd-num(next.sourceOffset))<=.06;
  }

  // ---- readiness of everything needed for the next `win` seconds starting at t ----
  function readiness(t,relaxed,activeOnly){
    const win=lookaheadFor(t),t1=t+win;

  // asked to play once; the player code then waits for readyState before playing -> frozen frame.
  // A short muted play() wakes the decoder; afterwards the element is put back exactly where it was.

  }
