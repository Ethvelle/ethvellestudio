async function changeMyCreditBalance(delta, entry){
  if(!(currentUser&&firebaseReady&&fbDb)) throw new Error('CREDITS_UNAVAILABLE');
  entry=entry&&typeof entry==='object'?entry:{};
  const amount=Math.trunc(Number(delta)||0);
  if(!amount) return {balance:myCreditsBalance,id:''};
  const uid=currentUser.uid;
  const userRef=fbDb.collection('users').doc(uid);
  const kind=String(entry.type||'usage');
  if(kind!=='usage'&&kind!=='refund')throw new Error('INVALID_CREDIT_TRANSACTION');
  if((kind==='usage'&&amount>=0)||(kind==='refund'&&amount<=0))throw new Error('INVALID_CREDIT_AMOUNT');
  const refundId=kind==='refund'&&entry.referenceId?'refund_'+String(entry.referenceId):'';
  if(kind==='refund'&&!refundId)throw new Error('REFUND_REFERENCE_REQUIRED');
  const ledgerRef=refundId?fbDb.collection('creditTransactions').doc(refundId):fbDb.collection('creditTransactions').doc();
  const createdAt=Date.now();
  let newBalance=0,alreadyApplied=false;
  await fbDb.runTransaction(async tx=>{
    // All reads precede writes so the wallet and its ledger are one atomic operation.
    const userSnap=await tx.get(userRef);
    const existingLedger=kind==='refund'?await tx.get(ledgerRef):null;
    if(existingLedger&&existingLedger.exists){
      alreadyApplied=true;
      const prior=existingLedger.data()||{};newBalance=Math.max(0,Number(prior.balanceAfter)||Number((userSnap.data()||{}).creditsBalance)||0);
      return;
    }
    if(kind==='refund'){
      const originalRef=fbDb.collection('creditTransactions').doc(String(entry.referenceId));
      const original=await tx.get(originalRef);
      if(!original.exists||String((original.data()||{}).uid)!==String(uid)||String((original.data()||{}).type)!=='usage')throw new Error('INVALID_REFUND_REFERENCE');
    }
    const data=userSnap.exists?(userSnap.data()||{}):{};
    const balance=Math.max(0,Number(data.creditsBalance)||0);
    if(amount<0&&balance<Math.abs(amount)){const err=new Error('INSUFFICIENT_CREDITS');err.code='ethvelle/insufficient-credits';throw err;}
    newBalance=balance+amount;
    tx.set(userRef,{uid,email:currentUser.email||null,displayName:currentUser.displayName||null,creditsBalance:newBalance,creditsUpdatedAt:createdAt,lastCreditTransactionId:ledgerRef.id},{merge:true});
    tx.set(ledgerRef,{uid,email:currentUser.email||null,displayName:currentUser.displayName||null,sessionId:(typeof getTrackingIdentity==='function'?(getTrackingIdentity()||{}).sessionId:null),amount,balanceAfter:newBalance,type:kind,feature:entry.feature||'export',description:entry.description||(kind==='refund'?'Export refund':'Video export'),quality:entry.quality||null,durationSeconds:entry.durationSeconds||null,referenceId:kind==='refund'?String(entry.referenceId):(entry.referenceId||null),createdAt,source:'main-site'});
  });
  myCreditsBalance=newBalance;
  return {balance:newBalance,id:ledgerRef.id,alreadyApplied};
}

// Whether the signed-in user has EVER placed a paid-plan order (any status — pending,
// approved, even rejected all count, since the point is "have they already gone through
// checkout before", not "do they currently have a paid plan"). Drives the "First-purchase
// offer" discount in getActivePlanDiscount() below: once this flips true it stays true for
// that account, so the offer only ever shows to a genuinely first-time buyer. Defaults to
// true (nothing to offer) until the listener actually reports in, so the UI never briefly
// flashes a discount it has to take away a moment later.
let myHasEverOrdered = true;
function applyApprovedOrders(list){
  // Keep the newest order status for EACH plan, not just the globally latest order, so a
  // pending badge stays attached to the correct plan card even after another plan is ordered.
  const previousPlanOrdersKey = JSON.stringify(Object.keys(myPlanOrderStates).sort().map(k=>[k,myPlanOrderStates[k].id,myPlanOrderStates[k].status]));
  const nextPlanOrders = Object.create(null);
  list.forEach(o=>{
    const key = String(o && o.planId || '');
    if(!key || nextPlanOrders[key]) return; // list is newest-first
    nextPlanOrders[key] = {
      id:o.id || '', planId:key, planName:o.planName || key,
      price:Number(o.price)||0, method:o.method||'',
      status:String(o.status||'pending').toLowerCase(), createdAt:o.createdAt||null, approvedAt:o.approvedAt||null, expiresAt:o.expiresAt||null, creditsGrantAmount:Number(o.creditsGrantAmount)||0, creditsGrantedAt:o.creditsGrantedAt||null
    };
  });
  myPlanOrderStates = nextPlanOrders;
  const currentPlanOrdersKey = JSON.stringify(Object.keys(myPlanOrderStates).sort().map(k=>[k,myPlanOrderStates[k].id,myPlanOrderStates[k].status]));
  const planOrdersChanged = previousPlanOrdersKey !== currentPlanOrdersKey;
  // The latest order is shown separately from the active plan: pending/rejected orders must
  // not silently activate a plan, while an older approved plan remains active until replaced.
  const previousOrderKey = myLatestOrderInfo ? [myLatestOrderInfo.id,myLatestOrderInfo.planId,myLatestOrderInfo.status].join('|') : '';
  const latestPaid = list.find(o=>Number(o.price) > 0);
  const latest = latestPaid || list[0] || null;
  myLatestOrderInfo = latest ? {
    id: latest.id || '', planId: latest.planId || '', planName: latest.planName || latest.planId || 'Plan',
    price: Number(latest.price) || 0, method: latest.method || '',
    status: String(latest.status || 'pending').toLowerCase(), createdAt: latest.createdAt || null
  } : null;
  const newOrderKey = myLatestOrderInfo ? [myLatestOrderInfo.id,myLatestOrderInfo.planId,myLatestOrderInfo.status].join('|') : '';
  const orderInfoChanged = previousOrderKey !== newOrderKey;

  // Only an approved order changes the plan currently enforced by the app.
  const approved = list.find(o=>String(o.status||'').toLowerCase()==='approved');
  const newId = approved ? approved.planId : 'free';
  const newName = approved ? (approved.planName || approved.planId) : 'Free';
  const hasOrdered = list.some(o=>Number(o.price) > 0);
  const orderedChanged = hasOrdered !== myHasEverOrdered;
  myHasEverOrdered = hasOrdered;
  if(newId !== myActivePlanId || newName !== myActivePlanName || orderedChanged || orderInfoChanged || planOrdersChanged){
    myActivePlanId = newId;
    myActivePlanName = newName;
    if(currentUser) saveCachedMyPlan(currentUser.uid);
    updateProfileUI(); // also refreshes plan-dependent profile features
    _plansSig = null; // force badges/plan cards to refresh when an order changes status
    renderPaymentPlans();
    resyncUserPlanToProfileDoc();
    if(typeof wirePromoPopups==='function') wirePromoPopups(true, activeScreenId());
  }
  if(typeof renderMyPlanScreen==='function') renderMyPlanScreen();
}

        list.sort((a,b)=>{
          const ta = (a.createdAt && a.createdAt.toMillis) ? a.createdAt.toMillis() : (a.createdAt ? new Date(a.createdAt).getTime() : 0);
          const tb = (b.createdAt && b.createdAt.toMillis) ? b.createdAt.toMillis() : (b.createdAt ? new Date(b.createdAt).getTime() : 0);
          return tb - ta;
        });
        applyApprovedOrders(list);
      }, err=>{
        console.warn('Ethvelle: could not read this user\u2019s orders from Firestore', err);
      });
  }catch(err){ console.warn('Ethvelle: my-orders sync failed to start', err); }
}
function stopMyOrdersSync(){
  if(myOrdersUnsub){ try{ myOrdersUnsub(); }catch(e){} myOrdersUnsub = null; }
  // Never carry an order badge across a sign-out or account switch.
  myLatestOrderInfo = null;
  myPlanOrderStates = Object.create(null);
}

/* ================= MY NOTIFICATIONS (admin-sent inbox, in-app only) =================
   Reads the shared `notifications` collection the admin panel writes to. Each doc targets
   either everyone (target:'all') or one specific uid (target:'<uid>'). Filtered client-side
   so a single listener covers both. Two per-device lists are kept in localStorage:
     - read ids    -> which messages have been seen (drives the red dot)
     - deleted ids -> messages this person removed from their own inbox
   The admin's messages are shared broadcasts, so deleting only hides them on this device. */
let myNotifsUnsub = null;
let myNotifications = [];
let notifHighlightIds = [];
function notifReadKey(uid){ return 'riggy.notifsRead.' + uid; }
function notifDeletedKey(uid){ return 'riggy.notifsDeleted.' + uid; }
function readIdList(key){
  try{ return JSON.parse(localStorage.getItem(key) || '[]'); }catch(e){ return []; }
}
function writeIdList(key, ids){
  try{ localStorage.setItem(key, JSON.stringify(ids)); }catch(e){}
}
function getReadNotifIds(){ return currentUser ? readIdList(notifReadKey(currentUser.uid)) : []; }
function getDeletedNotifIds(){ return currentUser ? readIdList(notifDeletedKey(currentUser.uid)) : []; }
// What the person actually sees: admin messages minus anything they deleted on this device.
function visibleNotifications(){
  if(!currentUser) return [];
  const deleted = getDeletedNotifIds();
  return myNotifications.filter(n => !deleted.includes(n.id));
}
function markAllNotifsSeen(){
  if(!currentUser) return;
  const ids = getReadNotifIds();
  visibleNotifications().forEach(n => { if(!ids.includes(n.id)) ids.push(n.id); });
  writeIdList(notifReadKey(currentUser.uid), ids);
}
function isNotifScreenOpen(){
  const s = document.getElementById('screen-notifications');
  return !!(s && s.classList.contains('active'));
}
function deleteNotif(id){
  if(!currentUser) return;
  const ids = getDeletedNotifIds();
  if(!ids.includes(id)) ids.push(id);
  writeIdList(notifDeletedKey(currentUser.uid), ids);
  renderNotifInbox();
}
// Runs every time the Notifications screen opens (home bell or Profile menu). It works out which
// messages are new first, then marks everything seen, so the red dot clears on open while the new
// cards keep their blue highlight for this visit.
function openNotificationsScreen(){
  const readNow = getReadNotifIds();
  notifHighlightIds = visibleNotifications().map(n => n.id).filter(id => !readNow.includes(id));
  markAllNotifsSeen();

}
function startMyNotificationsSync(uid){
  stopMyNotificationsSync();

      .onSnapshot(snap=>{
        const all = snap.docs.map(d=>({ id:d.id, ...d.data() }));
        myNotifications = all.filter(n => n.target === 'all' || n.target === uid);
        // A message that arrives while this screen is open counts as seen straight away.

      });

}
function stopMyNotificationsSync(){
  if(myNotifsUnsub){ try{ myNotifsUnsub(); }catch(e){} myNotifsUnsub = null; }
  myNotifications = [];
  notifHighlightIds = [];

function fmtNotifTime(ts){
  try{
    const d = ts && ts.toDate ? ts.toDate() : (ts ? new Date(ts) : null);
    if(!d || isNaN(d.getTime())) return '';
    return d.toLocaleDateString(undefined,{month:'short',day:'numeric'}) + ' \u00B7 ' + d.toLocaleTimeString(undefined,{hour:'numeric',minute:'2-digit'});
  }catch(e){ return ''; }
}

function renderNotifInbox(){
  const readIds = getReadNotifIds();
  const visible = visibleNotifications();

  const showBadges = readProfileNotificationPrefs().badges !== false;
  const badge = document.getElementById('notifMenuBadge');
  if(badge) badge.classList.toggle('show', showBadges && unreadCount > 0);
  const homeBadge = document.getElementById('homeNotifBadge');
  if(homeBadge) homeBadge.classList.toggle('show', showBadges && unreadCount > 0);

  const list = document.getElementById('notifInboxList');
  const label = document.getElementById('notifInboxLabel');

  if(visible.length === 0){
    list.innerHTML = '<p class="notif-empty">No notifications yet</p>';
    if(label) label.style.display = 'none';
    return;
  }
  if(label) label.style.display = '';
  list.innerHTML = visible.map(n=>{
    const highlight = notifHighlightIds.includes(n.id);
    return `
      <div class="notif-inbox-card${highlight ? ' unread' : ''}">
        <button class="notif-inbox-del" onclick="deleteNotif('${n.id}')" aria-label="Delete notification">

        <p class="notif-inbox-time">${escapeHtml(fmtNotifTime(n.createdAt))}</p>
      </div>`;
  }).join('');
}

// Sets an avatar div's fallback-letter text without nuking a photo <img> already inside it —
// textContent would otherwise wipe out that child, so this only touches a dedicated text node.
function setAvatarLetter(avatarEl, letter){
  if(!avatarEl) return;
  let textNode = null;
  for(const node of avatarEl.childNodes){ if(node.nodeType === Node.TEXT_NODE){ textNode = node; break; } }
  if(textNode){ textNode.nodeValue = letter; }
  else { avatarEl.insertBefore(document.createTextNode(letter), avatarEl.firstChild); }
}

// Updates every avatar element on the page that shares the same photo — currently the Profile
// screen's avatar and the Settings screen's avatar — so uploading from either place keeps both
// in sync without a page reload.
function renderProfileAvatarPhoto(){
  const photo = getLocalProfilePhoto();
  ['profileAvatar','settingsAvatar'].forEach(id=>{
    const avatarEl = document.getElementById(id);
    if(!avatarEl) return;
    const existing = avatarEl.querySelector('img');
    if(photo){
      if(existing){ existing.src = photo; }
      else { const img = document.createElement('img'); img.src = photo; img.alt = ''; avatarEl.appendChild(img); }
      // FIXED: the fallback-letter text node (e.g. "U") was left sitting in the DOM under the
      // <img> — now that the img is absolutely positioned it's fully covered visually, but we
      // still blank the letter out here too so nothing can ever peek through at the edges.
      avatarEl.dataset.letter = avatarEl.dataset.letter || avatarEl.textContent.trim();
      for(const node of Array.from(avatarEl.childNodes)){
        if(node.nodeType === Node.TEXT_NODE) node.textContent = '';
      }
    } else if(existing){
      existing.remove();
      // Restore the fallback letter now that there's no photo to show instead.
      if(avatarEl.dataset.letter) avatarEl.firstChild
        ? avatarEl.insertBefore(document.createTextNode(avatarEl.dataset.letter), avatarEl.firstChild)
        : avatarEl.appendChild(document.createTextNode(avatarEl.dataset.letter));
    }
  });
}

// Downscales to a small square JPEG before storing — an uncompressed phone photo can be
// several MB, which would blow past localStorage's ~5MB-per-origin ceiling after only a
// few changes. 256x256 is plenty for a 76px avatar at any screen density.
const PROFILE_PHOTO_MAX_DIM = 256;
function onProfilePhotoChosen(file){
  if(!file || !currentUser || !canUploadProfilePhoto()) return;
  if(!/^image\//.test(file.type)){ showToast('Please choose an image file', { icon: toastIcons.info }); return; }
  const reader = new FileReader();
  reader.onload = function(e){
    const img = new Image();
    img.onload = function(){
      const size = PROFILE_PHOTO_MAX_DIM;
      const canvas = document.createElement('canvas');
      canvas.width = size; canvas.height = size;
      const ctx = canvas.getContext('2d');
      // High-quality downscale: without this, small scaled-down images (avatars, tile
      // thumbnails, logos) come out with jagged/pixelated edges on some mobile browsers,
      // which default to a cheap filter instead of a smooth one.
      ctx.imageSmoothingEnabled = true;
      ctx.imageSmoothingQuality = 'high';
      // Cover-crop to a square from the image's center, so portrait/landscape photos both
      // fill the round avatar instead of being squeezed.
      const s = Math.min(img.width, img.height);
      const sx = (img.width - s) / 2, sy = (img.height - s) / 2;
      ctx.drawImage(img, sx, sy, s, s, 0, 0, size, size);
      let dataUrl;
      try{ dataUrl = canvas.toDataURL('image/jpeg', 0.85); }catch(err){
        showToast('Couldn\u2019t process that image', { icon: toastIcons.info }); return;
      }
      try{
        localStorage.setItem(profilePhotoKey(currentUser.uid), dataUrl);
        renderProfileAvatarPhoto();
        showToast('Profile photo updated', { icon: toastIcons.check, duration:1400 });
      }catch(err){
        showToast('Couldn\u2019t save photo \u2014 storage full?', { icon: toastIcons.info });
      }
    };
    img.onerror = function(){ showToast('Couldn\u2019t read that image', { icon: toastIcons.info }); };
    img.src = e.target.result;
  };
  reader.onerror = function(){ showToast('Couldn\u2019t read that file', { icon: toastIcons.info }); };
  reader.readAsDataURL(file);
}

// Small, local copy of the admin panel's friendlyAdminAuthError pattern — this file has no
// such helper yet, and pulling in the whole admin error-mapping table would be overkill for
// the handful of codes updateProfile/updatePassword can actually raise.
function friendlyAdminAuthErrorText(err){
  const code = (err && err.code) || '';
  if(code.indexOf('weak-password') > -1) return 'Password should be at least 6 characters';
  if(code.indexOf('wrong-password') > -1 || code.indexOf('invalid-credential') > -1) return 'Current password is incorrect';
  if(code.indexOf('network-request-failed') > -1) return 'Network error — check your connection and try again';
  return 'Could not save changes — try again';
}

/* ================= PASSWORD CHANGE — 2-STEP FLOW =================
   Per request: don't show current/new/confirm all on one page. Step 1 asks only for the
   current password and verifies it (reauthenticate) right there on "Next". Only once that
   succeeds does Step 2 appear, asking for the new password twice, with its own "Confirm"
   button that actually applies the change. "Back" returns to Step 1 without losing the
   already-verified current password (so the person isn't forced to retype it just to fix a
   typo in the new one) — re-verification only happens if they reopen Settings fresh or the
   session needs it again. */
// Holds the current password once Step 1 has verified it, so Step 2's Confirm can
// reauthenticate silently right before updatePassword without asking for it a second time.
// Cleared on screen entry, on successful password change, and on sign-out.
let pwVerifiedCurrentPassword = null;
function resetPasswordStepFlow(){

// Step 1 "Next": verify the typed current password by reauthenticating with it. A blank
// field is allowed through (matches "leave blank to keep current password" — the person just
// isn't changing it right now) and simply does nothing.
async function pwStepVerifyCurrent(){
  const currentPasswordEl = document.getElementById('settingsCurrentPasswordInput');
  const currentPassword = currentPasswordEl.value;
  if(!currentPassword) return;
  if(!currentUser || !(firebaseReady && fbAuth && fbAuth.currentUser)){
    showToast('Sign in to change your password', {icon: toastIcons.info});
    return;
  }
  const btn = document.getElementById('pwStep1NextBtn');
  btn.disabled = true; btn.textContent = 'Checking…';
  document.getElementById('settingsNameNote').textContent = '';
  try{
    const email = fbAuth.currentUser.email;
    const cred = firebase.auth.EmailAuthProvider.credential(email, currentPassword);
    await fbAuth.currentUser.reauthenticateWithCredential(cred);
    // Verified — remember it for Step 2's silent reauth right before updatePassword, then
    // swap the panel over to Step 2.
    pwVerifiedCurrentPassword = currentPassword;
    document.getElementById('pwStep1Card').style.display = 'none';
    document.getElementById('pwStep1Note').style.display = 'none';
    btn.style.display = 'none';
    document.getElementById('pwStep2Card').style.display = '';
    document.getElementById('pwStep2Note').style.display = '';
    document.getElementById('pwStep2Buttons').style.display = '';
    document.getElementById('settingsPasswordInput').focus();
  }catch(err){
    const code = (err && err.code) || '';
    if(code.indexOf('requires-recent-login') > -1){
      document.getElementById('settingsNameNote').textContent = 'Please sign in again to change your password, then retry.';
      openLoginScreen(()=>goTo('settings'));
    } else {
      document.getElementById('settingsNameNote').textContent = friendlyAdminAuthErrorText(err);
    }
  }
  btn.disabled = false; btn.textContent = 'Next';
}
// Step 2 "Back": return to Step 1. Keeps the already-verified current password in memory
// (pwVerifiedCurrentPassword) rather than wiping it, so correcting a new-password typo
// doesn't force retyping the current password too.
function pwStepBackToCurrent(){
  document.getElementById('settingsPasswordInput').value = '';
  document.getElementById('settingsPasswordConfirmInput').value = '';
  document.getElementById('pwStep2Card').style.display = 'none';
  document.getElementById('pwStep2Note').style.display = 'none';
  document.getElementById('pwStep2Buttons').style.display = 'none';
  document.getElementById('pwStep1Card').style.display = '';
  document.getElementById('pwStep1Note').style.display = '';
  document.getElementById('pwStep1NextBtn').style.display = '';
  document.getElementById('settingsNameNote').textContent = '';
}
// Step 2 "Confirm": validate the new password + confirmation, then actually apply it.
async function pwStepConfirmNew(){
  const newPasswordEl = document.getElementById('settingsPasswordInput');
  const confirmPasswordEl = document.getElementById('settingsPasswordConfirmInput');
  const newPassword = newPasswordEl.value;
  const confirmPassword = confirmPasswordEl.value;
  if(!newPassword || newPassword.length < 6){
    showToast('New password should be at least 6 characters', {icon: toastIcons.info});
    return;
  }
  if(newPassword !== confirmPassword){
    showToast('New password and confirmation don\u2019t match', {icon: toastIcons.info});
    return;
  }
  if(!pwVerifiedCurrentPassword || !currentUser || !(firebaseReady && fbAuth && fbAuth.currentUser)){
    // Verified password fell out of memory (e.g. long idle, or Settings was reopened) —
    // safest is to send them back to re-verify Step 1 rather than guess.
    showToast('Please re-enter your current password', {icon: toastIcons.info});
    resetPasswordStepFlow();
    return;
  }
  const btn = document.querySelector('#pwStep2Buttons .settings-step-btn:not(.secondary)');
  const backBtn = document.querySelector('#pwStep2Buttons .settings-step-btn.secondary');
  btn.disabled = true; btn.textContent = 'Saving…'; if(backBtn) backBtn.disabled = true;
  document.getElementById('settingsNameNote').textContent = '';
  try{
    // Reauthenticate again right before the actual change — the credential from Step 1 can
    // have gone stale if the person lingered on Step 2, and Firebase requires a fresh one
    // for a sensitive action like this regardless.
    const email = fbAuth.currentUser.email;
    const cred = firebase.auth.EmailAuthProvider.credential(email, pwVerifiedCurrentPassword);
    await fbAuth.currentUser.reauthenticateWithCredential(cred);
    await fbAuth.currentUser.updatePassword(newPassword);
    showToast('Password changed', {icon: toastIcons.check, duration:1600});
    resetPasswordStepFlow();
  }catch(err){
    const code = (err && err.code) || '';
    if(code.indexOf('wrong-password') > -1 || code.indexOf('invalid-credential') > -1){
      // The remembered current password no longer checks out — don't silently retry, send
      // them back to Step 1 to re-enter it.
      showToast('Current password no longer verified — please re-enter it', {icon: toastIcons.info});
      resetPasswordStepFlow();
      return;
    } else if(code.indexOf('requires-recent-login') > -1){
      document.getElementById('settingsNameNote').textContent = 'Please sign in again to change your password, then retry.';
      openLoginScreen(()=>goTo('settings'));
    } else {
      document.getElementById('settingsNameNote').textContent = friendlyAdminAuthErrorText(err);
    }
  }
  btn.disabled = false; btn.textContent = 'Confirm'; if(backBtn) backBtn.disabled = false;
}

   No push/email backend exists yet to actually deliver these, so for now they're a simple
   per-device preference store — flipping them just saves a flag locally (per the "soon but
   not yet wired up" scope this section was built to). */
const NOTIF_PREFS_KEY = 'riggy.notifPrefs.v1';
const NOTIF_PREF_DEFAULTS = { push:true, email:false, exportDone:true, billing:true };
function loadNotifPrefs(){
  try{
    const raw = localStorage.getItem(NOTIF_PREFS_KEY);
    if(raw) return { ...NOTIF_PREF_DEFAULTS, ...JSON.parse(raw) };
  }catch(e){}
  return { ...NOTIF_PREF_DEFAULTS };

let notifPrefs = loadNotifPrefs();
function saveNotifPrefs(){
  try{ localStorage.setItem(NOTIF_PREFS_KEY, JSON.stringify(notifPrefs)); }catch(e){}
}

  Object.keys(NOTIF_PREF_DEFAULTS).forEach(key=>{

    if(el) el.classList.toggle('on', !!notifPrefs[key]);
  });
}
function toggleNotifPref(key){
  notifPrefs[key] = !notifPrefs[key];
  saveNotifPrefs();
  renderNotifToggles();
}

async function hideMyCreditExpense(transactionId){
  const id=String(transactionId||'');
  if(!id||!(currentUser&&currentUser.uid)){showToast('Sign in to manage credit history.',{icon:toastIcons.info,duration:2400});return;}
  const entry=(Array.isArray(myCreditTransactions)?myCreditTransactions:[]).find(t=>String(t&&t.id)===id);
  if(!entry||String(entry.uid||'')!==String(currentUser.uid)||String(entry.type||'').toLowerCase()!=='usage'||!(Number(entry.amount)<0)){
    showToast('Only your credit-usage expenses can be removed.',{icon:toastIcons.info,duration:2600});return;
  }
  if(myHiddenCreditTransactionIds.includes(id))return;
  const confirmed=window.confirm('Delete this expense from your Credits History? Your credit balance and original accounting record will not change.');
  if(!confirmed)return;
  const uid=String(currentUser.uid);
  const next=[...new Set([...myHiddenCreditTransactionIds,id])];
  let synced=false;
  try{
    if(!(firebaseReady&&fbDb&&firebase&&firebase.firestore&&firebase.firestore.FieldValue))throw new Error('CREDIT_HISTORY_SYNC_UNAVAILABLE');
    await fbDb.collection('users').doc(uid).set({hiddenCreditTransactionIds:firebase.firestore.FieldValue.arrayUnion(id)},{merge:true});
    synced=true;
  }catch(err){console.warn('Ethvelle: could not sync hidden credit expense; keeping this removal on this device.',err);}
  const savedLocally=writeLocalHiddenCreditIds(uid,next);
  myHiddenCreditTransactionIds=next;
  if(typeof renderMyPlanScreen==='function')renderMyPlanScreen();
  if(synced){showToast('Expense removed from your visible history. Your credit balance was not changed.',{icon:toastIcons.check,duration:3000});}
  else if(savedLocally){showToast('Expense hidden on this device. Account sync is unavailable.',{icon:toastIcons.info,duration:3200});}
  else{showToast('Could not save this history change. Please check device storage and try again.',{icon:toastIcons.info,duration:3200});}
}

async function handleForgotPassword(){
  if(loginMode !== 'signin') return;
  const email = document.getElementById('loginEmailInput').value.trim();
  if(!email){ showLoginError('Enter your email address first to reset your password'); return; }
  if(!firebaseReady || !fbAuth){ showLoginError('Password reset is unavailable right now — try again later'); return; }
  const link = document.getElementById('forgotPasswordLink');
  if(link && link.dataset.busy === '1') return;
  if(link){ link.dataset.busy = '1'; link.textContent = 'Sending…'; link.style.pointerEvents = 'none'; }
  document.getElementById('loginError').classList.remove('show');
  showLoginStatus('');
  try{
    await fbAuth.sendPasswordResetEmail(email);
    showLoginStatus('Password reset email sent. Check your inbox and spam folder.');
  }catch(err){
    showLoginError(friendlyAuthError(err));
  }finally{
    if(link){ link.dataset.busy = '0'; link.textContent = 'Forgot password?'; link.style.pointerEvents = ''; }
  }

function friendlyAuthError(err){
  const code = err && err.code || '';
  if(code.indexOf('wrong-password') > -1 || code.indexOf('invalid-credential') > -1) return 'Wrong email or password';
  if(code.indexOf('user-not-found') > -1) return 'No account found with that email';
  if(code.indexOf('email-already-in-use') > -1) return 'That email is already registered — try signing in instead';
  if(code.indexOf('weak-password') > -1) return 'Password should be at least 6 characters';
  if(code.indexOf('invalid-email') > -1) return 'Enter a valid email address';
  if(code.indexOf('network-request-failed') > -1) return 'Network error — check your connection';
  if(code.indexOf('popup-closed-by-user') > -1) return 'Google sign-in was closed before finishing';
  return (err && err.message) || 'Something went wrong — try again';
}

async function performEmailAuth(){
  const wasSignup=loginMode==='signup';
  if(wasSignup&&(!loginLegalConsentAcceptedOnce||!requireLoginLegalConsent()))return;
  loginLegalConsentAcceptedOnce=false;
  const email=document.getElementById('loginEmailInput').value.trim();

  if(!email||!password){showLoginError('Enter both email and password');return;}
  if(!firebaseReady||!fbAuth){showLoginError('Sign-in is unavailable right now — try again later');return;}
  btn.disabled=true;btn.textContent=wasSignup?'Signing up…':'Signing in…';
  try{
    if(wasSignup){await fbAuth.createUserWithEmailAndPassword(email,password);trackEvent('user_signup');}
    else{await fbAuth.signInWithEmailAndPassword(email,password);trackEvent('user_login');}
  }catch(err){showLoginError(friendlyAuthError(err));}
  finally{btn.disabled=false;btn.textContent=wasSignup?'Sign up':'Sign in';}
}

async function performGoogleSignIn(){
  if(!loginLegalConsentAcceptedOnce||!requireLoginLegalConsent())return;
  loginLegalConsentAcceptedOnce=false;
  if(!firebaseReady||!fbAuth){showLoginError('Sign-in is unavailable right now — try again later');return;}
  const btn=document.getElementById('googleSignInBtn');
  btn.disabled=true;
  try{
    const provider=new firebase.auth.GoogleAuthProvider();
    const result=await fbAuth.signInWithPopup(provider);
    trackEvent(result&&result.additionalUserInfo&&result.additionalUserInfo.isNewUser?'user_signup':'user_login');
  }catch(err){
    if(err&&(err.code==='auth/popup-blocked'||err.code==='auth/operation-not-supported-in-this-environment')){
      try{
        const provider=new firebase.auth.GoogleAuthProvider();
        await fbAuth.signInWithRedirect(provider);
        return;
      }catch(err2){showLoginError(friendlyAuthError(err2));}
    }else{showLoginError(friendlyAuthError(err));}
  }finally{btn.disabled=false;}
}

/* ================= USERS DIRECTORY (per-user Firestore profile doc) =================
   Mirrors the signed-in user into users/{uid} so the admin panel can list, filter and
   message real accounts. Previously plan/activity only ever lived in this device's
   localStorage — nothing about a user existed in the cloud beyond Firebase Auth's own
   record (which the admin panel cannot list without the Admin SDK). This runs once per
   sign-in and again whenever the active plan changes, as a best-effort merge write —
   never blocks the UI and never throws if Firestore is unreachable. */
function syncUserProfileDoc(user){
  if(!(firebaseReady && fbDb) || !user) return;
  try{
    const ref = fbDb.collection('users').doc(user.uid);
    ref.set({
      uid: user.uid,
      email: user.email || null,
      displayName: user.displayName || null,
      photoURL: user.photoURL || null,
      planId: myActivePlanId,
      planName: myActivePlanName,
      lastSeen: firebase.firestore.FieldValue.serverTimestamp(),
      createdAt: firebase.firestore.FieldValue.serverTimestamp() // ignored by merge if the doc already has one — see below
    }, { merge:true }).catch(()=>{});
    // createdAt should only ever be set once (on first sync) — a plain merge:true above
    // would otherwise overwrite it with "now" on every later login. Firestore has no
    // "set only if absent" in a single merge call, so read-then-maybe-write once.
    ref.get().then(snap=>{
      const data = snap.data();
      if(data && data._createdOnce) return;
      try{window.evQueueAutoEvent&&window.evQueueAutoEvent('account_created',{},user);}catch(_e){}
      ref.set({ _createdOnce:true, createdAt: firebase.firestore.FieldValue.serverTimestamp() }, { merge:true }).catch(()=>{});
    }).catch(()=>{});
  }catch(e){ console.warn('Ethvelle: could not sync user profile doc', e); }
}

/* ================= SITE-CONTENT TRACKING (feeds the admin dashboard) =================
   Declared here (before goTo/trackEvent('visitor') below) since function declarations are
   hoisted but the const/let bindings they close over are not — calling trackEvent() from
   the bottom of the script before TRACKING_KEY had executed used to throw a
   "Cannot access before initialization" error that silently broke visitor counting.
   Every event bumps a localStorage counter immediately (so the dashboard always has
   something to show) and mirrors to Firestore in the background when configured, keyed by
   today's date so "today" counts reset naturally each day while totals keep accumulating.

   EXTENDED DATA MODEL (backward compatible):
   The original three fields (visitors/exports/projects) keep working exactly as before —
   every old call site (`trackEvent('visitor')`, `trackEvent('project')`, `trackEvent('export')`)
   still increments the same three counters, so historical totals are never broken or reset.
   New event kinds below just add MORE fields alongside those three, in the same dailyStats
   day-doc and the same stats/totals doc, using the same increment-based writes. A new
   monthlyStats/{YYYY-MM} doc is written alongside dailyStats so the admin dashboard can show
   month/quarter/year views without scanning every daily doc ever created.

   ANONYMOUS IDENTITY (privacy-conscious, no personal data):
   - visitorId: a random id persisted in localStorage — identifies "this browser" across visits,
     so the dashboard can distinguish page-load COUNT from actual unique visitors. Only a random
     string; nothing that identifies a person.
   - sessionId: a random id kept in sessionStorage — identifies "this tab session"; a fresh
     session starts whenever the app is opened in a new tab (sessionStorage is cleared then),
     so multiple tabs from the same visitor count as multiple sessions but the SAME visitorId.
   Both are anonymous identifiers with no link to a real identity unless the visitor separately
   signs in, in which case authenticatedUid (the existing Firebase UID) is used for user-level
   stats — never these anonymous ids. */
function todayKey(d){
  d = d || new Date();
  return d.getFullYear()+'-'+String(d.getMonth()+1).padStart(2,'0')+'-'+String(d.getDate()).padStart(2,'0');
}
function monthKey(d){
  d = d || new Date();
  return d.getFullYear()+'-'+String(d.getMonth()+1).padStart(2,'0');
}
function currentHour(){ return new Date().getHours(); } // 0-23, local time

/* ---- anonymous visitor/session identity ---- */
const VISITOR_ID_KEY = 'riggy.visitorId.v1';
const VISITOR_FIRSTSEEN_KEY = 'riggy.visitorFirstSeen.v1';
const SESSION_ID_KEY = 'riggy.sessionId.v1'; // sessionStorage — one per tab, cleared when the tab closes
function makeAnonId(){ return 'v_'+Date.now().toString(36)+Math.random().toString(36).slice(2,10); }
function getOrCreateVisitorId(){
  try{
    let id = localStorage.getItem(VISITOR_ID_KEY);
    if(!id){
      id = makeAnonId();
      localStorage.setItem(VISITOR_ID_KEY, id);
      localStorage.setItem(VISITOR_FIRSTSEEN_KEY, new Date().toISOString());
      return { id, isNew:true };
    }
    return { id, isNew:false };
  }catch(e){
    // localStorage unavailable (private browsing lockdown, etc) — fall back to a per-load id;
    // this visit will count as "unique" every time, which is the best available approximation.
    return { id: makeAnonId(), isNew:true };
  }
}
function getOrCreateSessionId(){
  try{
    let id = sessionStorage.getItem(SESSION_ID_KEY);
    if(!id){
      id = makeAnonId();
      sessionStorage.setItem(SESSION_ID_KEY, id);
      return { id, isNew:true };
    }
    return { id, isNew:false };
  }catch(e){
    return { id: makeAnonId(), isNew:true };
  }
}
// Tracks, once per calendar day per browser, whether THIS visitorId has already been counted as
// a "unique visitor" today — so refreshing the page 20 times in one day still shows as 1 unique
// visitor (though 20 plain "visitors"/page-loads), without needing any server-side lookup.
const VISITOR_LASTDAY_KEY = 'riggy.visitorLastCountedDay.v1';
function isFirstVisitToday(){
  try{
    const last = localStorage.getItem(VISITOR_LASTDAY_KEY);
    const today = todayKey();
    if(last === today) return false;
    localStorage.setItem(VISITOR_LASTDAY_KEY, today);
    return true;
  }catch(e){
    return true; // can't dedupe without storage — treat every load as unique rather than undercount
  }
}

/* Per-tab session activity log. It uses the same anonymous visitor/session IDs as the
   roll-up counters, stores no IP/location/device fingerprint, and logs only short action
   metadata (never payment proof contents or transaction IDs). Each event is a small child
   document so a session can have a bounded, queryable timeline without growing one Firestore
   document forever. */
let _ethvelleTrackingIdentity = null;
function getTrackingIdentity(){
  if(_ethvelleTrackingIdentity) return _ethvelleTrackingIdentity;
  const visitor = getOrCreateVisitorId();
  const session = getOrCreateSessionId();
  _ethvelleTrackingIdentity = { visitorId:visitor.id, sessionId:session.id };
  return _ethvelleTrackingIdentity;
}
function trackingPlanSnapshot(){
  const signedIn = !!currentUser;
  if(!signedIn) return { uid:null, displayName:null, email:null, planId:'guest', planName:'Guest' };
  const id = (myActivePlanId && String(myActivePlanId)!=='') ? String(myActivePlanId) : 'free';
  let name = myActivePlanName || '';
  try{
    const p = typeof getPlans==='function' ? getPlans().find(x=>String(x.id)===id) : null;
    if(p && p.name) name = p.name;
  }catch(e){}
  if(!name) name = id==='free' ? 'Free' : id;
  return { uid:currentUser.uid || null, displayName:currentUser.displayName || null,
    email:currentUser.email || null, planId:id, planName:name };
}

function trackOncePerTab(key, kind, details){
  const storageKey = 'riggy.session.once.' + key;
  try{
    if(sessionStorage.getItem(storageKey)==='1') return;
    sessionStorage.setItem(storageKey,'1');
  }catch(e){}
  trackEvent(kind, details);
}
// kind: any key of EVENT_FIELDS above (includes the original 'visitor' | 'export' | 'project')
let _lastPaymentPageViewAt = 0;
function trackEvent(kind, details){
  // Payment entry could be requested twice in quick succession by overlapping navigation
  // callbacks (notably promo-driven navigation). Count one page-view per actual entry, while
  // still allowing a genuine later return to Payment to count again. The in-memory guard also
  // covers restricted WebViews where sessionStorage can throw.
  if(kind === 'payment_page_view'){
    const stampKey = 'riggy.paymentPageView.lastAt.v1';
    const now = Date.now();
    if(_lastPaymentPageViewAt && now - _lastPaymentPageViewAt < 1800) return;
    try{
      const last = Number(sessionStorage.getItem(stampKey) || 0);
      if(last && now - last < 1800){ _lastPaymentPageViewAt = last; return; }
      sessionStorage.setItem(stampKey, String(now));
    }catch(e){ /* in-memory debounce remains active if storage is unavailable */ }
    _lastPaymentPageViewAt = now;
  }
  const fields = EVENT_FIELDS[kind];
  if(!fields || !fields.length) return; // unknown kind — no-op rather than throw, keeps callers safe
  const t = loadTracking();
  const key = todayKey();
  const hour = currentHour();
  if(!t.days[key]) t.days[key] = blankDay();
  // Older saved localStorage days (from before this update) may be missing newer fields —
  // backfill them in place so incrementing never throws on `undefined + 1`.
  const day = t.days[key];
  Object.keys(blankDay()).forEach(f=>{ if(f!=='hours' && day[f]===undefined) day[f]=0; });
  if(!day.hours) day.hours = {};
  if(!day.hours[hour]) day.hours[hour] = {};
  fields.forEach(field=>{
    day[field] = (day[field]||0) + 1;
    day.hours[hour][field] = (day.hours[hour][field]||0) + 1;
    t.totals[field] = (t.totals[field]||0) + 1;
  });
  saveTracking(t);
  // Detailed visitor timeline is independent of the roll-up counters. Failures are swallowed
  // inside the recorder so normal editing, payment and navigation continue as usual.
  recordVisitorSessionEvent(kind, details || {});
  // Mirrors this event to Firestore (dailyStats/{today}, monthlyStats/{YYYY-MM} + stats/totals)
  // so the separate admin panel's dashboard can show live, global counts covering every visitor
  // — not just this device. Uses FieldValue.increment (atomic on Firestore's servers) rather than
  // a read-modify-write, so two people using the app at the same moment both land correctly
  // instead of one overwriting the other's count. No-ops safely if Firestore is unreachable.
  if(firebaseReady && fbDb){
    try{
      const inc = firebase.firestore.FieldValue.increment(1);
      const dayPatch = {};
      fields.forEach(field=>{ dayPatch[field] = inc; });
      dayPatch.hours = {};
      fields.forEach(field=>{ dayPatch.hours[hour] = Object.assign(dayPatch.hours[hour]||{}, { [field]: inc }); });
      const dayRef = fbDb.collection('dailyStats').doc(key);
      dayRef.set(dayPatch, { merge:true }).catch(()=>{});

      const totalsPatch = {};
      fields.forEach(field=>{ totalsPatch[field] = inc; });
      const totalsRef = fbDb.collection('stats').doc('totals');
      totalsRef.set(totalsPatch, { merge:true }).catch(()=>{});

      // Monthly rollup — same fields, no hourly breakdown (the hourly chart only ever needs
      // "today", so monthlyStats skips that cost entirely). This lets the dashboard show a
      // month/quarter/year view by reading ~12-36 small docs instead of scanning every day.
      const mPatch = {};
      fields.forEach(field=>{ mPatch[field] = inc; });
      const monthRef = fbDb.collection('monthlyStats').doc(monthKey());
      monthRef.set(mPatch, { merge:true }).catch(()=>{});
    }catch(e){}
  }
}

/* ---- session/visitor lifecycle: called once per page load, right before the legacy
   trackEvent('visitor') call further down, so a single page view always produces at least
   the original 'visitor' count, PLUS a 'session_start' and (once per day per browser) a
   'unique_visitor' count. Kept as a separate small function so the legacy call site below
   stays a clean one-liner. ---- */
function detectTrafficSource(){
  // Prefer explicit campaign tags; otherwise classify the referring hostname. Store only the
  // category, never a full referrer URL or search query, so no unnecessary user data is logged.
  let params;
  try{ params = new URLSearchParams(window.location.search || ''); }catch(e){ params = null; }
  const utmSource = params ? String(params.get('utm_source') || params.get('source') || '').toLowerCase() : '';
  const utmMedium = params ? String(params.get('utm_medium') || '').toLowerCase() : '';
  let refHost = '';
  try{ refHost = document.referrer ? new URL(document.referrer).hostname.toLowerCase().replace(/^www\./,'') : ''; }catch(e){}
  const tagged = (utmSource + ' ' + utmMedium).trim();
  const hostAndTag = (tagged + ' ' + refHost).toLowerCase();
  if(/facebook|(^|[\s._-])fb($|[\s._-])|fb\.com|meta\.com/.test(hostAndTag)) return 'facebook';
  if(/youtube|youtu\.be|youtube-nocookie/.test(hostAndTag)) return 'youtube';
  if(/google|bing|duckduckgo|yahoo|yandex|ecosia|search/.test(hostAndTag)) return 'search';
  if(/instagram|tiktok|twitter|x\.com|linkedin|pinterest|reddit|threads\.net/.test(hostAndTag)) return 'social';
  if(!refHost || refHost === String(location.hostname || '').toLowerCase().replace(/^www\./,'')) return 'direct';
  return 'other';
}
function trackVisitAndSession(){
  getOrCreateVisitorId();       // ensures a persistent id exists; return value unused here
  const session = getOrCreateSessionId();
  if(session.isNew) trackEvent('session_start');
  if(isFirstVisitToday()) trackEvent('unique_visitor');
  const trafficSource = detectTrafficSource();
  // sessionStorage survives same-tab refreshes, preventing repeated attribution on reload.
  trackOncePerTab('traffic-source', 'traffic_source_' + trafficSource, { source:trafficSource });
}

      trackOncePerTab('payment-page', 'payment_page_session', { screen:'payment' });

const STORAGE_KEY = 'riggy.projects.v1';
let projects = loadProjectsFromStorage();
let pendingRatio="9:16";

function loadProjectsFromStorage(){
  try{
    const raw = localStorage.getItem(STORAGE_KEY);
    if(!raw) return [];
    const parsed = JSON.parse(raw);
    return Array.isArray(parsed) ? parsed : [];
  }catch(err){
    console.warn('Ethvelle: could not read saved projects', err);
    return [];
  }
}
// BUGFIX ("it show a use smaller background for save notification all time" — the toast
// kept reappearing on every single edit once a project was over quota): persistProjectState
// calls this on almost every interaction (every drag, every trim, every keyframe, many with
// skipHistory:true for live-updates-while-dragging), so once localStorage's ~5-10MB quota
// was exceeded, EVERY one of those calls failed and re-showed the exact same toast — it
// wasn't stuck on screen, it was being re-triggered dozens of times a minute, which reads
// identically to "won't go away". lastQuotaToastAt throttles it to once per 8 seconds no
// matter how many saves fail in that window, so the person can keep editing without a wall
// of repeat toasts, while still being told the truth the moment they pause.
let lastQuotaToastAt = 0;
function saveProjectsToStorage(){
  try{
    localStorage.setItem(STORAGE_KEY, JSON.stringify(projects));
    return true;
  }catch(err){
    // A background image, or any of several background/overlay images/videos now that both
    // rows support multiple clips, is stored as base64 (see the various upload handlers'
    // BUGFIX notes) rather than a tiny blob: URL reference, which makes hitting
    // localStorage's quota (~5-10MB depending on browser) a real possibility once a project
    // has accumulated enough media. Surfacing this with a toast instead of only a
    // console.warn matters: silently failing to save would mean edits look fine in the live
    // editor but quietly never persist, only to vanish on next reopen.
    console.warn('Ethvelle: could not save projects', err);
    const now = Date.now();
    if(now - lastQuotaToastAt > 8000){
      lastQuotaToastAt = now;
      // BUGFIX: no longer blames "background image" specifically — with multiple
      // backgrounds AND multiple overlay clips now possible, the actual cause just as often
      // is several images/videos added over time rather than one large background, so the
      // message names the real, fixable options instead of pointing at only one of them.
      showToast("Storage is full — try removing an unused clip/background or using a smaller image", {icon: toastIcons.info, duration:3200});
    }
    return false;
  }
}

function getAvailableModelAssets(){
  const all=[...(Array.isArray(modelAssets)?modelAssets:[]),...(Array.isArray(publishedModelAssets)?publishedModelAssets:[])];
  const seen=new Set();return all.filter(a=>{const id=String(a&&a.id||'');if(!id||seen.has(id))return false;seen.add(id);return true;}).sort((a,b)=>(Number(b.updatedAt||b.createdAt)||0)-(Number(a.updatedAt||a.createdAt)||0));
}
function loadModelAssetsFromStorage(){
  try{
    const raw = localStorage.getItem(MODEL_ASSETS_KEY);
    if(!raw) return [];
    const parsed = JSON.parse(raw);
    return Array.isArray(parsed) ? parsed : [];
  }catch(err){
    console.warn('Ethvelle: could not read saved model assets', err);
    return [];
  }
}
let lastModelAssetQuotaToastAt = 0;
function saveModelAssetsToStorage(){
  try{
    localStorage.setItem(MODEL_ASSETS_KEY, JSON.stringify(modelAssets));
    return true;
  }catch(err){
    console.warn('Ethvelle: could not save model assets', err);
    const now = Date.now();
    if(now - lastModelAssetQuotaToastAt > 8000){
      lastModelAssetQuotaToastAt = now;
      showToast("Storage is full — try deleting an unused saved model", {icon: toastIcons.info, duration:3200});
    }
    return false;
  }
}

// Saves a character's rig art (auto-rig source image + joints, exactly what's needed to
// rebuild it later — see applyRigOverride/PART_SLOTS-style `parts`) as a standalone, named
// asset in the library, independent of the project it was created in. Called once right
// after a brand-new custom character finishes rigging (see finalizeAutoRiggedCharacter) —
// every uploaded model is a reusable asset by default, not an opt-in extra step.
function saveCharacterAsModelAsset(character, opts){
  const asset = {
    id: makeModelAssetId(),
    name: character.name || 'Model',
    type: 'custom',
    modelType: character.modelType || (character.sheet && character.sheet.modelType) || 'humanoid',
    parts: character.parts ? JSON.parse(JSON.stringify(character.parts)) : undefined,
    rigOverride: character.rigOverride ? JSON.parse(JSON.stringify(character.rigOverride)) : undefined,
    autoRig: character.autoRig ? JSON.parse(JSON.stringify(character.autoRig)) : undefined,
    sheet: character.sheet ? JSON.parse(JSON.stringify(character.sheet)) : undefined,
    thumb: character.thumb || (character.sheet && character.sheet.thumb) || undefined,
    defaultAnimKey: character.clip ? character.clip.animKey : null,
    createdAt: Date.now(),
    updatedAt: Date.now()
  };
  modelAssets.push(asset);
  character.assetId = asset.id;
  const saved = saveModelAssetsToStorage();
  trackEvent('model_saved');
  if(!(opts && opts.silent) && saved) showToast('Saved to your models', {icon: toastIcons.save, duration:1400});

}
// Builds a brand-new character (own rig + own default animation clip) from a saved model
// asset and adds it to the CURRENT project — this is the "use this model in any project"
// action. Mirrors finalizeAutoRiggedCharacter's own character-construction path exactly, just
// sourced from a saved asset's data instead of a freshly-uploaded image.
function addCharacterFromModelAsset(assetOrId){
  // Resolve the live library entry again at click time. The card can outlive an edit, and a
  // published copy can be stale; the locally edited asset is always the authoritative version.
  const id=String(assetOrId && typeof assetOrId==='object' ? assetOrId.id : assetOrId || '');
  const local=modelAssets.find(a=>String(a.id)===id);
  const published=Array.isArray(publishedModelAssets)?publishedModelAssets.find(a=>String(a.id)===id):null;
  const source=local || published || (assetOrId && typeof assetOrId==='object' ? assetOrId : null);
  if(!source) return null;
  const asset=JSON.parse(JSON.stringify(source));
  // The Lab's sheet override lives inside `sheet`; never feed it into the main site's legacy rig.
  const c = makeCharacter(asset.name, asset.sheet ? undefined : asset.rigOverride);
  c.type = 'custom';
  c.modelType = (asset.sheet && asset.sheet.modelType) || asset.modelType || 'humanoid';
  c.thumb = asset.thumb || (asset.sheet && asset.sheet.thumb) || undefined;
  c.assetId = asset.id;
  c.sheet = asset.sheet ? JSON.parse(JSON.stringify(asset.sheet)) : undefined;
  ensureSpeciesTailBone(c);
  c.expr = c.sheet ? (typeof sheetExprFresh === 'function' ? sheetExprFresh() : {}) : undefined;
  c.parts = c.sheet ? {} : (asset.parts ? JSON.parse(JSON.stringify(asset.parts)) : undefined);
  c.autoRig = c.sheet ? undefined : (asset.autoRig ? JSON.parse(JSON.stringify(asset.autoRig)) : undefined);
  if(c.sheet){ c.rigOverride = undefined; sheetPlace(c); }
  // UPDATE (default placeholder clip on add): makeCharacter() now gives every new character
  // a 'default' (static, no-motion) clip automatically — see makeCharacter's own comment —
  // so a model pulled in from the library shows up on the timeline immediately too, same as
  // plain addCharacter()/finalizeAutoRiggedCharacter. No forced 'walk'/remembered-default
  // animation here, just the same static placeholder every other creation path gets.
  characters.push(c);
  activeCharIdx = characters.length-1;
  syncActiveCharacterRefs();
  selectedJoint = null;
  if(navigator.vibrate) navigator.vibrate(12);
  trackEvent('model_reused');
  showToast('Model added', { duration:1400 });
  preloadCustomCharacterImages(ctx, [c]).then(()=>c.sheet ? undefined : rebuildSkinMeshesForCharacters([c])).then(()=>{
    // c.clip is the 'default' placeholder (see makeCharacter) — a static pose, so this just
    // draws the model in its raw rest pose, same as plain addCharacter().
    drawStick();
    renderCharacterLayers(); syncTransformSliders(); persistProjectState();
    renderTimelineClip(); syncTimelineInnerWidth(); renderRuler();
  });
  return c;
}

  }
  return '';
}

// A project cover represents only the content visible at absolute time zero. Compare that
// slice of the saved state before scheduling a refresh, so routine timeline scrubs, selection
// changes, keyframe edits away from zero, etc. never queue a cover capture.
function ethvelleFrameZeroCoverContentView(state){
  if(!state || typeof state!=='object') return null;
  const atZero=item=>!!item && (Number(item.trimStart)||0)<=0.0001 && Number(item.trimEnd)>0.0001;
  const pick=(item,keys)=>{
    const out={};
    keys.forEach(key=>{if(item && Object.prototype.hasOwnProperty.call(item,key))out[key]=item[key];});
    return out;
  };
  const backgroundRows=Array.isArray(state.backgrounds)?state.backgrounds:[];
  const overlaysRows=Array.isArray(state.overlays)?state.overlays:[];
  const charactersRows=Array.isArray(state.characters)?state.characters:[];
  return {
    backgrounds:backgroundRows.filter(atZero).map(b=>pick(b,[
      'id','src','trimStart','trimEnd','transform','keyframes','ethvelleMediaKeyframes','mediaPerspectiveX','mediaPerspectiveY','mediaWide','mediaLength'
    ])),
    legacyBackground:backgroundRows.length?null:pick(state,['currentBg','bgTransform','bgDuration']),
    overlays:overlaysRows.filter(atZero).map(o=>pick(o,[
      'id','type','src','trimStart','trimEnd','sourceDuration','sourceOffset','speed',
      'reversePlayback','__ethvelleBehindBackground','transform','opacity','adjustments','filterPreset','chromaKey','mediaPerspectiveX','mediaPerspectiveY','mediaWide','mediaLength',
      'keyframes','ethvelleMediaKeyframes','textContent','fontFamily','fontSize',
      'fontWeight','fontStyle','textAlign','textColor','stroke'
    ])),
    characters:charactersRows.map(c=>({
      id:c.id,type:c.type,modelType:c.modelType,parts:c.parts,rigOverride:c.rigOverride,
      autoRig:c.autoRig,expr:c.expr,handProps:c.handProps,
      restTransform:c.restTransform||c.modelTransform,
      clip:atZero(c.clip)?c.clip:null,
      segments:(Array.isArray(c.segments)?c.segments:[]).filter(atZero)
    }))
  };
}
function ethvelleFrameZeroCoverContentEqual(a,b){
  if(a===b)return true;
  if(a==null||b==null||typeof a!==typeof b)return false;
  if(typeof a!=='object')return a===b;
  const aa=Array.isArray(a),bb=Array.isArray(b);
  if(aa!==bb)return false;
  const ak=Object.keys(a),bk=Object.keys(b);
  if(ak.length!==bk.length)return false;
  for(let i=0;i<ak.length;i++){
    const key=ak[i];
    if(!Object.prototype.hasOwnProperty.call(b,key)||!ethvelleFrameZeroCoverContentEqual(a[key],b[key]))return false;
  }
  return true;
}
// Build isolated character objects for first-frame cover rendering. Bone records and model
// transforms are cloned, so reset/pose evaluation never mutates the live editor's rig.
function ethvelleCloneCharactersForFirstFrameCover(){
  return characters.map(char=>{
    const rest={...(char.restTransform||char.modelTransform||{})};
    const copy={
      ...char,
      bones:Object.fromEntries(Object.entries(char.bones||{}).map(([key,bone])=>[key,{...bone}])),
      baseAngles:{...(char.baseAngles||{})},
      modelTransform:{...rest},
      restTransform:{...rest}
    };
    resetCharacterForTimelineFrame(copy);
    if(copy.clip){
      const pose=poseAtClipTime(0,copy.clip,copy.segments||[]);
      if(pose)applyPoseToBones(pose,copy.bones,copy.baseAngles,copy.hipRestY);
      const clipTransform=transformAtClipTime(0,copy.clip,copy.segments||[]);
      if(clipTransform)applyTransformToModel(clipTransform,copy.modelTransform);
    }
    return copy;
  });
}

// Persist the live editor state (pose, clip, transform, bg, overlays) into the open project
function persistProjectState(opts){
  if(!currentProject) return;
  const previousSavedStateForCover=currentProject.savedState;
  currentProject.savedState = {
    currentAnimKey,
    timelineDuration,
    // Preserve the timeline position so time-based text overlays are still visible after reload.
    playheadTimeSec: Number.isFinite(Number(currentAbsoluteTimeSec)) ? Math.max(0, Number(currentAbsoluteTimeSec)) : 0,
    // Every character's own Transform (size/position/rotation) AND its own animation
    // clip/segments — bone pose itself is never persisted (same as the original
    // single-character app), only rebuilt fresh at rest pose on reopen. modelTransform
    // (singular, top-level) is intentionally no longer written — openEditor's migration path
    // reads it only from OLD saves that predate this array. clip/segments are saved per
    // character now (not as one shared top-level activeClip/splitStickmanSegments) so every
    // character keeps its own animation independently across save/reload, not just whichever
    // one happened to be active when you saved.
    characters: characters.map(c=>({
      id:c.id, name:c.name, modelTransform:{...c.modelTransform},
      restTransform:{...(c.restTransform || c.modelTransform)},
      clip: c.clip ? JSON.parse(JSON.stringify(c.clip)) : null,
      segments: JSON.parse(JSON.stringify(c.segments||[])),
      // Custom (image-based) character data — see PART_SLOTS/finishCustomCharacterBuild.
      // `parts` holds each uploaded part's base64 image data URL directly (same pattern as
      // overlays' `src` and currentBg above), so a custom character's artwork survives a
      // reload exactly like an uploaded background/clip image already does. Undefined for
      // an ordinary stickman character, so old saves and new plain characters both stay the
      // same tiny size.
      type: c.type,
      parts: c.parts ? JSON.parse(JSON.stringify(c.parts)) : undefined,
      // Auto-rig data (see buildRigOverrideFromJoints/applyRigOverride and autorig-detect's
      // ported detectCharacter): `rigOverride` is the bone length/angle/attachAt fractions
      // openEditor's restore path re-applies on top of a fresh buildFreshRig() (bone pose is
      // never persisted, per the note above, but these PROPORTIONS must be, or an auto-rigged
      // character would revert to the generic stickman proportions on reload). `autoRig` keeps
      // the original uploaded image + detected/adjusted joints so "Rig my art" can re-open this
      // character straight into the Auto Rig Preview for further correction later (section 16/17
      // of the feature — re-detect must start from the ORIGINAL image, not a re-compressed part
      // crop). Both undefined for a stickman or an old manual-parts custom character.
      rigOverride: c.rigOverride ? JSON.parse(JSON.stringify(c.rigOverride)) : undefined,
      autoRig: c.autoRig ? JSON.parse(JSON.stringify(c.autoRig)) : undefined,
      modelType: c.modelType || undefined,
      thumb: c.thumb || undefined,
      sheet: c.sheet ? JSON.parse(JSON.stringify(c.sheet)) : undefined,
      expr: c.expr ? JSON.parse(JSON.stringify(c.expr)) : undefined,
      handProps: c.handProps ? JSON.parse(JSON.stringify(c.handProps)) : undefined,
      assetId: c.assetId || undefined
    })),
    activeCharIdx,
    // FEATURE (multiple sequential backgrounds): `backgrounds` is now the real source of
    // truth — an array of clips, same shape as `overlays`, each with its own src/transform/
    // trimStart/trimEnd. bgTransform/currentBg/bgDuration below are kept ADDITIONALLY,
    // written from the currently-active clip, purely so a project saved by an older version
    // of this file (or opened by code that hasn't been updated) still has something sane to
    // read — they are never the authoritative data once `backgrounds` is present (see
    // openEditor's migration path below, which only falls back to them when `backgrounds`
    // is missing entirely from an old save).
    backgrounds: backgrounds.map(b=>({...b, transform:{...b.transform}})),
    bgTransform: {...bgTransform},
    currentBg,
    bgDuration,
    // BUGFIX (this was the "clips aren't saving after refreshing" bug — see the fileInput
    // 'change' handler's matching BUGFIX note for the fix itself): `src` now always holds a
    // `url(data:...)`-free base64 data: URL string too (image/video/audio alike), exactly
    // like currentBg above — never a `url(blob:...)` reference. Blob URLs die the moment the
    // page session ends, so a saved clip's actual picture/video/audio used to go blank on
    // reopen even though its trim/position data (and thumbs/peaks, already plain data)
    // survived fine — this spread now carries a `src` that survives right along with them.
    overlays: overlays.map(o=>({...o, thumbs:o.thumbs?[...o.thumbs]:undefined, peaks:o.peaks?[...o.peaks]:undefined}))
  };
  currentProject.duration = activeClip ? +(combinedStickmanDuration() - activeClip.trimStart).toFixed(1) : currentProject.duration;
  const idx = projects.findIndex(p=>p.id===currentProject.id);
  if(idx>-1) projects[idx] = currentProject;
  saveProjectsToStorage();
  const previousFrameZeroCoverContent=ethvelleFrameZeroCoverContentView(previousSavedStateForCover);
  const nextFrameZeroCoverContent=ethvelleFrameZeroCoverContentView(currentProject.savedState);
  if(!ethvelleFrameZeroCoverContentEqual(previousFrameZeroCoverContent,nextFrameZeroCoverContent))queueProjectFirstFrameCover(1100);
  if(!opts || !opts.skipHistory) pushHistory();
}

/* ---------------- Undo / redo history ---------------- */
let historyStack = [];
let historyIndex = -1;
let restoringHistory = false;
const HISTORY_LIMIT = 40;

function snapshotState(){
  return JSON.stringify({
    currentAnimKey,
    activeClip,
    splitStickmanSegments,
    timelineDuration,
    bgDuration,
    characters,
    activeCharIdx,
    bgTransform,
    currentBg,
    backgrounds,
    overlays
  });
}
function pushHistory(){
  if(restoringHistory) return;
  const snap = snapshotState();
  if(historyIndex>=0 && historyStack[historyIndex]===snap) return; // no real change
  historyStack = historyStack.slice(0, historyIndex+1);
  historyStack.push(snap);
  if(historyStack.length>HISTORY_LIMIT) historyStack.shift();
  historyIndex = historyStack.length-1;
  updateHistoryButtons();
}
function resetHistory(){
  historyStack = [snapshotState()];
  historyIndex = 0;
  updateHistoryButtons();
}
function updateHistoryButtons(){
  const undoBtn = document.getElementById('undoBtn');
  const redoBtn = document.getElementById('redoBtn');
  if(!undoBtn || !redoBtn) return;
  const canUndo = historyIndex > 0;
  const canRedo = historyIndex < historyStack.length-1;
  undoBtn.classList.toggle('history-available', canUndo);
  redoBtn.classList.toggle('history-available', canRedo);
  undoBtn.style.opacity = canUndo ? '1' : '0.4';
  redoBtn.style.opacity = canRedo ? '1' : '0.4';
  undoBtn.setAttribute('aria-disabled', canUndo ? 'false' : 'true');
  redoBtn.setAttribute('aria-disabled', canRedo ? 'false' : 'true');
}
function applySnapshot(snap){
  const s = JSON.parse(snap);
  restoringHistory = true;
  // Undo/redo mutates activeClip/segments/keyframes directly — never do that while the
  // playback loop might be mid-tick reading them (PART 36: history must apply cleanly,
  // not race a running animation).
  pausePlayback();
  currentAnimKey = s.currentAnimKey;
  timelineDuration = s.timelineDuration;
  // Full rig objects were captured verbatim in this snapshot (unlike the lighter
  // Transform-only shape used for disk saves) — undo/redo restores bone pose exactly as it
  // was at that point in history, since it's an in-memory-only structure with no migration
  // concerns the way older disk saves have. Each character's own clip/segments live nested
  // on the character object itself (see makeCharacter) and are restored as part of this.
  characters = s.characters && s.characters.length ? s.characters : characters;
  // Old history entries saved before per-character clips existed kept activeClip/
  // splitStickmanSegments only at the top level, not nested on any character — patch those
  // onto the (single, pre-multi-character) character here so undo/redo across that boundary
  // never silently drops the clip that was on the timeline at the time.
  characters.forEach(c=>{ if(c.clip===undefined) c.clip = null; if(c.segments===undefined) c.segments = []; });
  activeCharIdx = Math.min(s.activeCharIdx ?? 0, characters.length-1);
  if(s.activeClip !== undefined && characters[activeCharIdx] && !characters[activeCharIdx].clip){
    characters[activeCharIdx].clip = s.activeClip;
    characters[activeCharIdx].segments = s.splitStickmanSegments || [];
  }
  syncActiveCharacterRefs();
  // FEATURE (multiple sequential backgrounds): `backgrounds` is the real data now. Old
  // history entries captured before this array existed only have the single legacy
  // currentBg/bgTransform/bgDuration fields — migrate those into a single one-clip
  // `backgrounds` array on the fly so undo/redo across that boundary never silently loses
  // whatever background was set at that point in history.
  if(s.backgrounds){
    backgrounds = s.backgrounds;
  } else if(s.currentBg){
    backgrounds = [{ id:'bg'+Date.now(), src:s.currentBg, trimStart:0, trimEnd: s.bgDuration ?? s.timelineDuration, transform: s.bgTransform ? {rotation:0, ...s.bgTransform} : {scale:1, offsetX:0, offsetY:0, rotation:0} }];
  } else {
    backgrounds = [];
  }
  refreshCurrentBgFromPlayhead();
  overlays = s.overlays || [];
  applyBackground();
  deselectAllClips();
  frame = 0;
  currentAbsoluteTimeSec = 0;
  renderOverlayLayers();
  renderBackgroundTrack();
  renderCharacterLayers(); // rebuilds every character row AND the active one's clip markup
  renderFrame();
  syncTransformSliders();
  persistProjectState({skipHistory:true});
  restoringHistory = false;
}
function undoHistory(){
  if(historyIndex<=0) return;
  historyIndex--;
  applySnapshot(historyStack[historyIndex]);
  updateHistoryButtons();
  showToast('Undo', {icon: toastIcons.info, duration:900});
}
function redoHistory(){
  if(historyIndex>=historyStack.length-1) return;
  historyIndex++;
  applySnapshot(historyStack[historyIndex]);
  updateHistoryButtons();
  showToast('Redo', {icon: toastIcons.info, duration:900});
}

function openRatioModal(){
  // Real project-count gating: if the signed-in user's plan caps how many projects they can
  // have and they're already at (or over) that cap, block here with an upgrade prompt
  // instead of opening the "new project" sheet at all.

// Renders the Projects → Models grid. The grid always opens with an "Add model" tile (so
// uploading a new model never requires starting a project first — see the "add model's
// batton" request), followed by every saved model asset. Tapping a saved model's thumbnail/
// name opens it straight in the project-free model editor (openModelAssetEditor) so it can be
// re-posed/re-rigged in place; the small pencil/trash icons on the corner rename or delete the
// asset itself, same as the in-project "My models" library sheet already does.
// Model-grid preview helpers. The sheet compiler's legacy `thumb` is intentionally just a
// small head portrait. For sheet-based models we render the saved neutral/rest rig once into an
// offscreen canvas and crop to the visible alpha bounds, producing a full-body preview without
// changing any rig data or the thumbnail used elsewhere in the editor.
const modelGridPreviewPending = new Map();
// v299: model cards now show ONLY the model's face (its neutral head part) rendered from the
// original full-resolution head artwork, not a full-body crop. The result is cached in memory
// only (never written to localStorage) so large high-quality previews cannot bloat saved data.
const modelFacePreviewCache = new Map();
const modelFacePreviewFailed = new Set();
const modelFacePreviewPending = new Map();
function modelFaceSourceFor(asset){
  try{
    if(asset && asset.sheet && asset.sheet.parts && asset.sheet.parts.head && asset.sheet.parts.head.src) return String(asset.sheet.parts.head.src);
    if(asset && asset.parts && asset.parts.head && asset.parts.head.src) return String(asset.parts.head.src);
  }catch(_e){}
  return '';
}
function modelGridThumbSrc(asset){
  const faceSrc = modelFaceSourceFor(asset);
  if(faceSrc){
    if(modelFacePreviewCache.has(faceSrc)) return modelFacePreviewCache.get(faceSrc);
    if(modelFacePreviewFailed.has(faceSrc)) return (asset && asset.thumb) || modelAssetThumbSrc(asset);
    return ''; // wait for the high-quality face render instead of flashing a full-body image
  }
  if(asset && asset.autoRig && asset.autoRig.sourceImage) return asset.autoRig.sourceImage;
  if(asset && asset.thumb) return asset.thumb;
  return asset ? modelAssetThumbSrc(asset) : '';
}
// Face tiles are rendered as a 2048px (2K) square so they stay sharp on high-density phone screens.
// The head artwork is enlarged in gentle <=2x steps (one big jump looks soft or jagged) and then given a
// light unsharp-mask pass. The result is held in memory as a blob URL only - never written to localStorage.
const MODEL_FACE_PREVIEW_PX = 2048;
let modelFaceQueue = Promise.resolve();
function modelFaceResizeSteps(src, tw, th){
  let cur = src, guard = 0;
  while((cur.width !== tw || cur.height !== th) && guard++ < 14){
    const nw = tw > cur.width ? Math.min(tw, cur.width*2) : Math.max(tw, Math.ceil(cur.width/2));
    const nh = th > cur.height ? Math.min(th, cur.height*2) : Math.max(th, Math.ceil(cur.height/2));
    const c = document.createElement('canvas'); c.width = Math.max(1,nw); c.height = Math.max(1,nh);
    const x = c.getContext('2d'); x.imageSmoothingEnabled = true; x.imageSmoothingQuality = 'high';
    x.drawImage(cur,0,0,cur.width,cur.height,0,0,c.width,c.height);
    cur = c;
  }
  return cur;
}
function modelFaceUnsharp(canvas, radius, amount){
  try{
    const ctx = canvas.getContext('2d');
    if(!('filter' in ctx)) return;
    const w = canvas.width, h = canvas.height;
    const blur = document.createElement('canvas'); blur.width = w; blur.height = h;
    const bc = blur.getContext('2d'); bc.filter = 'blur('+radius+'px)'; bc.drawImage(canvas,0,0);
    const a = ctx.getImageData(0,0,w,h), b = bc.getImageData(0,0,w,h), ad = a.data, bd = b.data;
    for(let i=0;i<ad.length;i+=4){
      if(ad[i+3] < 40 || bd[i+3] < 90) continue;   // leave the transparent edge alone: no halos
      for(let c=0;c<3;c++){
        const o = ad[i+c], diff = o-bd[i+c];
        if(diff > 2 || diff < -2){ const v = o+diff*amount; ad[i+c] = v < 0 ? 0 : v > 255 ? 255 : v; }
      }
    }
    ctx.putImageData(a,0,0);
  }catch(err){ console.warn('Ethvelle: face sharpen skipped',err); }
}
// Name kept for existing callers; it now builds the face-only preview.
function generateModelGridFullBodyPreview(asset){
  const faceSrc = modelFaceSourceFor(asset);
  if(!faceSrc) return Promise.resolve('');
  if(modelFacePreviewCache.has(faceSrc)) return Promise.resolve(modelFacePreviewCache.get(faceSrc));
  if(modelFacePreviewPending.has(faceSrc)) return modelFacePreviewPending.get(faceSrc);
  const task = (modelFaceQueue = modelFaceQueue.then(()=>new Promise(r=>setTimeout(r,0))).then(()=>new Promise(resolve=>{
    const img = new Image();
    img.onload = ()=>{
      try{
        const iw = img.naturalWidth, ih = img.naturalHeight;
        if(!iw || !ih){ resolve(''); return; }
        // Crop to the visible (non-transparent) pixels so the face fills the card.
        let sx=0, sy=0, sw=iw, sh=ih;
        if(iw*ih <= 9000000){
          const probe = document.createElement('canvas'); probe.width=iw; probe.height=ih;
          const pc = probe.getContext('2d',{willReadFrequently:true});
          pc.drawImage(img,0,0);
          const px = pc.getImageData(0,0,iw,ih).data;
          let minX=iw, minY=ih, maxX=-1, maxY=-1;
          for(let y=0;y<ih;y++){
            const row=y*iw*4;
            for(let x=0;x<iw;x++){
              if(px[row+x*4+3]>12){ if(x<minX)minX=x; if(x>maxX)maxX=x; if(y<minY)minY=y; if(y>maxY)maxY=y; }
            }
          }
          if(maxX>=minX && maxY>=minY){
            const pad = Math.max(2,Math.round(Math.max(maxX-minX+1,maxY-minY+1)*.03));
            minX=Math.max(0,minX-pad); minY=Math.max(0,minY-pad);
            maxX=Math.min(iw-1,maxX+pad); maxY=Math.min(ih-1,maxY+pad);
            sx=minX; sy=minY; sw=maxX-minX+1; sh=maxY-minY+1;
          }
        }
        // Square 2K output. The face is fitted to ~96% of the tile and ENLARGED when the artwork is smaller,
        // so every card gets a full-resolution tile instead of a small bitmap the browser has to stretch.
        const SIZE = MODEL_FACE_PREVIEW_PX, longest = Math.max(sw,sh);
        const scale = (SIZE*.96)/longest;
        const tw = Math.max(1,Math.round(sw*scale)), th = Math.max(1,Math.round(sh*scale));
        const crop = document.createElement('canvas'); crop.width = sw; crop.height = sh;
        crop.getContext('2d').drawImage(img,sx,sy,sw,sh,0,0,sw,sh);
        const big = modelFaceResizeSteps(crop,tw,th);
        const out = document.createElement('canvas'); out.width = SIZE; out.height = SIZE;
        const oc = out.getContext('2d'); oc.imageSmoothingEnabled = true; oc.imageSmoothingQuality = 'high';
        oc.drawImage(big,Math.round((SIZE-tw)/2),Math.round((SIZE-th)/2));
        if(scale > 1.25) modelFaceUnsharp(out, Math.min(4,Math.max(1.2,scale*.4)), Math.min(1.1,.45+scale*.08));
        else modelFaceUnsharp(out, 1, .35);
        if(out.toBlob) out.toBlob(blob=>resolve(blob ? URL.createObjectURL(blob) : out.toDataURL('image/png')),'image/png');
        else resolve(out.toDataURL('image/png'));
      }catch(err){
        console.warn('Ethvelle: could not build face model preview',err);
        resolve('');
      }
    };
    img.onerror = ()=>resolve('');
    img.src = faceSrc;
  }))).then(url=>{
    modelFacePreviewPending.delete(faceSrc);
    if(url) modelFacePreviewCache.set(faceSrc,url); else modelFacePreviewFailed.add(faceSrc);
    return url;
  });
  modelFacePreviewPending.set(faceSrc,task);
  return task;
}

   Long-press any project card (~450ms) to enter select mode. In select mode a tap toggles the

const MOVE_TOLERANCE = 10; // px of finger drift that cancels a long-press (so scrolling doesn't select)

function updateSelectUI(){
  const screen = document.getElementById('screen-projects');
  screen.classList.toggle('select-mode', selectMode);
  const n = selectedProjectIds.size;
  const title = document.getElementById('selectCount');
  if(title) title.textContent = 'Select projects';
  const deleteButton = document.getElementById('selectDeleteBtn');
  if(deleteButton) deleteButton.disabled = n === 0;
}
function enterSelectMode(firstId){
  selectMode = true;
  selectedProjectIds.clear();
  if(firstId != null) selectedProjectIds.add(firstId);
  if(navigator.vibrate){ try{ navigator.vibrate(18); }catch(e){} }
  renderProjects();
  updateSelectUI();
}
function exitSelectMode(){
  selectMode = false;
  selectedProjectIds.clear();
  renderProjects();
  updateSelectUI();
}
function toggleProjectSelected(id){
  if(selectedProjectIds.has(id)) selectedProjectIds.delete(id); else selectedProjectIds.add(id);
  // update just this card so the tap feels instant (no full re-render / flicker)
  const card = document.querySelector('.project-card[data-pid="'+id+'"]');
  if(card) card.classList.toggle('selected', selectedProjectIds.has(id));
  if(selectedProjectIds.size === 0){ exitSelectMode(); return; }
  updateSelectUI();
}
function toggleSelectAll(){
  if(selectedProjectIds.size === projects.length) selectedProjectIds.clear();
  else projects.forEach(p=>selectedProjectIds.add(p.id));
  if(selectedProjectIds.size === 0){ exitSelectMode(); return; }
  renderProjects();
  updateSelectUI();
}
function openBulkDeleteConfirm(){
  const n = selectedProjectIds.size;
  if(n === 0) return;
  document.getElementById('bulkDeleteTitle').textContent = n === 1 ? 'Delete 1 project?' : 'Delete ' + n + ' projects?';
  document.getElementById('bulkDeleteBtn').textContent = n === 1 ? 'Delete project' : 'Delete ' + n + ' projects';
  document.getElementById('bulkDeleteBackdrop').classList.add('active');
}
function closeBulkDeleteConfirm(){
  document.getElementById('bulkDeleteBackdrop').classList.remove('active');
}
function deleteSelectedConfirmed(){
  const n = selectedProjectIds.size;
  if(n === 0){ closeBulkDeleteConfirm(); return; }
  projects = projects.filter(p=>!selectedProjectIds.has(p.id));
  saveProjectsToStorage();
  for(let i=0;i<n;i++) trackEvent('project_deleted'); // count each deleted project individually
  closeBulkDeleteConfirm();
  selectMode = false;
  selectedProjectIds.clear();
  renderProjects();
  updateSelectUI();
  showToast(n === 1 ? 'Project deleted' : n + ' projects deleted', {icon: toastIcons.check, duration:1400});
}

(function wireProjectGesture(){
  const grid = document.getElementById('projectGrid');
  let timer = null, startX = 0, startY = 0, pressedId = null, longFired = false;

  function clear(){ clearTimeout(timer); timer = null; }
  grid.addEventListener('pointerdown', e=>{
    if(e.target.closest('.project-more')) return;
    const card = e.target.closest('.project-card');
    if(!card) return;
    pressedId = Number(card.dataset.pid);
    startX = e.clientX; startY = e.clientY; longFired = false;
    clear();
    if(!selectMode){
      timer = setTimeout(()=>{
        longFired = true;
        enterSelectMode(pressedId);
      }, LONG_PRESS_MS);
    }
  });
  grid.addEventListener('pointermove', e=>{
    if(timer && (Math.abs(e.clientX-startX) > MOVE_TOLERANCE || Math.abs(e.clientY-startY) > MOVE_TOLERANCE)) clear();
  });
  ['pointerup','pointercancel','pointerleave'].forEach(t=>grid.addEventListener(t, clear));
  // click fires after pointerup: a long-press must NOT also count as a tap/open
  grid.addEventListener('click', e=>{
    if(e.target.closest('.project-more')) return;
    const card = e.target.closest('.project-card');
    if(!card) return;
    if(longFired){ longFired = false; return; }
    const id = Number(card.dataset.pid);
    if(selectMode) toggleProjectSelected(id);
    else openEditorById(id);
  });
  // stop the browser's long-press context menu / image-save menu on cards
  grid.addEventListener('contextmenu', e=>{ if(e.target.closest('.project-card')) e.preventDefault(); });
})();

let selectedPlan = null;
function selectPlan(el, userAction){
  if(!el) return;
  document.querySelectorAll('#screen-payment .plan-card').forEach(c=>c.classList.remove('selected'));
  el.classList.add('selected');
  selectedPlan = el.dataset.plan;
  try{ localStorage.setItem('riggy.plan.selected', selectedPlan); }catch(e){}
  if(userAction){
    const p = (typeof getPlans==='function' ? getPlans().find(x=>String(x.id)===String(selectedPlan)) : null) || {id:selectedPlan, name:el.dataset.name||selectedPlan, price:''};
    const detail={ planId:p.id, planName:p.name||p.id, planPrice:p.price };
    trackEvent('plan_selection', withPromoAttribution(detail));
    trackPromoAttributedEvent('promo_plan_card_click', detail);
  }
}
function confirmPlan(plan){
  const card = document.querySelector('#screen-payment .plan-card[data-plan="'+plan+'"]');
  if(card && !card.classList.contains('selected')) selectPlan(card);
  let planData = getPlans().find(p=>p.id===plan);
  if(!planData){
    // Rare fallback: confirmPlan fired before renderPaymentPlans() replaced the placeholder
    // cards with the real Firestore-backed ones (e.g. a very early tap). Read price straight
    // off the card's own rendered price text so a paid plan never gets mistaken for free.
    const priceText = card ? card.querySelector('.plan-price') : null;
    const priceNum = priceText ? parseInt((priceText.textContent||'').replace(/[^\d]/g,''), 10) || 0 : 0;
    const descText = card ? card.querySelector('.plan-desc') : null;
    planData = { id:plan, name:(card?card.dataset.name:plan) || plan, price:priceNum, desc:(descText?descText.textContent:'') };
  }
  // Track each explicit plan confirmation click, including the final plan choice for this
  // session. The Pro-specific counter is driven by the actual plan configured in Firestore.
  const planEventDetail={ planId:planData.id, planName:planData.name, planPrice:planData.price };
  trackEvent('plan_cta_click', withPromoAttribution(planEventDetail));
  trackPromoAttributedEvent('promo_plan_cta_click', planEventDetail);
  trackEvent('plan_confirm_click', withPromoAttribution(planEventDetail));
  trackPromoAttributedEvent('promo_plan_confirm_click', planEventDetail);
  const chosenName = String(planData.name || '').trim().toLowerCase();
  if(String(planData.id).toLowerCase()==='pro' || chosenName==='pro'){
    trackEvent('upgrade_to_pro_click', withPromoAttribution(planEventDetail));
    trackPromoAttributedEvent('promo_upgrade_to_pro_click', planEventDetail);
  }
  if(Number(planData.price) === 0){
    showToast((planData.name||'Free') + ' plan selected', {icon: toastIcons.check, duration:1600});
    return;
  }
  requireLoginThen(()=>{
    // Apply whatever discount (admin/first-purchase/promo) is currently active for this plan
    // — same computation renderPaymentPlans() used to show the struck-through price — so
    // checkout always charges exactly what the card displayed. originalPrice/discountLabel are
    // carried through so the checkout screen and the submitted order record both show it.
    const disc = computePlanDiscount(planData);
    const checkoutData = disc
      ? { ...planData, price: disc.finalPrice, originalPrice: disc.originalPrice, discountLabel: disc.label, isPromoDiscount: !!disc.isPromo }
      : { ...planData, originalPrice: null, discountLabel: null, isPromoDiscount:false };
    openCheckout(checkoutData);
  });
}

  const card = document.querySelector('#screen-payment .plan-card[data-plan="'+(saved||'free')+'"]');
  if(card){ card.classList.add('selected'); selectedPlan = card.dataset.plan; }
})();

   Opened from confirmPlan() for any paid plan, gated behind requireLoginThen() — the same
   login-gate Export already uses — so every order document always has a real signed-in uid.
   Flow: pick bKash or Nagad -> customer sends money to the fixed number below -> uploads a
   screenshot as proof (compressed locally and stored inline in Firestore) -> submits ->
   one document is written to the top-level `orders` Firestore
   collection for the admin panel's Orders page to review/approve/reject. This app never
   marks its own order as paid — only the admin changing `status` does that. */
const CHECKOUT_BKASH_NAGAD_NUMBER = '01947913700';
let checkoutPlan = null;
let checkoutMethod = null;
const CHECKOUT_MAX_PROOFS = 5;
let checkoutProofs = [];   // [{file, url}] — url is an object URL used only for the thumbnail
let checkoutSubmitting = false;

    + (!planData.isPromoDiscount && planData.discountLabel ? ' \u2014 ' + escapeHtml(planData.discountLabel) : '');
  document.getElementById('checkoutPlanPrice').innerHTML = (planData.originalPrice ? '<span class="plan-price-strike">\u09F3'+planData.originalPrice+'</span> ' : '') + '\u09F3' + (planData.price||0) + '<span>/mo</span>';

  document.getElementById('checkoutForm').style.display = 'block';

}

function onCheckoutProofPicked(e){
  const files = Array.from(e.target.files || []);
  e.target.value = '';
  if(!files.length) return;
  document.getElementById('checkoutError').style.display = 'none';
  let skippedNonImage = false, hitLimit = false;
  files.forEach(file=>{
    if(!/^image\//i.test(file.type)){ skippedNonImage = true; return; }
    const dup = checkoutProofs.some(p=> p.file.name === file.name && p.file.size === file.size && p.file.lastModified === file.lastModified);
    if(dup) return;
    if(checkoutProofs.length >= CHECKOUT_MAX_PROOFS){ hitLimit = true; return; }
    checkoutProofs.push({ file, url: URL.createObjectURL(file) });
    const proofDetail={ proofIndex:checkoutProofs.length, planId:checkoutPlan && checkoutPlan.id, planName:checkoutPlan && checkoutPlan.name };
    trackEvent('payment_proof_selected', withPromoAttribution(proofDetail));
    trackPromoAttributedEvent('promo_proof_image_selected', proofDetail);
  });
  renderCheckoutProofs();
  if(hitLimit) showCheckoutError('সর্বোচ্চ ' + toBnDigits(CHECKOUT_MAX_PROOFS) + 'টি ছবি আপলোড করা যাবে');
  else if(skippedNonImage) showCheckoutError('শুধু ছবির ফাইল বাছাই করুন');
}
function toBnDigits(n){
  return String(n).replace(/\d/g, d=>'\u09E6\u09E7\u09E8\u09E9\u09EA\u09EB\u09EC\u09ED\u09EE\u09EF'[d]);
}
function removeCheckoutProof(i){
  const p = checkoutProofs[i];
  if(!p) return;
  try{ URL.revokeObjectURL(p.url); }catch(e){}
  checkoutProofs.splice(i, 1);
  document.getElementById('checkoutError').style.display = 'none';
  renderCheckoutProofs();
}

  const n = checkoutProofs.length;
  area.style.display = n ? 'none' : 'flex';
  grid.style.display = n ? 'grid' : 'none';
  if(count) count.textContent = n ? '(' + toBnDigits(n) + '/' + toBnDigits(CHECKOUT_MAX_PROOFS) + ')' : '';
  grid.innerHTML = '';
  checkoutProofs.forEach((p, i)=>{
    const tile = document.createElement('div');
    tile.className = 'cp-tile';
    const img = document.createElement('img');
    img.src = p.url;
    img.alt = 'Payment proof ' + (i + 1);
    const rm = document.createElement('div');
    rm.className = 'cp-remove';
    rm.innerHTML = '<svg viewBox="0 0 24 24" fill="none" stroke-width="2.4" stroke-linecap="round"><path d="M18 6L6 18M6 6l12 12"/></svg>';
    rm.onclick = ()=> removeCheckoutProof(i);
    tile.appendChild(img);
    tile.appendChild(rm);
    grid.appendChild(tile);
  });
  if(n < CHECKOUT_MAX_PROOFS){

}

// Compress payment proof screenshots locally so they fit safely inside a Firestore order.
// The image data is saved in proofImage / proofImages; no third-party image host is used.
function checkoutUploadProof(file, idx){
  return new Promise((resolve)=>{
    const url=URL.createObjectURL(file),img=new Image();
    img.onload=()=>{
      try{
        let scale=Math.min(1,960/Math.max(img.naturalWidth||1,img.naturalHeight||1));
        let canvas=document.createElement('canvas'),data='';
        for(let attempt=0;attempt<8;attempt++){
          canvas.width=Math.max(1,Math.round((img.naturalWidth||1)*scale));
          canvas.height=Math.max(1,Math.round((img.naturalHeight||1)*scale));
          const ctx=canvas.getContext('2d');ctx.imageSmoothingEnabled=true;ctx.imageSmoothingQuality='high';ctx.fillStyle='#fff';ctx.fillRect(0,0,canvas.width,canvas.height);ctx.drawImage(img,0,0,canvas.width,canvas.height);
          const quality=Math.max(.42,.70-attempt*.045);data=canvas.toDataURL('image/jpeg',quality);
          if(data.length<=125000)break;
          scale*=.84;
        }
        URL.revokeObjectURL(url);resolve(data&&data.length<=150000?data:'');
      }catch(err){URL.revokeObjectURL(url);resolve('');}
    };
    img.onerror=()=>{URL.revokeObjectURL(url);resolve('');};
    img.src=url;
  });
}
async function submitCheckoutOrder(){
  if(checkoutSubmitting || !checkoutMethod || !checkoutProofs.length || !checkoutPlan) return;
  if(!currentUser){ requireLoginThen(()=>openCheckout(checkoutPlan)); return; }
  document.getElementById('checkoutError').style.display = 'none';
  checkoutSubmitting = true;
  updateCheckoutSubmitState();
  const btn = document.getElementById('checkoutSubmitBtn');
  const originalLabel = btn.textContent;
  const total = checkoutProofs.length;
  let uploaded = 0;
  const showProgress = ()=>{ btn.textContent = 'প্রুফ আপলোড হচ্ছে (' + toBnDigits(uploaded) + '/' + toBnDigits(total) + ')…'; };
  showProgress();
  try{
    const results = await Promise.all(checkoutProofs.map((p, i)=>
      checkoutUploadProof(p.file, i).then(u=>{
        uploaded++; showProgress();
        if(u){
          const proofDetail={ proofIndex:i+1, planId:checkoutPlan && checkoutPlan.id, planName:checkoutPlan && checkoutPlan.name };
          trackEvent('payment_proof_uploaded', withPromoAttribution(proofDetail));
          trackPromoAttributedEvent('promo_proof_image_uploaded', proofDetail);
        }
        return u;
      })
    ));
    const proofUrls = results.filter(Boolean);
    if(!proofUrls.length) throw new Error('proof upload failed');
    // Payment proof images are stored directly in Firestore. Keep a safe margin below its 1 MiB document limit.
    const inlineSize = proofUrls.reduce((sum, u)=> sum + (u.indexOf('data:') === 0 ? u.length : 0), 0);
    if(inlineSize > 720000){
      showCheckoutError('ছবিগুলো অনেক বড় — কম সংখ্যক ছবি দিয়ে আবার চেষ্টা করুন');
      return;
    }
    const orderPromoAttribution=getPromoAttribution();
    const orderTrackingIdentity=getTrackingIdentity();
    const order = {
      uid: currentUser.uid,
      email: currentUser.email || '',
      visitorId: orderTrackingIdentity.visitorId,
      visitorSessionId: orderTrackingIdentity.sessionId,
      promoPopupId: orderPromoAttribution ? orderPromoAttribution.popupId : null,
      promoAttributionSource: orderPromoAttribution ? 'promo_popup' : null,
      planId: checkoutPlan.id,
      planName: checkoutPlan.name,
      price: Number(checkoutPlan.price) || 0,
      // originalPrice/discountLabel are only set when a discount was actually applied (see
      // confirmPlan) — null otherwise, so the admin Orders page can tell a discounted order
      // from a full-price one at a glance.
      originalPrice: checkoutPlan.originalPrice || null,
      discountLabel: checkoutPlan.discountLabel || null,
      method: checkoutMethod,
      proofImage: proofUrls[0],      // first image — kept so the existing admin Orders page keeps working
      proofImages: proofUrls,        // all uploaded proof images
      trxId: (document.getElementById('checkoutTrxInput').value || '').trim(),
      status: 'pending',
      createdAt: (firebaseReady && firebase.firestore) ? firebase.firestore.FieldValue.serverTimestamp() : new Date().toISOString()
    };
    if(!(firebaseReady && fbDb)) throw new Error('offline');
    btn.textContent = 'সাবমিট হচ্ছে…';
    const submittedOrderRef = await fbDb.collection('orders').add(order);
    try{window.evQueueAutoEvent&&window.evQueueAutoEvent('order_placed',{plan:order.planName,price:order.price,method:order.method});}catch(_e){}
    myLatestOrderInfo = {
      id: submittedOrderRef.id,
      planId: String(checkoutPlan.id),
      planName: checkoutPlan.name || String(checkoutPlan.id),
      price: Number(checkoutPlan.price) || 0,
      method: checkoutMethod || '',
      status: 'pending',
      createdAt: new Date().toISOString()
    };
    myPlanOrderStates[String(checkoutPlan.id)] = myLatestOrderInfo;
    _plansSig = null;
    renderPaymentPlans();
    if(typeof renderMyPlanScreen === 'function') renderMyPlanScreen();
    const submittedDetails={ planId:checkoutPlan.id, planName:checkoutPlan.name, planPrice:checkoutPlan.price };
    trackEvent('order_submitted', withPromoAttribution(submittedDetails));
    if(orderPromoAttribution){
      trackEvent('promo_order_submitted', {...submittedDetails, popupId:orderPromoAttribution.popupId, source:'promo_popup'});
      clearPromoAttribution();
    }
    document.getElementById('checkoutPendingPlanName').textContent = checkoutPlan.name;
    document.getElementById('checkoutForm').style.display = 'none';
    document.getElementById('checkoutPendingView').style.display = 'block';
  }catch(err){
    console.warn('Ethvelle: order submission failed', err);
    showCheckoutError('এখন সাবমিট করা যায়নি — ইন্টারনেট সংযোগ দেখে আবার চেষ্টা করুন');
  } finally {
    checkoutSubmitting = false;
    btn.textContent = originalLabel;
    updateCheckoutSubmitState();
  }
}

/* ================= HOME BANNER: swipe <-> dots sync =================
   Refactored into a named, re-callable function (was a run-once IIFE) so it can be re-wired
   after renderHomeBanners() replaces the slide/dot markup with admin-configured banners —
   otherwise the dots would keep listening to detached elements from the very first render. */
let _bannerDotsObserver = null, _bannerScrollHandler = null, _bannerAutoTimer = null;
const BANNER_AUTOPLAY_MS = 5000;
function wireBannerDots(){
  const scroller = document.getElementById('homeBannerScroll');
  const dotsWrap = document.getElementById('homeBannerDots');
  if(!scroller || !dotsWrap) return;
  const dots = Array.from(dotsWrap.children);
  const banners = Array.from(scroller.querySelectorAll('.home-banner-slide'));
  function setActive(i){ dots.forEach((d,k)=>d.classList.toggle('active', k===i)); }
  function currentIndex(){
    // pick the banner whose centre is closest to the scroller's centre
    const mid = scroller.scrollLeft + scroller.clientWidth/2;
    let best = 0, bestDist = Infinity;
    banners.forEach((b,i)=>{
      const c = b.offsetLeft + b.offsetWidth/2;
      const d = Math.abs(c - mid);
      if(d < bestDist){ bestDist = d; best = i; }
    });
    return best;
  }
  function goToSlide(i){
    const b = banners[i]; if(!b) return;
    scroller.scrollTo({ left: b.offsetLeft, behavior:'smooth' });
  }
  let raf = null;
  // The scroller element lives for the whole session while its slides/dots are re-created,
  // so drop the previous handler first — otherwise every re-render stacked another listener.
  if(_bannerScrollHandler) scroller.removeEventListener('scroll', _bannerScrollHandler);
  _bannerScrollHandler = ()=>{
    if(raf) return;
    raf = requestAnimationFrame(()=>{ raf = null; setActive(currentIndex()); });
  };
  scroller.addEventListener('scroll', _bannerScrollHandler, {passive:true});
  // tap a dot to jump to that banner
  dots.forEach((d,i)=>d.addEventListener('click', ()=>{ goToSlide(i); restartBannerAutoplay(); }));
  // The home screen is display:none until goTo('home') runs, so layout numbers are 0 at load.
  // Re-sync whenever it becomes visible / resizes so dots start on the right banner.
  if(_bannerDotsObserver) _bannerDotsObserver.disconnect();
  if(window.ResizeObserver){
    _bannerDotsObserver = new ResizeObserver(()=>setActive(currentIndex()));
    _bannerDotsObserver.observe(scroller);
  }

  /* ---- Auto-rotate: advance to the next banner every 5s ----
     A manual swipe/tap restarts the 5s timer so the banner doesn't jump away right after the
     person interacts with it. Paused while the tab/page is hidden so it doesn't burn cycles
     or jump slides while the person isn't looking. */
  function stopBannerAutoplay(){ if(_bannerAutoTimer){ clearInterval(_bannerAutoTimer); _bannerAutoTimer = null; } }
  function startBannerAutoplay(){
    stopBannerAutoplay();
    if(banners.length < 2) return;
    if(document.hidden) return;
    _bannerAutoTimer = setInterval(()=>{
      const next = (currentIndex() + 1) % banners.length;
      goToSlide(next);
    }, BANNER_AUTOPLAY_MS);
  }
  window.restartBannerAutoplay = function(){ startBannerAutoplay(); };
  // Manual swipe should also reset the timer so autoplay doesn't fight the person's gesture.
  scroller.addEventListener('touchstart', ()=>stopBannerAutoplay(), {passive:true});
  scroller.addEventListener('touchend', ()=>startBannerAutoplay(), {passive:true});
  document.removeEventListener('visibilitychange', _bannerVisibilityHandler);
  _bannerVisibilityHandler = ()=>{ if(document.hidden) stopBannerAutoplay(); else startBannerAutoplay(); };
  document.addEventListener('visibilitychange', _bannerVisibilityHandler);
  startBannerAutoplay();
}
let _bannerVisibilityHandler = ()=>{};
wireBannerDots();

function setupCanvasStage(ratio){
  const wrap = document.querySelector('.canvas-wrap');
  const stage = document.getElementById('canvasStage');
  const [rw, rh] = ratio.split(':').map(Number);
  // Portrait projects use a tighter canvas gutter, like CapCut, so they can use more of the
  // preview area. Measure the actual content box after applying the ratio class: the old
  // fixed width/height deductions could leave unused space or make WebView squeeze the stage.
  wrap.classList.toggle('canvas-wrap-portrait', rh > rw);
  const wrapRect = wrap.getBoundingClientRect();
  const wrapCS = getComputedStyle(wrap);
  const wrapInnerW = wrap.clientWidth - (parseFloat(wrapCS.paddingLeft)||0) - (parseFloat(wrapCS.paddingRight)||0);
  const wrapInnerH = wrap.clientHeight - (parseFloat(wrapCS.paddingTop)||0) - (parseFloat(wrapCS.paddingBottom)||0);
  const maxW = wrapInnerW > 0 ? wrapInnerW : Math.max(1, wrapRect.width - 32);
  const maxH = wrapInnerH > 0 ? wrapInnerH : Math.max(1, wrapRect.height - 20);
  let w = maxW, h = w * (rh/rw);
  if(window.__ethvelleInlineZoomFullscreen && (rw/rh)>=1.70){
    // Keep the project ratio fixed; fit the unrotated 16:9 frame inside the current
    // viewport. Device rotation changes the available bounds, never the canvas axes.
    const aspect=rw/rh;
    w=Math.min(maxW,maxH*aspect);
    h=w/aspect;
    stage.style.transform='none';
    stage.style.transformOrigin='50% 50%';
  } else if(window.__ethvelleCustomAnimationMode && (rw/rh) >= 1.70 && window.innerHeight > window.innerWidth){
    // The retired custom studio also uses the same stable, unrotated project frame.
    w = Math.min(maxW, maxH * (rw/rh));
    h = w * (rh/rw);
    stage.style.transform = 'none';
    stage.style.transformOrigin = '50% 50%';
  } else {
    if(h > maxH){ h = maxH; w = h * (rw/rh); }
    stage.style.transform = 'none';
    stage.style.transformOrigin = '50% 50%';
  }
  // Keep the preview frame locked to the selected project ratio. Explicitly disable
  // flex shrinking: Android WebView can otherwise reduce the width while retaining the
  // computed height during an orientation change, making a landscape frame look vertically stretched.
  stage.style.flex = '0 0 auto';
  stage.style.aspectRatio = rw+' / '+rh;
  stage.style.width = w+'px';
  stage.style.height = h+'px';
  applyBackground();

  const canvas = document.getElementById('stickCanvas');
  // Use a crisp high-DPI backing surface while capping very large mobile DPRs to avoid
  // memory spikes during transforms. Round backing dimensions to whole device pixels.
  const deviceDpr = Math.max(1, Number(window.devicePixelRatio) || 1);
  const zoomPreview = !!(wrap.classList.contains('evca-inline-zoom') || window.__ethvelleInlineZoomFullscreen);
  const requestedDpr = zoomPreview ? Math.min(1.65, deviceDpr) : Math.min(4, Math.max(2.5, deviceDpr));
  const pixelBudgetDpr = Math.sqrt((zoomPreview ? 7000000 : 18000000) / Math.max(1, w * h));
  const stageDpr = Math.max(1, Math.min(requestedDpr, pixelBudgetDpr));
  canvas.width = Math.max(1, Math.round(w * stageDpr));
  canvas.height = Math.max(1, Math.round(h * stageDpr));
  ctx = canvas.getContext('2d', { alpha:true });
  const stageScaleX = canvas.width / Math.max(1, w);
  const stageScaleY = canvas.height / Math.max(1, h);
  ctx.setTransform(stageScaleX,0,0,stageScaleY,0,0);
  ctx.imageSmoothingEnabled = true;
  try { ctx.imageSmoothingQuality = 'high'; } catch(_e) {}
  ctx.lineCap = 'round';
  ctx.lineJoin = 'round';
  ctx.miterLimit = 2;
  stageW = w; stageH = h;
  updateSelBox(); // stage changed size/position — re-anchor the selection box to it
}

// FEATURE ("I dident able to add multiple background"): the Background row used to hold
// exactly one image/color for the ENTIRE project (`currentBg`, a single string) — there was
// no way to place a second background anywhere on the timeline at all. `backgrounds` is now
// an array of clips, same shape as an overlay clip — {id, src|color, trimStart, trimEnd,

// project can switch from one background image to another at a chosen point in time instead
// of being stuck with a single background forever. `currentBg` and `bgTransform` below are
// now DERIVED, read-only views of "whichever background clip covers the current playhead
// time" (see currentBackgroundClip()/refreshCurrentBgFromPlayhead()) — kept as real variables
// (rather than deleted outright) purely so the ~25 existing render/export/persist call sites
// that already read `currentBg`/`bgTransform` keep working completely unchanged; every WRITE
// to background data now goes through `backgrounds[]` instead of assigning these directly.
let backgrounds = [];
let currentBg = null;
// Pan/zoom applied to the background IMAGE only (irrelevant for solid-color backgrounds).
// Independent of modelTransform, which only moves the stick figure. `rotation` added so the
// same Transform panel (size/position/rotation) that already edits the stickman can drive
// the Background too — see updateTransform's selectedClip-aware BUGFIX note. Now the ACTIVE
// background clip's own transform object (see currentBackgroundClip) rather than one single

let bgTransform = { scale:1, offsetX:0, offsetY:0, rotation:0 };
// True for the exact duration of a background-row edge drag (see wireBackgroundTrackInteractions)
// — lets syncTimelineInnerWidth() know to skip rebuilding the background track while that
// drag's pointer capture is still live. See the BUGFIX notes at both call sites for why.
let bgHandleDragActive = false;
const MIN_CLIP_SEC = 0.1;
// Default length for a newly-added background clip (matches DEFAULT_CLIP_LEN's role for
// animation clips) — used when appending a new background after the last existing one.
// UPDATED: every newly-added background clip is now 5 seconds long, exactly like a newly-added
// animation clip (DEFAULT_CLIP_LEN, 5s) — previously 2s, so a background looked much shorter
// than the character clip sitting right under it on the timeline. Kept as its own constant
// (rather than aliasing DEFAULT_CLIP_LEN directly) because DEFAULT_CLIP_LEN is declared later

const BG_DEFAULT_DURATION = 5.0;
// Legacy mirror of "the active background clip's own length" — no code should treat this as
// authoritative anymore (see backgrounds[] above); kept in sync by refreshCurrentBgFromPlayhead
// purely so old-format save/export/history code paths that still read it see a sane value.
let bgDuration = BG_DEFAULT_DURATION;
// Returns the background clip that actually covers a given absolute timeline second (the
// last one whose trimStart is at/before `t`, matching "sequential, switches as playback
// reaches each one" — the same rule a video track uses to decide what's "playing" at time t).
// Falls back to the first clip if `t` is before everything (e.g. the very start of the
// project), and to null if there are no background clips at all yet.
function currentBackgroundClip(t){
  if(!backgrounds.length) return null;
  if(!isPlaying && selectedClip!=null){
    const selectedEnd=backgrounds.find(b=>String(b.id)===String(selectedClip)&&Math.abs(Number(t)-Number(b.trimEnd))<=0.001);
    if(selectedEnd)return selectedEnd;
  }
  const sorted = backgrounds; // kept sorted by trimStart at all times (see resolveBgClipOverlaps)
  let match = sorted[0];
  for(const b of sorted){ if(b.trimStart <= t + 0.0001) match = b; else break; }
  return match;
}
// Re-points the legacy currentBg/bgTransform variables at whichever background clip covers
