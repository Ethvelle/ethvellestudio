(function(){
  'use strict';
  /* Several older layers rebuild the colour row (open, tab change, Transparent tap) and each rebuild
     dropped the eyedropper button and re-created it a frame later, so the row visibly jumped
     (eyedropper / wheel / transparent changed places). This guard runs as a MutationObserver
     microtask, i.e. BEFORE the browser paints, and puts the eyedropper straight back in front of
     the wheel, with transparent right after it. It is idempotent, so it never loops. */
  let proto=null, observed=null, observer=null, busy=false;
  const $=id=>document.getElementById(id);
  function inComposer(d){return d.classList.contains('evpa-in-composer')||!!($('ethvelleTextComposer')&&$('ethvelleTextComposer').classList.contains('active'));}
  function fix(){
    if(busy)return;
    const d=$('ethvellePicsartTextDock'),pal=$('evpaPalette');
    if(!d||!pal||inComposer(d))return;
    const live=pal.querySelector('.ev-eyedrop-text');
    if(live){proto=live.cloneNode(true);proto.setAttribute('aria-pressed','false');}
    busy=true;
    try{
      pal.querySelectorAll('.evpa-color-wheel').forEach(wheel=>{
        const parent=wheel.parentElement;if(!parent)return;
        let eye=parent.querySelector(':scope > .ev-eyedrop-text');
        if(!eye&&proto)eye=proto.cloneNode(true);
        if(eye&&eye.nextElementSibling!==wheel)parent.insertBefore(eye,wheel);
        const clear=parent.querySelector(':scope > .evpa-transparent');
        if(clear&&wheel.nextElementSibling!==clear)wheel.after(clear);
      });
    }catch(_e){}
    busy=false;
  }
  function attach(){
    const d=$('ethvellePicsartTextDock');
    if(!d||typeof MutationObserver!=='function')return false;
    if(observed===d)return true;
    if(observer)observer.disconnect();
    observer=new MutationObserver(fix);
    observer.observe(d,{childList:true,subtree:true});
    observed=d;fix();return true;
  }
  function boot(){
    if(attach())return;
    const wait=new MutationObserver(()=>{if(attach())wait.disconnect();});
    wait.observe(document.body,{childList:true,subtree:true});
  }
  ['openEthvelleTextStylePanel','openEthvelleTextComposer'].forEach(name=>{
    const old=window[name];
    if(typeof old!=='function'||old.__ev219Order)return;

  });
