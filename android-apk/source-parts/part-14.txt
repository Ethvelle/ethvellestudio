    });

  }

  function sameSourceContinuation(a,b){
    if(!a||!b||a===b||a.type!==b.type||a.reversePlayback||b.reversePlayback)return false;
    if(String(a.src||'')!==String(b.src||''))return false;
    const ra=clipRate(a),rb=clipRate(b);
    if(Math.abs(ra-rb)>.0005)return false;
    if(Math.abs((Number(a.trimEnd)||0)-(Number(b.trimStart)||0))>.006)return false;
    const endSrc=(Number(a.sourceOffset)||0)+((Number(a.trimEnd)||0)-(Number(a.trimStart)||0))*ra;
    if(Math.abs(endSrc-(Number(b.sourceOffset)||0))>.02)return false;
    if(Math.abs(clipVolume(a,Number(a.trimEnd)||0)-clipVolume(b,Number(b.trimStart)||0))>.01)return false;
    return true;
  }

    }

  // Internal decoder rebuffer: freeze the AudioContext and canonical timeline together.
  // Unlike a user pause, this deliberately does NOT stop, disconnect, or recreate any
  // scheduled AudioBufferSourceNode. AudioContext suspension preserves each source's sample.
  let internalRebufferHold=null;
  async function suspendForRebuffer(){
    if(internalRebufferHold)return internalRebufferHold;
    const ctx=audioContext();
    const startTime=(isPlaying&&typeof getCurrentAbsoluteTime==='function')
      ?getCurrentAbsoluteTime():(Number(currentAbsoluteTimeSec)||0);
    const hold={ctx,run:runSerial,time:startTime,suspendFailed:false};

    // Stop the performance-clock loop immediately, without entering the public pause wrapper
    // (the v165 public pause wrapper intentionally calls stopScheduledAll()).
    if(playbackRaf!==null){try{cancelAnimationFrame(playbackRaf);}catch(_e){}playbackRaf=null;}
    setCurrentAbsoluteTime(startTime);
    isPlaying=false;
    playbackStartedAtMs=0;
    playbackStartTimeSec=startTime;
    internalRebufferHold=hold;

    // Suspend first, then read currentTime after the suspend promise resolves. The context may
    // finish its current render quantum before stopping; this calculation advances the held
    // timeline to the exact audio sample at which the source nodes were frozen.
    if(ctx&&ctx.state==='running'){
      try{const pending=ctx.suspend();if(pending&&typeof pending.then==='function')await pending;}
      catch(err){hold.suspendFailed=true;if(window.__evVideoDebug)console.warn('[evAudio] rebuffer suspend:',err);}
    }
    // A real user pause/seek/edit may invalidate the run while suspend() is pending.
    // In that case the public action owns the clock and this internal hold must not rewind it.
    if(internalRebufferHold!==hold||runSerial!==hold.run)return hold;
    let frozen=startTime;
    if(ctx&&anchor&&anchor.run===hold.run&&Number.isFinite(ctx.currentTime)){
      frozen=anchor.tl+(ctx.currentTime-anchor.ctx-anchor.lead);
    }
    frozen=Math.max(0,Number(frozen)||0);
    setCurrentAbsoluteTime(frozen);
    playbackStartTimeSec=frozen;
    hold.time=frozen;
    try{if(typeof renderFrame==='function')renderFrame();}catch(_e){}
    return hold;
  }
  async function resumeAfterRebuffer(){
    const hold=internalRebufferHold;
    const ctx=(hold&&hold.ctx)||audioContext();
    // Resume the same suspended context: previously scheduled sources continue at their
    // exact frozen sample. Do not call startPlayback(), which is a new-run rescheduler.
    if(ctx&&ctx.state==='suspended'){
      const pending=ctx.resume();if(pending&&typeof pending.then==='function')await pending;
    }
    let t=Math.max(0,Number(currentAbsoluteTimeSec)||(hold&&hold.time)||0);
    // If the browser rejected suspend(), keep the clock aligned with the still-running
    // context at resume instead of replaying from an older timeline timestamp.
    if(hold&&hold.suspendFailed&&ctx&&ctx.state==='running'&&anchor&&anchor.run===hold.run){
      t=Math.max(0,anchor.tl+(ctx.currentTime-anchor.ctx-anchor.lead));
      setCurrentAbsoluteTime(t);
    }
    // Rebase both clocks at the just-resumed sample while keeping runSerial and all source
    // nodes intact. Future clip starts will now be scheduled from this matching clock pair.
    if(ctx){
      anchor={ctx:ctx.currentTime,tl:t,lead:0,run:runSerial,badTicks:0};
      try{if(window.__evPlaybackDiagnostics)window.__evPlaybackDiagnostics.recordAnchor();}catch(_e){}
      lastTimelineContextTime=ctx.currentTime;
    }else{
      anchor=null;
      lastTimelineContextTime=null;
    }
    lastTimelineTime=t;
    playbackStartTimeSec=t;
    playbackStartedAtMs=performance.now();
    isPlaying=true;
    internalRebufferHold=null;
    if(playbackRaf!==null){try{cancelAnimationFrame(playbackRaf);}catch(_e){}}
    playbackRaf=requestAnimationFrame(playbackTick);
    return true;
  }
  /*
   * v200 stop-path audit (all calls in this playback/audio scheduler were checked):
   * stopScheduledOne:
   * - scheduleBufferClip's existing-entry call only replaces a stale entry when its clip
   *   signature or playback-run token no longer matches; it is not a trim-boundary stop.
   * - syncScheduledAudio's preferred-video call evaluates preference within this clip's
   *   own trim span, so a following split cannot win merely because the playhead passed
   *   this clip's trimEnd.
   * - The scheduled-entry sweep immediately removes deleted clips, gives present clips
   *   trimEnd + 0.4s for their scheduled source to finish, and removes only stale entries
   *   lying before the active scheduling window.
   * stopScheduledAll:
   * - sync while playback is stopped; sustained clock-drift re-anchoring; starting a new
   *   playback run; pause; project-end stop; seek; speed edit; and audio-setting edit/reprime.
   *   These are transport/resync/edit invalidations, not automatic per-clip end cutoffs.
   */

  function scheduleBufferClip(ov,buffer,ctx,time){
    const id=String(ov.id),rate=clipRate(ov);
    const clipStart=Number(ov.trimStart)||0,clipEnd=Number(ov.trimEnd)||0;
    let timelineStart=Math.max(time,clipStart);
    if(timelineStart>=clipEnd)return;
    const signature=[String(ov.src||''),clipStart,clipEnd,Number(ov.sourceOffset)||0,rate,ov.type].join('|');
    const existing=scheduled.get(id);
    if(existing&&existing.signature===signature&&existing.run===runSerial){
      // Already queued on the shared clock: only keep its live gain/fade/pan values current.
      const vol=clipVolume(ov,timelineStart);
      if(existing.route==='video-graph'){
        const n=existing.nodes;
        try{n.gain.gain.setTargetAtTime(vol,ctx.currentTime,.015);}catch(_e){n.gain.gain.value=vol;}
        if(n.fade){const fade=fadeAmount(ov,timelineStart);try{n.fade.gain.setTargetAtTime(fade,ctx.currentTime,.015);}catch(_e){n.fade.gain.value=fade;}}
        if(n.pan){let cfg={};try{cfg=typeof getEthvelleAudioSettings==='function'?getEthvelleAudioSettings(ov)||{}:{};}catch(_e){}try{n.pan.pan.setTargetAtTime(clamp(cfg.pan,-100,100)/100,ctx.currentTime,.02);}catch(_e){}}
      }else if(existing.route==='audio-graph'){
        const n=existing.nodes;
        if(n&&n.gain&&n.gain.gain){try{n.gain.gain.setTargetAtTime(vol,ctx.currentTime,.015);}catch(_e){n.gain.gain.value=vol;}}
        if(n&&n.__evcaV67&&n.__evcaV67.fade){const fade=fadeAmount(ov,timelineStart);try{n.__evcaV67.fade.gain.setTargetAtTime(fade,ctx.currentTime,.015);}catch(_e){n.__evcaV67.fade.gain.value=fade;}}
      }
      return;
    }else if(existing)stopScheduledOne(id);

    const A=ensureAnchor(ctx);
    let when=A.ctx+A.lead+(timelineStart-A.tl);
    let offset=Math.max(0,Number(ov.sourceOffset)||0)+(timelineStart-clipStart)*rate;
    let sourceDuration=Math.min((clipEnd-timelineStart)*rate,buffer.duration-offset);
    const nowCtx=ctx.currentTime+.003;
    if(when<nowCtx){
      // Scheduler ran late (frame hitch): skip the part that should already have played
      // instead of starting it from its beginning, which would put it behind the picture.
      const late=nowCtx-when;
      offset+=late*rate;sourceDuration-=late*rate;timelineStart+=late;when=nowCtx;
    }
    if(offset>=buffer.duration-.005||sourceDuration<=.005)return;

    /* v221: clip speed must not change pitch. Use a time-stretched copy (pitch untouched) and play it at the
       pitch-setting rate only; timeline mapping below stays in original-source seconds. */
    let playBuffer=buffer,playRate=rate,bufScale=1,bufOrigin=0;
    const stretchSpeed=Math.max(.05,Number(ov.speed)||1);
    if(Math.abs(stretchSpeed-1)>.001&&window.ethvellePitchSafeStretch){
      const stretched=window.ethvellePitchSafeStretch.getReady(buffer,stretchSpeed,Math.max(0,Number(ov.sourceOffset)||0),(Number(ov.sourceOffset)||0)+(clipEnd-clipStart)*rate+.1);
      if(stretched===null)return; /* still computing: stay silent for a moment, the next sync tick schedules it */
      if(stretched&&stretched.buffer){playBuffer=stretched.buffer;playRate=rate/stretchSpeed;bufScale=1/stretchSpeed;bufOrigin=stretched.originSec;}
    }
    const src=ctx.createBufferSource();try{if(window.__evPlaybackDiagnostics)window.__evPlaybackDiagnostics.recordSourceNode();}catch(_e){}src.buffer=playBuffer;src.playbackRate.value=playRate;
    // Per-source envelope: a few-ms fade in/out at every real cut removes the click you hear
    // when a clip is cut through the middle of a waveform. Plain splits of the same source
    // (continuous audio) are detected and left unfaded so they stay seamless.
    const env=ctx.createGain();src.connect(env);
    const route=connectBufferSource(ov,env,ctx);
    if(route.route==='video-graph'){
      const n=route.nodes;
      try{n.gain.gain.setValueAtTime(clipVolume(ov,timelineStart),when);}catch(_e){n.gain.gain.value=clipVolume(ov,timelineStart);}
      if(n.fade){const fade=fadeAmount(ov,timelineStart);try{n.fade.gain.setValueAtTime(fade,when);}catch(_e){n.fade.gain.value=fade;}}
      if(n.pan){let cfg={};try{cfg=typeof getEthvelleAudioSettings==='function'?getEthvelleAudioSettings(ov)||{}:{};}catch(_e){}try{n.pan.pan.setValueAtTime(clamp(cfg.pan,-100,100)/100,when);}catch(_e){}}
    }else if(route.nodes&&route.nodes.gain&&route.nodes.gain.gain){
      const gain=route.nodes.gain.gain;
      try{gain.setValueAtTime(clipVolume(ov,timelineStart),when);}catch(_e){gain.value=clipVolume(ov,timelineStart);}
      if(route.nodes.__evcaV67&&route.nodes.__evcaV67.fade){const f=route.nodes.__evcaV67.fade.gain;try{f.setValueAtTime(fadeAmount(ov,timelineStart),when);}catch(_e){f.value=fadeAmount(ov,timelineStart);}}
    }
    try{
      const wall=sourceDuration/rate,endAt=when+wall,f=Math.min(FADE_SEC,wall/3);
      const prev=continuationNeighbor(ov,-1),next=continuationNeighbor(ov,1);
      const joinsPrev=!!prev&&timelineStart<=clipStart+.004&&scheduled.has(String(prev.id));
      const g=env.gain;
      if(!joinsPrev){g.setValueAtTime(0,when);g.linearRampToValueAtTime(1,when+f);}
      if(!next&&wall>f*3){g.setValueAtTime(1,endAt-f);g.linearRampToValueAtTime(0,endAt);}
    }catch(_e){}
    const entry={source:src,env,signature,run:runSerial,when,timelineStart,rate,route:route.route,nodes:route.nodes};
    scheduled.set(id,entry);
    src.onended=()=>{if(scheduled.get(id)===entry)scheduled.delete(id);try{src.disconnect();}catch(_e){}try{env.disconnect();}catch(_e){}};
    try{src.start(when,Math.max(0,(offset-bufOrigin)*bufScale),Math.max(.001,sourceDuration*bufScale));}catch(err){
      scheduled.delete(id);try{src.disconnect();}catch(_e){}try{env.disconnect();}catch(_e){}
      const failed=audioCache.get(String(ov.src||''));if(failed){failed.status='failed';failed.error='AudioBufferSource scheduling failed: '+(err&&err.message?err.message:String(err));}
      if(window.__evVideoDebug)console.warn('[evAudio] buffer schedule fallback',id,err);
      prepareAudioClipElementFallback(ov);
    }
  }
  function syncScheduledAudio(){
    if(!isPlaying){if(scheduled.size)stopScheduledAll('pause');applyVideoMuteState();return;}
    const ctx=audioContext();if(!ctx)return;
    if(ctx.state==='suspended'){try{ctx.resume().catch(()=>{});}catch(_e){}}
