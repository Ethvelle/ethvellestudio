<script id="ethvelle-v432-effect-device">
(function(){
  'use strict';
  /* Effects are fetched from the admin-published link ONCE. The file is then stored in the browser's device media
     store (the same store imported videos use) and every later use of that effect plays from the device. */
  const MAP_KEY='ethvelle.effectDeviceKeys.v1',SRC_DB='ethvelle-source-media-v1';
  const pending=new Map();
  const readMap=()=>{try{return JSON.parse(localStorage.getItem(MAP_KEY)||'{}')||{};}catch(_e){return {};}};
  const writeMap=m=>{try{localStorage.setItem(MAP_KEY,JSON.stringify(m));}catch(_e){}};
  window.ethvelleFxOnDevice=function(fx){const s=String(fx&&fx.src||'');return !!(s&&readMap()[s]);};
  function getRecord(key){
    return new Promise(function(resolve){
      try{
        const r=indexedDB.open(SRC_DB,1);
        r.onupgradeneeded=function(){if(!r.result.objectStoreNames.contains('assets'))r.result.createObjectStore('assets');};
        r.onsuccess=function(){try{const q=r.result.transaction('assets','readonly').objectStore('assets').get(key);q.onsuccess=function(){resolve(q.result||null);};q.onerror=function(){resolve(null);};}catch(_e){resolve(null);}};
        r.onerror=function(){resolve(null);};
      }catch(_e){resolve(null);}
    });
  }
  const typeByExt={webm:'video/webm',png:'image/png',gif:'image/gif'};
  function extOf(src,type){
    const t=String(type||'').toLowerCase();
    if(t.indexOf('webm')>-1)return 'webm';if(t.indexOf('png')>-1)return 'png';if(t.indexOf('gif')>-1)return 'gif';
    const m=String(src).split(/[?#]/)[0].match(/\.([a-z0-9]{2,5})$/i);return m?m[1].toLowerCase():'png';
  }
  async function fetchBlob(src,onProgress){
    const res=await fetch(src,{mode:'cors',credentials:'omit'});
    if(!res.ok)throw new Error('HTTP '+res.status);
    const total=Number(res.headers.get('content-length'))||0;
    let type=String(res.headers.get('content-type')||'').split(';')[0].trim().toLowerCase();
    const ext0=extOf(src,type);if(!/^(video|image)\//.test(type))type=typeByExt[ext0]||'application/octet-stream';
    if(!res.body||!res.body.getReader){const b=await res.blob();onProgress(1);return new Blob([b],{type:type});}
    const reader=res.body.getReader(),chunks=[];let got=0;
    for(;;){const r=await reader.read();if(r.done)break;chunks.push(r.value);got+=r.value.length;onProgress(total?Math.min(.98,got/total):-1);}
    onProgress(1);return new Blob(chunks,{type:type});
  }
  /* Download popup (same look as the audio one) */
  let popEl=null,popHide=0;
  function ensurePop(){
    if(popEl&&popEl.isConnected)return popEl;
    popEl=document.createElement('div');popEl.id='evEffectDl';popEl.setAttribute('role','status');popEl.setAttribute('aria-live','polite');
    popEl.innerHTML='<div class="evad-card"><div class="evad-ring"><svg class="evad-arrow" viewBox="0 0 24 24" aria-hidden="true"><path d="M12 4v11M7.5 10.5 12 15l4.5-4.5"/></svg><svg class="evad-check" viewBox="0 0 24 24" aria-hidden="true"><path d="M5 12.5l4.5 4.5L19 7.5"/></svg><i class="evad-tray"></i></div><div class="evad-title">Downloading effect\u2026</div><div class="evad-name"></div><div class="evad-bar"><b></b></div><div class="evad-pct">0%</div></div>';
    document.body.appendChild(popEl);return popEl;
  }
  function openPop(name){
    clearTimeout(popHide);const el=ensurePop();
    el.classList.remove('done','indet');el.querySelector('.evad-title').textContent='Downloading effect\u2026';
    el.querySelector('.evad-name').textContent=String(name||'');el.querySelector('.evad-bar b').style.width='0%';el.querySelector('.evad-pct').textContent='0%';
    void el.offsetWidth;el.classList.add('show');
  }
  function setPop(frac){
    const el=popEl;if(!el)return;
    if(frac<0){el.classList.add('indet');el.querySelector('.evad-pct').textContent='';return;}
    el.classList.remove('indet');const pct=Math.max(0,Math.min(100,Math.round(frac*100)));
    el.querySelector('.evad-bar b').style.width=pct+'%';el.querySelector('.evad-pct').textContent=pct+'%';
  }
  function donePop(){
    const el=popEl;if(!el)return Promise.resolve();
    el.classList.remove('indet');el.classList.add('done');el.querySelector('.evad-title').textContent='Saved to your device';
    el.querySelector('.evad-bar b').style.width='100%';el.querySelector('.evad-pct').textContent='100%';
    return new Promise(function(resolve){popHide=setTimeout(function(){el.classList.remove('show');resolve();},850);});
  }
  function closePop(){clearTimeout(popHide);if(popEl)popEl.classList.remove('show');}
  /* Resolves { blob, key }. key is the device-media key; null when the browser could not store the file. */
  window.ethvelleFxEnsureLocal=function(fx){
    const src=String(fx&&fx.src||'');
    if(!src)return Promise.reject(new Error('No file'));
    if(pending.has(src))return pending.get(src);
    const p=(async function(){
      const map=readMap(),have=map[src];
      if(have){
        const rec=await getRecord(have);
        if(rec&&rec.blob)return {blob:rec.blob,key:have};
        delete map[src];writeMap(map);
      }
      openPop(fx.name||'Effect');
      let blob;
      try{blob=await fetchBlob(src,setPop);}catch(err){closePop();throw err;}
      const file=new File([blob],(String(fx.name||'effect').replace(/[\\/:*?"<>|\u0000-\u001f]+/g,'').trim().replace(/\s+/g,'-')||'effect')+'.'+extOf(src,blob.type),{type:blob.type,lastModified:Date.now()});
      let key=null;
      if(typeof window.__ethvelleStoreMediaFile==='function'){try{key=await window.__ethvelleStoreMediaFile(file);}catch(err){console.warn('Ethvelle effect device save failed:',err);}}
      if(key){
        const m=readMap();m[src]=key;writeMap(m);
        try{if(navigator.storage&&navigator.storage.persist)navigator.storage.persist();}catch(_e){}
        await donePop();
      }else{closePop();}
      return {blob:file,key:key};
    })().finally(function(){pending.delete(src);});
    pending.set(src,p);return p;
  };
})();
