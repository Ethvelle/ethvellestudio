let characters = []; // [{ id, name, bones, baseAngles, modelTransform, hipRestY }]
let activeCharIdx = 0;
function activeCharacter(){ return characters[activeCharIdx] || null; }
function makeCharacterId(){ return 'char_'+Date.now().toString(36)+Math.random().toString(36).slice(2,7); }

// Bone hierarchy: each has a length, base angle (radians), and current angle offset (posable)
let bones; // always === activeCharacter().bones — see syncActiveCharacterRefs()
let baseAngles = {}; // rest-pose angles, used as the anchor for animation offsets — always === activeCharacter().baseAngles
let selectedJoint = null;
let dragging = false;

// Model transform (size / position / rotation) applied on top of the rig — always === activeCharacter().modelTransform
let modelTransform = { scale:1, offsetX:0, offsetY:0, rotation:0 };

// Builds a brand-new rig (fresh bones + baseAngles) at the default rest pose — used both for
// the very first character and for every character added afterward, so each one starts
// posable and independent rather than sharing bone objects with any other character.
//
// RIG SHAPE (v2 — elbow/knee split): each arm and leg used to be a single straight bone
// (armL/armR/legL/legR). To support image-based custom characters (see the 9-part upload
// system — buildCustomRigFromParts()) that need a real elbow and knee to bend at, every limb
// is now TWO bones: an upper segment (armL/armR/legL/legR, unchanged names/parenting so old
// stickman code keeps working) plus a new lower segment that attaches at the upper segment's
// END (attachAt:1) — armL_lo/armR_lo/legL_lo/legR_lo. The stickman's rendering, posing, and
// joint-drag code below all walk `Object.keys(bones)` / an explicit bone-key list, so these
// new keys need to be added to those lists too (see drawStick, getJointList, boneEnd/boneStart
// walkers, and the custom-character image-stamping path).
function buildFreshRig(){
  const cx = stageW/2;
  const topY = stageH*0.28;
  const freshBones = {
    hip:     { x: cx, y: topY + stageH*0.22, angle: 0, parent:null, length:0 },
    spine:   { parent:'hip', length: stageH*0.16, angle: -Math.PI/2 },
    head:    { parent:'spine', length: stageH*0.09, angle: -Math.PI/2, isHead:true },
    armL:    { parent:'spine', length: stageH*0.065, angle: Math.PI*0.85, attachAt:0.85 },
    armL_lo: { parent:'armL', length: stageH*0.065, angle: Math.PI*0.85, attachAt:1 },
    armR:    { parent:'spine', length: stageH*0.065, angle: -Math.PI*0.15, attachAt:0.85 },
    armR_lo: { parent:'armR', length: stageH*0.065, angle: -Math.PI*0.15, attachAt:1 },
    legL:    { parent:'hip', length: stageH*0.09, angle: Math.PI/2 + 0.35 },
    legL_lo: { parent:'legL', length: stageH*0.09, angle: Math.PI/2 + 0.35, attachAt:1 },
    legR:    { parent:'hip', length: stageH*0.09, angle: Math.PI/2 - 0.35 },
    legR_lo: { parent:'legR', length: stageH*0.09, angle: Math.PI/2 - 0.35, attachAt:1 },
  };
  // Add a small, branched hand rig. Gemini supplies the per-character rest-pose points when
  // available; these defaults keep the bones valid for older/local-only projects too.
  ['L','R'].forEach(side=>{
    const left = side==='L';
    const arm = left ? 'armL_lo' : 'armR_lo';
    const dir = left ? Math.PI : 0;
    freshBones['hand'+side] = { parent:arm, length:stageH*0.022, angle:dir, attachAt:1 };
    const fingers = [
      {name:'thumb', at:0.12, off:left?-0.010:0.010, spread:left?-0.70:0.70},
      {name:'index', at:0.34, off:left?-0.006:0.006, spread:left?-0.18:0.18},
      {name:'middle',at:0.50, off:0, spread:0},
      {name:'ring',  at:0.67, off:left?0.005:-0.005, spread:left?0.12:-0.12},
      {name:'pinky', at:0.84, off:left?0.010:-0.010, spread:left?0.32:-0.32},
    ];
    fingers.forEach(f=>{
      const k=f.name+side;
      freshBones[k+'_1']={ parent:'hand'+side, length:stageH*0.012, angle:dir+f.spread, attachAt:f.at, attachOffsetX:0, attachOffsetY:stageH*f.off };
      freshBones[k+'_2']={ parent:k+'_1', length:stageH*0.010, angle:dir+f.spread*0.65, attachAt:1 };
    });
  });
  const freshBaseAngles = {};
  Object.keys(freshBones).forEach(k => freshBaseAngles[k] = freshBones[k].angle);
  return { bones: freshBones, baseAngles: freshBaseAngles };
}
// Merges an auto-rig's computed length/angle/attachAt fractions onto a freshly-built rig (see
// buildFreshRig). Lengths are fractions of the uploaded character's own detected height, scaled
// here to a fixed target on-stage figure height so different uploads default to a similar size
// (the user can always rescale afterward via the existing pinch/drag model transform). Used both
// when creating a new auto-rigged character and when restoring one from a saved project, since
// openEditor() always rebuilds bones fresh from buildFreshRig() and never persists live bone
// angles/lengths (see the comment on that in persistProjectState).
const AUTO_RIG_FIGURE_HEIGHT_FRAC = 0.5;
function applyRigOverride(rig, override){
  if(!override) return;
  const scale = stageH*AUTO_RIG_FIGURE_HEIGHT_FRAC;
  Object.keys(override.lengths||{}).forEach(k=>{
    if(rig.bones[k]) rig.bones[k].length = override.lengths[k]*scale;
  });
  Object.keys(override.angles||{}).forEach(k=>{
    if(rig.bones[k]) rig.bones[k].angle = override.angles[k];
  });
  if(override.attachAt){
    Object.keys(override.attachAt).forEach(k=>{
      if(rig.bones[k] && override.attachAt[k]!=null) rig.bones[k].attachAt = Math.max(0,Math.min(1,override.attachAt[k]));
    });
  }
  if(override.attachOffsets){
    Object.keys(override.attachOffsets).forEach(k=>{
      if(!rig.bones[k]) return;
      const off=override.attachOffsets[k]||{};
      rig.bones[k].attachOffsetX=(Number(off.x)||0)*scale;
      rig.bones[k].attachOffsetY=(Number(off.y)||0)*scale;
    });
  }
  // baseAngles must equal the (possibly overridden) rest angle exactly, same as buildFreshRig
  // does — animation/pose deltas are added on top of this (see applyPoseToBones).
  Object.keys(rig.bones).forEach(k=> rig.baseAngles[k] = rig.bones[k].angle);
}
// Every posable limb-END key that owns an upper+lower bone pair, in a fixed order used
// throughout the rig code (drawing, joints, custom-character part mapping). 'head' and
// 'spine' are handled separately since they aren't paired limbs.
const LIMB_KEYS = ['armL','armR','legL','legR'];
function lowerKey(limbKey){ return limbKey+'_lo'; }
// Creates one new character object (own rig + own transform), does NOT add it to the
// `characters` array or select it — callers decide that (see addCharacter()/initStickman()).
// `rigOverride` (optional) is an auto-rig's computed bone lengths/angles/attachAt — see
// applyRigOverride/buildRigOverrideFromJoints — used for auto-rigged custom characters so their
// skeleton actually matches the uploaded artwork's proportions instead of the generic default.
function makeCharacter(name, rigOverride){
  const rig = buildFreshRig();
  if(rigOverride) applyRigOverride(rig, rigOverride);
  const defaultTransform = { scale:1, offsetX:0, offsetY:0, rotation:0 };
  return {
    id: makeCharacterId(),
    name: name || ('Character '+(characters.length+1)),
    bones: rig.bones,
    baseAngles: rig.baseAngles,
    modelTransform: { ...defaultTransform },
    restTransform: { ...defaultTransform },
    hipRestY: rig.bones.hip.y,
    // FEATURE (default placeholder clip on add): a brand new character now starts with a
    // real clip already sitting on its timeline row — the 'default' preset (see animPresets),
    // which holds a static rest pose and drives no actual animation — instead of clip:null.
    // This gives every new character something visible on the timeline immediately, and
    // means deleteSelectedClip()'s existing "deleting a character's last/only clip deletes
    // the character" behavior (see its own FEATURE comment) is reachable right away instead
    // of only after the user manually picks a real animation first.
    clip: {
      animKey: 'default',
      trimStart: 0,
      trimEnd: DEFAULT_CLIP_LEN,
      keyframes: [],
      speed: 1
    },
    segments: [],          // split-off / appended clips, same shape as clip
    rigOverride: rigOverride || undefined
  };
}
// Re-points the plain `bones`/`baseAngles`/`modelTransform`/`hipRestY`/`activeClip`/
// `splitStickmanSegments` variables at the currently-active character's own objects. MUST be
// called after activeCharIdx changes (addCharacter, selectCharacter, deleteCharacter, undo/
// redo, project load) — every drawing/posing/transform/timeline function below keeps working
// unmodified because it only ever reads these plain names, never `characters[activeCharIdx].X`
// directly. Because activeClip/splitStickmanSegments are re-pointed (not copied), mutating them
// in place — e.g. `activeClip.trimStart = x` — writes straight through to the character object,
// so no separate "save back to character" step is ever needed.
function syncActiveCharacterRefs(){
  invalidateCharacterRestBoundsCache();
  const c = activeCharacter();
  if(!c) return;
  bones = c.bones;
  baseAngles = c.baseAngles;
  modelTransform = c.modelTransform;
  hipRestY = c.hipRestY;
  activeClip = c.clip;
  splitStickmanSegments = c.segments;
}
// Whenever activeClip or splitStickmanSegments is REASSIGNED outright (activeClip = {...},
// activeClip = null, splitStickmanSegments = [], activeClip = splitStickmanSegments.shift())
// rather than mutated in place, the active character's own `clip`/`segments` fields must be
// pointed at the new value too, or the change is silently lost the next time
// syncActiveCharacterRefs() runs (e.g. switching characters and back). In-place mutations
// (activeClip.trimStart = x, splitStickmanSegments.push(...)) never need this, since the
// character object already shares the same underlying array/object reference.
function syncClipRefsToCharacter(){
  const c = activeCharacter();
  if(!c) return;
  c.clip = activeClip;
  c.segments = splitStickmanSegments;
}
// Adds a brand-new character (own rig, own transform, default rest pose), makes it the
// active one, and refreshes the canvas/UI to show it. This is the "add a layer" action.
function addCharacter(){
  const c = makeCharacter();
  characters.push(c);
  activeCharIdx = characters.length - 1;
  syncActiveCharacterRefs();
  selectedJoint = null;
  drawStick();
  renderCharacterLayers();
  syncTransformSliders();
  persistProjectState();
  if(navigator.vibrate) navigator.vibrate(12);
  showToast('Character added', {duration:1200});
  return c;
}
// Switches which character is being posed/transformed/drawn-selected. Does not change any
// OTHER character's pose or transform — it only changes which one the shared bones/
// baseAngles/modelTransform names currently point at.
function selectCharacter(idx){
  if(idx<0 || idx>=characters.length) return;
  const previousIdx=activeCharIdx;
  activeCharIdx = idx;
  syncActiveCharacterRefs();
  selectedJoint = null;
  drawStick();
  renderCharacterLayerSelection(previousIdx,idx);
  syncTransformSliders();
}
// Removes a character entirely. Always keeps at least one character in the project — the
// last remaining character can't be deleted, same as the timeline never being left with a
// completely empty stickman track.
function deleteCharacter(idx){
  if(characters.length<=1) { showToast("Can't delete the only character", {duration:1400}); return; }
  if(idx<0 || idx>=characters.length) return;
  characters.splice(idx,1);
  if(activeCharIdx >= characters.length) activeCharIdx = characters.length-1;
  else if(activeCharIdx > idx) activeCharIdx--;
  syncActiveCharacterRefs();
  selectedJoint = null;
  drawStick();
  renderCharacterLayers();
  syncTransformSliders();
  persistProjectState();
}
// FEATURE ("move any layer or animation character or clip forward or backward... bring one
// clip above another or send it back"): moves the SELECTED clip/character one step in its
// own array's paint order — for overlays/backgrounds, later array position = paints on top
// (see renderOverlayMedia/canvas compositing, which already paints in array order), so this
// simply swaps the selected item with its array-neighbor in the given direction. For a
// character LAYER, the same swap happens in `characters[]`, which is what drawOtherCharacters
// iterates in order — the active character still visually draws on top of everyone while
// it's selected (a deliberate, separate "keep what you're editing visible/grabbable"
// convenience — see drawStick's own comment), but its underlying stacking position among the
// OTHER characters, and what happens once a different character becomes active, does move.
// `dir` is +1 (Forward — swap with the next-higher item, paints more on top) or -1
// (Backward — swap with the next-lower item, paints further back).
function reorderSelectedLayer(dir){
  if(!selectedClip) return;
  const isStickman = selectedClip==='main' || String(selectedClip).startsWith('seg');
  if(isStickman) return; // see syncBottomToolsMode's canReorder — shouldn't be reachable
  const isCharLayer = typeof selectedClip==='string' && selectedClip.startsWith('char') && !isStickman;
  if(isCharLayer){
    const idx = parseInt(selectedClip.slice(4), 10);
    const swapWith = idx + dir;
    if(isNaN(idx) || swapWith<0 || swapWith>=characters.length) {
      showToast(dir>0 ? "Already at the front" : "Already at the back", {duration:1200});
      return;
    }
    [characters[idx], characters[swapWith]] = [characters[swapWith], characters[idx]];
    // The active character follows its own object, not its old index, so selecting/editing
    // continuity is preserved across the swap.
    if(activeCharIdx===idx) activeCharIdx = swapWith;
    else if(activeCharIdx===swapWith) activeCharIdx = idx;
    selectedClip = 'char'+swapWith;
    syncActiveCharacterRefs();
    drawStick();
    renderCharacterLayers();
    if(typeof syncBottomToolsMode==='function')syncBottomToolsMode();
    persistProjectState();
    return;
  }
  const bg = backgroundClipForSelection();
  if(bg){
    // Background clips are laid out strictly by trimStart (see resolveBgClipOverlaps) —
    // there's no independent "paint order" separate from their time position, so Forward/
    // Backward here means swap TIME SLOTS with the adjacent clip (the one immediately after/
    // before it), which is the only reordering that's meaningful for a sequential row.
    const sorted = backgrounds; // kept sorted by trimStart
    const idx = sorted.findIndex(b=>b.id===bg.id);
    const swapWith = idx + dir;
    if(swapWith<0 || swapWith>=sorted.length){
      showToast(dir>0 ? "Already the last background" : "Already the first background", {duration:1200});
      return;
    }
    const other = sorted[swapWith];
    const bgOldStart=Number(bg.trimStart)||0,otherOldStart=Number(other.trimStart)||0;
    const bgOldKeyTimes=Array.isArray(bg.ethvelleMediaKeyframes)?bg.ethvelleMediaKeyframes.map(k=>Number(k&&k.t)||0):[];
    const otherOldKeyTimes=Array.isArray(other.ethvelleMediaKeyframes)?other.ethvelleMediaKeyframes.map(k=>Number(k&&k.t)||0):[];
    const aLen = bg.trimEnd-bg.trimStart, bLen = other.trimEnd-other.trimStart;
    if(dir>0){
      bg.trimStart = other.trimStart; bg.trimEnd = bg.trimStart+aLen;
      other.trimStart = bg.trimEnd; other.trimEnd = other.trimStart+bLen;
    } else {
      other.trimStart = bg.trimStart; other.trimEnd = other.trimStart+bLen;
      bg.trimStart = other.trimEnd; bg.trimEnd = bg.trimStart+aLen;
    }
    shiftTimelineKeyframesOnce(bg.ethvelleMediaKeyframes,bgOldKeyTimes,(Number(bg.trimStart)||0)-bgOldStart);
    shiftTimelineKeyframesOnce(other.ethvelleMediaKeyframes,otherOldKeyTimes,(Number(other.trimStart)||0)-otherOldStart);
    backgrounds.sort((a,b)=>a.trimStart-b.trimStart);
    renderBackgroundTrack();
    refreshCurrentBgFromPlayhead();
    applyBackground();
    if(typeof syncBottomToolsMode==='function')syncBottomToolsMode();
    persistProjectState();
    return;
  }
  const ovIdx = overlays.findIndex(o=>String(o.id)===String(selectedClip));
  if(ovIdx===-1) return;
  const selectedOverlay=overlays[ovIdx];
  const visualStackTypes=new Set(['image','video','text','shape','color']);
  // Audio has no canvas paint order. Preserve its previous simple array-order behavior
  // instead of applying the visual Background-layer state to an audio-only clip.
  if(!visualStackTypes.has(selectedOverlay.type)){
    const swapWith=ovIdx+dir;
    if(swapWith<0||swapWith>=overlays.length){
      showToast(dir>0 ? 'Already at the front' : 'Already at the back', {duration:1200});
      return;
    }
    [overlays[ovIdx],overlays[swapWith]]=[overlays[swapWith],overlays[ovIdx]];
    renderOverlayLayers();renderFrame();if(typeof syncBottomToolsMode==='function')syncBottomToolsMode();persistProjectState();
    return;
  }
  const startOf=o=>Number.isFinite(Number(o&&o.trimStart))?Number(o.trimStart):0;
  const endOf=o=>Number.isFinite(Number(o&&o.trimEnd))?Number(o.trimEnd):startOf(o);
  const selectedStart=startOf(selectedOverlay), selectedEnd=endOf(selectedOverlay);
  // Only clips that actually overlap this clip in time can affect its visible stacking.
  // Non-overlapping clips are skipped, so one tap always changes the visible order instead
  // of silently swapping with a clip that is not on the canvas at the same time.
  const overlappingVisuals=overlays.map((item,index)=>({item,index})).filter(entry=>{
    if(entry.index===ovIdx||!entry.item||!visualStackTypes.has(entry.item.type))return false;
    const overlapStart=Math.max(selectedStart,startOf(entry.item));
    const overlapEnd=Math.min(selectedEnd,endOf(entry.item));
    return overlapEnd>overlapStart+0.0001;
  });
  if(dir<0){
    const lower=overlappingVisuals.filter(entry=>entry.index<ovIdx).pop();
    if(lower){
      [overlays[ovIdx],overlays[lower.index]]=[overlays[lower.index],overlays[ovIdx]];
      selectedOverlay.__ethvelleBehindBackground=false;
    }else if(!selectedOverlay.__ethvelleBehindBackground){
      // There is no lower visual overlay at any overlapping time. One Backward action
      // therefore sends the selected clip directly behind the Background layer.
      selectedOverlay.__ethvelleBehindBackground=true;
    }else{
      showToast('Already behind the background', {duration:1200});
      return;
    }
  }else{
    if(selectedOverlay.__ethvelleBehindBackground){
      // Forward from behind the background returns it as the lowest normal overlay.
      selectedOverlay.__ethvelleBehindBackground=false;
      const [item]=overlays.splice(ovIdx,1);
      overlays.unshift(item);
    }else{
      const higher=overlappingVisuals.find(entry=>entry.index>ovIdx);
      if(higher){
        [overlays[ovIdx],overlays[higher.index]]=[overlays[higher.index],overlays[ovIdx]];
      }else if(ovIdx<overlays.length-1){
        // No higher overlapping visual clip: send it to the top in a single action,
        // skipping unrelated clips that are not visible at the same time.
        const [item]=overlays.splice(ovIdx,1);
        overlays.push(item);
      }else{
        showToast('Already at the front', {duration:1200});
        return;
      }
    }
  }
  renderOverlayLayers();
  renderFrame(); // repaint immediately so the updated stack is visible on the canvas
  if(typeof syncBottomToolsMode==='function')syncBottomToolsMode();
  persistProjectState();
}
// FEATURE ("I want you made I can duplicate my layer, animation clip or background"): clones
// the selected clip/character, placed immediately adjacent to (and selected in place of) the
// original — same "spread the object with a fresh id" pattern splitSelectedClipAtPlayhead
// already uses for its second half.
function duplicateSelectedLayer(){
  if(!selectedClip) return;
  const isStickman = selectedClip==='main' || String(selectedClip).startsWith('seg');
  if(isStickman){
    showToast("Select the character layer to duplicate its animation", {icon: toastIcons.info, duration:1800});
    return;
  }
  const isCharLayer = typeof selectedClip==='string' && selectedClip.startsWith('char') && !isStickman;
  if(isCharLayer){
    const idx = parseInt(selectedClip.slice(4), 10);
    const src = characters[idx];
    if(!src) return;
    const clone = {
      ...src,
      id: makeCharacterId(),
      name: (src.name||'Character')+' copy',
      bones: JSON.parse(JSON.stringify(src.bones)),
      baseAngles: {...src.baseAngles},
      modelTransform: {...src.modelTransform},
      clip: src.clip ? JSON.parse(JSON.stringify(src.clip)) : null,
      segments: (src.segments||[]).map(s=>({...s, keyframes:(s.keyframes||[]).map(k=>({...k}))})),
      parts: src.parts ? JSON.parse(JSON.stringify(src.parts)) : undefined
    };
    characters.splice(idx+1, 0, clone);
    activeCharIdx = idx+1;
    syncActiveCharacterRefs();
    drawStick();
    renderCharacterLayers();
    syncTransformSliders();
    onClipSelected('char'+activeCharIdx,'add');
    showToast('Character duplicated', {icon: toastIcons.check, duration:1200});
    persistProjectState();
    return;
  }
  const bg = backgroundClipForSelection();
  if(bg){
    // Placed immediately AFTER the original in time, same "append after existing content"
    // placement addBackgroundClip already uses, then everything from there shifts later via
    // resolveBgClipOverlaps if needed — here it can't collide with anything since it's
    // inserted into the exact gap right after the original ends (or pushes into the next
    // clip's own space, which the clamp then resolves the same hard-wall way any drag would).
    const len = bg.trimEnd - bg.trimStart;
    const clone = { ...bg, id:'bg'+Date.now(), transform:{...bg.transform}, trimStart: bg.trimEnd, trimEnd: bg.trimEnd+len };
    backgrounds.push(clone);
    backgrounds.sort((a,b)=>a.trimStart-b.trimStart);
    resolveBgClipOverlaps(clone.id);
    renderBackgroundTrack();
    onClipSelected(clone,'add');
    showToast('Background duplicated', {icon: toastIcons.check, duration:1200});
    persistProjectState();
    return;
  }
  const ov = overlays.find(o=>o.id===selectedClip);
  if(ov){
    // Overlay clips already tolerate free time-overlap on their own row (see
    // packOverlayGroup) — placed immediately after the original by default so it's visibly
    // distinct rather than sitting exactly on top of its own source, but the person can drag
    // it wherever they like right after, same as any other clip.
    const len = ov.trimEnd - ov.trimStart;
    const clone = { ...ov, id:'ov'+Date.now(), transform: ov.transform?{...ov.transform}:ov.transform, trimStart: ov.trimEnd, trimEnd: ov.trimEnd+len };
    overlays.push(clone);
    renderOverlayLayers();
    renderFrame();
    onClipSelected(clone,'add');
    showToast((ov.type==='audio'?'Audio':ov.type==='video'?'Video':'Image')+' duplicated', {icon: toastIcons.check, duration:1200});
    persistProjectState();
    return;
  }
}
// Draws every OTHER character (all except the active one) underneath the active character's
// own interactive drawStick() pass, so every character is visible on stage at once but only
// the active one shows joint handles / gets posed. Uses each character's own bones/
// baseAngles/modelTransform without touching the shared plain variables (so it never
// disturbs whatever the active character's live pose currently is).
// BUGFIX ("adding multiple custom models — selecting one made another show as a plain stick
// figure"): this used to have its OWN drawing branch that unconditionally stroked a bare
// blue-line skeleton for every non-active character, regardless of whether that character was
// a custom/auto-rigged model with real uploaded artwork (c.type==='custom' + c.parts). Only
// the ACTIVE character's own drawStick() ever checked for custom parts and drew the real
// image — so the moment a custom model stopped being the active one, it visibly reverted to
// generic stickman until re-selected, exactly the reported bug. The export path
// (drawCharacterToContext, used for video export/thumbnails) already draws every character
// correctly — custom art when present, plain stick lines otherwise — so this now simply
// reuses that exact same function instead of keeping a second, incomplete copy of the same
// logic here. Same transform math, same bone walking, same result; one fewer place for the
// two to drift out of sync.
function drawOtherCharacters(){
  if(!ctx) return;
  characters.forEach((c, idx)=>{
    if(idx === activeCharIdx) return;
    drawCharacterToContext(ctx, c);
  });
}
// Renders the character layer rows (one per character, each tappable to select it, with a
// delete button and an Add-character button after the list). Each row's own track now shows
// THAT character's own animation clip(s) directly — Character 1's clips render under
// Character 1's row, Character 2's under Character 2's row, and so on, instead of every
// character's clips being dumped onto one shared track. Only the currently-ACTIVE
// character's track gets the live interactive markup (drag/trim handles, keyframe diamonds)
// via renderTimelineClip/wireClipInteractions, matching how only the active character's rig
// is posable; other characters show a simpler read-only preview of their own clip(s) so you
// can still see at a glance what's on their track without it being editable until you select
// that row first.
function renderCharacterLayers(){
  const container = document.getElementById('characterRows');
  if(!container) return;
  // The active/read-only clip markup is rebuilt when a character row is selected. Keep the
  // timeline viewport anchored so the browser cannot jump to a newly inserted first clip or
  // retarget the release/click to the Add-character row underneath the finger.
  const timelineScroller = document.getElementById('timelineScroll');
  const pageScroller = document.scrollingElement || document.documentElement;
  const scrollSnapshot = {
    timelineLeft: timelineScroller ? timelineScroller.scrollLeft : 0,
    timelineTop: timelineScroller ? timelineScroller.scrollTop : 0,
    pageLeft: pageScroller ? pageScroller.scrollLeft : 0,
    pageTop: pageScroller ? pageScroller.scrollTop : 0
  };
  const restoreScroll = ()=>{
    if(timelineScroller && timelineScroller.isConnected){
      const maxLeft=Math.max(0,timelineScroller.scrollWidth-timelineScroller.clientWidth);
      const maxTop=Math.max(0,timelineScroller.scrollHeight-timelineScroller.clientHeight);
      timelineScroller.scrollLeft=Math.min(scrollSnapshot.timelineLeft,maxLeft);
      timelineScroller.scrollTop=Math.min(scrollSnapshot.timelineTop,maxTop);
    }
    if(pageScroller){
      pageScroller.scrollLeft=scrollSnapshot.pageLeft;
      pageScroller.scrollTop=scrollSnapshot.pageTop;
    }
  };
  container.innerHTML = characters.map((c, idx)=>{
    const isActive = idx===activeCharIdx;
    // BUGFIX (extra delete button sitting beside the row): a red trash icon used to be
    // pinned to the right edge of every character row (whenever more than one character
    // existed), as a second, separate delete affordance sitting right on the timeline
    // itself. Removed — deleting a character layer now works exactly like deleting any
    // clip/layer already does: select the row (arms the contextual bottom toolbar, see
    // selectClipForChrome/syncBottomToolsMode), then tap Delete down there. One consistent
    // place for every delete action instead of a scattered inline button per row-type.
    const chromeSel = selectedClip===('char'+idx);
    // FIX (rows only ever extended all together): a non-active character's row must size
    // itself from ITS OWN clip overlap too (not just the active row) — otherwise switching
    // which character is active would make a row that was already tall (because IT has
    // overlapping clips) snap back to a single fixed lane height the moment it's no longer
    // the active one. readOnlyCharClipMarkup below now returns both the markup and the lane
    // count it used, so the row height here always matches that same packing.
    const readOnly = c.clip ? readOnlyCharClipMarkup(c, idx) : { html:'', laneCount:1 };
    const rowH = charRowHeightForLanes(readOnly.laneCount || 1);
    return `
    <div class="layer-row char-row ${isActive?'selected':''} ${chromeSel?'chrome-selected':''}" data-char-idx="${idx}" style="--ev-char-row-height:${rowH+4}px; height:${rowH+4}px;">
      <div class="layer-badge" title="${c.name}" aria-hidden="true" data-model-type="${window.evCharModelType(c)}">
        ${window.evCharBadgeIconSvg(window.evCharModelType(c))}
      </div>
      <div class="layer-track char-row-track" id="track-char-${idx}" data-char-idx="${idx}" onclick="onCharTrackClick(event, ${idx})" style="--ev-char-track-height:${rowH}px; height:${rowH}px;">
        ${isActive ? '' : (c.clip ? readOnly.html : `<span class="char-row-name">${c.name}</span>`)}
      </div>
    </div>
  `;
  }).join('');  // The active character's row needs the full interactive clip markup (handles, keyframes,
  // drag-to-move) — same rendering path as before, just targeting that character's own row
  // element instead of the old always-present shared #track-stickman row.
  renderTimelineClip();
  restoreScroll();
  requestAnimationFrame(restoreScroll);
}
// Fast path for switching the active character. The old path rebuilt the entire
// #characterRows subtree (including every read-only clip) for a one-row selection change.
// That caused visible layout/scroll anchoring and unnecessary work on slower Android phones.
function renderCharacterLayerSelection(previousIdx,nextIdx){
  const snapshot=captureEthvelleRowScroll();
  const timeline=document.getElementById('timelineScroll');
  const left=timeline?timeline.scrollLeft:0;
  const paintRow=(idx,isActive)=>{
    const c=characters[idx];
    const row=document.querySelector(`.char-row[data-char-idx="${idx}"]`);
    const track=document.getElementById('track-char-'+idx);
    if(!c||!row||!track)return false;
    row.classList.toggle('selected',!!isActive);
    row.classList.toggle('chrome-selected',selectedClip===('char'+idx));
    const readOnly=c.clip?readOnlyCharClipMarkup(c,idx):{html:'',laneCount:1};
    setCharacterLaneGeometry(row,track,readOnly.laneCount||1);
    if(isActive){
      track.innerHTML='';
    }else{
      track.innerHTML=c.clip?readOnly.html:`<span class="char-row-name">${escapeHtml(c.name||'Character')}</span>`;
    }
    return true;
  };
  if(previousIdx!==nextIdx)paintRow(previousIdx,false);
  const ok=paintRow(nextIdx,true);
  if(!ok){renderCharacterLayers();return;}
  renderTimelineClip({skipOverlayRender:true});
  restoreEthvelleRowScroll(snapshot);
  if(timeline&&timeline.isConnected)timeline.scrollLeft=left;
}

// Tapping a non-active character's row/track selects that character (so its clips become
// editable and its rig becomes posable) rather than doing nothing. Tapping the ACTIVE
// character's empty background area (not one of its clip elements) falls through to
// selectCharacter too.
// BUGFIX (selecting the animation clip immediately re-selected the character badge
// instead): the clip body/handles only ever stopped propagation on their OWN
// pointerdown/onpointerdown handlers (see wireClipInteractions / wireSplitSegmentInteractions).
// stopPropagation() on a pointerdown does NOT stop the separate, later 'click' event the
// browser still synthesizes and bubbles up from that same tap — so every tap on the clip
// correctly called selectClipForChrome('main') first, then this handler's own onclick fired
// right after (bubbled from the same tap) and immediately overwrote it with
// selectClipForChrome('char'+idx), leaving the character row selected instead of the clip.
// Fixed with the same deny-list technique the timeline scroller's own pointerdown handler
// already uses (see initTimelineScrub): if the click actually originated on a clip, handle,
// or keyframe marker inside this track, step aside entirely and let that element's own
// selection stand — only a click on genuinely empty track space falls through to select the
// character/row itself.
// Also arms the contextual bottom toolbar for this character layer (selectClipForChrome),
// which is what now surfaces Delete — see the .char-row-delete removal note above.
function onCharTrackClick(e, idx){
  // Any tap within the visible animation clip belongs to that clip, including its outer edge
  // and selection bars. Never let it fall through and select the character row/badge instead.
  if(e.target.closest('.anim-clip, .anim-clip-readonly, .anim-clip-body, .anim-clip-handle, .kf-marker')) return;
  // Keep a character-row tap inside its own selection handler. This prevents any ancestor or
  // delegated timeline click handler from treating it as a tap on the following Add row.
  if(e.preventDefault)e.preventDefault();
  if(e.stopPropagation)e.stopPropagation();
  selectCharacter(idx);
  selectClipForChrome('char'+idx);
}
// Non-active characters show a plain, non-interactive strip for each of their own clips
// (main + segments) so their track is never blank/misleading — just not draggable/trimmable
// until you tap that row to make it the active character. FIX (rows only extended all
// together): packed into the same sub-lanes the active row would use, and returns the lane
// count alongside the markup so the caller (renderCharacterLayers) can size THIS row to
// match — a non-active character with two overlapping clips still shows both, each on its
// own line, at that row's own height, independent of every other row.
function readOnlyCharClipMarkup(c, characterIndex){
  if(!c.clip) return {html:'',laneCount:1};
  const pps=pxPerSecond();
  const all=getAllStickmanClips(c.clip,c.segments||[]);
  const layout=characterClipLaneLayout(c.clip,c.segments||[]);
  const html=all.map(({ref,clip})=>{
    const preset=animPresets[clip.animKey];
    const left=clip.trimStart*pps,width=Math.max(MIN_CLIP_SEC*pps,(clip.trimEnd-clip.trimStart)*pps);
    const top=charClipLaneTop(layout.laneOf.get(ref)||0);
    const keyframes=(Array.isArray(clip.keyframes)?clip.keyframes:[]).map(kf=>{
      const clipWidth=Math.max(MIN_CLIP_SEC*pps,(Number(clip.trimEnd)-Number(clip.trimStart))*pps);
      const raw=(Number(kf.t)-Number(clip.trimStart))*pps;
      const x=Math.max(9,Math.min(Math.max(9,clipWidth-9),Number.isFinite(raw)?raw:9));
      const under=Math.abs(Number(kf.t)-Number(currentAbsoluteTimeSec)) <= (typeof keyframeContactWindow==='function'?keyframeContactWindow():0.05);
      return `<div class="kf-marker ev-readonly-kf-marker ${under?'under-playhead':''}" style="left:${x}px;" aria-hidden="true"></div>`;
    }).join('');
    return `<div class="anim-clip anim-clip-readonly" data-readonly-clip-ref="${escapeHtml(ref)}" data-readonly-char-idx="${characterIndex}" role="button" tabindex="0" aria-label="Select ${escapeHtml(c.name||'character')} animation clip" onclick="event.stopPropagation(); selectCharacter(${characterIndex}); selectClipForChrome('${ref}');" onkeydown="if(event.key==='Enter'||event.key===' '){event.preventDefault();event.stopPropagation();selectCharacter(${characterIndex});selectClipForChrome('${ref}');}" style="left:${left}px; width:${width}px; --ev-char-lane-top:${top}px; top:${top}px; height:${CHAR_SUB_LANE_H}px;">
      <span class="anim-clip-label">${preset?escapeHtml(preset.name):'Animation'}</span>
      <div class="keyframe-track ev-readonly-kf-track" aria-hidden="true">${keyframes}</div>
    </div>`;
  }).join('');
  return {html,laneCount:layout.laneCount};
}

function getBoneEndpointPair(key){
  const b = bones[key];
  if(!b.parent) return {s:{x:b.x,y:b.y},e:{x:b.x,y:b.y}};
  const pp=getBoneEndpointPair(b.parent), p0=pp.s, p1=pp.e;
  const t=b.attachAt===undefined?1:Math.max(0,Math.min(1,b.attachAt));
  let x=p0.x+(p1.x-p0.x)*t, y=p0.y+(p1.y-p0.y)*t;
  const dx=p1.x-p0.x, dy=p1.y-p0.y, a=Math.atan2(dy,dx);
  const ox=Number(b.attachOffsetX)||0, oy=Number(b.attachOffsetY)||0;
  x+=Math.cos(a)*ox-Math.sin(a)*oy; y+=Math.sin(a)*ox+Math.cos(a)*oy;
  return {s:{x,y},e:{x:x+Math.cos(b.angle)*b.length,y:y+Math.sin(b.angle)*b.length}};
}
function getBoneStart(key){ return getBoneEndpointPair(key).s; }
function getBoneEnd(key){ return getBoneEndpointPair(key).e; }

/* ================= CUSTOM CHARACTER (9-part image upload + rig) =================
   Lets a user build a posable character from their own artwork instead of only the built-in
   green stickman. The user uploads 9 separate transparent PNGs (head, torso, 2x upper-arm,
   2x forearm+hand, 2x thigh, 2x shin+foot) — one per PART_SLOTS entry below — then, for each
   part, drags a pivot dot onto a preview of that part to mark where it hinges (see
   openPivotEditorForSlot). The result is stored on the character as:
     char.type = 'custom'
     char.parts = { head:{src, pivotX, pivotY, w, h}, torso:{...}, armL_upper:{...}, ... }
   pivotX/pivotY are 0..1 FRACTIONS of that image's own width/height (not pixels), so they
   stay correct regardless of how large the source photo was. w/h are the image's natural
   pixel dimensions, captured at upload time so the draw path doesn't need to re-decode the
   image just to know its aspect ratio.
   Each part is drawn stamped onto its matching bone segment (see PART_SLOTS' `bone` field
   mapping into the rig built by buildFreshRig): the image's pivot point is placed at the
   bone's START joint, then the image is rotated to the bone's current angle and drawn out
   along the bone's length — so posing the rig (dragging joints, playing an animation) moves
   the artwork exactly like the green stickman, just with real art instead of lines. */

// Every uploadable body-part slot: id (used as the parts{} key + DOM id), a human label,
// which rig bone it stamps onto, and whether the pivot should default to the top (shoulder/
// hip end) or bottom (elbow/knee end) of the uploaded image — just a sane starting position
// for the drag-to-place pivot editor, the user can always move it.
const PART_SLOTS = [
  { id:'head',       label:'Head',              bone:'head',    defaultPivot:'bottom' },
  { id:'torso',      label:'Torso',             bone:'spine',   defaultPivot:'bottom' },
  { id:'armL_upper', label:'Left upper arm',    bone:'armL',    defaultPivot:'top' },
  { id:'armL_lower', label:'Left forearm+hand', bone:'armL_lo', defaultPivot:'top' },
  { id:'armR_upper', label:'Right upper arm',   bone:'armR',    defaultPivot:'top' },
  { id:'armR_lower', label:'Right forearm+hand',bone:'armR_lo', defaultPivot:'top' },
  { id:'legL_upper', label:'Left thigh',        bone:'legL',    defaultPivot:'top' },
  { id:'legL_lower', label:'Left shin+foot',    bone:'legL_lo', defaultPivot:'top' },
  { id:'legR_upper', label:'Right thigh',       bone:'legR',    defaultPivot:'top' },
  { id:'legR_lower', label:'Right shin+foot',   bone:'legR_lo', defaultPivot:'top' },
];
// NOTE: torso stamps onto 'spine' and head onto 'head' — everything else onto its matching
// limb bone (see buildFreshRig's LIMB_KEYS + lowerKey()). 10 slots total (not 9) since each
// arm/leg is now genuinely two separate images (upper+lower) rather than one image covering

// In-progress upload state for the custom-character builder modal — holds each slot's part
// data until the user taps "Create character"/"Save changes", at which point it's converted
// into (or written back onto) a real character (see finishCustomCharacterBuild()). Kept
// separate from any real character object so backing out of the builder
// (closeCustomCharBuilder) never half-creates or half-edits a character.
let customBuilderParts = {}; // { slotId: {src, pivotX, pivotY, w, h} }
let customBuilderActiveSlot = null; // slot id currently open in the pivot editor
// True while the builder is editing an ALREADY-CREATED custom character in place (opened via
// openCustomCharBuilder(true) from the Bones panel's "Rig my art" tile) rather than building a
// brand new one (opened via the plain openCustomCharBuilder() call from the Add Character
// chooser). Read by finishCustomCharacterBuild to decide whether to update characters
// [activeCharIdx] or push a new character.
let customBuilderEditingExisting = false;
// Set to a modelAssets[].id while the builder is editing a SAVED MODEL directly from the
// Projects → Models grid (openModelAssetEditor) — i.e. with no project/character open at all.
// Mutually exclusive with customBuilderEditingExisting: finishCustomCharacterBuild checks this
// first and, when set, writes the edited parts back onto that modelAssets entry instead of
// onto characters[activeCharIdx], then returns to the Models grid instead of a canvas.
let customBuilderEditingAssetId = null;

// forActiveCharacter: pass true to edit the CURRENTLY ACTIVE character's existing custom art
// (if it has any) instead of always starting from 10 blank slots. Used by the Bones panel's
// "Rig my art" tile so a model that's already been built can be reopened, have any part
// replaced or its pivot/rotation/mirror adjusted, and saved back onto the same character —
// see the "made I can reedit or adjust my existing model" request. The Add Character chooser's
// "Custom character" tile keeps calling this with no argument, so it always starts blank even

function openCustomCharBuilder(forActiveCharacter){
  if(forActiveCharacter)edShowLoading('Preparing character editor');
  const activeChar = forActiveCharacter ? characters[activeCharIdx] : null;
  const editingExisting = !!(activeChar && activeChar.type==='custom' && activeChar.parts);
  customBuilderEditingExisting = editingExisting;
  customBuilderEditingAssetId = null; // plain character edit — never a bare-asset edit
  customBuilderParts = {};

    // (closeCustomCharBuilder) never mutates the live character — only finishCustomCharacter

        src:p.src, w:p.w, h:p.h, pivotX:p.pivotX, pivotY:p.pivotY,
        contentBox: p.contentBox ? {...p.contentBox} : null
      };
    });
  }
  customBuilderActiveSlot = null;
  closeAddCharModal();
  renderCustomBuilderGrid();
  if(forActiveCharacter)edHideLoadingAfterPaint();
  document.getElementById('customBuilderTitle').textContent = editingExisting ? 'Edit character' : 'Build custom character';
  document.getElementById('customBuilderSub').textContent = editingExisting
    ? 'Tap any part to replace it, rotate it, mirror it, or move its pin.'
    : 'Upload a transparent PNG for each part, then pin its joint. All 10 parts are needed.';
  document.getElementById('customBuilderDoneBtn').textContent = editingExisting ? 'Save changes' : 'Create character';
  document.getElementById('customCharBuilderBackdrop').classList.add('active');
}

// Wired to #partFileInput's change event — reads the picked file, captures its natural
// pixel size (needed for correct aspect-ratio drawing later), sets a sane default pivot, and
// opens the pivot editor so the user can drag it into place before the slot counts as filled.
// FIX (uploaded/cropped part images render as the wrong shape/size — see
// drawCustomCharacterOnContext notes below): a part image straight off a phone crop tool
// almost always has extra transparent (or solid-color) margin around the actual artwork —
// the crop rectangle rarely lines up pixel-perfect with the drawn content. Scaling and
// pivoting against the RAW image canvas (old behavior) then measures against mostly-empty
// space, so the same bone length produces wildly different visible-art sizes from one part
// to the next, and the default pivot (a fixed 6%/94% down the raw canvas) usually lands
// nowhere near the actual joint. detectOpaqueBounds scans the image's alpha channel (falling
// back to non-background-color pixels for a flattened JPEG with no alpha at all) and returns
// the tight pixel bounding box of the actual visible content, so every downstream
// calculation — default pivot placement, and the draw-time scale in
// drawCustomCharacterOnContext — is measured against what's actually drawn, not the
// surrounding canvas.
function detectOpaqueBounds(img){
  const w = img.naturalWidth, h = img.naturalHeight;
  const cv = document.createElement('canvas');
  cv.width = w; cv.height = h;
  const cctx = cv.getContext('2d', {willReadFrequently:true});
  cctx.drawImage(img, 0, 0);
  let data;
  try{ data = cctx.getImageData(0,0,w,h).data; }
  catch(err){ return {x:0,y:0,w,h}; } // tainted canvas (cross-origin) — fall back to full image
  let minX=w, minY=h, maxX=-1, maxY=-1;
  const ALPHA_THRESH = 10; // ignore near-fully-transparent antialiasing fringe pixels
  for(let y=0; y<h; y++){
    for(let x=0; x<w; x++){
      const a = data[(y*w+x)*4+3];
      if(a > ALPHA_THRESH){
        if(x<minX) minX=x; if(x>maxX) maxX=x;
        if(y<minY) minY=y; if(y>maxY) maxY=y;
      }
    }
  }
  if(maxX < minX || maxY < minY) return {x:0,y:0,w,h}; // fully transparent image — fall back
  return { x:minX, y:minY, w:(maxX-minX+1), h:(maxY-minY+1) };
}
document.getElementById('partFileInput').addEventListener('change', (e)=>{
  const file = e.target.files[0];
  const slotId = customBuilderActiveSlot;
  if(!file || !slotId){ e.target.value=''; return; }
  const reader = new FileReader();
  reader.onload = ()=>{
    const dataUrl = reader.result;
    const img = new Image();
    img.onload = ()=>{
      const slot = PART_SLOTS.find(s=>s.id===slotId);
      const box = detectOpaqueBounds(img);
      // Default pivot is now placed at the top/bottom-center of the DETECTED CONTENT box,
      // converted back to a fraction of the full raw image (since pivotX/pivotY are stored
      // as fractions of the raw canvas everywhere else in the file — drawImage, the pivot
      // dot's on-screen position, etc). A small 6% inset from the content edge (not the
      // canvas edge) keeps the pivot just inside the visible art instead of sitting exactly
      // on its outermost, often-antialiased edge.
      const insetFrac = 0.06;
      const defaultPivotY = slot.defaultPivot==='top'
        ? (box.y + box.h*insetFrac) / img.naturalHeight
        : (box.y + box.h*(1-insetFrac)) / img.naturalHeight;
      customBuilderParts[slotId] = {
        src: dataUrl, w: img.naturalWidth, h: img.naturalHeight,
        contentBox: box,
        pivotX: (box.x + box.w*0.5) / img.naturalWidth,
        pivotY: defaultPivotY
      };
      openPivotEditorForSlot(slotId);
    };
    img.onerror = ()=>{ showToast("Couldn't read that image", {icon: toastIcons.info, duration:1800}); };
    img.src = dataUrl;
  };
  reader.onerror = ()=>{ showToast("Couldn't read that file", {icon: toastIcons.info, duration:1800}); };
  reader.readAsDataURL(file);
  e.target.value='';
});


/* ---- Pivot editor: drag a dot onto the uploaded part image to mark its hinge point ---- */
function openPivotEditorForSlot(slotId){
  customBuilderActiveSlot = slotId;
  const part = customBuilderParts[slotId];
  const slot = PART_SLOTS.find(s=>s.id===slotId);
  if(!part || !slot) return;
  document.getElementById('pivotEditorTitle').textContent = 'Pin the joint — '+slot.label;
  const img = document.getElementById('pivotEditorImg');
  img.src = part.src;
  document.getElementById('pivotEditorBackdrop').classList.add('active');
  // Positioning the dot needs the rendered image's on-screen box, which only exists after
  // the image has actually painted (or instantly if it's already cached) — a plain
  // synchronous call right after setting .src would read stale/zero layout on first open.
  const place = ()=>{ positionPivotDot(part.pivotX, part.pivotY); positionContentBoxOutline(part); };
  if(img.complete) place(); else img.onload = place;
}
// Draws the dashed outline showing exactly what detectOpaqueBounds found as this part's real
// artwork — lets the user SEE why a pivot/scale might look off (e.g. a crop with a big
// stray margin on one side) instead of it being invisible math. Purely visual; doesn't
// affect drawCustomCharacterOnContext, which reads part.contentBox directly.
function positionContentBoxOutline(part){
  const box = document.getElementById('pivotContentBox');
  if(!box) return;
  if(!part.contentBox){ box.style.display = 'none'; return; }
  const img = document.getElementById('pivotEditorImg');
  const rect = img.getBoundingClientRect();
  const wrapRect = img.parentElement.getBoundingClientRect();
  const { x, y, w, h } = part.contentBox;
  box.style.display = 'block';
  box.style.left = (rect.left - wrapRect.left + rect.width*(x/part.w)) + 'px';
  box.style.top = (rect.top - wrapRect.top + rect.height*(y/part.h)) + 'px';
  box.style.width = (rect.width*(w/part.w)) + 'px';
  box.style.height = (rect.height*(h/part.h)) + 'px';
}

function setupPivotDrag(){
  const dot=document.getElementById('pivotDot');
  const img=document.getElementById('pivotEditorImg');
  if(!dot||!img)return;
  let dragging=false,frame=0,pending=null,geom=null,captureTarget=null,pointerId=null;
  function applyPoint(x,y){
    if(!geom||!geom.img.width||!geom.img.height)return;
    const fx=Math.max(0,Math.min(1,(x-geom.img.left)/geom.img.width));
    const fy=Math.max(0,Math.min(1,(y-geom.img.top)/geom.img.height));
    const part=customBuilderParts[customBuilderActiveSlot];
    if(part){part.pivotX=fx;part.pivotY=fy;}
    dot.style.left=(geom.img.left-geom.wrap.left+geom.img.width*fx)+'px';
    dot.style.top=(geom.img.top-geom.wrap.top+geom.img.height*fy)+'px';
  }
  function flush(){
    if(frame){cancelAnimationFrame(frame);frame=0;}
    const p=pending;pending=null;
    if(p)applyPoint(p.x,p.y);
  }
  function queue(e){pending={x:e.clientX,y:e.clientY};if(!frame)frame=requestAnimationFrame(flush);}
  function finish(e){
    if(!dragging|| (e&&pointerId!=null&&e.pointerId!==pointerId))return;
    flush();dragging=false;geom=null;pointerId=null;captureTarget=null;
  }
  dot.onpointerdown=e=>{
    if(e.pointerType==='mouse'&&e.button!==0)return;
    dragging=true;pointerId=e.pointerId;
    const wrap=img.parentElement;
    geom={img:img.getBoundingClientRect(),wrap:wrap.getBoundingClientRect()};
    captureTarget=dot;
    try{dot.setPointerCapture(e.pointerId);}catch(_e){}
    e.preventDefault();e.stopPropagation();
  };
  dot.onpointermove=e=>{if(dragging&&e.pointerId===pointerId){queue(e);e.preventDefault();}};
  dot.onpointerup=finish;dot.onpointercancel=finish;dot.onlostpointercapture=finish;
  // A tap on the picture still places the pivot immediately, then continues as a drag.
  img.onpointerdown=e=>{
    if(e.pointerType==='mouse'&&e.button!==0)return;
    dragging=true;pointerId=e.pointerId;
    const wrap=img.parentElement;
    geom={img:img.getBoundingClientRect(),wrap:wrap.getBoundingClientRect()};
    applyPoint(e.clientX,e.clientY);
    captureTarget=img;
    try{img.setPointerCapture(e.pointerId);}catch(_e){}
    e.preventDefault();
  };
  img.onpointermove=e=>{if(dragging&&captureTarget===img&&e.pointerId===pointerId){queue(e);e.preventDefault();}};
  img.onpointerup=finish;img.onpointercancel=finish;img.onlostpointercapture=finish;
}

  });

// ---- Rotate / Mirror the part currently open in the pivot editor ----
// Both bake their transform directly into the part's own stored image pixels (via an offscreen
// canvas) and algebraically remap its pivot fraction + detected content box to match, rather
// than tracking a separate rotation/flip flag. That means drawCustomCharacterOnContext, saving,
// and export all need zero changes — they just draw whatever image/pivot/box is stored, exactly
// as before; rotating/mirroring twenty times over stays lossless since it's PNG re-encoding,
// not JPEG. See applyBakedPartTransform for the shared finish-up step.
function rotatePivotEditorPart(){
  const part = customBuilderParts[customBuilderActiveSlot];
  const img = document.getElementById('pivotEditorImg');
  if(!part || !img.complete || !img.naturalWidth) return;
  const oldW = part.w, oldH = part.h;
  const cv = document.createElement('canvas');
  cv.width = oldH; cv.height = oldW; // 90° rotation swaps width/height
  const cctx = cv.getContext('2d');
  cctx.translate(cv.width, 0);
  cctx.rotate(Math.PI/2);
  cctx.drawImage(img, 0, 0, oldW, oldH);
  // A 90° CW rotation maps raw-image fraction (x,y) -> (1-y, x) in the new (swapped) image —
  // apply the same mapping to the pivot and to the four corners of the content box (in pixels
  // of the OLD image, since that's what contentBox is stored in).
  const box = part.contentBox;
  const newBox = box ? { x: oldH-(box.y+box.h), y: box.x, w: box.h, h: box.w } : null;
  applyBakedPartTransform(cv.toDataURL('image/png'), oldH, oldW, 1-part.pivotY, part.pivotX, newBox);
}
function mirrorPivotEditorPart(){
  const part = customBuilderParts[customBuilderActiveSlot];
  const img = document.getElementById('pivotEditorImg');
  if(!part || !img.complete || !img.naturalWidth) return;
  const w = part.w, h = part.h;
  const cv = document.createElement('canvas');
  cv.width = w; cv.height = h; // dimensions unchanged by a horizontal flip
  const cctx = cv.getContext('2d');
  cctx.translate(w, 0);
  cctx.scale(-1, 1);
  cctx.drawImage(img, 0, 0, w, h);
  const box = part.contentBox;
  const newBox = box ? { x: w-box.x-box.w, y: box.y, w: box.w, h: box.h } : null;
  applyBakedPartTransform(cv.toDataURL('image/png'), w, h, 1-part.pivotX, part.pivotY, newBox);
}
// Shared tail for rotate/mirror above: stores the newly-baked image + remapped pivot/content
// box on the slot that's currently open, then reopens the pivot editor for that same slot so
// the on-screen dot and dashed outline immediately reflect the new orientation.
function applyBakedPartTransform(newSrc, newW, newH, newPivotX, newPivotY, newBox){
  const slotId = customBuilderActiveSlot;
  customBuilderParts[slotId] = { src:newSrc, w:newW, h:newH, pivotX:newPivotX, pivotY:newPivotY, contentBox:newBox };
  if(navigator.vibrate) navigator.vibrate(8);
  openPivotEditorForSlot(slotId);
}

/* ============================================================================
   SKELETAL MESH SKINNING (replaces rigid cut-and-stamp part rendering for
   auto-rigged characters — see finalizeAutoRiggedCharacter/buildSkinMesh).
   ----------------------------------------------------------------------------
   OLD BEHAVIOR (still kept below, in drawCustomCharacterOnContext, as a
   fallback for characters saved before this existed / manually-built 10-slot
   characters that have no single source image): the source art was CUT into
   10 rectangular crops (see generateAutoRigParts/computePartRegions), and
   each crop was pasted onto its own bone independently. Because a crop is a
   hard rectangle, the art visibly separated at the joints once limbs rotated
   away from the pose they were cut in — exactly the "cut body parts" problem
   reported.

   NEW BEHAVIOR: the character's full original photo (char.autoRig.sourceImage)
   is kept as ONE texture, never cut. A triangle mesh is laid over it once, and
   every vertex is given a WEIGHTED blend of influence from the 1-2 nearest
   bones (a "vertex group", same idea as Blender's Armature modifier). Each
   frame, every bone's CURRENT transform (translation+rotation, derived from
   getBoneStart/getBoneEnd) is compared against its REST transform (the pose
   the mesh was built in) to get a delta transform; a vertex's new position is
   the weighted blend of every influencing bone's delta applied to that
   vertex's rest position (linear blend skinning). The deformed mesh is then
   drawn by warping the source texture one triangle at a time (drawImage's
   affine-transform trick — canvas has no native triangle-texture-mapping, but
   any triangle-to-triangle mapping IS an affine transform, so each triangle
   is clipped and drawn with ctx.setTransform solved for its 3 corners). This
   is what makes the art stretch/bend continuously FROM the bone points,
   instead of snapping between disconnected rigid pieces.
   ============================================================================ */

// One skinning bone per rig bone key, defined by its REST (detection-time) start/end
// in source-image pixel space — mirrors AUTORIG_BONE_FOR_PART/buildRigOverrideFromJoints'
// own J-point pairs exactly, just kept as explicit segments instead of folded into angle/
// length fractions.
const SKIN_BONE_DEFS = [
  { key:'head',    a:'neck',      b:'headTop' },
  { key:'spine',   a:'hip',       b:'neck' },
  { key:'armL',    a:'shoulderL', b:'elbowL' },
  { key:'armL_lo', a:'elbowL',    b:'wristL' },
  { key:'handL',   a:'wristL',    b:'palmL' },
  { key:'armR',    a:'shoulderR', b:'elbowR' },
  { key:'armR_lo', a:'elbowR',    b:'wristR' },
  { key:'handR',   a:'wristR',    b:'palmR' },
  { key:'legL',    a:'hip',      b:'kneeL' },
  { key:'legL_lo', a:'kneeL',    b:'ankleL' },
  { key:'legR',    a:'hip',      b:'kneeR' },
  { key:'legR_lo', a:'kneeR',    b:'ankleR' },
];
['L','R'].forEach(side=>['thumb','index','middle','ring','pinky'].forEach(finger=>{
  SKIN_BONE_DEFS.push({key:finger+side+'_1',a:finger+'Base'+side,b:finger+'Mid'+side});
  SKIN_BONE_DEFS.push({key:finger+side+'_2',a:finger+'Mid'+side,b:finger+'Tip'+side});
}));
const SKIN_MESH_GRID = 22; // denser than the old 14x14 grid; extra grid lines are inserted around detected joints.

// Builds the once-per-character skin mesh: a regular grid of vertices over the character's
// opaque silhouette (source-image pixel space), each vertex weighted against every bone
// segment in SKIN_BONE_DEFS by inverse distance (closer bone = more influence), normalized to
// sum to 1, then narrowed to its strongest 1-2 bones so a vertex deep inside one limb doesn't
// get a diluting pull from a distant, unrelated one. Returns null if srcCanvas has no visible
// pixels (nothing to build a mesh over).
function buildSkinMesh(srcCanvas, joints, srcW, srcH){
  let data;
  try{ data = srcCanvas.getContext('2d').getImageData(0,0,srcW,srcH).data; }
  catch(e){ return null; }
  const J = {};
  for(const k in joints){
    const pt=joints[k];
    if(pt && Number.isFinite(pt.x) && Number.isFinite(pt.y)) J[k] = { x:pt.x*srcW, y:pt.y*srcH };
  }
  const defs = SKIN_BONE_DEFS.filter(def=>J[def.a] && J[def.b] && Math.hypot(J[def.b].x-J[def.a].x,J[def.b].y-J[def.a].y)>0.5);
  if(defs.length<8) return null;
  const height=Math.max(1,Math.max(J.ankleL?.y||0,J.ankleR?.y||0)-(J.headTop?.y||0));
  const bones = defs.map(def=>({ key:def.key, a:J[def.a], b:J[def.b], len:Math.hypot(J[def.b].x-J[def.a].x,J[def.b].y-J[def.a].y) }));
  let minX=srcW, minY=srcH, maxX=-1, maxY=-1;
  for(let y=0;y<srcH;y++) for(let x=0;x<srcW;x++) if(data[(y*srcW+x)*4+3] > AUTORIG_ALPHA_THRESH){
    if(x<minX)minX=x; if(x>maxX)maxX=x; if(y<minY)minY=y; if(y>maxY)maxY=y;
  }
  if(maxX<minX || maxY<minY) return null;
  const padX=(maxX-minX)*0.025, padY=(maxY-minY)*0.025;
  minX=Math.max(0,minX-padX); maxX=Math.min(srcW-1,maxX+padX);
  minY=Math.max(0,minY-padY); maxY=Math.min(srcH-1,maxY+padY);
  function distToSeg(p,a,b){
    const dx=b.x-a.x,dy=b.y-a.y,l2=dx*dx+dy*dy||1e-6;
    const t=Math.max(0,Math.min(1,((p.x-a.x)*dx+(p.y-a.y)*dy)/l2));
    return Math.hypot(p.x-(a.x+dx*t),p.y-(a.y+dy*t));
  }
  function radiusFor(key,len){
    if(/^((thumb|index|middle|ring|pinky)[LR]_[12])$/.test(key)) return Math.max(2,height*0.008);
    if(/^hand[LR]$/.test(key)) return Math.max(3,height*0.018);
    if(key==='spine') return Math.max(5,height*0.105);
    if(key==='head') return Math.max(4,height*0.075);
    if(key.startsWith('leg')) return Math.max(3,height*0.035);
    return Math.max(2,height*0.024);
  }
  const keySet=new Set(defs.map(d=>d.key));
  function adjacent(a,b){
    if(a===b) return true;
    const fixed={
      head:['spine'],spine:['head','armL','armR','legL','legR'],
      armL:['spine','armL_lo'],armL_lo:['armL','handL'],handL:['armL_lo'],
      armR:['spine','armR_lo'],armR_lo:['armR','handR'],handR:['armR_lo'],
      legL:['spine','legL_lo'],legL_lo:['legL'],legR:['spine','legR_lo'],legR_lo:['legR']
    };
    if((fixed[a]||[]).includes(b)||(fixed[b]||[]).includes(a)) return true;
    const ma=a.match(/^(thumb|index|middle|ring|pinky)(L|R)_([12])$/), mb=b.match(/^(thumb|index|middle|ring|pinky)(L|R)_([12])$/);
    if(ma&&mb) return ma[1]===mb[1]&&ma[2]===mb[2]&&Math.abs(Number(ma[3])-Number(mb[3]))===1;
    const md=a.match(/^(thumb|index|middle|ring|pinky)(L|R)_1$/), mh=b.match(/^hand([LR])$/);
    const nd=b.match(/^(thumb|index|middle|ring|pinky)(L|R)_1$/), nh=a.match(/^hand([LR])$/);
    return !!((md&&mh&&md[2]===mh[1])||(nd&&nh&&nd[2]===nh[1]));
  }
  function axisStops(min,max,size,axis){
    const span=max-min, values=[];
    for(let i=0;i<=SKIN_MESH_GRID;i++) values.push({v:min+span*i/SKIN_MESH_GRID,pinned:false});
    for(const key in J){
      const v=J[key][axis];
      if(v>min+1 && v<max-1) values.push({v,pinned:true});
      const nudge=Math.max(1,size*0.004);
      if(v-nudge>min+1 && v-nudge<max-1) values.push({v:v-nudge,pinned:true});
      if(v+nudge>min+1 && v+nudge<max-1) values.push({v:v+nudge,pinned:true});
    }
    values.sort((a,b)=>a.v-b.v);
    let out=[];
    values.forEach(item=>{
      const prev=out[out.length-1];
      if(prev && Math.abs(prev.v-item.v)<Math.max(0.75,size*0.0008)){ prev.pinned=prev.pinned||item.pinned; if(item.pinned) prev.v=item.v; }
      else out.push({...item});
    });
    // Keep the landmarks, thin only uniform/non-landmark stops if a very hand-detailed input creates many columns.
    while(out.length>42){
      let best=-1,bestGap=Infinity;
      for(let i=1;i<out.length-1;i++) if(!out[i].pinned){
        const gap=Math.min(out[i].v-out[i-1].v,out[i+1].v-out[i].v);
        if(gap<bestGap){bestGap=gap;best=i;}
      }
      if(best<0) break;
      out.splice(best,1);
    }
    return [...new Set(out.map(x=>x.v))].sort((a,b)=>a-b);
  }
  const xs=axisStops(minX,maxX,srcW,'x'), ys=axisStops(minY,maxY,srcH,'y');
  const verts=[];
  for(let ry=0;ry<ys.length;ry++) for(let rx=0;rx<xs.length;rx++){
    const x=xs[rx],y=ys[ry], point={x,y};
    const ds=bones.map(bn=>({key:bn.key,d:distToSeg(point,bn.a,bn.b)/radiusFor(bn.key,bn.len)})).sort((a,b)=>a.d-b.d);
    const first=ds[0], second=ds.find(item=>item.key!==first.key&&keySet.has(item.key)&&adjacent(first.key,item.key));
    let w2=0;
    // Blend only a narrow joint seam between connected bones. Distant body parts never pull on one another.
    if(second){
      const gap=second.d-first.d;
      if(gap<0.78) w2=Math.min(0.28,Math.max(0,(0.78-gap)/0.78)*0.28);
    }
    const weights=w2>0.015?[{key:first.key,w:1-w2},{key:second.key,w:w2}]:[{key:first.key,w:1}];
    verts.push({x,y,u:x/srcW,v:y/srcH,weights});
  }
  const tris=[];
  for(let ry=0;ry<ys.length-1;ry++) for(let rx=0;rx<xs.length-1;rx++){
    const i0=ry*xs.length+rx,i1=i0+1,i2=i0+xs.length,i3=i2+1;
    tris.push([i0,i1,i2],[i1,i3,i2]);
  }
  const restBones={};
  bones.forEach(bn=>{
    const dx=bn.b.x-bn.a.x,dy=bn.b.y-bn.a.y;
    restBones[bn.key]={ax:bn.a.x,ay:bn.a.y,angle:Math.atan2(dy,dx),len:Math.max(1e-6,Math.hypot(dx,dy))};
  });
  return {verts,tris,restBones,boneDefs:defs,xs,ys,srcW,srcH,bounds:{minX,minY,maxX,maxY}};
}

// Re-derives skinMesh for every restored custom character that has autoRig data but no mesh
// yet (see openEditor's restore path — skinMesh is deliberately never persisted since it's
// fully rebuildable from autoRig.sourceImage+joints, which ARE persisted). Decodes each
// character's own source image into a throwaway canvas (buildSkinMesh needs real pixel data
// for its silhouette bounds, not just an <img> element) and builds the mesh from that. Old
// projects/characters with no autoRig at all (manual 10-slot builds, or pre-autorig saves)
// are left untouched and keep rendering via the old cut-part path, same as before this
// feature existed.
function rebuildSkinMeshesForCharacters(chars){
  const targets = chars.filter(c=>c.type==='custom' && c.autoRig && c.autoRig.sourceImage && !c.skinMesh);
  if(!targets.length) return Promise.resolve();
  return Promise.all(targets.map(c=>new Promise(resolve=>{
    const img = new Image();
    img.onload = ()=>{
      try{
        const cnv = document.createElement('canvas');
        cnv.width = c.autoRig.srcW; cnv.height = c.autoRig.srcH;
        cnv.getContext('2d').drawImage(img, 0, 0, c.autoRig.srcW, c.autoRig.srcH);
        c.skinMesh = buildSkinMesh(cnv, c.autoRig.joints, c.autoRig.srcW, c.autoRig.srcH);
      }catch(e){ c.skinMesh = null; } // falls back to the old cut-part path for this character
      resolve();
    };
    img.onerror = ()=>resolve(); // leave skinMesh null; old cut-part path still renders it
    img.src = c.autoRig.sourceImage;
  })));
}

// Per-frame: deforms every mesh vertex from its rest position using RIGID single-bone posing
// (each vertex now has exactly one weight of 1 — see buildSkinMesh). The vertex's rest-relative
// offset from its bone's rest start is rotated by that bone's rest->current rotation delta,
// scaled only by THAT bone's own length-change (so a joint drag that lengthens/shortens one
// segment still moves with it), then placed at the bone's CURRENT start position. Because each
// vertex belongs to only one bone, a joint move never pulls on skin belonging to a neighboring
// limb — the character reads as rigid cut-out pieces hinged at the joints, not a stretchy blend.
// `boneStartFn`/`boneEndFn` default to the live global getBoneStart/getBoneEnd (already pointed
// at the active character), matching drawCustomCharacterOnContext's own pattern, so export can
// pass bound closures for a non-active character exactly like it already does for the old
// cut-part path.
function poseSkinMesh(mesh, boneStartFn, boneEndFn){
  const bStart=boneStartFn||getBoneStart, bEnd=boneEndFn||getBoneEnd, cur={};
  (mesh.boneDefs||SKIN_BONE_DEFS).forEach(def=>{
    if(!mesh.restBones[def.key]) return;
    const s=bStart(def.key),e=bEnd(def.key),dx=e.x-s.x,dy=e.y-s.y;
    cur[def.key]={x:s.x,y:s.y,angle:Math.atan2(dy,dx),len:Math.max(1e-6,Math.hypot(dx,dy))};
  });
  function safeScale(key,rest,now){
    const raw=now.len/rest.len;
    if(key==='spine'||key==='head') return Math.max(0.985,Math.min(1.015,raw));
    if(/^((thumb|index|middle|ring|pinky)[LR]_[12])$/.test(key)) return Math.max(0.82,Math.min(1.12,raw));
    return Math.max(0.92,Math.min(1.08,raw));
  }
  function transformPoint(v,key,forceRigid){
    const rest=mesh.restBones[key],now=cur[key];
    if(!rest||!now) return {x:v.x,y:v.y};
    const dx=v.x-rest.ax,dy=v.y-rest.ay,cr=Math.cos(-rest.angle),sr=Math.sin(-rest.angle);
    let lx=dx*cr-dy*sr,ly=dx*sr+dy*cr;
    const sc=forceRigid?1:safeScale(key,rest,now); lx*=sc;ly*=sc;
    const cc=Math.cos(now.angle),ss=Math.sin(now.angle);
    return {x:now.x+lx*cc-ly*ss,y:now.y+lx*ss+ly*cc};
  }
  const posed=mesh.verts.map(v=>{
    let x=0,y=0;
    v.weights.forEach(w=>{const q=transformPoint(v,w.key,false);x+=q.x*w.w;y+=q.y*w.w;});
    return {x,y};
  });
  posed._currentBones=cur;
  posed._rigidMapper=(v,key)=>transformPoint(v,key,true);
  return posed;
}

// Draws the deformed mesh by warping the source texture one triangle at a time. Any mapping
// from one triangle to another is an affine transform, so each triangle is clipped (so
// neighboring triangles never bleed into each other at the seam) and the transform that sends
// the SOURCE triangle's 3 corners onto the DEFORMED triangle's 3 corners is solved and applied
// before drawing the full image — canvas clips to the path first, so only that triangle's
// pixels actually land on the stage. This is the standard 2D canvas equivalent of GPU texture-
// mapped triangle rendering. Since each vertex now belongs rigidly to a single bone (see
// buildSkinMesh/poseSkinMesh), a triangle whose 3 corners all share one bone just rotates/
// translates as a solid piece; only the handful of seam triangles that straddle two different
// bones' vertex sets get a (mild, joint-local) shear — this is what keeps the joints from
// visibly tearing apart while the rest of each limb stays perfectly rigid.
function drawSkinnedMesh(targetCtx, img, mesh, posedVerts){
  mesh.tris.forEach(tri=>{
    const [i0,i1,i2] = tri;
    const s0 = mesh.verts[i0], s1 = mesh.verts[i1], s2 = mesh.verts[i2];
    let d0 = posedVerts[i0], d1 = posedVerts[i1], d2 = posedVerts[i2];
    // Prevent inverted/near-collapsed triangles (the usual visual cause of a folded belly or a pinched seam).
    const srcArea=(s1.x-s0.x)*(s2.y-s0.y)-(s2.x-s0.x)*(s1.y-s0.y);
    let dstArea=(d1.x-d0.x)*(d2.y-d0.y)-(d2.x-d0.x)*(d1.y-d0.y);
    const badTriangle=srcArea*dstArea<=0 || Math.abs(dstArea)<Math.abs(srcArea)*0.045;
    if(badTriangle && posedVerts._rigidMapper){
      const votes={};
      [s0,s1,s2].forEach(v=>v.weights.forEach(w=>votes[w.key]=(votes[w.key]||0)+w.w));
      const rigidKey=Object.keys(votes).sort((a,b)=>votes[b]-votes[a])[0];
      d0=posedVerts._rigidMapper(s0,rigidKey); d1=posedVerts._rigidMapper(s1,rigidKey); d2=posedVerts._rigidMapper(s2,rigidKey);
      dstArea=(d1.x-d0.x)*(d2.y-d0.y)-(d2.x-d0.x)*(d1.y-d0.y);
      if(srcArea*dstArea<=0 || Math.abs(dstArea)<1e-5) return;
    }
    // Source triangle in image pixel space.
    const sx0=s0.x, sy0=s0.y, sx1=s1.x, sy1=s1.y, sx2=s2.x, sy2=s2.y;
    // Solve the 2x3 affine matrix [a c e; b d f] mapping (sx,sy)->(dx,dy) for all 3 corners.
    const denom = (sx1-sx0)*(sy2-sy0) - (sx2-sx0)*(sy1-sy0);
    if(Math.abs(denom) < 1e-8) return; // degenerate triangle (shouldn't happen on a regular grid) — skip rather than divide by ~0
    const a = ((d1.x-d0.x)*(sy2-sy0) - (d2.x-d0.x)*(sy1-sy0)) / denom;
    const b = ((d1.y-d0.y)*(sy2-sy0) - (d2.y-d0.y)*(sy1-sy0)) / denom;
    const c = ((d2.x-d0.x)*(sx1-sx0) - (d1.x-d0.x)*(sx2-sx0)) / denom;
    const d = ((d2.y-d0.y)*(sx1-sx0) - (d1.y-d0.y)*(sx2-sx0)) / denom;
    const e = d0.x - (a*sx0 + c*sy0);
    const f = d0.y - (b*sx0 + d*sy0);
    targetCtx.save();
    targetCtx.beginPath();
    // FIX (visible black crosshatch/"graph" over the character): canvas anti-aliases every
    // clip path edge, so two triangles whose edges sit pixel-adjacent (not overlapping) each
    // fade out right at that shared edge — the two independent fades don't cancel, they add
    // up as a thin dark seam. This was barely visible under the old blended skinning (adjacent
    // triangles moved almost identically), but rigid single-bone posing (see buildSkinMesh) can
    // rotate/shear neighboring triangles differently enough that the seam grid became a clearly
    // visible line pattern across flat-colored art like a shirt. Fix: nudge each of the
    // triangle's 3 clip corners outward from the triangle's own center by ~0.75px before
    // clipping, so neighboring triangles now overlap by a fraction of a pixel instead of
    // butting exactly edge-to-edge — invisible on the art itself, but enough to hide the seam.
    const cx = (d0.x+d1.x+d2.x)/3, cy = (d0.y+d1.y+d2.y)/3;
    const OVERLAP = 0.75;
    function pushOut(p){
      const vx = p.x-cx, vy = p.y-cy;
      const len = Math.hypot(vx,vy) || 1;
      return { x: p.x + (vx/len)*OVERLAP, y: p.y + (vy/len)*OVERLAP };
    }
    const p0 = pushOut(d0), p1 = pushOut(d1), p2 = pushOut(d2);
    targetCtx.moveTo(p0.x, p0.y);
    targetCtx.lineTo(p1.x, p1.y);
    targetCtx.lineTo(p2.x, p2.y);
    targetCtx.closePath();
    targetCtx.clip();
    targetCtx.transform(a, b, c, d, e, f);
    targetCtx.drawImage(img, 0, 0);
    targetCtx.restore();
  });
}

// Draws every uploaded body-part image stamped onto its matching bone segment (see
// PART_SLOTS' `bone` field). Each part's own pivot point (dragged into place in the pivot
// editor) is placed at that bone's START joint, then rotated to the bone's current angle —
// so posing the rig moves the artwork exactly like the stroke-stickman's lines would.
function drawCustomCharacterOnContext(targetCtx, char, boneStartFn, boneEndFn){
  // Model Lab part-sheet models use their own 21-bone hierarchical puppet renderer.
  // All legacy auto-rig and older custom-part models continue through their original paths.
  if(char && char.sheet && typeof drawSheetCharacter === 'function'){
    drawSheetCharacter(targetCtx, char);
    return;
  }
  // Live drawStick() calls this with no boneStartFn/boneEndFn — fall back to the global
  // getBoneStart/getBoneEnd, which read the shared `bones` (already pointed at this
  // character via syncActiveCharacterRefs since this only ever runs for the active
  // character in that path). Export's drawCharacterToContext passes its own closures bound
  // to that specific (possibly non-active) character's bones instead.
  const bStart = boneStartFn || getBoneStart;
  const bEnd = boneEndFn || getBoneEnd;

  // NEW: skeletal mesh skinning path (see the SKELETAL MESH SKINNING block above) — used for
  // every auto-rigged character that has a built skin mesh (char.skinMesh, built once in
  // finalizeAutoRiggedCharacter from the full original photo). This replaces the old cut-part
  // stamping below for those characters entirely, so the art stretches/bends continuously from
  // the bone points instead of being built from separately-rotating rigid crops. Characters
  // saved before this feature existed have no skinMesh yet and fall through to the old path
  // unchanged, so old projects keep opening/rendering exactly as before.
  if(char.skinMesh && char.autoRig && char.autoRig.sourceImage){
    if(!targetCtx.__skinImgCache) targetCtx.__skinImgCache = {};
    let simg = targetCtx.__skinImgCache[char.autoRig.sourceImage];
    if(!simg){
      simg = new Image();
      simg.src = char.autoRig.sourceImage;
      targetCtx.__skinImgCache[char.autoRig.sourceImage] = simg;
    }
    if(simg.complete && simg.naturalWidth){
      const posed = poseSkinMesh(char.skinMesh, bStart, bEnd);
      drawSkinnedMesh(targetCtx, simg, char.skinMesh, posed);
    }
    return;
  }

  PART_SLOTS.forEach(slot=>{
    const part = char.parts[slot.id];
    if(!part || !part.src) return;
    const s = bStart(slot.bone);
    const e = bEnd(slot.bone);
    const dx = e.x-s.x, dy = e.y-s.y;
    const boneLen = Math.hypot(dx, dy) || 1;
    const angle = Math.atan2(dy, dx);
    // FIX (uploaded parts render as the wrong/distorted shape): everything below is now
    // measured against the part's DETECTED CONTENT BOX (see detectOpaqueBounds) instead of
    // the raw uploaded image canvas. A phone crop almost never lines up pixel-perfect with
    // the actual artwork — there's near-always leftover transparent margin — so scaling
    // "the whole image" to fit a bone length was really scaling "mostly empty space plus a
    // bit of art" to fit it, which is what produced the squashed/oversized/misplaced limbs.
    // Parts saved before this fix have no contentBox yet; box falls back to the full raw
    // image so those old characters keep rendering exactly as before (no surprise resize).
    const box = part.contentBox || {x:0, y:0, w:part.w, h:part.h};
    // pivotX/pivotY are stored as fractions of the RAW image (see the pivot dot's own
    // coordinate space in positionPivotDot/setupPivotDrag) — convert to a fraction of the
    // content box so the "distance from pivot to the box's far edge" below lines up with
    // where the art actually starts and ends, not the canvas edges.
    const pivotPxX = part.pivotX*part.w, pivotPxY = part.pivotY*part.h;
    const pivotInBoxX = (pivotPxX - box.x)/box.w, pivotInBoxY = (pivotPxY - box.y)/box.h;
    // Scale so the pixel distance from the pivot to whichever content-box edge is farther
    // (top or bottom of the ACTUAL ART — the pivot isn't always exactly at 0 or 1) equals
    // the bone's current length, so posing/resizing the rig scales the art proportionally.
    const pivotToFar = Math.max(box.h*pivotInBoxY, box.h*(1-pivotInBoxY)) || box.h/2;
    const scale = boneLen / pivotToFar;
    const drawW = part.w*scale, drawH = part.h*scale;
    if(!targetCtx.__partImgCache) targetCtx.__partImgCache = {};
    let img = targetCtx.__partImgCache[part.src];
    if(!img){
      img = new Image();
      img.src = part.src;
      targetCtx.__partImgCache[part.src] = img;
    }
    if(!img.complete || !img.naturalWidth) return; // not decoded yet; next frame will catch it
    targetCtx.save();
    targetCtx.translate(s.x, s.y);
    // Bone angle 0 points along +x (see getBoneEnd's cos/sin usage: end = start + cos*len,
    // sin*len). The image's own "down the bone" direction (pivot near the top, art hanging
    // toward the far edge) is the image's LOCAL +y axis. After ctx.rotate(theta), local +y
    // maps to world direction (-sin(theta), cos(theta)) — for that to equal the bone's
    // direction (cos(angle), sin(angle)) we need theta = angle - 90deg, NOT angle + 90deg
    // (the previous +90deg had the sign backwards, which was rotating every limb to the
    // wrong side — e.g. an arm meant to hang outward instead swinging up behind the torso).
    targetCtx.rotate(angle - Math.PI/2);
    // Draw the FULL raw image (so any margin around the art is preserved exactly as
    // uploaded — we're not cropping the pixels, just correcting what the scale/pivot math
    // measures against), positioned so the pivot point itself — not the raw canvas corner —
    // lands at the bone's start joint.
    targetCtx.drawImage(img, -drawW*part.pivotX, -drawH*part.pivotY, drawW, drawH);
    targetCtx.restore();
  });
}
// Decodes every custom character's part images into targetCtx's own __partImgCache ahead of
// time, so the very first drawCustomCharacterOnContext() call against that context already
// has every image ready (see startExport's call to this, and initCustomCharacterPreload's
// live-canvas equivalent below). Returns a promise that resolves once every image has either
// loaded or failed (a failed part just never draws, same as any other missing-source part).
function preloadCustomCharacterImages(targetCtx, chars){
  if(!targetCtx.__partImgCache) targetCtx.__partImgCache = {};
  if(!targetCtx.__skinImgCache) targetCtx.__skinImgCache = {};
  const cache = targetCtx.__partImgCache;
  const skinCache = targetCtx.__skinImgCache;
  const waits = [];
  chars.forEach(c=>{
    if(c.type!=='custom') return;
    // Skinned characters (see drawCustomCharacterOnContext's skinMesh branch) preload the ONE
    // full source photo instead of per-part crops.
    if(c.skinMesh && c.autoRig && c.autoRig.sourceImage && !skinCache[c.autoRig.sourceImage]){
      const simg = new Image();
      skinCache[c.autoRig.sourceImage] = simg;
      waits.push(new Promise(res=>{ simg.onload = res; simg.onerror = res; }));
      simg.src = c.autoRig.sourceImage;
    }
    if(c.sheet && c.sheet.parts){
      if(!targetCtx.__sheetImg) targetCtx.__sheetImg = {};
      Object.values(c.sheet.parts).forEach(part=>{
        if(!part || !part.src || targetCtx.__sheetImg[part.src]) return;
        const img = new Image();
        targetCtx.__sheetImg[part.src] = img;
        waits.push(new Promise(res=>{ img.onload = res; img.onerror = res; }));
        img.src = part.src;
      });
    }
    if(!c.parts) return;
    PART_SLOTS.forEach(slot=>{
      const part = c.parts[slot.id];
      if(!part || !part.src || cache[part.src]) return;
      const img = new Image();
      cache[part.src] = img;
      waits.push(new Promise(res=>{ img.onload = res; img.onerror = res; }));
      img.src = part.src;
    });
  });
  return Promise.all(waits);
}

/* ============================================================================
   AUTO-RIG: single full-body image -> automatic bone skeleton + body-part rig.
   Replaces the 10-part manual upload as the default "Custom character" flow
   (see openAutoRigUploader / the Add Character modal's "Custom character"
   tile, and openAutoRigEditorForActive / the Bones panel's "Rig my art"
   tile). The old manual builder (openCustomCharBuilder / PART_SLOTS pivot
   editor above) is left fully intact and reachable via a secondary "manual
   mode" link, both so old saved projects keep editing exactly as before and
   as a fallback for art the auto-detector struggles with.

   HONESTY NOTE: this is a heuristic, alpha-silhouette image-analysis
   pipeline (row-by-row foreground-run scanning), not a machine-learning
   model — see detectCharacterSilhouette below. It's designed around a
   front-facing character with arms extended out to the sides and a
   transparent background (the classic T/A-pose rigging reference pose);
   it degrades gracefully (with an on-screen warning + a sane fallback
   guess) when that assumption doesn't hold, rather than failing silently
   or fabricating confidence. Every stage is its own function
   (detectCharacterSilhouette -> buildRigOverrideFromJoints /
   computePartRegions -> generateAutoRigParts) so a real ML segmentation or
   pose model could replace just the detection stage later without
   touching the rig-building, cropping, or rendering code.

   OUTPUT SHAPE: generateAutoRigParts() below produces exactly the same
   {src, pivotX, pivotY, w, h, contentBox} shape per PART_SLOTS id that the
   manual pivot editor already produces, and buildRigOverrideFromJoints()
   produces {lengths, angles, attachAt} consumed by applyRigOverride() (see
   makeCharacter/openEditor above) — so drawCustomCharacterOnContext,
   preloadCustomCharacterImages, export, and project save/load all work
   completely unchanged for an auto-rigged character.
   ============================================================================ */

const AUTORIG_ALPHA_THRESH = 20;
const AUTORIG_MAX_SRC_DIM = 1100;     // normalized working/stored image cap
const AUTORIG_ANALYSIS_MAX_DIM = 420; // downscaled resolution used for detection scanning
// Default on-stage figure height (fraction of stageH) is AUTO_RIG_FIGURE_HEIGHT_FRAC, defined
// alongside applyRigOverride() above, next to buildFreshRig — that's the single place scaling
// from detected-image-space to stage-space actually happens.

function autorigIsFg(data, w, x, y){ return data[(y*w+x)*4+3] > AUTORIG_ALPHA_THRESH; }

// ---- STEP: detect character foreground/silhouette + calculate character bounds ----
function autorigGlobalBounds(data, w, h){
  let minX=w, minY=h, maxX=-1, maxY=-1;
  for(let y=0;y<h;y++){
    for(let x=0;x<w;x++){
      if(autorigIsFg(data,w,x,y)){
        if(x<minX)minX=x; if(x>maxX)maxX=x;
        if(y<minY)minY=y; if(y>maxY)maxY=y;
      }
    }
  }
  if(maxX<minX || maxY<minY) return null; // fully transparent
  return { minX, minY, maxX, maxY };
}
// Per-row contiguous foreground runs, only within bounds. rows[y] = [[x0,x1], ...].
function autorigComputeRowRuns(data, w, bounds){
  const { minX, maxX, minY, maxY } = bounds;
  const rows = new Array(maxY+1);
  for(let y=minY;y<=maxY;y++){
    const runs=[]; let runStart=-1;
    for(let x=minX;x<=maxX;x++){
      const fg = autorigIsFg(data,w,x,y);
      if(fg && runStart===-1) runStart=x;
      if(!fg && runStart!==-1){ runs.push([runStart,x-1]); runStart=-1; }
    }
    if(runStart!==-1) runs.push([runStart,maxX]);
    rows[y]=runs;
  }
  return rows;
}
function autorigRowSpanWidth(runs){ if(!runs||!runs.length) return 0; return runs[runs.length-1][1]-runs[0][0]+1; }
// Persistent leg-split finder (see STEP: detect major body regions) — a single noisy row
// (fringe pixel gap) doesn't trigger a false split; it must hold for a short run of rows.
function autorigFindSplitY(rows, startY, endY){
  const persist = Math.max(3, Math.round((endY-startY)*0.03));
  for(let y=startY;y<=endY;y++){
    const runs = rows[y];
    if(runs && runs.length>=2){
      let ok=true;
      for(let k=1;k<=persist && y+k<=endY;k++){ if(!rows[y+k]||rows[y+k].length<2){ ok=false; break; } }
      if(ok) return y;
    }
  }
  return -1;
}
// Local minimum of row width between headTopY and armBandTopY — typically the neck (narrower
// than the head above and the shoulders/chest below). Both ends of the search range are
// trimmed a little first, since a rounded head/hair outline tapers toward zero width right at
// its own top and bottom edges, which would otherwise look like an even narrower "neck".
function autorigFindNeckY(rows, headTopY, armBandTopY){
  const span = armBandTopY-headTopY;
  const margin = Math.max(2, Math.round(span*0.12));
  let searchStart = headTopY+margin;
  const searchEnd = armBandTopY-margin;
  // Anchor the search start to the head's own widest row (hair/ears/cheeks) instead of trusting
  // a blind percentage of the head-to-shoulder span. A large/rounded cartoon head can still be
  // tapering toward its chin well past that fixed 12% margin — without this, the plain global
  // minimum below used to land ON THE CHIN (part of the face), not the actual neck below it,
  // which is what put a stray joint dot visibly on the character's face.
  let headMaxW = 0;
  for(let y=headTopY; y<=searchEnd; y++) headMaxW = Math.max(headMaxW, autorigRowSpanWidth(rows[y]));
  if(headMaxW>0){
    for(let y=searchStart; y<=searchEnd; y++){
      if(autorigRowSpanWidth(rows[y]) <= headMaxW*0.62){ searchStart = y; break; }
    }
  }
  let minW=Infinity, minY=-1;
  for(let y=searchStart;y<=searchEnd;y++){
    const width = autorigRowSpanWidth(rows[y]);
    if(width>0 && width<minW){ minW=width; minY=y; }
  }
  if(minY===-1) return { y: Math.round(headTopY+span*0.55), guessed:true };
  return { y:minY, guessed:false };
}

// ---- STEPS: detect body regions -> estimate skeleton -> estimate joints ----
// Ported 1:1 from a standalone, unit-tested reference implementation (synthetic T-pose,
// dress/merged-legs, tiny-character, and fully-transparent test cases) — see this feature's
// summary for where that lives. Returns null only for a fully transparent image. Otherwise
// always returns a complete, usable joint set (falling back to a proportion-based guess +
// a warning wherever detection is ambiguous — see the "Do NOT silently produce a broken rig"
// requirement this is built around) plus `warnings`, user-facing strings for anything guessed.
function detectCharacterSilhouette(imageData){
  const { width:w, height:h, data } = imageData;
  const warnings = [];
  const gb = autorigGlobalBounds(data,w,h);
  if(!gb) return null;
  const { minX,minY,maxX,maxY } = gb;
  const figH = maxY-minY+1, figW = maxX-minX+1;
  if(figH < h*0.15 || figW < w*0.08){
    warnings.push('The character looks very small in this image — results may be less accurate.');
  }
  const rows = autorigComputeRowRuns(data,w,gb);

  // Arm band: rows whose span width is close to the overall max span (arms extended out to
  // the sides make those rows far wider than the head/torso alone).
  let maxSpan=0;
  for(let y=minY;y<=maxY;y++) maxSpan=Math.max(maxSpan, autorigRowSpanWidth(rows[y]));
  let armBandTop=-1, armBandBottom=-1;
  for(let y=minY;y<=maxY;y++){
    if(autorigRowSpanWidth(rows[y]) >= maxSpan*0.85){ if(armBandTop===-1) armBandTop=y; armBandBottom=y; }
  }
  const torsoProbeY = Math.min(maxY, minY+Math.round(figH*0.35));
  const torsoProbeWidth = autorigRowSpanWidth(rows[torsoProbeY]) || 1;
  const armsExtended = maxSpan >= torsoProbeWidth*1.25;
  if(!armsExtended){
    warnings.push("Couldn't confidently detect arms extended out to the sides — using an estimated arm pose. Drag the shoulder/elbow/wrist points to fix.");
  }

  // Leg split (hip/crotch): scan downward from the bottom of the arm band for the silhouette
  // splitting into two separate leg runs.
  const legScanStart = Math.min(maxY, armBandBottom+1);
  let hipY = autorigFindSplitY(rows, legScanStart, maxY);
  let legsDetected = true;
  if(hipY===-1){
    legsDetected = false;
    warnings.push("Couldn't clearly detect separate legs — splitting the lower body down the middle. Adjust the knee/ankle points if needed.");
    hipY = minY+Math.round(figH*0.55); // common full-body hip-height proportion
    if(hipY<=armBandBottom) hipY = armBandBottom+1;
    if(hipY>=maxY) hipY = maxY-1;
  }

  const neckInfo = autorigFindNeckY(rows, minY, Math.max(minY+1, armBandTop===-1?Math.round(minY+figH*0.25):armBandTop));
  const neckY = neckInfo.y;
  if(neckInfo.guessed) warnings.push("Couldn't clearly find the neck line — using an estimated head size. Adjust the neck point if needed.");

  // Body centerline: measured at the row just above the hip split (still a single torso run
  // there) — the most stable symmetric reference for a front-facing character.
  const waistY = Math.max(minY, hipY-1);
  const waistRuns = rows[waistY] && rows[waistY].length ? rows[waistY] : rows[neckY];
  const bodyCenterX = waistRuns && waistRuns.length
    ? (waistRuns[0][0]+waistRuns[waistRuns.length-1][1])/2
    : (minX+maxX)/2;

  // Arms: leftmost/rightmost foreground pixel within the arm band.
  let armMinX=maxX, armMaxX=minX;
  const armBandLo = armBandTop===-1?neckY:armBandTop, armBandHi = armBandBottom===-1?neckY:armBandBottom;
  for(let y=armBandLo;y<=armBandHi;y++){
    const runs=rows[y]; if(!runs||!runs.length) continue;
    armMinX=Math.min(armMinX, runs[0][0]); armMaxX=Math.max(armMaxX, runs[runs.length-1][1]);
  }
  if(armMinX>armMaxX){ armMinX=minX; armMaxX=maxX; }
  const armY = Math.round((armBandLo+armBandHi)/2);
  const shoulderY = armY; // pure-horizontal-arm simplification (matches the T/A-pose reference)

  // attachAt: fraction from hip up to neck where the shoulder sits on the spine (see
  // applyRigOverride) — data-driven instead of the stickman's fixed 0.85.
  let attachAtFrac = (hipY-shoulderY)/Math.max(1,(hipY-neckY));
  attachAtFrac = Math.max(0.5, Math.min(0.98, attachAtFrac));

  // Legs: split the region below hipY into left/right blobs (or, if no split was detected,
  // divide each row's own full span down the middle instead — see legsDetected above).
  let leftMinX=Infinity, leftMaxX=-Infinity, leftBottomY=hipY;
  let rightMinX=Infinity, rightMaxX=-Infinity, rightBottomY=hipY;
  for(let y=hipY;y<=maxY;y++){
    const runs=rows[y]; if(!runs||!runs.length) continue;
    if(legsDetected && runs.length>=2){
      const left=runs[0], right=runs[runs.length-1];
      leftMinX=Math.min(leftMinX,left[0]); leftMaxX=Math.max(leftMaxX,left[1]); leftBottomY=y;
      rightMinX=Math.min(rightMinX,right[0]); rightMaxX=Math.max(rightMaxX,right[1]); rightBottomY=y;
    } else {
      const first=runs[0][0], last=runs[runs.length-1][1];
      leftMinX=Math.min(leftMinX,first); leftMaxX=Math.max(leftMaxX,bodyCenterX); leftBottomY=y;
      rightMinX=Math.min(rightMinX,bodyCenterX); rightMaxX=Math.max(rightMaxX,last); rightBottomY=y;
    }
  }
  if(leftMaxX<leftMinX){ leftMinX=minX; leftMaxX=bodyCenterX; }
  if(rightMaxX<rightMinX){ rightMinX=bodyCenterX; rightMaxX=maxX; }

  const ankleL = { x:(leftMinX+leftMaxX)/2, y:leftBottomY };
  const ankleR = { x:(rightMinX+rightMaxX)/2, y:rightBottomY };
  const kneeL = { x:(bodyCenterX+ankleL.x)/2, y:(hipY+ankleL.y)/2 };
  const kneeR = { x:(bodyCenterX+ankleR.x)/2, y:(hipY+ankleR.y)/2 };

  // Left/right by screen position (smaller x = "L") — matches this app's existing stickman
  // rig convention (see buildFreshRig: armL/legL default to the -x side), not anatomy, so
  // there's no photo-vs-mirror ambiguity to correct for.
  const shoulder = { x:bodyCenterX, y:shoulderY };
  const wristL = { x:armMinX, y:armY }, wristR = { x:armMaxX, y:armY };
  const elbowL = { x:(bodyCenterX+wristL.x)/2, y:(shoulderY+armY)/2 };
  const elbowR = { x:(bodyCenterX+wristR.x)/2, y:(shoulderY+armY)/2 };
  const headTop = { x:bodyCenterX, y:minY };
  const neck = { x:bodyCenterX, y:neckY };
  const hip = { x:bodyCenterX, y:hipY };

  // Sanity clamps so a downstream rig never gets a degenerate (zero-length/inverted) bone.
  const MIN_GAP = Math.max(2, Math.round(figH*0.02));
  const hipYc = Math.max(hip.y, neck.y+MIN_GAP);
  const feetLYc = Math.max(ankleL.y, hipYc+MIN_GAP);
  const feetRYc = Math.max(ankleR.y, hipYc+MIN_GAP);

  // Torso half-width, measured well below the shoulder/arm line (55% of the way down to the
  // hip) so the extended arms — which only exist at shoulder height — never widen this reading.
  // computePartRegions uses this to keep the 'torso' crop to the body's own actual width instead
  // of defaulting to the full figure width (arms included): without it, the torso crop bakes in
  // a static copy of both arms UNDERNEATH the separately-rotated arm parts, which is what made
  // parts like the hands appear to render more than once in the finished rig.
  const torsoWidthProbeY = Math.max(minY, Math.min(maxY, Math.round(shoulderY + (hipY-shoulderY)*0.55)));
  const torsoWidthProbePx = autorigRowSpanWidth(rows[torsoWidthProbeY]) || torsoProbeWidth || figW*0.4;
  const torsoHalfWidthFrac = Math.max(2, torsoWidthProbePx/2) / w;

  // How far down neck sits between headTop and the shoulder line, as a 0-1 fraction — the UI no
  // longer exposes neck as its own draggable point (see AUTORIG_JOINT_KEYS), so this ratio is
  // what keeps it sensibly anchored while the user drags headTop/shoulder/hip instead.
  const neckRatio = Math.max(0, Math.min(1, (neckY-minY) / Math.max(1, (shoulderY-minY))));

  const toFrac = p => ({ x:p.x/w, y:p.y/h });
  return {
    warnings, srcW:w, srcH:h, bounds: gb,
    diagnostics: { armsExtended, legsDetected },
    torsoHalfWidthFrac, neckRatio,
    joints: {
      headTop: toFrac(headTop), neck: toFrac(neck),
      shoulderL: toFrac(shoulder), shoulderR: toFrac(shoulder),
      elbowL: toFrac(elbowL), elbowR: toFrac(elbowR),
      wristL: toFrac(wristL), wristR: toFrac(wristR),
      hip: toFrac(hip),
      kneeL: toFrac(kneeL), kneeR: toFrac(kneeR),
      ankleL: toFrac({ x:ankleL.x, y:feetLYc }), ankleR: toFrac({ x:ankleR.x, y:feetRYc }),
    },
    attachAt: { armL: attachAtFrac, armR: attachAtFrac },
  };
}

// ---- STEP: generate bones ----
// Turns detected/user-adjusted joints (fractions of the source image) into the bone
// length/angle/attachAt fractions applyRigOverride() merges onto a fresh buildFreshRig() rig.
// Lengths are fractions of the character's OWN detected height (headTop-to-ankle), so
// applyRigOverride scales them to a fixed target on-stage figure height, independent of the
// source image's own resolution — a tiny upload and a huge one both come in at a sane size.
const AUTORIG_MIN_LEN_FRAC = 0.03;
function buildRigOverrideFromJoints(joints, srcW, srcH, attachAt){
  const J = {};
  for(const k in joints) J[k] = { x: joints[k].x*srcW, y: joints[k].y*srcH };
  const dist = (a,b) => Math.max(AUTORIG_MIN_LEN_FRAC, Math.hypot(b.x-a.x, b.y-a.y));
  const ang = (a,b) => Math.atan2(b.y-a.y, b.x-a.x);
  const charHeightPx = Math.max(1, Math.max(J.ankleL.y, J.ankleR.y) - J.headTop.y);
  const lenFrac = (a,b,min=AUTORIG_MIN_LEN_FRAC) => Math.max(min,Math.hypot(b.x-a.x,b.y-a.y)/charHeightPx);
  const lengths = {
    spine: lenFrac(J.hip, J.neck), head: lenFrac(J.neck, J.headTop),
    armL: lenFrac(J.shoulderL, J.elbowL), armL_lo: lenFrac(J.elbowL, J.wristL),
    armR: lenFrac(J.shoulderR, J.elbowR), armR_lo: lenFrac(J.elbowR, J.wristR),
    legL: lenFrac(J.hip, J.kneeL), legL_lo: lenFrac(J.kneeL, J.ankleL),
    legR: lenFrac(J.hip, J.kneeR), legR_lo: lenFrac(J.kneeR, J.ankleR),
  };
  const angles = {
    spine: ang(J.hip, J.neck), head: ang(J.neck, J.headTop),
    armL: ang(J.shoulderL, J.elbowL), armL_lo: ang(J.elbowL, J.wristL),
    armR: ang(J.shoulderR, J.elbowR), armR_lo: ang(J.elbowR, J.wristR),
    legL: ang(J.hip, J.kneeL), legL_lo: ang(J.kneeL, J.ankleL),
    legR: ang(J.hip, J.kneeR), legR_lo: ang(J.kneeR, J.ankleR),
  };
  const at = { armL: attachAt.armL, armR: attachAt.armR };
  const attachOffsets = {};
  // Place each upper arm's root at its detected shoulder instead of the torso centre line.
  // Offsets are stored in tangent/normal coordinates so they follow torso rotation without
  // stretching the belly or dragging the chest when an arm bends.
  ['L','R'].forEach(side=>{
    const key='arm'+side, shoulder=J['shoulder'+side], t=Math.max(0,Math.min(1,Number(at[key])||0));
    const base={x:J.hip.x+(J.neck.x-J.hip.x)*t,y:J.hip.y+(J.neck.y-J.hip.y)*t};
    const dx=shoulder.x-base.x,dy=shoulder.y-base.y,a=angles.spine;
    attachOffsets[key]={x:(dx*Math.cos(a)+dy*Math.sin(a))/charHeightPx,y:(-dx*Math.sin(a)+dy*Math.cos(a))/charHeightPx};
  });
  ['L','R'].forEach(side=>{
    const wrist=J['wrist'+side], palm=J['palm'+side];
    if(wrist && palm){
      lengths['hand'+side]=lenFrac(wrist,palm,0.006);
      angles['hand'+side]=ang(wrist,palm);
      at['hand'+side]=1;
      const hx=palm.x-wrist.x, hy=palm.y-wrist.y, h2=hx*hx+hy*hy||1;
      ['thumb','index','middle','ring','pinky'].forEach(finger=>{
        const base=J[finger+'Base'+side], mid=J[finger+'Mid'+side], tip=J[finger+'Tip'+side];
        if(!base || !mid || !tip) return;
        const k=finger+side, t=Math.max(0,Math.min(1,((base.x-wrist.x)*hx+(base.y-wrist.y)*hy)/h2));
        const px=wrist.x+hx*t, py=wrist.y+hy*t;
        const ux=hx/Math.sqrt(h2), uy=hy/Math.sqrt(h2);
        const nx=-uy, ny=ux;
        at[k+'_1']=t;
        attachOffsets[k+'_1']={x:((base.x-px)*ux+(base.y-py)*uy)/charHeightPx, y:((base.x-px)*nx+(base.y-py)*ny)/charHeightPx};
        lengths[k+'_1']=lenFrac(base,mid,0.0035); angles[k+'_1']=ang(base,mid);
        lengths[k+'_2']=lenFrac(mid,tip,0.0035); angles[k+'_2']=ang(mid,tip); at[k+'_2']=1;
      });
    }
  });
  return { lengths, angles, attachAt:at, attachOffsets };
}
// Maps every PART_SLOTS id to the rig bone whose START joint is that part's pivot (see
// PART_SLOTS' own bone/defaultPivot fields above — this mirrors them exactly).
const AUTORIG_BONE_FOR_PART = {
  head:'head', torso:'spine',
  armL_upper:'armL', armL_lower:'armL_lo', armR_upper:'armR', armR_lower:'armR_lo',
  legL_upper:'legL', legL_lower:'legL_lo', legR_upper:'legR', legR_lower:'legR_lo',
};

// ---- STEPS: detect cut areas -> generate joint overlap areas ----
// Padded pixel-space region rectangles (same coordinate system as the joints passed in) for
// the 10 PART_SLOTS crop targets, generous enough around each joint that tightenRegion()
// below can crop tight to the REAL silhouette without clipping any actual artwork. `overlapPx`
// is how far each region reaches PAST its nominal joint boundary into the neighboring part —
// since the source art is one continuous drawing with no real gap at a joint, this overlap
// naturally captures a bit of the same artwork in both neighboring crops, which is what hides
// the seam once the parts rotate independently (the "stretchy/natural joint" effect) — see
// generateAutoRigParts/makeRotatedPartCanvas for where that overlap actually gets drawn.
// Fresh, pixel-accurate torso half-width for generateAutoRigParts to pass into
// computePartRegions at final-cut time (see torsoHalfWidthFrac in detectCharacterSilhouette,
// which this mirrors, but measured against the CURRENT — possibly user-dragged — joints and
// full-resolution art rather than a value cached from initial detection).
function autorigTorsoHalfWidthAt(data, w, h, J){
  const probeY = Math.max(0, Math.min(h-1, Math.round(J.shoulderL.y + (J.hip.y-J.shoulderL.y)*0.55)));
  let lo=-1, hi=-1;
  for(let x=0;x<w;x++){ if(autorigIsFg(data,w,x,probeY)){ if(lo===-1) lo=x; hi=x; } }
  if(lo===-1) return Math.max(2, Math.abs(J.elbowL.x-J.hip.x)*0.4);
  return Math.max(2, (hi-lo)/2);
}
function computePartRegions(joints, srcW, srcH, bounds, overlapPx, charHeightPx, torsoHalfWidthPx){
  const J = {};
  for(const k in joints) J[k] = { x: joints[k].x*srcW, y: joints[k].y*srcH };
  const cx = J.hip.x;
  const armPad = charHeightPx*0.14;
  // Torso's own half-width (falls back to the old arm-span estimate if the caller doesn't have
  // a measured value handy — see renderAutoRigOverlay's cheap cached copy vs
  // generateAutoRigParts' fresh one). Clamped so the torso region never reaches out to where
  // the extended arms are — see this function's block comment above for why that mattered.
  const torsoHW = Math.max(2, torsoHalfWidthPx || Math.abs(J.elbowL.x-cx)*0.4);
  const torsoX0 = Math.max(bounds.minX, cx-torsoHW-overlapPx);
  const torsoX1 = Math.min(bounds.maxX, cx+torsoHW+overlapPx);
  const R = {
    head: { x0:bounds.minX, y0:bounds.minY, x1:bounds.maxX, y1:J.neck.y+overlapPx },
    torso: { x0:torsoX0, y0:J.neck.y-overlapPx, x1:torsoX1, y1:J.hip.y+overlapPx },
    armL_upper: { x0:J.elbowL.x-overlapPx, y0:J.shoulderL.y-armPad, x1:cx+overlapPx, y1:J.shoulderL.y+armPad },
    armL_lower: { x0:bounds.minX, y0:J.elbowL.y-armPad, x1:J.elbowL.x+overlapPx, y1:J.elbowL.y+armPad },
    armR_upper: { x0:cx-overlapPx, y0:J.shoulderR.y-armPad, x1:J.elbowR.x+overlapPx, y1:J.shoulderR.y+armPad },
    armR_lower: { x0:J.elbowR.x-overlapPx, y0:J.elbowR.y-armPad, x1:bounds.maxX, y1:J.elbowR.y+armPad },
  };
  const legPadL = Math.abs(J.kneeL.x-cx)+overlapPx*3;
  const legPadR = Math.abs(J.kneeR.x-cx)+overlapPx*3;
  R.legL_upper = { x0:Math.min(J.ankleL.x,cx)-legPadL, y0:J.hip.y-overlapPx, x1:cx+overlapPx, y1:J.kneeL.y+overlapPx };
  R.legL_lower = { x0:Math.min(J.ankleL.x,cx)-legPadL, y0:J.kneeL.y-overlapPx, x1:cx+overlapPx, y1:bounds.maxY };
  R.legR_upper = { x0:cx-overlapPx, y0:J.hip.y-overlapPx, x1:Math.max(J.ankleR.x,cx)+legPadR, y1:J.kneeR.y+overlapPx };
  R.legR_lower = { x0:cx-overlapPx, y0:J.kneeR.y-overlapPx, x1:Math.max(J.ankleR.x,cx)+legPadR, y1:bounds.maxY };

  const pivotAbs = {
    head: J.neck, torso: J.hip,
    armL_upper: { x:cx, y:J.shoulderL.y }, armL_lower: J.elbowL,
    armR_upper: { x:cx, y:J.shoulderR.y }, armR_lower: J.elbowR,
    legL_upper: J.hip, legL_lower: J.kneeL,
    legR_upper: J.hip, legR_lower: J.kneeR,
  };
  Object.keys(R).forEach(k=>{
    const r = R[k];
    r.x0 = Math.max(bounds.minX, Math.min(r.x0, bounds.maxX));
    r.x1 = Math.max(bounds.minX, Math.min(r.x1, bounds.maxX));
    r.y0 = Math.max(bounds.minY, Math.min(r.y0, bounds.maxY));
    r.y1 = Math.max(bounds.minY, Math.min(r.y1, bounds.maxY));
    if(r.x1<=r.x0) r.x1 = r.x0+1;
    if(r.y1<=r.y0) r.y1 = r.y0+1;
  });
  return { regions:R, pivotAbs };
}
// Tightens a requested pixel rectangle down to the real foreground bounds found inside it —
// this is what keeps a crop following the actual artwork instead of being a crude fixed box
// (falls back to the requested rectangle if nothing foreground is found there, and always
// expands to contain `mustContain` — this part's own pivot — so a pivot never lands outside
// its own crop).
function tightenRegion(data, w, h, rect, mustContain){
  const x0 = Math.max(0, Math.floor(rect.x0)), y0 = Math.max(0, Math.floor(rect.y0));
  const x1 = Math.min(w-1, Math.ceil(rect.x1)), y1 = Math.min(h-1, Math.ceil(rect.y1));
  let minX=x1, minY=y1, maxX=x0, maxY=y0, found=false;
  for(let y=y0;y<=y1;y++){
    for(let x=x0;x<=x1;x++){
      if(data[(y*w+x)*4+3] > AUTORIG_ALPHA_THRESH){
        found=true;
        if(x<minX)minX=x; if(x>maxX)maxX=x;
        if(y<minY)minY=y; if(y>maxY)maxY=y;
      }
    }
  }
  if(!found){ minX=x0; minY=y0; maxX=x1; maxY=y1; }
  if(mustContain){
    minX=Math.min(minX, Math.floor(mustContain.x)); maxX=Math.max(maxX, Math.ceil(mustContain.x));
    minY=Math.min(minY, Math.floor(mustContain.y)); maxY=Math.max(maxY, Math.ceil(mustContain.y));
  }
  if(maxX<=minX) maxX=minX+1;
  if(maxY<=minY) maxY=minY+1;
  return { x:minX, y:minY, w:maxX-minX+1, h:maxY-minY+1 };
}
// Tight alpha bounds of a whole (small, already-cropped) canvas — same idea as this file's
// existing detectOpaqueBounds, just operating on a canvas directly instead of an <img> (no
// decode round-trip needed since generateAutoRigParts already has the pixels in hand).
function canvasTightBounds(canvas){
  const w = canvas.width, h = canvas.height;
  let data;
  try{ data = canvas.getContext('2d').getImageData(0,0,w,h).data; }
  catch(e){ return { x:0, y:0, w, h }; }
  let minX=w, minY=h, maxX=-1, maxY=-1;
  for(let y=0;y<h;y++) for(let x=0;x<w;x++){
    if(data[(y*w+x)*4+3] > AUTORIG_ALPHA_THRESH){ if(x<minX)minX=x; if(x>maxX)maxX=x; if(y<minY)minY=y; if(y>maxY)maxY=y; }
  }
  if(maxX<minX||maxY<minY) return { x:0, y:0, w, h };
  return { x:minX, y:minY, w:maxX-minX+1, h:maxY-minY+1 };
}

// ---- STEP: create rigged part images (with joint overlap baked in) ----
// drawCustomCharacterOnContext (above) scales/rotates every part by measuring ONLY its content
// box's local-Y extent from the pivot, then rotating the whole image so local +Y points along
// the bone's CURRENT direction — i.e. it expects every part image to be drawn as if its bone
// pointed straight down (angle = +90deg), with rotation to the bone's real angle applied at
// render time on top of that. A crop taken directly from the source image is NOT drawn that
// way in general (e.g. a horizontal arm's long axis runs sideways in the source, not down) —
// so each crop is rotated here, once, by (90deg - thisBone'sRestAngle) around its own pivot,
// before being handed off. This is the piece that makes reusing the existing manual-character
// renderer/exporter unchanged actually work for art in an arbitrary rest pose.
function makeRotatedPartCanvas(srcCanvas, cropRect, pivotAbs, phi){
  const pivotLocalX = pivotAbs.x-cropRect.x, pivotLocalY = pivotAbs.y-cropRect.y;
  const corners = [
    {x:0,y:0},{x:cropRect.w,y:0},{x:0,y:cropRect.h},{x:cropRect.w,y:cropRect.h}
  ].map(c=>({ x:c.x-pivotLocalX, y:c.y-pivotLocalY }));
  const cos = Math.cos(phi), sin = Math.sin(phi);
  const rotated = corners.map(c=>({ x:c.x*cos-c.y*sin, y:c.x*sin+c.y*cos }));
  const minX = Math.min(...rotated.map(p=>p.x)), maxX = Math.max(...rotated.map(p=>p.x));
  const minY = Math.min(...rotated.map(p=>p.y)), maxY = Math.max(...rotated.map(p=>p.y));
  const outW = Math.max(1, Math.ceil(maxX-minX)), outH = Math.max(1, Math.ceil(maxY-minY));
  const out = document.createElement('canvas');
  out.width = outW; out.height = outH;
  const octx = out.getContext('2d');
  octx.save();
  octx.translate(-minX, -minY);
  octx.rotate(phi);
  octx.drawImage(srcCanvas, cropRect.x, cropRect.y, cropRect.w, cropRect.h, -pivotLocalX, -pivotLocalY, cropRect.w, cropRect.h);
  octx.restore();
  return { canvas: out, pivotX: (-minX)/outW, pivotY: (-minY)/outH };
}
// Orchestrates STEPS 9-10 (generate body-part masks + joint overlap areas) into the 10
// PART_SLOTS-shaped parts finalizeAutoRiggedCharacter() saves onto the character. Reads the
// FULL working-resolution srcCanvas's pixels once and reuses that single ImageData for every
// part's tightenRegion() call rather than re-decoding per part.
// ==== FIX: per-pixel bone ownership (stops one part from swallowing pixels of its neighbours) ====
// PROBLEM this fixes: the old code cut every body part out of the source image with a plain
// RECTANGLE (computePartRegions) and kept EVERYTHING inside it. A T-pose arm rectangle is tall
// (armPad = 14% of body height above AND below the arm) and reaches to the body centre-line, so it
// also captured half the torso, the neck and the chin. Those foreign pixels then rotated together
// with the arm -> "hand cut includes chest/neck", and each foreign piece shows up again as a
// duplicated body part whenever the arm moves.
//
// HOW it is fixed: every opaque pixel is assigned to exactly ONE part - the bone it is closest to,
// measured relative to that bone's own thickness (radius). The rectangle is still used as a coarse
// bounding box, but pixels that belong to another part are erased from the crop. A small dilation
// (`joint overlap`) is added back afterwards ONLY inside the original artwork so joints still hide
// their seam when parts rotate (the original "stretchy joint" effect is preserved).
function autorigDistToSegment(px, py, ax, ay, bx, by){
  const dx = bx-ax, dy = by-ay;
  const L2 = dx*dx + dy*dy;
  if(L2 < 1e-6) return Math.hypot(px-ax, py-ay);
  let t = ((px-ax)*dx + (py-ay)*dy) / L2;
  t = t < 0 ? 0 : (t > 1 ? 1 : t);
  return Math.hypot(px-(ax+t*dx), py-(ay+t*dy));
}
// Returns { owner:Int8Array(w*h) (-1 = transparent, otherwise index into keys), keys }.
//
// HOW a pixel's owner is decided:  NEAREST BONE, measured to the bone SEGMENT and divided by that
// bone's own thickness (radius) so thick parts (torso, head) and thin parts (arms) compete fairly.
// This deliberately does NOT depend on hard-coded body zones (e.g. "below hip = leg"), because the
// auto-detector's hip/shoulder joints are only approximate on real cartoon art (both shoulders can
// sit on the body centre-line, the hip can sit above the real waistline). Nearest-bone degrades
// gracefully when a joint is a little off; zone rules do not.
//
// Torso/legs are split AT the hip joint (that is where the rig itself joins them) and arms start
// at the shoulder, so a sleeve at the shoulder cap goes to the arm.
function autorigBuildOwnershipMap(data, w, h, J, charHeightPx, torsoHalfWidthPx){
  const armR  = Math.max(8,  charHeightPx*0.055);
  const legR  = Math.max(8,  charHeightPx*0.06);
  const headR = Math.max(10, Math.abs(J.neck.y-J.headTop.y)*0.5);
  const torsoR= Math.max(10, torsoHalfWidthPx*0.95);
  const cx = (J.hip.x + J.neck.x)/2;
  // The rig (buildFreshRig/applyRigOverride) joins torso and legs AT THE HIP JOINT, and the part
  // renderer draws the torso only as far as its bone length (neck -> hip). So the cut must split
  // the body at the hip too, otherwise pixels below the hip that the torso "owns" are drawn past
  // the end of the spine bone and leave a visible gap where the legs start.
  const torsoTop = { x:cx, y:J.neck.y + (J.hip.y-J.neck.y)*0.10 };
  const bones = {
    head:       { a:J.neck,       b:J.headTop,   r:headR },
    // torso bone ends a LITTLE below the hip joint (8% of neck->hip). The auto-detected hip is
    // usually a bit above the real shirt hem, so a hard split exactly at the hip hands the
    // lowest strip of shirt to the legs. Kept small so the part image stays close to the
    // spine bone length the renderer scales it by (a large extension leaves a gap - tested).
    torso:      { a:torsoTop,     b:{x:cx, y:J.hip.y + (J.hip.y-J.neck.y)*0.08}, r:torsoR },
    armL_upper: { a:J.shoulderL,  b:J.elbowL,    r:armR },
    armL_lower: { a:J.elbowL,     b:J.wristL||J.elbowL, r:armR },
    armR_upper: { a:J.shoulderR,  b:J.elbowR,    r:armR },
    armR_lower: { a:J.elbowR,     b:J.wristR||J.elbowR, r:armR },
    legL_upper: { a:J.hip,        b:J.kneeL,     r:legR },
    legL_lower: { a:J.kneeL,      b:J.ankleL,    r:legR },
    legR_upper: { a:J.hip,        b:J.kneeR,     r:legR },
    legR_lower: { a:J.kneeR,      b:J.ankleR,    r:legR },
  };
  const keys = Object.keys(bones);
  const idx = {}; keys.forEach((k,i)=>idx[k]=i);
  const owner = new Int8Array(w*h).fill(-1);
  const shoulderY = Math.min(J.shoulderL.y, J.shoulderR.y);
  for(let y=0;y<h;y++){
    for(let x=0;x<w;x++){
      if(data[(y*w+x)*4+3] <= AUTORIG_ALPHA_THRESH) continue;
      let o = -1, bestD = Infinity;
      for(let k=0;k<keys.length;k++){
        const bn = bones[keys[k]];
        const d = autorigDistToSegment(x, y, bn.a.x, bn.a.y, bn.b.x, bn.b.y) / bn.r;
        if(d < bestD){ bestD = d; o = k; }
      }
      // A pixel clearly BELOW the shoulder line can never belong to the head: the head bone's
      // large radius otherwise grabs the collar/shoulders of a big-headed cartoon character.
      if(o === idx.head && y > shoulderY + armR*0.6) o = idx.torso;
      owner[y*w+x] = o;
    }
  }
  return { owner, keys };
}
// Builds the final alpha mask for ONE part inside its (tightened) crop rectangle:
//  1) start from the pixels this part OWNS,
//  2) grow that by `overlapPx` so neighbouring parts overlap a little at joints (hides the seam),
//  3) never grow into transparent pixels, and never grab pixels owned by a DIFFERENT limb of the
//     same kind (e.g. left arm must not pick up right arm) - only its direct chain neighbours.
const AUTORIG_JOINT_NEIGHBOURS = {
  head:['torso'], torso:['head','armL_upper','armR_upper','legL_upper','legR_upper'],
  armL_upper:['torso','armL_lower'], armL_lower:['armL_upper'],
  armR_upper:['torso','armR_lower'], armR_lower:['armR_upper'],
  legL_upper:['torso','legL_lower'], legL_lower:['legL_upper'],
  legR_upper:['torso','legR_lower'], legR_lower:['legR_upper'],
};
function autorigApplyPartMask(srcData, w, h, crop, own, keys, partKey, overlapPx){
  const partIdx = keys.indexOf(partKey);
  const allowed = new Set([partIdx].concat((AUTORIG_JOINT_NEIGHBOURS[partKey]||[]).map(n=>keys.indexOf(n))));
  const cw = crop.w, ch = crop.h;
  // pass 1: exactly-owned pixels inside the crop
  let mask = new Uint8Array(cw*ch);
  for(let y=0;y<ch;y++){
    const sy = crop.y+y; if(sy<0||sy>=h) continue;
    for(let x=0;x<cw;x++){
      const sx = crop.x+x; if(sx<0||sx>=w) continue;
      if(own[sy*w+sx] === partIdx) mask[y*cw+x] = 1;
    }
  }
  // pass 2: grow `overlapPx` steps, but only into opaque pixels owned by this part or its
  // direct joint neighbours (so the seam is covered, yet a hand never re-grabs the neck/chest
  // from far away).
  const steps = Math.max(1, Math.round(overlapPx));
  for(let s=0;s<steps;s++){
    const next = mask.slice();
    for(let y=0;y<ch;y++){
      for(let x=0;x<cw;x++){
        if(mask[y*cw+x]) continue;
        const sx = crop.x+x, sy = crop.y+y;
        if(sx<0||sx>=w||sy<0||sy>=h) continue;
        const o = own[sy*w+sx];
        if(o<0 || !allowed.has(o)) continue;
        if( (x>0 && mask[y*cw+x-1]) || (x<cw-1 && mask[y*cw+x+1]) ||
            (y>0 && mask[(y-1)*cw+x]) || (y<ch-1 && mask[(y+1)*cw+x]) ) next[y*cw+x] = 1;
      }
    }
    mask = next;
  }
  return mask;
}
// Orchestrates STEPS 9-10 (generate body-part masks + joint overlap areas) into the 10
// PART_SLOTS-shaped parts finalizeAutoRiggedCharacter() saves onto the character. Reads the
// FULL working-resolution srcCanvas's pixels once and reuses that single ImageData for every
// part's tightenRegion() call rather than re-decoding per part.
function generateAutoRigParts(srcCanvas, joints, override){
  const w = srcCanvas.width, h = srcCanvas.height;
  const cctx = srcCanvas.getContext('2d', { willReadFrequently:true });
  const data = cctx.getImageData(0,0,w,h).data;
  const bounds = autorigGlobalBounds(data,w,h) || { minX:0, minY:0, maxX:w-1, maxY:h-1 };
  const J = {}; for(const k in joints) J[k] = { x:joints[k].x*w, y:joints[k].y*h };
  const charHeightPx = Math.max(1, Math.max(J.ankleL.y, J.ankleR.y) - J.headTop.y);
  const overlapPx = Math.max(4, Math.min(40, charHeightPx*0.035));
  const torsoHalfWidthPx = autorigTorsoHalfWidthAt(data, w, h, J);
  const { regions, pivotAbs } = computePartRegions(joints, w, h, bounds, overlapPx, charHeightPx, torsoHalfWidthPx);

  // FIX: decide, once, which part every opaque pixel belongs to.
  const { owner, keys } = autorigBuildOwnershipMap(data, w, h, J, charHeightPx, torsoHalfWidthPx);
  // joint overlap is intentionally SMALLER than before (was a whole extra rectangle margin) -
  // now it is a thin, opaque-only rim around the part.
  const rimPx = Math.max(3, Math.min(14, charHeightPx*0.014));

  const parts = {};
  Object.keys(regions).forEach(key=>{
    // The rectangle is only a coarse search window now; the true shape comes from the mask.
    const rough = tightenRegion(data, w, h, regions[key], pivotAbs[key]);
    const mask = autorigApplyPartMask(data, w, h, rough, owner, keys, key, rimPx);

    // Build the part image: copy ONLY masked pixels out of the source (everything else transparent).
    const partCanvas = document.createElement('canvas');
    partCanvas.width = rough.w; partCanvas.height = rough.h;
    const pctx = partCanvas.getContext('2d', { willReadFrequently:true });
    const out = pctx.createImageData(rough.w, rough.h);
    for(let y=0;y<rough.h;y++){
      for(let x=0;x<rough.w;x++){
        if(!mask[y*rough.w+x]) continue;
        const sx = rough.x+x, sy = rough.y+y;
        if(sx<0||sx>=w||sy<0||sy>=h) continue;
        const si = (sy*w+sx)*4, di = (y*rough.w+x)*4;
        out.data[di]=data[si]; out.data[di+1]=data[si+1]; out.data[di+2]=data[si+2]; out.data[di+3]=data[si+3];
      }
    }
    pctx.putImageData(out,0,0);

    // Re-tighten around what is REALLY left after masking, so the pivot/contentBox stay correct.
    const od = out.data;
    let mnX=rough.w, mnY=rough.h, mxX=-1, mxY=-1;
    for(let y=0;y<rough.h;y++) for(let x=0;x<rough.w;x++){
      if(od[(y*rough.w+x)*4+3] > AUTORIG_ALPHA_THRESH){ if(x<mnX)mnX=x; if(x>mxX)mxX=x; if(y<mnY)mnY=y; if(y>mxY)mxY=y; }
    }
    const pvx = pivotAbs[key].x-rough.x, pvy = pivotAbs[key].y-rough.y;   // pivot must stay inside its own crop
    if(mxX<mnX){ mnX=0; mnY=0; mxX=rough.w-1; mxY=rough.h-1; }            // nothing left -> keep whole crop (safety)
    mnX=Math.min(mnX,Math.floor(pvx)); mxX=Math.max(mxX,Math.ceil(pvx));
    mnY=Math.min(mnY,Math.floor(pvy)); mxY=Math.max(mxY,Math.ceil(pvy));
    mnX=Math.max(0,mnX); mnY=Math.max(0,mnY); mxX=Math.min(rough.w-1,mxX); mxY=Math.min(rough.h-1,mxY);
    const tight = { x:rough.x+mnX, y:rough.y+mnY, w:mxX-mnX+1, h:mxY-mnY+1 };

    // makeRotatedPartCanvas now receives the MASKED canvas (coordinates shifted by `rough`) instead of the raw source.
    const shiftedCrop = { x:mnX, y:mnY, w:tight.w, h:tight.h };
    const shiftedPivot = { x:pvx, y:pvy };
    const theta = override.angles[AUTORIG_BONE_FOR_PART[key]];
    const phi = Math.PI/2 - theta;
    const { canvas, pivotX, pivotY } = makeRotatedPartCanvas(partCanvas, shiftedCrop, shiftedPivot, phi);
    const box = canvasTightBounds(canvas);
    parts[key] = { src: canvas.toDataURL('image/png'), w: canvas.width, h: canvas.height, pivotX, pivotY, contentBox: box };
  });
  return parts;
}

// ---- STEPS 1-2: load + normalize image ----
// Accepts either an <img> or a <canvas> as `source` (Re-detect re-normalizes from the already-
// normalized working canvas rather than re-decoding the original upload).
function autorigNormalizeToCanvas(source, maxDim){
  const w = source.naturalWidth || source.width, h = source.naturalHeight || source.height;
  const scale = Math.min(1, maxDim/Math.max(w,h));
  const outW = Math.max(1, Math.round(w*scale)), outH = Math.max(1, Math.round(h*scale));
  const cv = document.createElement('canvas');
  cv.width = outW; cv.height = outH;
  cv.getContext('2d', { willReadFrequently:true }).drawImage(source, 0, 0, outW, outH);
  return cv;
}
// Runs the full detection pipeline (STEPS 1-10) against a freshly loaded/normalized image and
// returns either { error } or the data openAutoRigPreview needs. Detection itself runs against
// a small downscaled analysis canvas for speed — its output joints are plain fractions of
// width/height, so they apply unchanged to the larger stored srcCanvas.
function runAutoRigDetection(img){
  const srcCanvas = autorigNormalizeToCanvas(img, AUTORIG_MAX_SRC_DIM);
  const analysisCanvas = autorigNormalizeToCanvas(img, AUTORIG_ANALYSIS_MAX_DIM);
  let imageData;
  try{ imageData = analysisCanvas.getContext('2d', { willReadFrequently:true }).getImageData(0,0,analysisCanvas.width, analysisCanvas.height); }
  catch(e){ return { error: "Couldn't analyze that image." }; }
  const det = detectCharacterSilhouette(imageData);
  if(!det) return { error: 'This image looks empty or fully transparent — try a different one.' };
  return { srcCanvas, srcW:srcCanvas.width, srcH:srcCanvas.height, joints:det.joints, attachAt:det.attachAt, warnings:det.warnings, torsoHalfWidthFrac:det.torsoHalfWidthFrac, neckRatio:det.neckRatio };
}

// UI-facing joint list: shoulderL/shoulderR (which the rig always keeps pinned to the spine's
// own centerline — see applyRigOverride's attachAt) are combined into a single "shoulder"
// handle so dragging it doesn't imply a horizontal offset the rig has no field for; every
// other joint is independently 2D-draggable. See autoRigJointFrac/autoRigSetJointFrac.
// `neck` is deliberately NOT in this list (11 points, not 12): the pixel-width scan that finds
// it can land inside the face on some art (see autorigFindNeckY), which put a stray, confusing
// point directly on the character's face. It's still a real joint internally (bone lines still
// connect through it, and it still defines the head/torso crop boundary) — it's just no longer
// its own draggable handle. recomputeAutoRigNeck() keeps it sensibly positioned, proportionally
// between headTop and the shoulder line (using the ratio detection first found — see neckRatio
// in detectCharacterSilhouette), whenever headTop/shoulder/hip are dragged instead.
const AUTORIG_JOINT_KEYS = ['headTop','shoulder','elbowL','wristL','elbowR','wristR','hip','kneeL','ankleL','kneeR','ankleR'];
const AUTORIG_BONE_PAIRS = [
  ['neck','headTop'], ['neck','hip'],
  ['shoulder','elbowL'], ['elbowL','wristL'],
  ['shoulder','elbowR'], ['elbowR','wristR'],
  ['hip','kneeL'], ['kneeL','ankleL'],
  ['hip','kneeR'], ['kneeR','ankleR'],
];
function autoRigJointFrac(key){
  const j = autoRigState.joints;
  if(key==='shoulder') return { x:j.hip.x, y:j.shoulderL.y };
  return j[key];
}
// Keeps the (no longer draggable) neck proportionally placed between headTop and the shoulder
// line, and on the spine's centerline — called after headTop/shoulder/hip move. Falls back to
// the midpoint if this session has no neckRatio yet (shouldn't normally happen).
function recomputeAutoRigNeck(){
  const j = autoRigState.joints;
  const ratio = (typeof autoRigState.neckRatio === 'number') ? autoRigState.neckRatio : 0.5;
  j.neck.x = j.hip.x;
  j.neck.y = j.headTop.y + (j.shoulderL.y - j.headTop.y) * ratio;
}
function autoRigSetJointFrac(key, fx, fy){
  const j = autoRigState.joints;
  if(key==='shoulder'){
    j.shoulderL.y=fy; j.shoulderR.y=fy; j.shoulderL.x=j.hip.x; j.shoulderR.x=j.hip.x;
    recomputeAutoRigNeck();
    return;
  }
  j[key].x = fx; j[key].y = fy;
  if(key==='headTop' || key==='hip') recomputeAutoRigNeck();
}
function setupAutoRigJointDrag(dot, key){
  const img = document.getElementById('autoRigStageImg');
  let dragging=false,frame=0,pending=null,rect=null,pointerId=null;
  function applyPending(){
    frame=0;
    const p=pending;pending=null;
    if(!dragging||!p||!rect||!rect.width||!rect.height)return;
    const fx=Math.max(0,Math.min(1,(p.x-rect.left)/rect.width));
    const fy=Math.max(0,Math.min(1,(p.y-rect.top)/rect.height));
    autoRigSetJointFrac(key,fx,fy);
    renderAutoRigOverlay();
  }
  function queue(e){
    pending={x:e.clientX,y:e.clientY};
    if(!frame)frame=requestAnimationFrame(applyPending);
  }
  function finish(){
    if(frame){cancelAnimationFrame(frame);frame=0;}
    if(pending)applyPending();
    dragging=false;pending=null;rect=null;pointerId=null;dot.classList.remove('dragging');
  }
  dot.onpointerdown=(e)=>{
    if(e.pointerType==='mouse'&&e.button!==0)return;
    dragging=true;pointerId=e.pointerId;rect=img.getBoundingClientRect();
    dot.classList.add('dragging');
    try{dot.setPointerCapture(e.pointerId);}catch(_e){}
    e.preventDefault();e.stopPropagation();
  };
  dot.onpointermove=(e)=>{if(dragging&&e.pointerId===pointerId){queue(e);e.preventDefault();}};
  dot.onpointerup=(e)=>{if(!dragging||e.pointerId!==pointerId)return;queue(e);finish();};
  dot.onpointercancel=finish;
  dot.onlostpointercapture=finish;
}
// Redraws the bone-line/region overlay from the CURRENT joints (called after every drag, reset,
// and re-detect). Joint dots are created once and repositioned in place on every later call
// (never destroyed/recreated) so an in-progress drag's pointer capture never gets invalidated
// mid-gesture. Region rectangles are cheap pure arithmetic (computePartRegions) — no pixel
// scanning happens until "Create Rigged Character" actually crops the final artwork.
function renderAutoRigOverlay(){
  if(!autoRigState||!autoRigState.joints)return;
  const boneLayer=document.getElementById('autoRigBoneLayer');
  const jointLayer=document.getElementById('autoRigJointLayer');
  const regionLayer=document.getElementById('autoRigRegionLayer');
  const img=document.getElementById('autoRigStageImg');
  const wrap=document.getElementById('autoRigStageWrap');
  if(!boneLayer||!jointLayer||!regionLayer||!img||!wrap)return;
  // Read the two layout boxes once. The previous painter re-read both boxes for every
  // joint, bone endpoint and region on every drag event, forcing repeated synchronous layout.
  const ir=img.getBoundingClientRect(),wr=wrap.getBoundingClientRect();
  window.__ethvelleAutoRigFrameGeometry={left:ir.left-wr.left,top:ir.top-wr.top,width:ir.width,height:ir.height};
  const px=(fx,fy)=>({x:ir.left-wr.left+ir.width*fx,y:ir.top-wr.top+ir.height*fy});
  const w=autoRigState.srcW,h=autoRigState.srcH;
  if(!(w>0&&h>0&&ir.width>0&&ir.height>0))return;

  if(!boneLayer.__evv109Nodes){boneLayer.innerHTML='';boneLayer.__evv109Nodes=[];}
  if(!regionLayer.__evv109Nodes){regionLayer.innerHTML='';regionLayer.__evv109Nodes=new Map();}
  const boneNodes=boneLayer.__evv109Nodes;
  const regionNodes=regionLayer.__evv109Nodes;

  try{
    const J={};for(const k in autoRigState.joints)J[k]={x:autoRigState.joints[k].x*w,y:autoRigState.joints[k].y*h};
    const charHeightPx=Math.max(1,Math.max(J.ankleL?.y||0,J.ankleR?.y||0)-(J.headTop?.y||0));
    const overlapPx=Math.max(4,Math.min(40,charHeightPx*.035));
    const bounds={minX:0,minY:0,maxX:w-1,maxY:h-1};
    const torsoHalfWidthPx=(autoRigState.torsoHalfWidthFrac||.12)*w;
    const result=computePartRegions(autoRigState.joints,w,h,bounds,overlapPx,charHeightPx,torsoHalfWidthPx);
    const regions=result&&result.regions?result.regions:{};
    const activeRegionKeys=new Set();
    Object.entries(regions).forEach(([key,r])=>{
      activeRegionKeys.add(key);
      let el=regionNodes.get(key);
      if(!el){el=document.createElement('div');el.className='autorig-region-rect-el';el.dataset.regionKey=key;regionLayer.appendChild(el);regionNodes.set(key,el);}
      const p0=px(r.x0/w,r.y0/h),p1=px(r.x1/w,r.y1/h);
      el.style.display='block';el.style.left=p0.x+'px';el.style.top=p0.y+'px';
      el.style.width=Math.max(0,p1.x-p0.x)+'px';el.style.height=Math.max(0,p1.y-p0.y)+'px';
    });
    regionNodes.forEach((el,key)=>{if(!activeRegionKeys.has(key))el.style.display='none';});
  }catch(err){/* Region preview is cosmetic; keep joint dragging alive if it fails. */}

  AUTORIG_BONE_PAIRS.forEach(([a,b],index)=>{
    let el=boneNodes[index];
    if(!el){el=document.createElement('div');el.className='autorig-bone-line-el';boneLayer.appendChild(el);boneNodes[index]=el;}
    const pa=autoRigJointFrac(a),pb=autoRigJointFrac(b),p0=px(pa.x,pa.y),p1=px(pb.x,pb.y);
    const dx=p1.x-p0.x,dy=p1.y-p0.y;
    el.style.display='block';el.style.left=p0.x+'px';el.style.top=p0.y+'px';
    el.style.width=Math.hypot(dx,dy)+'px';el.style.transform='translateY(-50%) rotate('+Math.atan2(dy,dx)+'rad)';
  });
  for(let i=AUTORIG_BONE_PAIRS.length;i<boneNodes.length;i++)boneNodes[i].style.display='none';

  AUTORIG_JOINT_KEYS.forEach(key=>{
    const f=autoRigJointFrac(key),p=px(f.x,f.y);
    let dot=jointLayer.querySelector('[data-joint="'+key+'"]');
    if(!dot){dot=document.createElement('div');dot.className='autorig-joint-dot';dot.dataset.joint=key;setupAutoRigJointDrag(dot,key);jointLayer.appendChild(dot);}
    dot.style.left=p.x+'px';dot.style.top=p.y+'px';
  });
}
function resetAutoRig(){
  if(!autoRigState || !autoRigState.originalJoints) return;
  autoRigState.joints = JSON.parse(JSON.stringify(autoRigState.originalJoints));
  renderAutoRigOverlay();
  if(navigator.vibrate) navigator.vibrate(8);
}
// Re-runs detection from scratch against the SAME uploaded image (never a re-compressed part
// crop), overwriting any manual joint adjustments made so far. No confirmation step (matches
// this screen's other one-tap action, Reset Auto Rig) — the original upload is never touched,
// so nothing here is destructive to anything outside this preview session.
function rebuildAutoRigDetection(){
  if(!autoRigState || !autoRigState.srcCanvas) return;
  const analysisCanvas = autorigNormalizeToCanvas(autoRigState.srcCanvas, AUTORIG_ANALYSIS_MAX_DIM);
  let imageData;
  try{ imageData = analysisCanvas.getContext('2d', { willReadFrequently:true }).getImageData(0,0,analysisCanvas.width, analysisCanvas.height); }
  catch(e){ showToast("Couldn't re-analyze this image", { icon: toastIcons.info, duration:1800 }); return; }
  const det = detectCharacterSilhouette(imageData);
  if(!det){ showToast('This image looks empty or fully transparent', { icon: toastIcons.info, duration:1800 }); return; }
  autoRigState.joints = det.joints;
  autoRigState.originalJoints = JSON.parse(JSON.stringify(det.joints));
  autoRigState.attachAt = det.attachAt;
  autoRigState.torsoHalfWidthFrac = det.torsoHalfWidthFrac;
  autoRigState.neckRatio = det.neckRatio;
  autoRigState.warnings = det.warnings;
  renderAutoRigWarnings();
  renderAutoRigOverlay();
  showToast('Rig re-detected', { duration:1400 });
  if(navigator.vibrate) navigator.vibrate(12);
}
// Converts the current joints into a real character: generates the 10 rigged part images
// (STEPS 9-10, see generateAutoRigParts) and the bone-length/angle rig override (STEP 4-5, see
// buildRigOverrideFromJoints) from the CURRENT joint positions — every preview line/region the
// user has been looking at corresponds to this exact same data, not a placeholder. Either
// updates the character being edited in place (mode:'edit' — existing clip/segments untouched,
// per this feature's "keep existing animation clips while editing the rig" requirement) or adds
// a brand new one (mode:'new'), matching finishCustomCharacterBuild's own two branches above.
function finalizeAutoRiggedCharacter(){
  if(!autoRigState || !autoRigState.srcCanvas) return;
  const override = buildRigOverrideFromJoints(autoRigState.joints, autoRigState.srcW, autoRigState.srcH, autoRigState.attachAt);
  let newParts;
  try{
    newParts = generateAutoRigParts(autoRigState.srcCanvas, autoRigState.joints, override);
  }catch(err){
    trackEvent('autorig_failed');
    showToast("Couldn't finish rigging this character — try adjusting the points and try again.", { icon: toastIcons.info, duration:2400 });
    return;
  }
  // NEW: build the skeletal skin mesh (see buildSkinMesh) from the SAME full source canvas +
  // joints used above — this is what drawCustomCharacterOnContext actually renders from now
  // (see its skinMesh branch), so the character's art is never cut into parts. newParts above
  // is still generated and stored as a fallback ONLY (used if mesh-building fails here, or if
  // this project is later opened by a version of the app that predates skin meshes) — it is
  // not what normally draws anymore for a character that has a skinMesh.
  let skinMesh = null;
  try{
    skinMesh = buildSkinMesh(autoRigState.srcCanvas, autoRigState.joints, autoRigState.srcW, autoRigState.srcH);
  }catch(err){
    skinMesh = null; // falls back to the old cut-part rendering below — better a working rigid character than none
  }
  const autoRigData = {
    sourceImage: autoRigState.srcCanvas.toDataURL('image/png'),
    srcW: autoRigState.srcW, srcH: autoRigState.srcH,
    joints: JSON.parse(JSON.stringify(autoRigState.joints)),
    attachAt: { ...autoRigState.attachAt },
    torsoHalfWidthFrac: autoRigState.torsoHalfWidthFrac,
    neckRatio: autoRigState.neckRatio,
    face: autoRigState.face || ethvelleEstimateFace(autoRigState.srcCanvas, autoRigState.joints),
    handRig: !!(autoRigState.joints.palmL && autoRigState.joints.palmR && ['L','R'].some(side=>['thumb','index','middle','ring','pinky'].some(f=>autoRigState.joints[f+'Base'+side]&&autoRigState.joints[f+'Mid'+side]&&autoRigState.joints[f+'Tip'+side]))),
    aiProvider: autoRigState.aiProvider || 'local',
  };
  if(autoRigState.mode==='newAsset'){
    // Built entirely from a plain object — never touches `characters` or the canvas, since
    // neither exists when this is opened from the Models grid with no project open. Reuses
    // makeCharacter purely as a convenient shape-builder for name/parts/rigOverride, exactly
    // like saveCharacterAsModelAsset does when converting a real character.
    const tempChar = makeCharacter(undefined, override);
    tempChar.type = 'custom'; tempChar.parts = newParts; tempChar.autoRig = autoRigData;
    const asset = saveCharacterAsModelAsset(tempChar, { silent:true });
    closeAutoRigPreview();
    if(navigator.vibrate) navigator.vibrate(12);
    trackEvent('autorig_completed');
    trackEvent('model_added');
    showToast('Model added', { duration:1400 });
    if(document.getElementById('screen-projects').classList.contains('active')) renderModelsTab();
    return;
  }
  if(autoRigState.mode==='edit' && characters[autoRigState.editingCharIdx]){
    const c = characters[autoRigState.editingCharIdx];
    c.type = 'custom'; c.parts = newParts; c.rigOverride = override; c.autoRig = autoRigData; c.skinMesh = skinMesh;
    const rig = buildFreshRig();
    applyRigOverride(rig, override);
    c.bones = rig.bones; c.baseAngles = rig.baseAngles; c.hipRestY = rig.bones.hip.y;
    if(autoRigState.editingCharIdx === activeCharIdx) syncActiveCharacterRefs();
    selectedJoint = null;
    closeAutoRigPreview();
    if(navigator.vibrate) navigator.vibrate(12);
    trackEvent('autorig_completed');
    showToast('Character updated', { duration:1400 });
    preloadCustomCharacterImages(ctx, [c]).then(()=>{
      drawStick(); renderCharacterLayers(); syncTransformSliders(); persistProjectState();
    });
    return;
  }
  const c = makeCharacter(undefined, override);
  c.type = 'custom'; c.parts = newParts; c.autoRig = autoRigData; c.skinMesh = skinMesh;
  // UPDATE (default placeholder clip on add): makeCharacter() now gives every new character
  // a 'default' (static, no-motion) clip automatically — see makeCharacter's own comment —
  // so a freshly-rigged model's row shows a real clip on the timeline right away instead of
  // sitting empty, exactly like plain addCharacter() and every other creation path. Not a
  // walk cycle or any other real motion — just the same static placeholder every new
  // character gets until the user opens Animate and picks something themselves.
  characters.push(c);
  activeCharIdx = characters.length-1;
  syncActiveCharacterRefs();
  selectedJoint = null;
  closeAutoRigPreview();
  if(navigator.vibrate) navigator.vibrate(12);
  trackEvent('autorig_completed');
  trackEvent('model_added');
  showToast('Rigged character added', { duration:1400 });
  // FEATURE (reusable custom models): every freshly-rigged character is also saved as a
  // standalone model asset (see saveCharacterAsModelAsset) so it can be pulled into any OTHER
  // project later from the model library, without re-uploading/re-rigging the same art again.
  saveCharacterAsModelAsset(c, { silent:true });
  preloadCustomCharacterImages(ctx, [c]).then(()=>{
    // c.clip is the 'default' placeholder (see makeCharacter) — a static pose, so this just
    // draws the character in its raw rest pose, same as plain addCharacter().
    drawStick();
    renderCharacterLayers(); syncTransformSliders(); persistProjectState();
    renderTimelineClip(); syncTimelineInnerWidth(); renderRuler();
  });
}

function drawStick(){
  if(!ctx) return;
  updateSelBox();
  ctx.clearRect(0,0,stageW,stageH);
  const activeC=activeCharacter();
  const dragPreview=window.__ethvelleCharacterDragPreview;
  if(activeC && dragPreview && dragPreview.character===activeC){
    const dx=modelTransform.offsetX-dragPreview.startOffsetX;
    const dy=modelTransform.offsetY-dragPreview.startOffsetY;
    // Keep the original characters[] paint order during the cached drag path too.
    if(dragPreview.beforeCanvas)ctx.drawImage(dragPreview.beforeCanvas,0,0,stageW,stageH);
    else if(dragPreview.othersCanvas)ctx.drawImage(dragPreview.othersCanvas,0,0,stageW,stageH);
    ctx.save();ctx.translate(dx,dy);ctx.drawImage(dragPreview.canvas,0,0,stageW,stageH);ctx.restore();
    if(dragPreview.afterCanvas)ctx.drawImage(dragPreview.afterCanvas,0,0,stageW,stageH);
    return;
  }
  // Paint every character in the project's actual layer order. Selection changes only
  // which rig is editable; it must never change who visually sits in front of whom.
  characters.forEach(c=>{
    if(c&&c.bones&&c.modelTransform)drawCharacterToContext(ctx,c);
  });
  if(!activeC || !bones) return;
  // Draw selection guides after all character artwork, within the active character's
  // transform. This keeps handles easy to see without changing the artwork's paint order.
  ctx.save();
  const anchor=bones.hip;
  ctx.translate(anchor.x+modelTransform.offsetX,anchor.y+modelTransform.offsetY);
  ctx.rotate(modelTransform.rotation*Math.PI/180);
  ctx.scale(modelTransform.scale,modelTransform.scale);
  ctx.scale(activeC.mirrorX?-1:1,activeC.mirrorY?-1:1);
  ctx.translate(-anchor.x,-anchor.y);
  if(window.ethvelleApplyCanvasMaskClip)window.ethvelleApplyCanvasMaskClip(ctx,activeC.visualMask,stageW,stageH);
  if(window.ethvelleRigOverlayVisible && typeof window.drawEthvelleRigOverlay==='function'){
    window.drawEthvelleRigOverlay(ctx,activeC);
  }
  const mainClipIsSelected=selectedClip==='main' || String(selectedClip).startsWith('seg');
  if(!isPlaying && mainClipIsSelected && !window.ethvelleRigOverlayVisible && !activeC.sheet){
    ctx.fillStyle='#2F5CE8';
    getJointList().forEach(j=>{
      const isSelected=!!selectedJoint && j.key===selectedJoint.key;
      ctx.beginPath();ctx.arc(j.x,j.y,isSelected?4.2:3.1,0,Math.PI*2);ctx.fill();
      if(isSelected){
        ctx.strokeStyle='#fff';ctx.lineWidth=1.35;ctx.beginPath();ctx.arc(j.x,j.y,5.6,0,Math.PI*2);ctx.stroke();
      }
    });
  }
  ctx.restore();
}

function sheetCharacterJointList(c){
  // Every blue/white point drawn over a part-sheet rig must be returned here so the same
  // screen-space point participates in hit-testing. Include all visible hand/foot endpoints
  // and the intermediate spine joints instead of drawing decorative, non-draggable endpoints.
  try{
    if(!c || !c.sheet || typeof sheetRigForCharacter !== 'function' || typeof sheetSolveRig !== 'function') return null;
    const rig = sheetRigForCharacter(c), S = sheetSolveRig(rig.bones, rig.baseAngles);
    const required=['pelvis','spine_lower','spine_upper','chest','neck','head','clavicleL','clavicleR','upperArmL','upperArmR','forearmL','forearmR','handL','handR','thighL','thighR','shinL','shinR','footL','footR'];
    if(required.some(k=>!S[k])) return null;
    const point=(key,p)=>({key,x:p.x,y:p.y});
    return [
      point('hip',S.pelvis.s),
      point('pelvisTop',S.pelvis.e),
      point('spineMid',S.spine_lower.e),
      point('spineUpper',S.spine_upper.e),
      point('chest',S.chest.e),
      point('spine',S.neck.e),
      point('headTop',S.head.e),
      point('shoulderL',S.clavicleL.e), point('elbowL',S.upperArmL.e),
      point('wristL',S.forearmL.e), point('handL',S.handL.e),
      point('hipL',S.thighL.s), point('kneeL',S.thighL.e),
      point('ankleL',S.shinL.e), point('footL',S.footL.e),
      point('shoulderR',S.clavicleR.e), point('elbowR',S.upperArmR.e),
      point('wristR',S.forearmR.e), point('handR',S.handR.e),
      point('hipR',S.thighR.s), point('kneeR',S.thighR.e),
      point('ankleR',S.shinR.e), point('footR',S.footR.e)
    ];
  }catch(err){
    console.warn('Ethvelle: could not resolve imported model joints; using legacy handles',err);
    return null;
  }
}
function dragSheetCharacterJoint(key, raw){
  const c = activeCharacter();
  if(!c || !c.sheet || !bones) return false;
  const rig = sheetRigForCharacter(c), S = sheetSolveRig(rig.bones, rig.baseAngles);
  if(key === 'hip'){
    const sh = c.sheet, ov = sh.rigOverride || {};
    const rootScale = stageH*(Number(ov.figFrac)||.5)/Math.max(1,Number(sh.H)||1);
    const off = sh.rootOff || {x:0,y:0};
    bones.hip.x = raw.x - (Number(off.x)||0)*rootScale;
    bones.hip.y = raw.y - (Number(off.y)||0)*rootScale;
    c.hipRestY = bones.hip.y; hipRestY = c.hipRestY;
    return true;
  }
  // Hip starts are position controls: update the thigh's local offset in the same rotated,
  // figure-height-normalized coordinate system used by sheetSolveRig.
  if(key==='hipL' || key==='hipR'){
    const boneKey=key==='hipL'?'thighL':'thighR', parent=S.pelvis;
    if(!parent || !rig.bones[boneKey]) return false;
    const sh=c.sheet,ov=sh.rigOverride||(sh.rigOverride={});
    const offsets=ov.offsets||(ov.offsets={});
    const scale=stageH*(Number(ov.figFrac)||.5);
    const dx=raw.x-parent.s.x,dy=raw.y-parent.s.y,co=Math.cos(parent.D),si=Math.sin(parent.D);
    offsets[boneKey]={x:(dx*co+dy*si)/Math.max(1e-6,scale),y:(-dx*si+dy*co)/Math.max(1e-6,scale)};
    return 'override';
  }
  // Most familiar limb endpoints continue to edit the legacy animation angles, preserving
  // the existing keyframe workflow. Additional points write only the matching 21-bone
  // rest-rig angle, so shoulders, spine subdivisions and hand/foot tips are also movable.
  const legacyMap={
    spine:{segment:null,start:()=>S.pelvis.s,end:()=>S.neck.e,legacy:'spine'},
    headTop:{segment:S.head,legacy:'head'},
    elbowL:{segment:S.upperArmL,legacy:'armL'}, wristL:{segment:S.forearmL,legacy:'armL_lo'},
    elbowR:{segment:S.upperArmR,legacy:'armR'}, wristR:{segment:S.forearmR,legacy:'armR_lo'},
    kneeL:{segment:S.thighL,legacy:'legL'}, ankleL:{segment:S.shinL,legacy:'legL_lo'},
    kneeR:{segment:S.thighR,legacy:'legR'}, ankleR:{segment:S.shinR,legacy:'legR_lo'},
    handL:{segment:S.handL,legacy:'handL'}, handR:{segment:S.handR,legacy:'handR'}
  };
  const directMap={
    pelvisTop:'pelvis', spineMid:'spine_lower', spineUpper:'spine_upper', chest:'chest',
    shoulderL:'clavicleL', shoulderR:'clavicleR', footL:'footL', footR:'footR'
  };
  const m=legacyMap[key];
  if(m){
    const start=m.segment?m.segment.s:m.start();
    const currentAngle=m.segment?m.segment.a:Math.atan2(m.end().y-start.y,m.end().x-start.x);
    const targetAngle=Math.atan2(raw.y-start.y,raw.x-start.x);
    const delta=Math.atan2(Math.sin(targetAngle-currentAngle),Math.cos(targetAngle-currentAngle));
    const legacyBone=bones[m.legacy];
    if(!legacyBone || !Number.isFinite(legacyBone.angle)) return false;
    legacyBone.angle+=delta;
    return true;
  }
  const boneKey=directMap[key], segment=boneKey&&S[boneKey];
  if(!boneKey || !segment || !rig.baseAngles || !Number.isFinite(rig.baseAngles[boneKey])) return false;
  const targetAngle=Math.atan2(raw.y-segment.s.y,raw.x-segment.s.x);
  const delta=Math.atan2(Math.sin(targetAngle-segment.a),Math.cos(targetAngle-segment.a));
  const sh=c.sheet,ov=sh.rigOverride||(sh.rigOverride={});
  const angles=ov.angles||(ov.angles={});
  angles[boneKey]=rig.baseAngles[boneKey]+delta;
  return 'override';
}
function getJointList(){
  // Part-sheet imports must share their exact solved rig with the visible joint handles.
  const c = activeCharacter();
  if(c && c.type === 'custom' && c.sheet){
    const sheetJoints = sheetCharacterJointList(c);
    if(sheetJoints && sheetJoints.length) return sheetJoints;
  }
  // Legacy and auto-rig characters retain their original 11-bone joint list.
  const list = [];
  const hip = getBoneEnd('hip'); list.push({key:'hip', x:hip.x, y:hip.y});
  const spineEnd = getBoneEnd('spine'); list.push({key:'spine', x:spineEnd.x, y:spineEnd.y});
  LIMB_KEYS.forEach(k=>{
    const e = getBoneEnd(k); list.push({key:k, x:e.x, y:e.y});
    const loKey = lowerKey(k);
    const eLo = getBoneEnd(loKey); list.push({key:loKey, x:eLo.x, y:eLo.y});
  });
  const legacyChar=activeCharacter();
  if(legacyChar && legacyChar.type==='custom' && legacyChar.autoRig && legacyChar.autoRig.handRig){
    ['L','R'].forEach(side=>{
      if(legacyChar.autoRig.joints?.['palm'+side]){
        const palm=getBoneEnd('hand'+side); list.push({key:'hand'+side,x:palm.x,y:palm.y});
      }
      ['thumb','index','middle','ring','pinky'].forEach(f=>{
        const hasBase=!!legacyChar.autoRig.joints?.[f+'Base'+side], hasMid=!!legacyChar.autoRig.joints?.[f+'Mid'+side], hasTip=!!legacyChar.autoRig.joints?.[f+'Tip'+side];
        if(hasBase&&hasMid){const k=f+side+'_1';if(bones[k]){const e=getBoneEnd(k);list.push({key:k,x:e.x,y:e.y});}}
        if(hasMid&&hasTip){const k=f+side+'_2';if(bones[k]){const e=getBoneEnd(k);list.push({key:k,x:e.x,y:e.y});}}
      });
    });
  }
  return list;
}

// Convert a screen-space point into rig/bone space, accounting for modelTransform
function screenToRigSpace(px, py){
  const anchor = bones.hip;
  // inverse of: translate(anchor+offset) -> rotate -> scale -> translate(-anchor)
  let x = px - (anchor.x + modelTransform.offsetX);
  let y = py - (anchor.y + modelTransform.offsetY);
  const rad = -modelTransform.rotation * Math.PI/180;
  const cos = Math.cos(rad), sin = Math.sin(rad);
  const rx = x*cos - y*sin;
  const ry = x*sin + y*cos;
  const sx = rx / modelTransform.scale;
  const sy = ry / modelTransform.scale;
  const mirrored = activeCharacter();
  return {
    x: anchor.x + sx*(mirrored && mirrored.mirrorX ? -1 : 1),
    y: anchor.y + sy*(mirrored && mirrored.mirrorY ? -1 : 1)
  };
}

/* ================= ON-CANVAS SELECTION BOX (CapCut-style) =================
   FEATURE (\"in CapCut when I add any clip I can adjust it by my hand with dragging — I want that
   same feature\"): the selected Background / image-or-video Clip / Character now gets a white box
   on the preview stage. Drag INSIDE it to move; drag the bottom-right round handle to resize AND
   rotate around the item's centre (exactly what CapCut's corner handle does — pull out to grow,
   sweep sideways to rotate; it snaps to 0/90/180 so an un-rotated item is easy to keep straight);
   tap the top-left \u2715 to delete. Pinching with two fingers resizes too.

   It edits the SAME transform objects the Transform panel sliders edit (bg.transform,
   ov.transform, modelTransform — see transformTarget()), so the box, the sliders, undo/redo,
   save/load and export all stay in agreement with no second source of truth.

   selBoxTarget() answers \"what is draggable right now, and where is its un-transformed base
   rectangle?\"; selBoxGeom() then maps that through the item's transform to the on-screen box.
   Both transforms use the SAME order the renderers use — translate(offset) then rotate then
   scale about a pivot — so the frame lines up with the picture pixel-for-pixel:
     - Background / Clip: pivot = the base rect's centre (CSS transform-origin:center).
     - Character: pivot = the hip anchor (drawStick's translate/rotate/scale/translate(-anchor)). */
const SB_MIN_SCALE = 0.1, SB_MAX_SCALE = 6;
const SB_JOINT_HIT_R = 26; // keep in step with `minD=26` in setupDrag's joint hit-test
let selBoxEdit = null; // target whose transform was changed by the gesture currently in progress

// The character's box is measured from its REST pose (bone angles temporarily reset to
// baseAngles), not its live pose, so the frame stays put while you drag a joint to pose a limb
// instead of jittering/resizing under your finger. Bone positions are read relative to the live
// hip, so it still follows a jump-bounce. Returns rig-space {minX,minY,maxX,maxY}.
var __ethvelleRestBoundsCache = null;
var __ethvelleRestBoundsCacheCharacter = null;
var __ethvelleRestBoundsCacheValid = false;
var __ethvelleRestBoundsCacheRaf = 0;
function invalidateCharacterRestBoundsCache(){
  __ethvelleRestBoundsCache = null;
  __ethvelleRestBoundsCacheCharacter = null;
  __ethvelleRestBoundsCacheValid = false;
  if(__ethvelleRestBoundsCacheRaf){
    cancelAnimationFrame(__ethvelleRestBoundsCacheRaf);
    __ethvelleRestBoundsCacheRaf = 0;
  }
}
function characterRestBounds(){
  const character = typeof activeCharacter==='function' ? activeCharacter() : null;
  if(__ethvelleRestBoundsCacheCharacter !== character) invalidateCharacterRestBoundsCache();
  if(__ethvelleRestBoundsCacheValid) return __ethvelleRestBoundsCache;
  const saved = {};
  Object.keys(bones).forEach(k=>{
    saved[k] = bones[k].angle;
    if(baseAngles[k] !== undefined) bones[k].angle = baseAngles[k];
  });
  let minX=Infinity, minY=Infinity, maxX=-Infinity, maxY=-Infinity;
  const add = (pt)=>{ if(!isFinite(pt.x) || !isFinite(pt.y)) return; minX=Math.min(minX,pt.x); maxX=Math.max(maxX,pt.x); minY=Math.min(minY,pt.y); maxY=Math.max(maxY,pt.y); };
  Object.keys(bones).forEach(k=>{ add(getBoneStart(k)); add(getBoneEnd(k)); });
  if(bones.head){
    const hs = getBoneStart('head'), he = getBoneEnd('head');
    const hc = { x:(hs.x+he.x)/2, y:(hs.y+he.y)/2 }, hr = bones.head.length*0.6;
    add({x:hc.x-hr, y:hc.y-hr}); add({x:hc.x+hr, y:hc.y+hr});
  }
  Object.keys(saved).forEach(k=>{ bones[k].angle = saved[k]; });
  let result = null;
  if(isFinite(minX)){
    // Padding = the 26px joint grab radius used by setupDrag's hit-test + a 16px margin. A joint
    // handle takes priority over the box, and on a small character those 26px circles blanket
    // nearly the whole body — so with a tight box there was NO spot left that moved the character
    // instead of grabbing a joint. The extra margin leaves a finger-sized ring around the outside
    // that drags the whole character; the joints inside stay draggable for posing exactly as before.
    // (It also covers limb thickness / custom artwork extending past the bare bone lines.)
    const pad = SB_JOINT_HIT_R + 16;
    result = { minX:minX-pad, minY:minY-pad, maxX:maxX+pad, maxY:maxY+pad };
  }
  __ethvelleRestBoundsCache = result;
  __ethvelleRestBoundsCacheCharacter = character;
  __ethvelleRestBoundsCacheValid = true;
  if(!__ethvelleRestBoundsCacheRaf){
    __ethvelleRestBoundsCacheRaf = requestAnimationFrame(()=>{
      __ethvelleRestBoundsCacheRaf = 0;
      __ethvelleRestBoundsCache = null;
      __ethvelleRestBoundsCacheCharacter = null;
      __ethvelleRestBoundsCacheValid = false;
    });
  }
  return result;
}

// Returns the actual object-fit:contain content rectangle for an image/video overlay.
// The media element fills the stage, but the artwork may occupy only a centered portion of it.
function ethvelleOverlayContentSize(ov){
  const el=ov&&typeof overlayMediaEls!=='undefined'&&overlayMediaEls?overlayMediaEls[String(ov.id)]:null;
  let iw=Number(el&&(el.videoWidth||el.naturalWidth))||0;
  let ih=Number(el&&(el.videoHeight||el.naturalHeight))||0;
  if(!(iw>0&&ih>0)){
    iw=Number(ov&&(ov.sourceWidth||ov.mediaWidth||ov.originalWidth||ov.videoWidth||ov.width))||0;
    ih=Number(ov&&(ov.sourceHeight||ov.mediaHeight||ov.originalHeight||ov.videoHeight||ov.height))||0;
  }
  if(!(iw>0&&ih>0))return {w:stageW,h:stageH,left:0,top:0};
  const f=Math.min(stageW/iw,stageH/ih);
  const w=Math.max(1,iw*f),h=Math.max(1,ih*f);
  return {w,h,left:(stageW-w)/2,top:(stageH-h)/2};
}

// What can be dragged right now? null = nothing (no selection, playing, or the selected item
// isn't actually on screen at the playhead, in which case a box would point at nothing).
function selBoxTarget(){
  if(isPlaying || !stageW || !stageH) return null;
  const t = getCurrentAbsoluteTime();
  const centre = { x:stageW/2, y:stageH/2 };

  const bg = backgroundClipForSelection();
  if(bg){
    if(!bg.src || !String(bg.src).startsWith('url(')) return null;   // solid colour: nothing to drag
    if(currentBackgroundClip(t) !== bg) return null;                  // not the one on screen now
    const img = document.getElementById('bgImgLayer');
    const iw = img && img.naturalWidth, ih = img && img.naturalHeight;
    if(!iw || !ih) return null;                                       // not decoded yet
    const f = Math.min(stageW/iw, stageH/ih);                         // object-fit:contain size
    return { kind:'background', obj:ethvelleGetObjectKeyframeValue(bg,'transform',t)||bg.transform, cB:centre, P0:centre, w0:iw*f, h0:ih*f, canDelete:true };
  }

  if(typeof selectedClip === 'string'){
    const ov = overlays.find(o=>String(o.id)===String(selectedClip));
    if(ov){
      if(!ov.transform) return null;
      if(!(t >= ov.trimStart && t < ov.trimEnd)) return null;
      if(ov.type==='text'){
        // Text gets a snug selection frame around its real glyph bounds rather than the
        // full preview-stage rectangle used by image/video layers. This also enables the
        // existing delete + resize/rotate handles for a selected text overlay.
        let rendered=(typeof overlayMediaEls!=='undefined'&&overlayMediaEls[ov.id])||null;
        if(!rendered){
          const mediaLayer=document.getElementById('overlayMediaLayer');
          if(mediaLayer) rendered=Array.from(mediaLayer.children).find(n=>String(n.dataset.overlayId||n.dataset.ovId||'')===String(ov.id))||null;
        }
        const textNode=rendered&&rendered.querySelector('.ethvelle-overlay-text-content');
        if(!textNode) return null;
        return {kind:'overlay',obj:ethvelleGetObjectKeyframeValue(ov,'transform',t)||ov.transform,cB:centre,P0:centre,w0:Math.max(34,(textNode.offsetWidth||textNode.getBoundingClientRect().width)+12),h0:Math.max(30,(textNode.offsetHeight||textNode.getBoundingClientRect().height)+12),canDelete:true};
      }
      if(ov.type!=='image' && ov.type!=='video') return null;
      const mediaSize=ethvelleOverlayContentSize(ov);
      return { kind:'overlay', obj:ethvelleGetObjectKeyframeValue(ov,'transform',t)||ov.transform, cB:centre, P0:centre, w0:mediaSize.w, h0:mediaSize.h, canDelete:true };
    }
  }

  const isCharClip = selectedClip==='main' || (typeof selectedClip==='string' && selectedClip.startsWith('seg'));
  if(isCharClip && activeCharacter() && bones && bones.hip){
    const bb = characterRestBounds();
    if(!bb) return null;
    return {
      kind:'character', obj:modelTransform,
      cB:{ x:(bb.minX+bb.maxX)/2, y:(bb.minY+bb.maxY)/2 },
      P0:{ x:bones.hip.x, y:bones.hip.y },
      w0:bb.maxX-bb.minX, h0:bb.maxY-bb.minY,
      // UPDATED ("add a cross button on the top-left corner where I can delete the clip or layer"):
      // the ✕ now shows for a selected character clip too. It runs the same deleteSelectedClip()
      // as the toolbar's Delete (removes that clip; the character itself goes when its last clip does).
      canDelete:true
    };
  }
  return null;
}

// Maps a target's base rectangle through its transform. Returns stage-local px:
// centre (cx,cy), size (w,h), rotation in degrees, and the pivot (px,py) the transform turns about.
function selBoxGeom(tg){
  const tr = tg.obj, sc = tr.scale, rot = tr.rotation || 0, a = rot*Math.PI/180;
  const cs = Math.cos(a), sn = Math.sin(a);
  const dx = (tg.cB.x - tg.P0.x)*sc, dy = (tg.cB.y - tg.P0.y)*sc;
  return {
    cx: tg.P0.x + tr.offsetX + dx*cs - dy*sn,
    cy: tg.P0.y + tr.offsetY + dx*sn + dy*cs,
    w: tg.w0*sc, h: tg.h0*sc, rot,
    px: tg.P0.x + tr.offsetX, py: tg.P0.y + tr.offsetY
  };
}

// True (returning the target) if a client-space point lies inside the current box.
function selBoxHit(clientX, clientY){
  const tg = selBoxTarget();
  if(!tg) return null;
  const g = selBoxGeom(tg);
  const r = document.getElementById('stickCanvas').getBoundingClientRect();
  if(!r.width||!r.height)return null;
  const x = (clientX-r.left)*(stageW/r.width) - g.cx;
  const y = (clientY-r.top)*(stageH/r.height) - g.cy;
  const a = -g.rot*Math.PI/180, cs = Math.cos(a), sn = Math.sin(a);
  const lx = x*cs - y*sn, ly = x*sn + y*cs;
  return (Math.abs(lx) <= g.w/2 && Math.abs(ly) <= g.h/2) ? tg : null;
}

// Shows/hides and positions the frame + buttons. Cheap (a handful of style writes); called from
// drawStick() (which every render path already calls), applyBgTransform(), selection changes,
// stage resizes and when the background image finishes decoding.
function updateSelBox(){
  const layer = document.getElementById('selBoxLayer');
  if(!layer) return;
  const tg = selBoxTarget();
  if(!tg){ layer.style.display = 'none'; return; }
  const stage = document.getElementById('canvasStage');
  const g = selBoxGeom(tg);
  layer.style.display = 'block';
  const gestureGeometry=window.__ethvelleGestureGeometry;
  const useGestureGeometry=!!(gestureGeometry&&gestureGeometry.stage===stage&&gestureGeometry.rect);
  layer.style.left = (useGestureGeometry?gestureGeometry.offsetLeft:stage.offsetLeft)+'px';
  layer.style.top = (useGestureGeometry?gestureGeometry.offsetTop:stage.offsetTop)+'px';
  layer.style.width = stageW+'px'; layer.style.height = stageH+'px';
  const fr = document.getElementById('selBoxFrame');
  fr.style.width = g.w+'px'; fr.style.height = g.h+'px';
  fr.style.left = (g.cx-g.w/2)+'px'; fr.style.top = (g.cy-g.h/2)+'px';
  fr.style.transform = `rotate(${g.rot}deg)`;
  // Corner buttons are LOCKED to the frame's real rotated corners.
  // BUGFIX ("the resize/rotate arrow changes its position when I extend the box"): these used to be
  // clamped to the STAGE rectangle (+8px), so as soon as the box grew past the stage edge the
  // handle slid off its corner and stopped at the stage border — and because the drag maths then
  // started from that displaced spot, the box also jumped when the handle was grabbed. The handles
  // now always sit exactly on their corners, even outside the stage (this layer isn't clipped).
  // The only clamp left is the physical SCREEN edge, purely so a corner pushed off-screen can't
  // become unreachable.
  const a = g.rot*Math.PI/180, cs = Math.cos(a), sn = Math.sin(a);
  const corner = (sx, sy)=>{
    const lx = sx*g.w/2, ly = sy*g.h/2;
    return { x: g.cx + lx*cs - ly*sn, y: g.cy + lx*sn + ly*cs };
  };
  const stRect = useGestureGeometry?gestureGeometry.rect:stage.getBoundingClientRect();
  const SB_EDGE = 16;
  const scaleX=stRect.width/Math.max(1,stageW),scaleY=stRect.height/Math.max(1,stageH);
  const place = (el, pt)=>{
    const cx = Math.max(SB_EDGE, Math.min(window.innerWidth  - SB_EDGE, stRect.left + pt.x*scaleX));
    const cy = Math.max(SB_EDGE, Math.min(window.innerHeight - SB_EDGE, stRect.top  + pt.y*scaleY));
    el.style.left = ((cx - stRect.left)/Math.max(.001,scaleX))+'px';
    el.style.top  = ((cy - stRect.top)/Math.max(.001,scaleY))+'px';
  };
  const del = document.getElementById('sbDelete'), scl = document.getElementById('sbScale');
  del.style.display = tg.canDelete ? 'flex' : 'none';
  place(del, corner(-1,-1));
  place(scl, corner(1,1));
}

// Paint only the selected visual overlay during a direct canvas gesture. Calling renderFrame() /
// renderOverlayMedia() on every pointermove walks every clip, touches video/audio decoders and
// wakes several observers; on low-memory Android phones that makes the selected layer lag behind
// the finger. This writes the same live CSS transform to the existing element without rebuilding
// the media stack. The full renderer still runs for normal timeline playback and export.
function applyOverlayTransformPreviewFast(tg){
  const id=String(typeof selectedClip!=='undefined'&&selectedClip!=null?selectedClip:'');
  const list=(typeof overlays!=='undefined'&&Array.isArray(overlays))?overlays:[];
  const clip=list.find(o=>String(o.id)===id);
  if(!clip)return false;
  const elements=(typeof overlayMediaEls!=='undefined'&&overlayMediaEls)?overlayMediaEls:{};
  const el=elements[id];
  if(!el||!el.isConnected)return false;
  const tr=(tg&&tg.obj)||clip.transform||{scale:1,offsetX:0,offsetY:0,rotation:0};
  const scale=Math.max(.0001,Number(tr.scale)||1);
  const ox=Number(tr.offsetX)||0,oy=Number(tr.offsetY)||0,rotation=Number(tr.rotation)||0;
  // v100+ uses clip-path (not zoom-transform) for crop, so always use the plain transform here.
  // applyCropWithoutZoom() (called after every render) handles the visual crop via clip-path.
  // Using the old zoom-scaled transform while the clip-path crop is active caused the content
  // to jump/zoom whenever the clip was touched or dragged (the "zoom on tap" bug).
  let value=`translate(${ox}px,${oy}px) rotate(${rotation}deg) scale(${scale})`;
  if(clip.mirrorX)value+=' scaleX(-1)';
  if(clip.mirrorY)value+=' scaleY(-1)';
  value+=(window.ethvellePerspectiveSuffix?window.ethvellePerspectiveSuffix(clip):'');
  if(el.style.transform!==value)el.style.transform=value;
  const chroma=(typeof overlayChromaCanvases!=='undefined'&&overlayChromaCanvases)?overlayChromaCanvases[id]:null;
  if(chroma&&chroma.style.transform!==value)chroma.style.transform=value;

  // v235: the video compositor may keep a last-good frame canvas above the live <video>
  // while Android's decoder is seeking. That retained canvas used to keep its old CSS
  // transform when the user moved the video on the stage, leaving a frozen picture at the
  // previous position. Synchronize every retained stand-in with the live owner in this same
  // fast path; do not run the full compositor or recreate any decoder during a canvas drag.
  const paintTime=(typeof getCurrentAbsoluteTime==='function')?Number(getCurrentAbsoluteTime()):Number(currentAbsoluteTimeSec)||0;
  const synced=new Set();
  const syncRetained=node=>{
    if(!node||node===el||synced.has(node))return;
    synced.add(node);
    try{
      if(node.dataset)node.dataset.evMaskOwnerTime=String(paintTime);
      if(typeof window.ethvelleApplyOverlayVisualStyle==='function')window.ethvelleApplyOverlayVisualStyle(node,clip,paintTime,el,false);
      else if(node.style&&node.style.transform!==value)node.style.transform=value;
    }catch(_e){if(node.style)node.style.transform=value;}
  };
  if(el.__evScrubHoldCanvas)syncRetained(el.__evScrubHoldCanvas);
  const mediaLayer=document.getElementById('overlayMediaLayer');
  if(mediaLayer)mediaLayer.querySelectorAll('[data-ev-mask-owner-id]').forEach(node=>{
    if(String(node.dataset.evMaskOwnerId||'')===id)syncRetained(node);
  });
  const outgoing=(window.renderOverlayMedia&&window.renderOverlayMedia.__evHeldOutgoing)||{};
  Object.keys(outgoing).forEach(ownerId=>{
    if(String(ownerId)!==id)return;
    const entry=outgoing[ownerId];
    if(entry){syncRetained(entry.canvas);syncRetained(entry.fallbackVideo);}
  });

  // If this clip is the tracking source for other layers, repaint those followers immediately
  // too. Their stored transform stays independent; only the source's live delta is inherited.
  if(typeof window.ethvelleRefreshTrackingFollowersForSource==='function'){
    window.ethvelleRefreshTrackingFollowersForSource('ov:'+id,paintTime);
  }
  // A CSS transform leaves the existing clip-path intact. Crop is refreshed by the normal
  // render path; no all-clips scan or decoder work is introduced on this pointermove path.
  return true;
}

// Repaints whatever a drag just changed, then the box itself. Media overlays use the light
// path above so pointer movement never forces every video/audio layer to be rendered again.
let selBoxSliderSyncRaf=0;
function scheduleSelBoxSliderSync(){
  if(selBoxSliderSyncRaf)return;
  selBoxSliderSyncRaf=requestAnimationFrame(()=>{
    selBoxSliderSyncRaf=0;
    const panel=document.getElementById('toolPanel');
    if(panel&&panel.classList.contains('open')&&document.getElementById('sizeSlider'))syncTransformSliders();
  });
}
function selBoxApply(tg){
  if(tg.kind==='background'){
    refreshCurrentBgFromPlayhead();
    applyBgTransform(); // already refreshes the selection frame
  }else if(tg.kind==='overlay'){
    applyOverlayTransformPreviewFast(tg);
    updateSelBox();
  }else{
    drawStick(); // drawStick updates the selection frame after painting the rig
    const ch=typeof activeCharacter==='function'?activeCharacter():null;
    if(ch&&typeof window.ethvelleRefreshTrackingFollowersForSource==='function'){
      const key=ch.id!=null?String(ch.id):'index:'+String(activeCharIdx);
      window.ethvelleRefreshTrackingFollowersForSource('char:'+key,typeof getCurrentAbsoluteTime==='function'?getCurrentAbsoluteTime():currentAbsoluteTimeSec);
    }
  }
  // Range controls need at most one DOM sync per display frame, not one per raw pointer event.
  scheduleSelBoxSliderSync();
}
// End of a drag/pinch: one undo step + keyframe sync (character only — same rule updateTransform uses).
function finishSelBoxEdit(tg){
  if(tg.kind==='character'){
    const updatedKeyframe = updateCurrentKeyframeFromEditorState();
    if(!updatedKeyframe && activeCharacter()) activeCharacter().restTransform = { ...modelTransform };
  }
  persistProjectState();
}
function normDeg(d){ return ((d+180)%360+360)%360-180; }

(function wireSelBoxButtons(){
  const scl = document.getElementById('sbScale'), del = document.getElementById('sbDelete');
  if(!scl || !del) return;
  let st=null,moveRaf=0,movePoint=null;
  function applyScaleMove(){
    moveRaf=0;
    const p=movePoint;movePoint=null;
    if(!st||!p)return;
    const fx=st.cx0+(p.x-st.downX),fy=st.cy0+(p.y-st.downY);
    const d=Math.hypot(fx-st.px,fy-st.py)||1;
    const a=Math.atan2(fy-st.py,fx-st.px);
    st.tg.obj.scale=Math.max(SB_MIN_SCALE,Math.min(SB_MAX_SCALE,st.s0*d/st.d0));
    let rot=normDeg(st.r0+(a-st.a0)*180/Math.PI);
    for(const snap of [-180,-90,0,90,180])if(Math.abs(rot-snap)<3){rot=snap;break;}
    st.tg.obj.rotation=rot===-180?180:rot;
    selBoxEdit=st.tg;selBoxApply(st.tg);
  }
  function queueScaleMove(e){movePoint={x:e.clientX,y:e.clientY};if(!moveRaf)moveRaf=requestAnimationFrame(applyScaleMove);}
  function flushScaleMove(){if(moveRaf){cancelAnimationFrame(moveRaf);moveRaf=0;}if(movePoint)applyScaleMove();}
  scl.addEventListener('pointerdown',(e)=>{
    const tg=selBoxTarget();if(!tg)return;
    e.preventDefault();e.stopPropagation();
    try{scl.setPointerCapture(e.pointerId);}catch(_err){}
    const g=selBoxGeom(tg),stage=document.getElementById('canvasStage');
    const r=stage.getBoundingClientRect(),sx=r.width/Math.max(1,stageW),sy=r.height/Math.max(1,stageH);
    const px=r.left+g.px*sx,py=r.top+g.py*sy;
    const ra=g.rot*Math.PI/180,rc=Math.cos(ra),rs=Math.sin(ra);
    const isTextResizeCorner=tg.kind==='overlay'&&Array.isArray(overlays)&&overlays.some(o=>String(o.id)===String(selectedClip)&&o.type==='text');
    const sxCorner=isTextResizeCorner?-1:1;
    const cx0=r.left+(g.cx+(sxCorner*g.w/2)*rc-(g.h/2)*rs)*sx;
    const cy0=r.top+(g.cy+(sxCorner*g.w/2)*rs+(g.h/2)*rc)*sy;
    st={tg,px,py,downX:e.clientX,downY:e.clientY,cx0,cy0,d0:Math.hypot(cx0-px,cy0-py)||1,
      a0:Math.atan2(cy0-py,cx0-px),s0:tg.obj.scale,r0:tg.obj.rotation||0};
  });
  scl.addEventListener('pointermove',(e)=>{if(!st)return;e.preventDefault();queueScaleMove(e);},{passive:false});
  const end=(e)=>{
    if(!st)return;
    flushScaleMove();
    try{scl.releasePointerCapture(e.pointerId);}catch(_err){}
    const tg=st.tg;st=null;movePoint=null;
    if(selBoxEdit){selBoxEdit=null;finishSelBoxEdit(tg);}
  };
  scl.addEventListener('pointerup',end);
  scl.addEventListener('pointercancel',end);
  del.addEventListener('pointerdown', (e)=>{ e.preventDefault(); e.stopPropagation(); });
  del.addEventListener('click', (e)=>{ e.stopPropagation(); if(selectedClip) deleteSelectedClip(); });
  // The picture may finish decoding AFTER the box was first asked to draw (naturalWidth is 0
  // until then), so re-measure the moment it loads.
  const bgImg = document.getElementById('bgImgLayer');
  if(bgImg) bgImg.addEventListener('load', ()=>updateSelBox());
})();

/* ---------- TAP-TO-SELECT ON THE PREVIEW (CapCut-style) ----------
   FEATURE (\"if I tap the clip, layer or background in the project it should select automatically —
   I shouldn't have to tap its layer in the timeline\"): touching the stage now selects whatever is
   visibly under the finger, in real paint order, top to bottom:
     characters (the selected one first, then the rest, last-added on top)
       -> image/video clips (later in `overlays` paints over earlier)
         -> the background clip playing at the playhead.
   Selecting it arms the same toolbar / timeline highlight / on-canvas box as tapping its timeline
   clip. Touching the item that is ALREADY selected keeps the old behaviour (a tap toggles
   play/pause, a drag moves it). */
function pointInSelGeom(g, x, y){
  const dx = x-g.cx, dy = y-g.cy, a = -g.rot*Math.PI/180, cs = Math.cos(a), sn = Math.sin(a);
  const lx = dx*cs - dy*sn, ly = dx*sn + dy*cs;
  return Math.abs(lx) <= g.w/2 && Math.abs(ly) <= g.h/2;
}
// Snug (un-padded) on-screen box for ANY character, active or not. characterRestBounds() reads the
// shared `bones`/`baseAngles`, so those two are pointed at this character for the duration of the
// call and always put back.
function charPickGeom(c){
  if(!c || !c.bones || !c.bones.hip) return null;
  const savedBones = bones, savedBase = baseAngles;
  let bb = null;
  try{ bones = c.bones; baseAngles = c.baseAngles || {}; bb = characterRestBounds(); }
  finally{ bones = savedBones; baseAngles = savedBase; }
  if(!bb) return null;
  const trim = SB_JOINT_HIT_R + 16 - 14;   // undo the drag-ring padding, keep a 14px finger margin
  const minX = bb.minX+trim, minY = bb.minY+trim, maxX = bb.maxX-trim, maxY = bb.maxY-trim;
  return selBoxGeom({
    obj:c.modelTransform,
    cB:{ x:(minX+maxX)/2, y:(minY+maxY)/2 },
    P0:{ x:c.bones.hip.x, y:c.bones.hip.y },
    w0:Math.max(maxX-minX, 20), h0:Math.max(maxY-minY, 20)
  });
}
function pickCanvasItemAt(clientX, clientY){
  if(isPlaying || !stageW || !stageH) return null;
  const canvasEl=document.getElementById('stickCanvas');
  if(!canvasEl)return null;
  const r=canvasEl.getBoundingClientRect();
  if(!r.width||!r.height)return null;
  const x=(clientX-r.left)*(stageW/r.width),y=(clientY-r.top)*(stageH/r.height);
  if(x<0||y<0||x>stageW||y>stageH)return null;
  const t=getCurrentAbsoluteTime(),centre={x:stageW/2,y:stageH/2};

  // A touch on the visible selection-frame edge belongs to that selected object. The frame
  // itself ignores pointer events, so this narrow hit band makes its border usable without
  // letting the whole selection rectangle swallow taps on other overlapping media.
  const selectedTarget=selBoxTarget();
  if(selectedTarget){
    const g=selBoxGeom(selectedTarget),dx=x-g.cx,dy=y-g.cy,a=-g.rot*Math.PI/180,cs=Math.cos(a),sn=Math.sin(a);
    const lx=dx*cs-dy*sn,ly=dx*sn+dy*cs,halfW=g.w/2,halfH=g.h/2,band=11;
    const withinExpanded=Math.abs(lx)<=halfW+band&&Math.abs(ly)<=halfH+band;
    const nearEdge=Math.abs(Math.abs(lx)-halfW)<=band||Math.abs(Math.abs(ly)-halfH)<=band;
    if(withinExpanded&&nearEdge){
      if(selectedTarget.kind==='overlay'&&selectedClip!=null)return {kind:'overlay',id:String(selectedClip)};
      if(selectedTarget.kind==='background'&&selectedClip!=null)return {kind:'background',id:String(selectedClip)};
      if(selectedTarget.kind==='character')return {kind:'character',idx:activeCharIdx};
    }
  }

  // Overlay media is painted above the character canvas. Test highest z-order first and
  // constrain image/video hits to their real object-fit:contain content bounds, not the full stage.
  for(let i=overlays.length-1;i>=0;i--){
    const ov=overlays[i];
    if(!ov||!ov.transform||!(t>=Number(ov.trimStart)&&t<Number(ov.trimEnd)))continue;
    let g=null;
    const tr=(typeof window.ethvelleGetTrackedOverlayTransform==='function'?window.ethvelleGetTrackedOverlayTransform(ov,t):null)||ethvelleGetObjectKeyframeValue(ov,'transform',t)||ov.transform;
    if(ov.type==='text'){
      let host=null;
      try{host=typeof overlayMediaEls!=='undefined'&&overlayMediaEls?overlayMediaEls[String(ov.id)]:null;}catch(_e){}
      if(!host){const layer=document.getElementById('overlayMediaLayer');if(layer)host=Array.from(layer.children).find(n=>String(n.dataset.overlayId||n.dataset.ovId||'')===String(ov.id))||null;}
      const textNode=host&&host.querySelector?host.querySelector('.ethvelle-overlay-text-content'):null;
      if(!textNode)continue;
      const rect=textNode.getBoundingClientRect();
      g=selBoxGeom({obj:tr,cB:centre,P0:centre,w0:Math.max(34,(textNode.offsetWidth||rect.width)+12),h0:Math.max(30,(textNode.offsetHeight||rect.height)+12)});
    }else if(ov.type==='image'||ov.type==='video'){
      const size=ethvelleOverlayContentSize(ov);
      g=selBoxGeom({obj:tr,cB:centre,P0:centre,w0:size.w,h0:size.h});
    }else continue;
    if(g&&pointInSelGeom(g,x,y))return {kind:'overlay',id:ov.id};
  }

  // Character art is below overlay media; topmost character (last drawn) wins.
  for(let i=characters.length-1;i>=0;i--){
    const g=charPickGeom(characters[i]);
    if(g&&pointInSelGeom(g,x,y))return {kind:'character',idx:i};
  }

  // Background is the bottom visual layer. Restrict its hit area to the transformed image
  // rectangle, so letterbox space does not accidentally select an unrelated background clip.
  const bg=currentBackgroundClip(t);
  if(bg){
    const img=document.getElementById('bgImgLayer');
    const iw=Number(img&&img.naturalWidth)||0,ih=Number(img&&img.naturalHeight)||0;
    if(iw>0&&ih>0){
      const f=Math.min(stageW/iw,stageH/ih),tr=ethvelleGetObjectKeyframeValue(bg,'transform',t)||bg.transform||{scale:1,offsetX:0,offsetY:0,rotation:0};
      const g=selBoxGeom({obj:tr,cB:centre,P0:centre,w0:iw*f,h0:ih*f});
      if(pointInSelGeom(g,x,y))return {kind:'background',id:bg.id};
    }
  }
  return null;
}
// Which of the ACTIVE character's clips is under the playhead ('main' or 'segN')?
function charClipRefAtTime(t){
  if(activeClip && t >= activeClip.trimStart - 0.0001 && t < activeClip.trimEnd) return 'main';
  for(let i = 0; i < splitStickmanSegments.length; i++){
    const sg = splitStickmanSegments[i];
    if(t >= sg.trimStart - 0.0001 && t < sg.trimEnd) return 'seg'+i;
  }
  if(activeClip) return 'main';
  return 'char'+activeCharIdx;
}
function isCanvasPickSelected(p){
  if(p.kind === 'character'){
    if(p.idx !== activeCharIdx) return false;
    return selectedClip === 'main' || String(selectedClip).startsWith('seg') ||
           (!activeClip && selectedClip === 'char'+p.idx);
  }
  return selectedClip === p.id;
}
function applyCanvasPick(p){
  if(p.kind === 'character'){
    if(p.idx !== activeCharIdx) selectCharacter(p.idx);
    selectClipForChrome(charClipRefAtTime(getCurrentAbsoluteTime()));
  } else {
    selectClipForChrome(p.id);
  }
}

function setupDrag(){
  const canvas = document.getElementById('stickCanvas');
  // Tracks every pointer currently down on the canvas (client coords), so a second finger
  // touching down mid-gesture switches us into pinch-zoom for the background image.
  const activePointers = new Map();
  let pinchState = null; // { startDist, tg, startScale } — two-finger resize of the SELECTED item
  let sheetRigOverrideChanged = false;
  // One-finger drag that started INSIDE the selection box (see selBoxHit): moves the selected
  // Background / Clip / Character. Replaces the old "drag anywhere on the stage pans the
  // background", which moved the background even when something else was selected.
  let boxMove = null;    // { tg, startClientX, startClientY, startOffsetX, startOffsetY }
  // Read layout once when a gesture starts; reusing these rectangles avoids sync layout
  // work on every pointermove while posing joints or dragging a canvas layer.
  let pointerCanvasRect = null;
  let pointerStageRect = null;
  // Stage interaction is for selecting, posing, moving and resizing only. Playback must
  // be started explicitly with the transport Play button; a canvas tap must never start it.
  const TAP_MOVE_TOLERANCE = 6; // px

  function hasBgImage(){ return !!(currentBg && String(currentBg).startsWith('url(')); }
  function canvasPickIdentity(p){return !p?'':p.kind==='overlay'?'ov:'+String(p.id):p.kind==='background'?'bg:'+String(p.id):p.kind==='character'?'char:'+String(p.idx):'';}
  function findUnderlyingCanvasClip(clientX,clientY,coverage,visited){
    if(!coverage||!stageW||!stageH)return null;
    const rect=canvas.getBoundingClientRect();if(!rect.width||!rect.height)return null;
    const x=(clientX-rect.left)*(stageW/rect.width),y=(clientY-rect.top)*(stageH/rect.height);
    if(!pointInSelGeom(selBoxGeom(coverage),x,y))return null;
    const t=getCurrentAbsoluteTime(),centre={x:stageW/2,y:stageH/2};
    for(let i=overlays.length-1;i>=0;i--){
      const ov=overlays[i];if(!ov||!['image','video','text'].includes(ov.type)||!ov.transform)continue;
      const key='ov:'+String(ov.id);if(visited&&visited.has(key))continue;
      if(!(t>=Number(ov.trimStart)&&t<Number(ov.trimEnd)))continue;
      let g=null;
      const tr=(typeof window.ethvelleGetTrackedOverlayTransform==='function'?window.ethvelleGetTrackedOverlayTransform(ov,t):null)||ethvelleGetObjectKeyframeValue(ov,'transform',t)||ov.transform;
      if(ov.type==='text'){
        const el=(typeof overlayMediaEls!=='undefined'&&overlayMediaEls)?overlayMediaEls[ov.id]:null;
        const text=el&&el.querySelector?el.querySelector('.ethvelle-overlay-text-content'):null;
        if(!text)continue;
        const w=Math.max(34,(text.offsetWidth||text.getBoundingClientRect().width)+12),h=Math.max(30,(text.offsetHeight||text.getBoundingClientRect().height)+12);
        g=selBoxGeom({obj:tr,cB:centre,P0:centre,w0:w,h0:h});
      }else{
        const size=ethvelleOverlayContentSize(ov);
        g=selBoxGeom({obj:tr,cB:centre,P0:centre,w0:size.w,h0:size.h});
      }
      if(g&&pointInSelGeom(g,x,y))return {kind:'overlay',id:ov.id};
    }
    const bg=typeof currentBackgroundClip==='function'?currentBackgroundClip(t):null;
    if(bg&&bg.id!=null&&!(visited&&visited.has('bg:'+String(bg.id)))&&bg.src&&String(bg.src).startsWith('url('))return {kind:'background',id:bg.id};
    return null;
  }
  function handoffCanvasSelection(e){
    if(!boxMove||!boxMove.tg)return false;
    const visited=boxMove.visited||(boxMove.visited=new Set());
    const currentKey=canvasPickIdentity({kind:boxMove.tg.kind,id:selectedClip,idx:activeCharIdx});
    if(currentKey)visited.add(currentKey);
    const next=findUnderlyingCanvasClip(e.clientX,e.clientY,boxMove.tg,visited);
    if(!next)return false;
    const key=canvasPickIdentity(next);if(!key||visited.has(key))return false;
    visited.add(key);
    applyCanvasPick(next);
    const target=selBoxTarget();
    if(!target)return false;
    boxMove={tg:target,startClientX:e.clientX,startClientY:e.clientY,startOffsetX:Number(target.obj.offsetX)||0,startOffsetY:Number(target.obj.offsetY)||0,visited};
    try{if(typeof updateSelBox==='function')updateSelBox();}catch(_e){}
    return true;
  }

  canvas.onpointerdown = (e)=>{
    // While already playing, a stage touch may pause, but a stage touch can never start
    // playback. Starting playback is reserved for the explicit transport Play button.
    if(isPlaying){
      pausePlayback();
      return;
    }
    pointerCanvasRect=canvas.getBoundingClientRect();
    const stageElement=document.getElementById('canvasStage');
    pointerStageRect=stageElement?stageElement.getBoundingClientRect():pointerCanvasRect;
    window.__ethvelleGestureGeometry=stageElement?{stage:stageElement,rect:pointerStageRect,offsetLeft:stageElement.offsetLeft,offsetTop:stageElement.offsetTop}:null;
    try{ canvas.setPointerCapture(e.pointerId); }catch(err){}
    activePointers.set(e.pointerId, {x:e.clientX, y:e.clientY});

    if(activePointers.size >= 2){
      // A second pointer takes ownership as a pinch; a pending single-finger rest-rig edit
      // will be committed by the normal pointer cleanup path without leaving a cached preview.
      // Second finger landed — this is now a pinch, regardless of what the first finger
      // was doing (cancel any in-progress joint drag or one-finger pan).
      dragging = false; selectedJoint = null; boxMove = null;
      window.__ethvelleCharacterDragPreview = null;
      const pts = [...activePointers.values()];
      const dx = pts[0].x - pts[1].x, dy = pts[0].y - pts[1].y;
      // Pinch now resizes whatever is SELECTED (background, clip or character), not only the
      // background; with nothing draggable selected there is nothing to pinch.
      const pinchTg = selBoxTarget();
      pinchState = pinchTg ? { startDist: Math.hypot(dx,dy) || 1, tg: pinchTg, startScale: pinchTg.obj.scale } : null;
      drawStick();
      return;
    }

    // With no active character (a brand-new, still-empty project) there are no bones
    // or joints to hit-test, so skip joint selection and leave the stage tap inert.
    // BUGFIX (invisible draggable joints): joint handles are now only DRAWN when the
    // character's own animation clip is selected (see drawStick's mainClipIsSelected), so
    // hit-testing must require the exact same condition — otherwise a joint could still be
    // grabbed and dragged while its handle wasn't actually showing on screen.
    const rect = pointerCanvasRect || canvas.getBoundingClientRect();
    const mainClipSelectedForDrag = selectedClip==='main' || String(selectedClip).startsWith('seg');
    const activeRigCharacter=activeCharacter();
    const isSheetRig=!!(activeRigCharacter && activeRigCharacter.type==='custom' && activeRigCharacter.sheet);
    const selectedCharacterLayer=/^char\d+$/.test(String(selectedClip||''));
    const hasActiveRig=!!(activeRigCharacter && bones && (
      (mainClipSelectedForDrag && (!isSheetRig || window.ethvelleRigOverlayVisible)) ||
      (selectedCharacterLayer && window.ethvelleRigOverlayVisible)
    ));
    const stageX = rect.width ? (e.clientX-rect.left)*(stageW/rect.width) : 0;
    const stageY = rect.height ? (e.clientY-rect.top)*(stageH/rect.height) : 0;
    const raw = hasActiveRig ? screenToRigSpace(stageX,stageY) : null;
    const joints = hasActiveRig ? getJointList() : [];
    let closest=null, minD=26;
    joints.forEach(j=>{
      const d = Math.hypot(j.x-raw.x, j.y-raw.y);
      if(d<minD){ minD=d; closest=j; }
    });
    if(closest){
      // Tapping/dragging a joint handle is for posing, not play/pause — leave it alone.
      selectedJoint = closest;
      dragging = true;
      drawStick();
    } else {
      // NEW: touching a DIFFERENT item than the selected one (character / clip / background)
      // selects it on the spot — see pickCanvasItemAt. Because this happens on touch-down, the
      // same gesture can carry straight on into dragging that item. Such a touch is a selection,
      // not a play/pause tap.
      const pick = e.__ethvelleZoomPicked ? null : pickCanvasItemAt(e.clientX, e.clientY);
      if(pick && !isCanvasPickSelected(pick)) applyCanvasPick(pick);
      // Inside the selected item's box a one-finger drag moves it. A plain tap only
      // selects the item; it never toggles playback, including on a selected item.
      const tg = selBoxHit(e.clientX, e.clientY);
      if(tg){
        const visited=new Set();
        const currentKey=canvasPickIdentity({kind:tg.kind,id:selectedClip,idx:activeCharIdx});
        if(currentKey)visited.add(currentKey);
        boxMove = { tg, startClientX:e.clientX, startClientY:e.clientY, startOffsetX:tg.obj.offsetX, startOffsetY:tg.obj.offsetY, visited };
      }
    }
  };
  function processCanvasPointerMove(e){
    if(activePointers.has(e.pointerId)) activePointers.set(e.pointerId, {x:e.clientX, y:e.clientY});

    if(pinchState && activePointers.size >= 2){
      const pts = [...activePointers.values()];
      const dx = pts[0].x - pts[1].x, dy = pts[0].y - pts[1].y;
      const dist = Math.hypot(dx,dy) || 1;
      pinchState.tg.obj.scale = Math.max(SB_MIN_SCALE, Math.min(SB_MAX_SCALE, pinchState.startScale * (dist/pinchState.startDist)));
      selBoxEdit = pinchState.tg;
      selBoxApply(pinchState.tg);
      return;
    }
    if(boxMove){
      const moved = Math.hypot(e.clientX-boxMove.startClientX, e.clientY-boxMove.startClientY);
      if(moved <= TAP_MOVE_TOLERANCE && !selBoxEdit) return; // still a tap — don't nudge the item by finger jitter
      // Cache a custom sheet-model's current pixels once at the start of a real layer move.
      // Reuse the live canvas' decoded-image cache so this does not trigger image reloads.
      if(boxMove.tg.kind==='character' && !window.__ethvelleCharacterDragPreview){
        const previewCharacter=activeCharacter();
        if(previewCharacter && previewCharacter.sheet && typeof drawCharacterToContext==='function'){
          try{
            const makePreviewCanvas=()=>{
              const cv=document.createElement('canvas');
              cv.width=Math.max(1,Math.round(stageW));
              cv.height=Math.max(1,Math.round(stageH));
              return cv;
            };
            const previewCanvas=makePreviewCanvas();
            const previewCtx=previewCanvas.getContext('2d');
            const activeIndex=activeCharIdx;
            const beforeCanvas=activeIndex>0?makePreviewCanvas():null;
            const afterCanvas=activeIndex<characters.length-1?makePreviewCanvas():null;
            const beforeCtx=beforeCanvas?beforeCanvas.getContext('2d'):null;
            const afterCtx=afterCanvas?afterCanvas.getContext('2d'):null;
            // All contexts share decoded-image dictionaries; this snapshot is built once
            // at gesture start, not re-created on every pointermove.
            if(ctx){
              if(!ctx.__sheetImg)ctx.__sheetImg={};
              if(!ctx.__skinImgCache)ctx.__skinImgCache={};
              [previewCtx,beforeCtx,afterCtx].forEach(target=>{
                if(target){target.__sheetImg=ctx.__sheetImg;target.__skinImgCache=ctx.__skinImgCache;}
              });
            }
            if(previewCtx && (!beforeCanvas||beforeCtx) && (!afterCanvas||afterCtx)){
              drawCharacterToContext(previewCtx,previewCharacter);
              if(window.ethvelleRigOverlayVisible && typeof window.drawEthvelleRigOverlay==='function'){
                const anchor=previewCharacter.bones.hip;
                previewCtx.save();
                previewCtx.translate(anchor.x+modelTransform.offsetX,anchor.y+modelTransform.offsetY);
                previewCtx.rotate(modelTransform.rotation*Math.PI/180);
                previewCtx.scale(modelTransform.scale,modelTransform.scale);
                previewCtx.scale(previewCharacter.mirrorX?-1:1,previewCharacter.mirrorY?-1:1);
                previewCtx.translate(-anchor.x,-anchor.y);
                window.drawEthvelleRigOverlay(previewCtx,previewCharacter);
                previewCtx.restore();
              }
              characters.forEach((otherCharacter,index)=>{
                if(index<activeIndex && beforeCtx)drawCharacterToContext(beforeCtx,otherCharacter);
                else if(index>activeIndex && afterCtx)drawCharacterToContext(afterCtx,otherCharacter);
              });
              window.__ethvelleCharacterDragPreview={
                character:previewCharacter,canvas:previewCanvas,beforeCanvas,afterCanvas,
                startOffsetX:boxMove.startOffsetX,startOffsetY:boxMove.startOffsetY
              };
            }
          }catch(previewError){
            window.__ethvelleCharacterDragPreview=null;
          }
        }
      }
      // Movement confirms this is a transform gesture, not a transport command.
      const stageRect = pointerStageRect || document.getElementById('canvasStage').getBoundingClientRect();
      const scaleX = stageRect.width / Math.max(1, stageW);
      const scaleY = stageRect.height / Math.max(1, stageH);
      const localDX = (e.clientX - boxMove.startClientX) / Math.max(0.01, scaleX);
      const localDY = (e.clientY - boxMove.startClientY) / Math.max(0.01, scaleY);
      boxMove.tg.obj.offsetX = boxMove.startOffsetX + localDX;
      boxMove.tg.obj.offsetY = boxMove.startOffsetY + localDY;
      selBoxEdit = boxMove.tg;
      selBoxApply(boxMove.tg);
      // Restore the prior coverage-aware handoff: only a clip genuinely underneath
      // the moving selection box, with the pointer still inside the shared coverage area,
      // may take over. The handoff continues the same drag from the new clip's transform.
      handoffCanvasSelection(e);
      return;
    }
    if(!dragging || !selectedJoint) return;
    const rect = pointerCanvasRect || canvas.getBoundingClientRect();
    const stageX = rect.width ? (e.clientX-rect.left)*(stageW/rect.width) : 0;
    const stageY = rect.height ? (e.clientY-rect.top)*(stageH/rect.height) : 0;
    const raw = screenToRigSpace(stageX,stageY);
    const current = activeCharacter();
    if(current && current.type === 'custom' && current.sheet){
      const jointResult=dragSheetCharacterJoint(selectedJoint.key, raw);
      if(jointResult){
        if(jointResult==='override') sheetRigOverrideChanged=true;
        drawStick();
        return;
      }
    }
    const b = bones[selectedJoint.key];
    const start = getBoneStart(selectedJoint.key);
    b.angle = Math.atan2(raw.y-start.y, raw.x-start.x);
    drawStick();
  }
  let canvasMoveRaf=0,canvasMovePending=null;
  function flushCanvasPointerMove(event){
    if(canvasMoveRaf){cancelAnimationFrame(canvasMoveRaf);canvasMoveRaf=0;}
    const latest=event?{pointerId:event.pointerId,clientX:event.clientX,clientY:event.clientY}:canvasMovePending;
    canvasMovePending=null;
    if(latest)processCanvasPointerMove(latest);
  }
  function cancelCanvasPointerMove(){
    if(canvasMoveRaf)cancelAnimationFrame(canvasMoveRaf);
    canvasMoveRaf=0;canvasMovePending=null;
  }
  canvas.onpointermove=(e)=>{
    canvasMovePending={pointerId:e.pointerId,clientX:e.clientX,clientY:e.clientY};
    if(canvasMoveRaf)return;
    canvasMoveRaf=requestAnimationFrame(()=>{
      canvasMoveRaf=0;
      const latest=canvasMovePending;canvasMovePending=null;
      if(latest)processCanvasPointerMove(latest);
    });
  };
  const clearGestureState = (e)=>{
    if(e && e.pointerId!==undefined) activePointers.delete(e.pointerId);
    if(activePointers.size < 2) pinchState = null;
    if(activePointers.size === 0){
      const wasDraggingBone = dragging;
      const hadCharacterDragPreview = !!window.__ethvelleCharacterDragPreview;
      dragging = false;
      boxMove = null;
      pointerCanvasRect = null;
      pointerStageRect = null;
      window.__ethvelleGestureGeometry=null;
      // A move/resize gesture just finished: save ONE undo step (and, for a character, sync a
      // keyframe parked under the playhead — see finishSelBoxEdit).
      if(selBoxEdit){ const done = selBoxEdit; selBoxEdit = null; finishSelBoxEdit(done); }
      // A bone-pose drag just ended — same PART 17 rule as Transform edits: if the
      // playhead is sitting exactly on an existing keyframe, update it with the new pose
      // instead of leaving it stale.
      if(wasDraggingBone) updateCurrentKeyframeFromEditorState();
      if(sheetRigOverrideChanged){
        sheetRigOverrideChanged=false;
        persistProjectState();
      }
      if(hadCharacterDragPreview){
        window.__ethvelleCharacterDragPreview = null;
        drawStick();
      }
    }
  };
  canvas.onpointerup = (e)=>{
    // Flush the final finger position before committing so no last-frame movement is lost.
    flushCanvasPointerMove(e);
    clearGestureState(e);
  };
  canvas.onpointerleave = (e)=>{ cancelCanvasPointerMove();clearGestureState(e); };
  canvas.onpointercancel = (e)=>{ cancelCanvasPointerMove();clearGestureState(e); };
}

function lerp(a,b,t){ return a + (b-a)*t; }
// Copies a previously-snapshotted bones state back onto the LIVE bones object in place
// (mutating each joint's own angle/x/y) rather than reassigning the `bones` variable itself.
// This matters because `bones` is just a reference to the active character's own
// `c.bones` object (see syncActiveCharacterRefs) — applyPoseToBones already mutates that
// same shared object in place during a preview, so restoring correctly means writing the
// snapshotted values back into it the same way, not swapping in a detached clone that
// syncActiveCharacterRefs would immediately overwrite/ignore on its next call.
function restoreBonesSnapshot(snapshot, targetBones){
  const target = targetBones || bones;
  if(!snapshot || !target) return;
  Object.keys(target).forEach(k=>{
    if(snapshot[k]){
      if(snapshot[k].angle !== undefined) target[k].angle = snapshot[k].angle;
      if(snapshot[k].x !== undefined) target[k].x = snapshot[k].x;
      if(snapshot[k].y !== undefined) target[k].y = snapshot[k].y;
    }
  });
}
// Shortest-path angular interpolation in degrees (e.g. 350deg -> 10deg travels through
// 360/0, a 20deg turn, instead of the naive 340deg the long way around). Normalizes the
// a->b delta into -180..+180 before lerping, then lets the result run outside 0..360 —
// callers that display/store rotation don't need it wrapped back into range.
function lerpAngleDegrees(a,b,t){
  let delta = ((b-a+180) % 360 + 360) % 360 - 180;
  return a + delta*t;
}

function evalPreset(preset, tFrac){
  const source=Array.isArray(preset&&preset.keyframes)?preset.keyframes:[];
  const kfs=source.filter(k=>k&&Number.isFinite(Number(k.t))&&k.pose&&typeof k.pose==='object')
    .map(k=>({t:Math.max(0,Math.min(1,Number(k.t))),pose:k.pose})).sort((a,b)=>a.t-b.t);
  if(!kfs.length)return {};
  if(kfs.length===1)return {...kfs[0].pose};
  const samePose=(a,b)=>{
    const keys=new Set([...Object.keys(a||{}),...Object.keys(b||{})]);
    for(const k of keys)if(Math.abs((Number(a&&a[k])||0)-(Number(b&&b[k])||0))>1e-8)return false;
    return true;
  };
  // A repeated t=1 copy of the t=0 pose is a loop seam, not an extra stop point.
  if(kfs.length>2&&kfs[0].t<=1e-8&&kfs[kfs.length-1].t>=1-1e-8&&samePose(kfs[0].pose,kfs[kfs.length-1].pose))kfs.pop();
  if(kfs.length===1)return {...kfs[0].pose};
  const n=kfs.length,t=((Number(tFrac)||0)%1+1)%1;
  let ai=0,bi=1,t1=kfs[0].t,t2=kfs[1].t;
  if(t<kfs[0].t&&kfs[0].t>0){ai=n-1;bi=0;t1=kfs[ai].t-1;t2=kfs[0].t;}
  else if(t>=kfs[n-1].t&&kfs[n-1].t<1){ai=n-1;bi=0;t1=kfs[ai].t;t2=kfs[0].t+1;}
  else{
    ai=0;
    while(ai<n-2&&t>kfs[ai+1].t)ai++;
    bi=ai+1;t1=kfs[ai].t;t2=kfs[bi].t;
  }
  const prevIndex=(ai-1+n)%n,nextIndex=(bi+1)%n;
  const p0=kfs[prevIndex],p1=kfs[ai],p2=kfs[bi],p3=kfs[nextIndex];
  // Shift the neighbour timestamps into the same loop cycle as this segment.
  // There are two distinct seam cases: just BEFORE the first keyframe (previous
  // neighbour belongs to the prior cycle), and just AFTER the last keyframe
  // (next neighbour belongs to the following cycle).
  const beforeFirst=(ai===n-1&&t<kfs[0].t);
  const afterLast=(ai===n-1&&t>=kfs[n-1].t);
  const t0=(ai===0||beforeFirst)?p0.t-1:p0.t;
  const t3=(bi===n-1||afterLast)?p3.t+1:p3.t;
  const span=Math.max(1e-6,t2-t1),u=Math.max(0,Math.min(1,(t-t1)/span));
  const u2=u*u,u3=u2*u;
  const h00=2*u3-3*u2+1,h10=u3-2*u2+u,h01=-2*u3+3*u2,h11=u3-u2;
  const keys=new Set([...Object.keys(p0.pose||{}),...Object.keys(p1.pose||{}),...Object.keys(p2.pose||{}),...Object.keys(p3.pose||{})]);
  const value=(frame,key)=>{const v=Number(frame&&frame.pose&&frame.pose[key]);return Number.isFinite(v)?v:0;};
  const pose={};
  keys.forEach(key=>{
    const v0=value(p0,key),v1=value(p1,key),v2=value(p2,key),v3=value(p3,key);
    const m1=(v2-v0)/Math.max(1e-6,t2-t0),m2=(v3-v1)/Math.max(1e-6,t3-t1);
    pose[key]=h00*v1+h10*span*m1+h01*v2+h11*span*m2;
  });
  return pose;
}

// Give older/admin-authored presets natural follow-through even when their keyframes only
// animate the large limbs. Explicit elbow/wrist/knee keys always win over these derived values.
function completeNaturalJointPose(pose, loopT, animKey){
  const out={...(pose||{})};
  if(animKey==='default')return out;
  const D=Math.PI/180,phase=2*Math.PI*(Number(loopT)||0);
  const primary=['spine','head','armL','armR','legL','legR'];
  if(!primary.some(k=>Math.abs(Number(out[k])||0)>.0005))return out;
  const has=k=>Object.prototype.hasOwnProperty.call(out,k);
  ['L','R'].forEach(side=>{
    const phaseSide=phase+(side==='L'?0:Math.PI);
    const arm='arm'+side,forearm=arm+'_lo',hand='hand'+side;
    const leg='leg'+side,shin=leg+'_lo';
    const a=Number(out[arm])||0,l=Number(out[leg])||0;
    if(!has(forearm))out[forearm]=-a*.16+1.25*D*Math.sin(phaseSide);
    const f=Number(out[forearm])||0;
    if(!has(hand))out[hand]=-(f-a)*.10+.55*D*Math.sin(phaseSide*2);
    if(!has(shin))out[shin]=-l*.12+2.0*D*Math.max(0,Math.sin(phaseSide));
  });
  if(!has('head'))out.head=-(Number(out.spine)||0)*.14+.45*D*Math.sin(phase);
  if(!has('hipY')){
    const gait=Math.max(Math.abs(Number(out.legL)||0),Math.abs(Number(out.legR)||0));
    if(gait>.015)out.hipY=-Math.min(.012,gait*.012)*(.45+.55*Math.cos(phase*2));
  }
  return out;
}

// Applies a computed pose onto a rig's bones. Defaults to the shared bones/baseAngles/
// hipRestY (the active character) when called with no extra arguments, exactly as before —
// pass an explicit (targetBones, targetBaseAngles, targetHipRestY) to pose a DIFFERENT,
// non-active character's own rig instead (see animateAllCharacters), without touching the
// active character's live bones at all.
function applyPoseToBones(pose, targetBones, targetBaseAngles, targetHipRestY){
  const b=targetBones!==undefined?targetBones:bones;
  const ba=targetBaseAngles!==undefined?targetBaseAngles:baseAngles;
  const restY=targetHipRestY!==undefined?targetHipRestY:hipRestY;
  const clamp=(v,min,max)=>Math.max(min,Math.min(max,v));
  Object.keys(b).forEach(k=>{
    if(ba[k]===undefined) return;
    const delta=Number(pose[k])||0;
    let limit=165*Math.PI/180;
    if(k==='spine') limit=27*Math.PI/180;
    else if(k==='head') limit=38*Math.PI/180;
    else if(k.startsWith('leg')&&!k.endsWith('_lo')) limit=112*Math.PI/180;
    else if(k.endsWith('_lo')&&k.startsWith('leg')) limit=142*Math.PI/180;
    else if(/^hand[LR]$/.test(k)) limit=95*Math.PI/180;
    else if(/^((thumb|index|middle|ring|pinky)[LR]_[12])$/.test(k)) limit=105*Math.PI/180;
    b[k].angle=ba[k]+clamp(delta,-limit,limit);
  });
  // Gesture curves apply to fingertip bones without changing limb length or the torso bind pose.
  const fist=clamp(Number(pose.gestureFist)||0,0,1), point=clamp(Number(pose.gesturePoint)||0,0,1), peace=clamp(Number(pose.gesturePeace)||0,0,1), thumbs=clamp(Number(pose.gestureThumbsUp)||0,0,1);
  const fingerNames=['thumb','index','middle','ring','pinky'];
  ['L','R'].forEach(side=>fingerNames.forEach((f,i)=>{
    for(let seg=1;seg<=2;seg++){
      const k=f+side+'_'+seg;
      if(!b[k]||ba[k]===undefined) continue;
      const sign=side==='L'?1:-1;
      let bend=(seg===1?43:57)*Math.PI/180*fist*sign;
      if(f!=='index') bend += (seg===1?38:48)*Math.PI/180*point*sign;
      if(f!=='index'&&f!=='middle') bend += (seg===1?40:50)*Math.PI/180*peace*sign;
      if(f!=='thumb') bend += (seg===1?42:55)*Math.PI/180*thumbs*sign;
      if(f==='thumb') bend -= sign*95*Math.PI/180*thumbs;
      b[k].angle=ba[k]+clamp((Number(pose[k])||0)+bend,-120*Math.PI/180,120*Math.PI/180);
    }
  }));
  // Keep jumping on a readable arc: no extreme root travel that makes the character look detached.
  if(pose.hipY!==undefined && Number.isFinite(restY)) b.hip.y=restY+clamp(Number(pose.hipY)||0,-0.13,0.035)*stageH;
  else if(Number.isFinite(restY)) b.hip.y=restY;
}

// independent segment (own trim handles, own delete) rather than mutating activeClip in place.

let splitStickmanSegments = [];

// Resolves the actual clip object (activeClip or one of splitStickmanSegments) behind a
// selection ref like 'main' or 'seg1'. Returns null for background/overlay refs (or
// anything else that isn't a stickman clip) — those don't carry keyframes/transform.
function getClipByRef(ref){
  if(ref === 'main') return activeClip;
  if(typeof ref === 'string' && ref.startsWith('seg')){
    const idx = parseInt(ref.slice(3), 10);
    return splitStickmanSegments[idx] || null;
  }
  return null;
}
// Returns every stickman-track clip (the main clip plus every split-off/appended segment)
// in left-to-right timeline order. Each entry carries a stable `ref` ('main' or 'segN')
// so callers can re-select or mutate the exact clip object it came from. Defaults to the
// currently-ACTIVE character's clip/segments (the plain shared variables) when called with
// no arguments, exactly as before — pass an explicit (clip, segments) pair (e.g. from
// animateAllCharacters) to inspect a DIFFERENT, non-active character's own clips instead.
function getAllStickmanClips(clip, segments){
  const useClip = clip !== undefined ? clip : activeClip;
  const useSegments = segments !== undefined ? segments : splitStickmanSegments;
  const list = [];
  if(useClip) list.push({ ref:'main', clip:useClip });
  (useSegments||[]).forEach((seg, idx)=> list.push({ ref:'seg'+idx, clip:seg }));
  list.sort((a,b)=> a.clip.trimStart - b.clip.trimStart);
  return list;
}
// Total playable length across every stickman clip on the track (their combined span),
// used so Play actually runs through every clip back-to-back instead of just the first.
// Same optional (clip, segments) override as getAllStickmanClips.
function combinedStickmanDuration(clip, segments){
  const all = getAllStickmanClips(clip, segments);
  if(!all.length) return 2.0;
  return all.reduce((max, {clip})=> Math.max(max, clip.trimEnd), 0);
}
// Total playable length across EVERY character's own clips, not just the active one — this
// is what playback/export must loop/run against now that each character animates
// independently, so a longer clip on a non-active character is never cut short just because
// the active character's own clip happens to be shorter (or vice versa).
function combinedAllCharactersDuration(){
  let max = 0;
  characters.forEach(c=>{
    if(!c.clip) return;
    max = Math.max(max, combinedStickmanDuration(c.clip, c.segments));
  });
  return max || 2.0;
}
// BUGFIX (couldn't scrub/drag past 2s with no animation selected): setCurrentAbsoluteTime
// and timeFromDragAnchor used to clamp the playhead to combinedStickmanDuration()/
// combinedAllCharactersDuration() alone — BOTH of those hard-return a bare 2.0 the moment
// there's no character/animation clip at all, completely ignoring `timelineDuration` (which
// already tracks how far the Background/Clips/Audio rows actually extend — see
// syncTimelineInnerWidth, trimPanelSetEnd, addMediaOverlay, etc.). So on a project with,
// say, a 6s background/audio clip but zero animation yet, the timeline VISUALLY drew out to
// 6s+ (tracks/ruler both sized off timelineDuration) but the playhead itself could never be
// dragged, scrubbed, or tapped past the 2s mark — every gesture silently clamped back.
// totalProjectDuration() is now the one authoritative "how far can the playhead go" bound:
// the longer of (a) every character's own animation extent and (b) timelineDuration itself,
// so background/overlay/audio-only projects (no animation yet) are fully scrubbable across
// their real length, and adding a longer animation later still extends things correctly.
function totalProjectDuration(){
  return Math.max(combinedAllCharactersDuration(), timelineDuration);
}
// Exact playback endpoint: unlike timelineDuration this never includes ruler/track padding
// and never inherits the two-second fallback used by the legacy character-duration helpers.
// It includes every character's main/split clips and every background/overlay type.
function getProjectContentEndSec(){
  let endSec=0;
  const includeClip=clip=>{
    if(!clip||typeof clip!=='object')return;
    const trimEnd=Number(clip.trimEnd);
    if(Number.isFinite(trimEnd))endSec=Math.max(endSec,trimEnd);
  };
  (Array.isArray(characters)?characters:[]).forEach(character=>{
    if(!character)return;
    includeClip(character.clip);
    (Array.isArray(character.segments)?character.segments:[]).forEach(includeClip);
  });
  // Include the active aliases as a defensive fallback for older/restored projects whose
  // active character references have not yet been synchronized to the characters array.
  if(typeof activeClip!=='undefined')includeClip(activeClip);
  if(typeof splitStickmanSegments!=='undefined'&&Array.isArray(splitStickmanSegments))splitStickmanSegments.forEach(includeClip);
  (Array.isArray(backgrounds)?backgrounds:[]).forEach(includeClip);
  (Array.isArray(overlays)?overlays:[]).forEach(includeClip);
  return Math.max(0.1,endSec);
}
// One exact duration source for export estimates, credit charging, progress readouts and encoding.
function getExportContentDurationSec(){
  try{return Math.max(0.1,Number(typeof getProjectContentEndSec==='function'?getProjectContentEndSec():0)||0.1);}
  catch(_e){return 0.1;}
}
// Finds whichever stickman clip covers a given ABSOLUTE timeline second (falls back to the
// nearest clip if the time sits in a gap between clips, so playback never just goes blank).
// Same optional (clip, segments) override as getAllStickmanClips.
function stickmanClipAt(absSec, clip, segments){
  const all = getAllStickmanClips(clip, segments);
  if(!all.length) return null;
  let hit = all.find(({clip})=> absSec >= clip.trimStart && absSec < clip.trimEnd);
  if(hit) return hit;
  // gap: pick the nearest clip in time rather than showing nothing
  let nearest = all[0], bestDist = Infinity;
  all.forEach(entry=>{
    const d = absSec < entry.clip.trimStart ? entry.clip.trimStart - absSec : absSec - entry.clip.trimEnd;
    if(d < bestDist){ bestDist = d; nearest = entry; }
  });
  return nearest;
}

function setActiveAnimation(key){
  currentAnimKey = key;
  hipRestY = bones.hip.y;
  if(activeCharacter()) activeCharacter().hipRestY = hipRestY;
  activeClip = {
    animKey: key,
    trimStart: 0,
    trimEnd: DEFAULT_CLIP_LEN,
    keyframes: [],
    speed: 1
  };
  // activeClip was just reassigned to a brand new object, so the active character's own
  // `clip` field (and not just the plain shared variable) must be pointed at it too, or the
  // new clip would vanish the moment syncActiveCharacterRefs() runs again (e.g. switching
  // characters and back) since the character object would still show its old/empty clip.
  syncClipRefsToCharacter();
  timelineDuration = Math.max(2.0, activeClip.trimEnd);
  frame = 0;
  currentAbsoluteTimeSec = 0;
  updateTimeReadout();
  renderTimelineClip();
  movePlayhead();
  drawStick();
  onClipSelected(activeClip,'add');
}

// Add a new animation clip according to the playhead's half of the current clip:
// first half = insert before/left, second half = insert after/right. Clips after the
// insertion point shift as a group, keeping all character clips independent and sequential.
function appendAnimationClip(key){
  if(!activeClip){setActiveAnimation(key);return;}
  const all=getAllStickmanClips().sort((a,b)=>(Number(a.clip.trimStart)||0)-(Number(b.clip.trimStart)||0));
  if(!all.length){setActiveAnimation(key);return;}
  const lastEnd=all.reduce((m,x)=>Math.max(m,Number(x.clip.trimEnd)||0),0);
  let playhead=0;try{playhead=Math.max(0,Number(typeof getCurrentAbsoluteTime==='function'?getCurrentAbsoluteTime():currentAbsoluteTimeSec)||0);}catch(_e){}
  let insertAt=lastEnd,anchor=all.find(x=>playhead>=Number(x.clip.trimStart)-.001&&playhead<Number(x.clip.trimEnd)-.001);
  if(anchor){const st=Number(anchor.clip.trimStart)||0,en=Number(anchor.clip.trimEnd)||st;insertAt=playhead<(st+en)/2?st:en;}
  else{
    const next=all.find(x=>Number(x.clip.trimStart)>=playhead-.001);
    if(next)insertAt=Math.max(0,Number(next.clip.trimStart)||0);
  }
  const len=Math.max(MIN_CLIP_SEC,Number(DEFAULT_CLIP_LEN)||2);
  all.forEach(({clip})=>{
    if(Number(clip.trimStart)>=insertAt-.001){
      clip.trimStart=Number(clip.trimStart)+len;clip.trimEnd=Number(clip.trimEnd)+len;
      if(Array.isArray(clip.keyframes))clip.keyframes=clip.keyframes.map(k=>{const kt=Number(k.t);return Number.isFinite(kt)?{...k,t:kt+len}:{...k};});
    }
  });
  const newClip={animKey:key,trimStart:insertAt,trimEnd:insertAt+len,keyframes:[],speed:1};
  splitStickmanSegments.push(newClip);
  splitStickmanSegments.sort((a,b)=>(Number(a.trimStart)||0)-(Number(b.trimStart)||0));
  syncClipRefsToCharacter();
  timelineDuration=Math.max(Number(timelineDuration)||0,...getAllStickmanClips().map(x=>Number(x.clip.trimEnd)||0));
  renderTimelineClip();syncTimelineInnerWidth();renderRuler();
  const newIdx=splitStickmanSegments.indexOf(newClip),newSegId='seg'+newIdx;
  onClipSelected(newSegId,'add');
}

/* ==================================================================================
   CANONICAL TIMELINE CLOCK + CENTRAL PLAYBACK CONTROLLER
   ==================================================================================
   ROOT CAUSE #1 (playback/pause race): the old code had `isPlaying=false;
   clearInterval(playInterval)` copy-pasted into ~10 different handlers (clip tap, keyframe
   tap, seekFrame, split, export, ...), each re-implementing "pause" slightly differently,
   plus a completely separate setInterval(tickFrame, 1000/FPS) clock that free-runs on its
   own timer independent of real elapsed time. Any of those call sites could race the next
   interval tick, so a tap could land "one tick behind" or get silently overwritten.
   FIX: exactly one requestAnimationFrame-driven clock (startPlayback/pausePlayback below),
   and every other function in the file now calls pausePlayback()/togglePlayback() instead
   of touching isPlaying/playInterval directly.

   ROOT CAUSE #2 (time representation): `frame` used to mean "frames since
   activeClip.trimStart" — but activeClip is just whichever clip happens to be in the
   activeClip slot, NOT necessarily the clip that's selected/playing/being scrubbed. Once
   split segments exist, activeClip.trimStart is the WRONG offset for anything happening on
   a segment, which is what caused split-segment keyframes/taps to resolve against the
   wrong clip.
   FIX: currentAbsoluteTimeSec is now the one true clock (absolute timeline seconds, t=0 at
   the very start of the whole timeline, matching how keyframe.t and clip.trimStart/trimEnd
   are already stored). `frame` is kept as a derived, backward-compatible view of it
   (frame = round((currentAbsoluteTimeSec - activeClip.trimStart) * FPS)) so the many
   existing call sites that still read `frame` keep working, but every NEW/fixed call site
   below reads/writes currentAbsoluteTimeSec directly via the helpers.
   ==================================================================================== */
let isPlaying=false;
let frame=0; // DERIVED/LEGACY: frames since activeClip.trimStart. Kept in sync by setCurrentAbsoluteTime(). Do not assign this directly in new code — use setCurrentAbsoluteTime()/seekToFrame().
let currentAbsoluteTimeSec = 0; // CANONICAL CLOCK: absolute timeline seconds, t=0 at timeline start.
let playbackRaf = null;
let playbackStartedAtMs = 0;    // performance.now() at the moment playback last (re)started
let playbackStartTimeSec = 0;   // currentAbsoluteTimeSec at that same moment
let evVideoBufferStall = null;  // { id, boundarySec, startedAtMs, paused } — one clock gate at a time
let evVideoGateScanFromSec = 0;
const EV_VIDEO_BUFFER_STALL_CAP_MS = 200; // Safety ceiling for any legacy gate; playback no longer waits on incoming video frames.
const EV_VIDEO_CLOCK_EPS = 0.0005;
// v150 debug flag: when true, logs every LIVE<->HOLD overlay-video transition and every
// playback clock gate stall (with its reason and duration) to the console. Default off.
window.__evVideoDebug = window.__evVideoDebug || false;
// BUGFIX (video frame hidden + no audio after adding a video clip): every alignment gate used a
// flat 0.04s tolerance. A <video> that is genuinely playing always trails the rAF timeline clock by
// ~40-150ms (play() start latency + decoder/audio clock), so the gates failed on every frame and
// evSafePlayVideo paused/muted/re-seeked the clip ~15x/second. Exact alignment (0.04s) is only
// required while the element is NOT playing (start, cut boundary, scrub). Once it is playing, the

const EV_VIDEO_ALIGN_TOL_START = 0.04;

function evVideoAlignTolerance(el){
  return (isPlaying&&el&&el.tagName==='VIDEO'&&!el.paused&&!el.seeking)
    ?EV_VIDEO_ALIGN_TOL_PLAYING:EV_VIDEO_ALIGN_TOL_START;
}

  const el=evGetOverlayVideoElement(ov);
  const t=evVideoBufferStall&&String(evVideoBufferStall.id)===String(ov&&ov.id)
    ?Number(evVideoBufferStall.boundarySec):Number(currentAbsoluteTimeSec)||0;
  return evOverlayVideoFrameIsPresentable(ov,el,t);
}
function evFindUnreadyOverlayVideoBoundary(fromSec,toSec){
  if(!Array.isArray(overlays)||toSec<fromSec)return null;
  let found=null;
  for(let i=0;i<overlays.length;i++){
    const ov=overlays[i];
    if(!ov||ov.type!=='video'||ov.reversePlayback||!ov.src)continue;
    const start=Number(ov.trimStart)||0;
    const end=Number(ov.trimEnd)||0;
    if(end<=start||start<=fromSec+EV_VIDEO_CLOCK_EPS||start>toSec+EV_VIDEO_CLOCK_EPS)continue;
    if(evIsOverlayVideoPresentable(ov))continue;
    if(!found||start<found.boundarySec)found={clip:ov,boundarySec:start};
  }
  return found;
}

// Reads the true "now" of the timeline. During a video-buffer gate it returns the exact
// clip boundary, so pause/tap handlers cannot commit a time that has already run past it.
function getCurrentAbsoluteTime(){
  // The canonical clock never waits for an incoming decoder. Video compositing holds the
  // last outgoing frame until the new decoder is ready, while audio remains clock-scheduled.
  if(isPlaying)return playbackStartTimeSec+(performance.now()-playbackStartedAtMs)/1000;
  return currentAbsoluteTimeSec;
}
// Sets the canonical clock and derives the legacy `frame` counter from it. This is the
// ONLY function that should ever assign currentAbsoluteTimeSec outside of the rAF tick
// itself. Does not render — callers render explicitly so scrub/seek can batch a redraw.
function setCurrentAbsoluteTime(tSec){
  // BUGFIX: was combinedStickmanDuration() — see totalProjectDuration()'s comment above for
  // why that silently capped every seek/scrub at 2s whenever no animation was selected yet.
  const cachedScrubDuration = window.__ethvelleTimelineScrubbing ? Number(window.__ethvelleTimelineScrubMaxDuration) : NaN;
  const dur = Number.isFinite(cachedScrubDuration) && cachedScrubDuration>0 ? cachedScrubDuration : totalProjectDuration();
  const boundedTime=Math.max(0,Math.min(dur,Number(tSec)||0));
  // Media clips are active on [trimStart, trimEnd), so a finger released exactly at the
  // project end used to seek one instant PAST the final video frame and clear the canvas.
  // Settle end-of-project seeks on the last renderable frame (one frame step before trimEnd).
  // This is shared by tap-seek, drag-scrub, and playback-end paths, so none can expose a blank
  // canvas merely because the clock landed on an exclusive clip boundary.
  const contentEnd=(typeof getProjectContentEndSec==='function')?Number(getProjectContentEndSec()):dur;
  const frameStep=1/Math.max(1,Number(FPS)||24);
  const hasRealContent=Number.isFinite(contentEnd)&&contentEnd>0.100001;
  const lastVisibleTime=hasRealContent?Math.max(0,contentEnd-frameStep):boundedTime;
  const nextTime=(hasRealContent&&boundedTime>=contentEnd-0.000001)
    ?Math.min(boundedTime,lastVisibleTime):boundedTime;

  // An actual seek/scrub away from a gated boundary cancels that gate. A pause/resume at
  // the same boundary deliberately preserves it so the pending decoder can finish loading.
  if(evVideoBufferStall&&(window.__ethvelleTimelineScrubbing||Math.abs(nextTime-evVideoBufferStall.boundarySec)>EV_VIDEO_CLOCK_EPS)){
    evVideoBufferStall=null;
    evVideoGateScanFromSec=nextTime;
    evClearHeldOutgoingOverlayFrames();
  }

  currentAbsoluteTimeSec=nextTime;
  const trimStart = activeClip ? activeClip.trimStart : 0;
  frame = Math.max(0, Math.round((currentAbsoluteTimeSec - trimStart) * FPS));
}
function getFrameFromTime(tSec){
  const trimStart = activeClip ? activeClip.trimStart : 0;
  return Math.max(0, Math.round((tSec - trimStart) * FPS));
}
function getCurrentClipAtTime(tSec){
  return stickmanClipAt(tSec);
}
function getClipLocalTime(clip, tSec){
  if(!clip) return 0;
  return Math.max(0, tSec - clip.trimStart);
}

function currentPresetDuration(){
  // Total playable length across ALL stickman clips (main clip + every split/appended
  // segment) so Play runs the full timeline instead of stopping after just the first
  // clip/segment. `frame` is measured from activeClip.trimStart (existing convention used
  // throughout scrubbing/keyframes), so the playable span is measured from there too.
  if(activeClip) return Math.max(0.05, combinedStickmanDuration() - activeClip.trimStart);
  return 2.0;
}
function totalFramesForCurrent(){
  return Math.max(1, Math.round(currentPresetDuration()*FPS));
}

// ---- central playback controller ----
// Starts (or resumes) playback from whatever currentAbsoluteTimeSec currently is. Anchors
// the real-time clock (performance.now()) to that instant so every subsequent rAF tick
// derives elapsed time mathematically instead of assuming a fixed 1000/FPS interval —
// this is what keeps playback synchronized with real elapsed time even if a frame callback
// is late/dropped, and is why we no longer use setInterval as the playback clock.
function startPlayback(){
  // Media-only projects must play normally too; a character rig is not required for
  // image, video, text, background, or audio clips on the shared timeline.
  const hasPlayableMedia = (Array.isArray(overlays) && overlays.some(o=>o && ['image','video','text','audio','shape','color'].includes(o.type)))
    || (Array.isArray(backgrounds) && backgrounds.length > 0);
  if(!activeClip && !hasPlayableMedia){
    showToast('Add a clip or choose an animation before playback', {icon: toastIcons.info});
    return;
  }
  if(isPlaying) return; // idempotent: calling start while already playing is a no-op
  const projectEnd=getProjectContentEndSec();
  const endRestartTolerance=1/Math.max(1,Number(FPS)||30);
  if(Number(currentAbsoluteTimeSec)>=projectEnd-endRestartTolerance){
    evVideoBufferStall=null;
    evVideoGateScanFromSec=0;
    try{evClearHeldOutgoingOverlayFrames();}catch(_e){}
    setCurrentAbsoluteTime(0);
  }
  // Starting playback always clears any lingering clip/keyframe selection (see historical
  // note preserved from the old togglePlay: picking a keyframe then pressing Play used to
  // leave the contextual toolbar armed underneath a running playback).
  if(selectedClip || selectedKfIdx !== null) deselectAllClips();
  isPlaying = true;
  const startedAt=performance.now();
  playbackStartedAtMs=startedAt;
  playbackStartTimeSec=currentAbsoluteTimeSec;

  // Decoder readiness is deliberately non-blocking. The renderer retains the outgoing
  // frame at cuts and the Web Audio scheduler pre-schedules sound independently.
  evVideoBufferStall=null;
  evVideoGateScanFromSec=currentAbsoluteTimeSec;

  // Start pre-rolling immediately on the Play path, not after the first rendered frame.
  try{evWarmLastCheckMs=0;warmUpcomingOverlayVideos(currentAbsoluteTimeSec);}catch(_e){}

  const icon = document.getElementById('playIcon');
  if(icon) icon.innerHTML = '<path d="M6 5h4v14H6zM14 5h4v14h-4z"/>';
  playbackRaf = requestAnimationFrame(playbackTick);
}
// Stop on the final content frame. Keep the playhead one frame-step inside the last clip
// because visual clip activity is represented by [trimStart, trimEnd), not an inclusive end.
function stopPlaybackAtProjectEnd(endSec){
  const end=Math.max(0.1,Number(endSec)||getProjectContentEndSec());
  const frameStep=1/Math.max(1,Number(FPS)||30);
  const lastVisibleTime=Math.max(0,end-Math.min(frameStep,Math.max(0.001,end*0.25)));
  if(playbackRaf!==null){try{cancelAnimationFrame(playbackRaf);}catch(_e){}playbackRaf=null;}
  isPlaying=false;
  playbackStartedAtMs=0;
  playbackStartTimeSec=lastVisibleTime;
  evVideoBufferStall=null;
  evVideoGateScanFromSec=end;
  setCurrentAbsoluteTime(lastVisibleTime);
  try{renderFrame();}catch(err){console.warn('Ethvelle final-frame render:',err);}

  // Pause attached and detached media alike (audio overlays are created with new Audio()).
  const media=new Set();
  const addMedia=value=>{if(value&&typeof value.pause==='function')media.add(value);};
  try{document.querySelectorAll('video,audio').forEach(addMedia);}catch(_e){}
  [overlayMediaEls,overlayWarmEls,audioOverlayEls].forEach(group=>{
    if(group&&typeof group==='object')Object.keys(group).forEach(id=>addMedia(group[id]));
  });
  try{
    const held=typeof renderOverlayMedia==='function'&&renderOverlayMedia.__evHeldOutgoing;
    if(held)Object.keys(held).forEach(id=>addMedia(held[id]&&held[id].fallbackVideo));
  }catch(_e){}
  media.forEach(el=>{
    try{el.pause();}catch(_e){}
    if(el.dataset&&el.dataset.evAudioOverlayId){
      el.dataset.evAudioNeedsSync='1';
      el.dataset.evAudioTimelineActive='0';
      el.__evAudioRunToken=(Number(el.__evAudioRunToken)||0)+1;
    }
  });
  const icon=document.getElementById('playIcon');
  if(icon)icon.innerHTML='<path d="M8 5v14l11-7z"/>';
}
// The ONE playback loop in the file. Every tick derives time from real elapsed time and
// stops at the exact content endpoint; the timeline does not wrap or restart automatically.
function playbackTick(timestamp){
  if(!isPlaying)return;
  const nowMs=Number.isFinite(timestamp)?timestamp:performance.now();
  const t=playbackStartTimeSec+(nowMs-playbackStartedAtMs)/1000;
  const contentEnd=getProjectContentEndSec();
  if(contentEnd>0&&t>=contentEnd){
    stopPlaybackAtProjectEnd(contentEnd);
    return;
  }
  // Do not gate the canonical clock on a video frame. renderOverlayMedia() snapshots the
  // outgoing visual and keeps it on stage until the incoming decoder can present its frame.
  setCurrentAbsoluteTime(t);
  renderFrame();
  playbackRaf=requestAnimationFrame(playbackTick);
}
// Freezes playback at the EXACT current instant, calculated from the real clock first
// (not "whatever currentAbsoluteTimeSec happened to be on the last rAF tick", which can be
// up to one frame stale/behind the actual tap). This is what makes mobile taps feel
// deterministic: update time -> commit it -> stop the loop -> render that exact time once
// more -> flip the icon. Idempotent: calling this repeatedly (or while already paused) is
// always safe and a no-op beyond re-rendering the same frozen time.
function pausePlaybackAtCurrentTime(){
  const exactTime = getCurrentAbsoluteTime(); // 1. update current time from performance.now()
  setCurrentAbsoluteTime(exactTime);          // 2. commit currentAbsoluteTimeSec (+ legacy `frame`)
  if(playbackRaf !== null){                   // 3. stop RAF
    cancelAnimationFrame(playbackRaf);
    playbackRaf = null;
  }
  isPlaying = false;                          // 4. set isPlaying false
  if(evVideoBufferStall){
    // Keep the pending boundary and outgoing picture, but suspend the timeout while paused.
    evVideoBufferStall.paused=true;
    evVideoBufferStall.startedAtMs=0;
  }
  renderFrame();                              // 5. render that exact time one final time
  if(evVideoBufferStall)evVideoBufferStall.rendered=true;
  const icon = document.getElementById('playIcon'); // 6. update UI/play icon
  if(icon) icon.innerHTML = '<path d="M8 5v14l11-7z"/>';
  // 7. position is already preserved — currentAbsoluteTimeSec was committed in step 2
}
// Public pause entry point. Every system in the file that needs to stop playback calls
// this (never isPlaying=false/clearInterval directly) so pause behavior is identical and
// centralized everywhere: clip tap, background tap, overlay tap, keyframe tap, timeline
// tap, timeline seek, split, trim, export, editor navigation, transport controls.
function pausePlayback(){
  if(!isPlaying) return; // idempotent: safe to call even when already paused
  pausePlaybackAtCurrentTime();
}
// togglePlayback means exactly what it says — invert current state. It is intentionally
// the ONLY place isPlaying is inverted; nothing else should use this as a "pause" helper
// (see the historical `if(isPlaying) togglePlay();` anti-pattern this replaces).
function togglePlayback(){
  if(isPlaying) pausePlayback();
  else startPlayback();
}

// Evaluate the pose at an absolute timeline time (seconds from the very start of the
// timeline, t=0), looking up whichever clip covers that instant (main clip or any
// split/appended segment) so playback runs continuously across all of them, then
// blended with that clip's own user keyframes. Defaults to the currently-ACTIVE character's
// clip/segments when called with no extra arguments (exactly as before, for every existing
// call site) — pass an explicit (clip, segments) pair to evaluate a DIFFERENT character's
// own animation instead (see animateAllCharacters), without touching the active one's state.
function poseAtClipTime(tSecAbsolute, clipOverride, segmentsOverride){
  const useClip = clipOverride !== undefined ? clipOverride : activeClip;
  const useSegments = segmentsOverride !== undefined ? segmentsOverride : splitStickmanSegments;
  if(!useClip) return null;
  const hit = stickmanClipAt(tSecAbsolute, useClip, useSegments);
  if(!hit) return null;
  const clip = hit.clip;
  const preset = animPresets[clip.animKey];
  if(!preset) return null;
  // Time relative to the covering clip's own trimStart, looping the underlying preset.
  // Speed (see applyClipSpeed) scales how fast we move through the preset's own loop —
  // 2x speed means clipLocalT advances through the preset twice as fast (more loops per
  // second of timeline time), 0.5x means half as fast — without touching the preset's own
  // authored keyframe data at all.
  const speed = clip.speed || 1;
  const clipLocalT = Math.max(0, (tSecAbsolute - clip.trimStart) * speed);
  const loopT = (clipLocalT % preset.duration) / preset.duration;
  let pose = evalPreset(preset, loopT);

  // user-added keyframes override/blend on top, sorted by time. A single keyframe only
  // nudges the pose right at that instant (crossfading in/out over a short window) rather
  // than clamping the ENTIRE clip to one frozen pose — previously one keyframe made the
  // whole clip look "stopped" because both the before- and after-edges resolved to the
  // exact same static pose with nothing to blend toward.
  //
  // BUGFIX (2nd+ keyframe freezes the animation): with 2+ keyframes, everything BEFORE the
  // first keyframe and everything AFTER the last keyframe used to be permanently clamped to
  // that keyframe's frozen pose for the rest of the clip, not just a short window around it.
  // FIX: reuse the exact same hold-then-blend-back-to-preset treatment the single-keyframe
  // case already had, applied at both the first and last keyframe's outward-facing edge.
  //
  // BUGFIX (animation crawls to ~0.2-0.5x speed between two keyframes): the "between two
  // keyframes" branch used to linearly interpolate directly from kf A's frozen pose to kf B's
  // frozen pose across the ENTIRE span between them (localT going 0->1 over however many
  // seconds apart the two keyframes were). Each keyframe's pose is just a snapshot of
  // whatever instant of the walk/wave/jump cycle it was dropped on — so a full multi-phase
  // step cycle that normally plays out with its own timing got flattened into one slow,
  // linear glide from pose A to pose B, which reads as the whole animation suddenly playing
  // back in slow motion. FIX: apply the SAME short hold-near-each-keyframe treatment used at
  // the clip's outer edges to the gap BETWEEN two keyframes too — each keyframe only asserts
  // itself for a brief window (HOLD, then eased back out over another HOLD) around its own
  // timestamp; everywhere else in the gap just keeps sampling the normal preset loop at full
  // speed, exactly as if no keyframe were there. Two keyframes placed close together (closer
  // than 2*HOLD apart) still cross-fade smoothly between their own hold windows with no gap
  // of bare preset motion in between, so short nudges still feel continuous.
  if(clip.keyframes.length){
    const kfs = [...clip.keyframes].sort((a,b)=>a.t-b.t);
    // BUGFIX (animation freezes exactly at the keyframe): holdAndBlend used to fully clamp
    // the pose to the keyframe's exact snapshot — no motion at all — for a flat 2*HOLD-wide
    // window centered on the keyframe's own timestamp, then blend back out. That flat window
    // IS a freeze: the stickman visibly stops dead right at (and for a moment around) every
    // keyframe, while playing normally just before/after that window — exactly the reported
    // bug. FIX: there is no flat window anymore. Influence now falls off continuously (a
    // smooth cosine ease) from full strength at the single instant dist===0 down to zero at
    // the window edge, so the keyframe's pose is only ever fully hit for an instant, not held
    // — the rig keeps moving through it instead of pausing on it.
    const WINDOW = 0.36; // seconds of influence on each side of a keyframe (same reach as before)

    // Returns null when tSecAbsolute is further than WINDOW from kf.t (this keyframe has no
    // influence here) — otherwise blends basePose toward kf.pose, strongest exactly at kf.t.
    function holdAndBlend(kf, basePose){
      const dist = Math.abs(tSecAbsolute - kf.t);
      if(dist >= WINDOW) return null; // far enough away — this keyframe has no say here
      const t = dist / WINDOW; // 0 at the key, 1 at the edge of influence
      const weight = 0.5 * (1 + Math.cos(Math.PI * t)); // eases 1 -> 0, zero slope at both ends
      const blended = {};
      const keys = new Set([...Object.keys(kf.pose), ...Object.keys(basePose)]);
      keys.forEach(k=>{
        const base = basePose[k] ?? 0;
        blended[k] = lerp(base, kf.pose[k] ?? base, weight);
      });
      return { ...basePose, ...blended };
    }

    if(tSecAbsolute <= kfs[0].t){
      pose = holdAndBlend(kfs[0], pose) || pose;
    } else if(tSecAbsolute >= kfs[kfs.length-1].t){
      pose = holdAndBlend(kfs[kfs.length-1], pose) || pose;
    } else {
      let i=0;
      while(i < kfs.length-1 && tSecAbsolute > kfs[i+1].t) i++;
      const a = kfs[i], b = kfs[i+1];
      // Let whichever keyframe is CLOSER to this instant hold/blend first (so right at the
      // midpoint between two evenly-spaced keyframes, influence hands off cleanly from one
      // to the other rather than favoring whichever happens to come first in time). If the
      // closer one has no influence here (this instant is further than 2*HOLD from both),
      // fall through to the farther one — covers the case where the keyframes are close
      // enough together that only one of the two hold windows actually reaches this instant.
      const distA = Math.abs(tSecAbsolute - a.t), distB = Math.abs(tSecAbsolute - b.t);
      const near = distA <= distB ? a : b, far = distA <= distB ? b : a;
      pose = holdAndBlend(near, pose) || holdAndBlend(far, pose) || pose;
    }
  }
  pose = completeNaturalJointPose(pose, loopT, clip.animKey);
  return pose;
}

// Evaluates the model Transform (size / horizontal & vertical position / rotation) at an
// absolute timeline time, by interpolating between whichever of the covering clip's
// keyframes carry a `transform` snapshot (every keyframe made via toggleKeyframeAtPlayhead
// captures one automatically alongside its bone pose). This is what makes moving/resizing/
// rotating the character between two keyframes actually animate — e.g. a keyframe at 1s
// with horizontal position 10 and another at 3s with horizontal position 40 makes the
// character glide from 10 to 40 across that span, exactly like the bone-pose keyframes
// already did above. Unlike bone pose, Transform has no underlying preset animation to
// blend back into when you move away from a single keyframe, so this simply holds at
// whichever keyframe is nearest outside the keyframed range (before the first / after the
// last / with only one keyframe total) and linearly interpolates in between. Returns null
// when the covering clip has no transform-keyframes at all, so projects/clips that only use
// bone-pose keyframes keep the model's Transform completely untouched, exactly as before.
// Same optional (clip, segments) override as poseAtClipTime — defaults to the active
// character when omitted.
function transformAtClipTime(tSecAbsolute, clipOverride, segmentsOverride){
  const useClip = clipOverride !== undefined ? clipOverride : activeClip;
  const useSegments = segmentsOverride !== undefined ? segmentsOverride : splitStickmanSegments;
  if(!useClip) return null;
  const hit = stickmanClipAt(tSecAbsolute, useClip, useSegments);
  if(!hit) return null;
  const clip = hit.clip;
  const kfs = clip.keyframes.filter(kf=>kf.transform).sort((a,b)=>a.t-b.t);
  if(!kfs.length) return null;
  if(kfs.length === 1) return { ...kfs[0].transform };
  if(tSecAbsolute <= kfs[0].t) return { ...kfs[0].transform };
  if(tSecAbsolute >= kfs[kfs.length-1].t) return { ...kfs[kfs.length-1].transform };
  let i=0;
  while(i < kfs.length-1 && tSecAbsolute > kfs[i+1].t) i++;
  const a = kfs[i], b = kfs[i+1];
  const span = (b.t - a.t) || 1;
  const localT = (tSecAbsolute - a.t)/span;
  const keys = new Set([...Object.keys(a.transform), ...Object.keys(b.transform)]);
  const blended = {};
  keys.forEach(k=>{
    const av = a.transform[k], bv = b.transform[k];
    if(typeof av==='number' && typeof bv==='number'){
      // rotation needs shortest-path angular interpolation (350deg -> 10deg should travel
      // 20deg through 360/0, not 340deg the long way around); every other numeric transform
      // field (scale/offsetX/offsetY) is a plain linear lerp.
      blended[k] = (k==='rotation') ? lerpAngleDegrees(av, bv, localT) : lerp(av, bv, localT);
    } else {
      blended[k] = bv;
    }
  });
  return blended;
}
// Applies an (interpolated) transform snapshot onto a live, mutable modelTransform object —
// the same way applyPoseToBones overwrites bones[k].angle every frame from baseAngles+pose.
// Defaults to the shared modelTransform (the active character) when called with no second
// argument, exactly as before — pass an explicit target to update a DIFFERENT character's
// own modelTransform instead (see animateAllCharacters).
function applyTransformToModel(transform, target){
  if(!transform) return;
  const mt = target !== undefined ? target : modelTransform;
  if(transform.scale !== undefined) mt.scale = transform.scale;
  if(transform.offsetX !== undefined) mt.offsetX = transform.offsetX;
  if(transform.offsetY !== undefined) mt.offsetY = transform.offsetY;
  if(transform.rotation !== undefined) mt.rotation = transform.rotation;
}

const overlayMediaEls = {}; // ov.id -> <video>|<img>|<div> stage element
const overlayChromaCanvases = {}; // keyed video overlays -> pixel-processed transparent canvas
// The compositor's z-order only changes when the active overlay set/order changes.

let overlayMediaOrderKey = '';
/* FIX (2nd clip / 2nd half of a split stutters): upcoming video clips are pre-loaded up to
   twenty timeline seconds ahead in hidden, attached, paused elements at their exact source
   starts. At a boundary the pending decoder is adopted instead of discarded; a short clock
   gate prevents the playhead from running ahead while its first frame is being decoded. */
const overlayWarmEls = {}; // String(ov.id) -> hidden, attached and paused <video> pre-loaded for an upcoming clip
const EV_WARM_AHEAD_SEC = 20, EV_WARM_MAX = (window.__evDecoderBudget&&window.__evDecoderBudget.warm)||2;

// Every overlay-video play request passes through this gate. A video may start only
// after its exact source position has decoded and the timeline is no longer gated at
// that clip's boundary. Active overlapping videos may decode for compositing, but only
// the topmost one remains eligible for audio; every start is muted until `playing`
// confirms the same <=40ms alignment and its gain can be ramped in.
function evOverlayVideoExpectedLocalTime(ov,timeSec,el){
  if(!ov)return 0;
  const speed=Math.max(0.05,Number(ov.speed)||1);
  const start=Number(ov.trimStart)||0;
  const sourceOffset=Math.max(0,Number(ov.sourceOffset)||0);
  const elapsed=Math.max(0,(Number(timeSec)-start)*speed);
  const span=Math.max(0,(Number(ov.trimEnd)-start)*speed);
  const raw=ov.reversePlayback?sourceOffset+span-elapsed:sourceOffset+elapsed;
  const knownLength=Number(ov.sourceDuration)||Number(el&&el.duration)||Math.max(sourceOffset+span,0.05);
  const maxT=Math.max(0.05,knownLength-0.02);
  return Math.max(0,Math.min(maxT,raw));
}
function evOverlayVideoFrameIsPresentable(ov,el,timeSec){
  if(!ov||!el||el.tagName!=='VIDEO'||el.readyState<2||el.seeking)return false;
  const t=Number.isFinite(Number(timeSec))?Number(timeSec):Number(currentAbsoluteTimeSec)||0;
  const expected=evOverlayVideoExpectedLocalTime(ov,t,el);
  return Math.abs((Number(el.currentTime)||0)-expected)<=evVideoAlignTolerance(el);
}

function evStopOverlayVideo(el,pauseIt=true){
  if(!el||el.tagName!=='VIDEO')return;
  try{el.muted=true;}catch(_e){}
  try{el.defaultMuted=true;}catch(_e){}
  if(el.__evGainRampRaf){try{cancelAnimationFrame(el.__evGainRampRaf);}catch(_e){}el.__evGainRampRaf=0;}
  delete el.dataset.evGainRampActive;
  delete el.dataset.evGainRampDone;
  delete el.dataset.evStablePlayingSince;
  delete el.dataset.evPlayingEventSeen;
  if(pauseIt&&!el.paused){try{el.pause();}catch(_e){}}
}
function evMuteOtherOverlayVideos(keepEl){
  // v279: overlapping video layers (e.g. a split clip extended under its twin) must ALL be heard,
  // so one layer no longer silences the others.
  return;
  if(typeof overlayMediaEls==='undefined')return;
  Object.keys(overlayMediaEls).forEach(id=>{
    const other=overlayMediaEls[id];
    if(!other||other===keepEl||other.tagName!=='VIDEO')return;
    // Keep lower overlapping videos decoding for correct visual compositing, but never
    // let them contribute audio while the topmost active video owns the preview mix.
    if(other.__evGainRampRaf){try{cancelAnimationFrame(other.__evGainRampRaf);}catch(_e){}other.__evGainRampRaf=0;}
    delete other.dataset.evGainRampActive;
    try{other.muted=true;other.defaultMuted=true;other.volume=0;other.setAttribute('muted','');}catch(_e){}
  });
}
function evPreferredAudibleOverlayVideo(timeSec){
  if(!Array.isArray(overlays))return null;
  const t=Number.isFinite(Number(timeSec))?Number(timeSec):Number(currentAbsoluteTimeSec)||0;
  // The topmost visible overlay video owns preview audio; lower video layers remain
  // visual-only so two overlapping layers can never play sound simultaneously.
  for(let i=overlays.length-1;i>=0;i--){
    const ov=overlays[i];
    if(ov&&ov.type==='video'&&!ov.reversePlayback&&evOverlayVideoIsActiveAt(ov,t))return ov;
  }
  return null;
}
// v279: every active, forward-playing video layer contributes its own sound. Previously only the
// topmost layer was audible and every lower overlapping layer was force-muted, so overlapping copies
// of the same clip never sounded doubled and one of them was silenced.
function evOverlayVideoIsAudibleAt(ov,timeSec){
  if(!ov||ov.type!=='video'||ov.reversePlayback)return false;
  const t=Number.isFinite(Number(timeSec))?Number(timeSec):Number(currentAbsoluteTimeSec)||0;
  return t>=Number(ov.trimStart||0)&&t<Number(ov.trimEnd||0);
}
function evCanOverlayVideoBeAudible(el,ov,timeSec){
  if(!el||!ov||!isPlaying||ov.reversePlayback||!el.isConnected||el.readyState<2||el.seeking||el.dataset.evPlayingEventSeen!=='1')return false;
  if(evVideoBufferStall&&String(evVideoBufferStall.id)===String(ov.id))return false;
  const t=Number.isFinite(Number(timeSec))?Number(timeSec):Number(currentAbsoluteTimeSec)||0;
  if(!evOverlayVideoIsAudibleAt(ov,t)||!evOverlayVideoIsActiveAt(ov,t))return false;
  return Math.abs((Number(el.currentTime)||0)-evOverlayVideoExpectedLocalTime(ov,t,el))<=evVideoAlignTolerance(el);
}
function evRampOverlayVideoAudible(el,ov){
  const t=Number(currentAbsoluteTimeSec)||0;
  const aligned=!!el&&!!ov&&isPlaying&&!ov.reversePlayback&&el.isConnected&&el.readyState>=2&&!el.seeking&&
    el.dataset.evPlayingEventSeen==='1'&&! (evVideoBufferStall&&String(evVideoBufferStall.id)===String(ov.id))&&
    evOverlayVideoIsActiveAt(ov,t)&&
    Math.abs((Number(el.currentTime)||0)-evOverlayVideoExpectedLocalTime(ov,t,el))<=evVideoAlignTolerance(el);
  if(!aligned){evStopOverlayVideo(el,true);return false;}
  if(!evOverlayVideoIsAudibleAt(ov,t)){
    // Not an active forward-playing video layer at this moment: keep it silent.
    if(el.__evGainRampRaf){try{cancelAnimationFrame(el.__evGainRampRaf);}catch(_e){}el.__evGainRampRaf=0;}
    delete el.dataset.evGainRampActive;
    try{el.muted=true;el.defaultMuted=true;el.volume=0;el.setAttribute('muted','');}catch(_e){}
    return false;
  }
  if(el.__evGainRampRaf)return true;
  // BUGFIX: once the 110ms fade-in has finished this function used to start it AGAIN on the next
  // frame (the guard above only sees a ramp that is still running), forcing volume=0 and ramping
  // 0->1 over and over (~8 pulses/second). A completed ramp for this playback run is final.
  if(el.dataset.evGainRampDone==='1'&&!el.muted){
    const doneTarget=Math.max(0,Math.min(1,Number(el.dataset.evTargetVolume==null?(ov.volume==null?1:ov.volume):el.dataset.evTargetVolume)||0));
    if(!el.dataset.evGainRampActive&&el.volume!==doneTarget){try{el.volume=doneTarget;}catch(_e){}}
    return true;
  }
  const target=Math.max(0,Math.min(1,Number(el.dataset.evTargetVolume==null?(ov.volume==null?1:ov.volume):el.dataset.evTargetVolume)||0));
  const started=performance.now();
  el.volume=0;
  try{el.muted=false;el.removeAttribute('muted');}catch(_e){el.muted=false;}
  el.dataset.evGainRampActive='1';
  const step=()=>{
    el.__evGainRampRaf=0;
    const now=Number(currentAbsoluteTimeSec)||0;
    if(!evCanOverlayVideoBeAudible(el,ov,now)){evStopOverlayVideo(el,true);return;}
    const amount=Math.min(1,Math.max(0,(performance.now()-started)/110));
    try{el.volume=target*amount;}catch(_e){}
    if(amount<1){el.__evGainRampRaf=requestAnimationFrame(step);}
    else{delete el.dataset.evGainRampActive;el.dataset.evGainRampDone='1';try{el.volume=target;}catch(_e){}}
  };
  el.__evGainRampRaf=requestAnimationFrame(step);
  return true;
}
function evSafePlayVideo(el,ov){
  if(!el||el.tagName!=='VIDEO')return Promise.resolve(false);
  if(!ov&&Array.isArray(overlays))ov=overlays.find(item=>item&&String(item.id)===String(el.dataset.evV72OverlayId||''));
  if(!ov)return Promise.resolve(false);
  const now=Number(currentAbsoluteTimeSec)||0;
  if(ov.reversePlayback||!evOverlayVideoIsActiveAt(ov,now)||
     (evVideoBufferStall&&String(evVideoBufferStall.id)===String(ov.id))){
    evStopOverlayVideo(el,true);return Promise.resolve(false);
  }
  const expected=evOverlayVideoExpectedLocalTime(ov,now,el);
  const drift=Math.abs((Number(el.currentTime)||0)-expected);
  // Once a decoder is already running, transient readyState drops / in-flight frame
  // refreshes must not pause it. Pausing here reset the audio clock and produced audible
  // clicks/stutters on Android when the browser briefly reported < HAVE_FUTURE_DATA.
  if(el.seeking){
    return Promise.resolve(!el.paused);
  }
  if(el.readyState<3){
    return Promise.resolve(!el.paused);
  }
  // A playing element (or one that just finished our exact seek) is allowed the playing tolerance;
  // only a paused, not-just-seeked element must be exact. This is what stops the pause/seek loop.
  const evSeekDoneAt=Number(el.dataset.evPlaySeekAt)||0;
  const evJustSeeked=evSeekDoneAt>0&&performance.now()-evSeekDoneAt<1500;
  const syncTol=(!el.paused||evJustSeeked)?EV_VIDEO_ALIGN_TOL_PLAYING:EV_VIDEO_ALIGN_TOL_START;
  if(drift>syncTol){
    // A changed trim/sourceOffset or a stale element is corrected with one exact seek,
    // never a playbackRate nudge. This path deliberately bypasses the old 1000ms limit.
    evStopOverlayVideo(el,true);
    if(!el.seeking&&el.readyState>=1){
      const lastTarget=Number(el.dataset.evPreciseSyncTarget);
      if(!Number.isFinite(lastTarget)||Math.abs(lastTarget-expected)>0.005||drift>syncTol){
        // Aim slightly ahead by the measured seek latency so the clip lands in sync instead of
        // landing behind the clock and triggering another seek.
        const evLatencyMs=Math.max(0,Math.min(1200,Number(el.dataset.evSeekLatencyMs)||0));
        const evSpeed=Math.max(0.05,Number(ov.speed)||1);
        const evMaxT=Math.max(0.05,(Number(ov.sourceDuration)||Number(el.duration)||expected+0.05)-0.02);
        const evTarget=Math.max(0,Math.min(evMaxT,expected+(evLatencyMs/1000)*evSpeed));
        el.dataset.evPreciseSyncTarget=String(evTarget);
        el.dataset.evSeekRequestAt=String(performance.now());
        try{el.currentTime=evTarget;}catch(_e){delete el.dataset.evPreciseSyncTarget;}
      }
    }
    return Promise.resolve(false);
  }
  delete el.dataset.evPreciseSyncTarget;
  if(!el.paused){
    if(isPlaying&&el.dataset.evPlayingEventSeen==='1')evRampOverlayVideoAudible(el,ov);
    return Promise.resolve(true);
  }
  // A new attempt always starts muted. The playing listener performs the alignment
  // check and the gain ramp, so a browser autoplay fallback can never leak early audio.
  try{el.muted=true;el.defaultMuted=true;el.setAttribute('muted','');}catch(_e){}
  if(el.__evSafePlayAttempt)return el.__evSafePlayAttempt;
  let nativePromise;
  try{nativePromise=el.play();}catch(err){nativePromise=Promise.reject(err);}
  const attempt=Promise.resolve(nativePromise).then(()=>true).catch(()=>false).finally(()=>{
    if(el.__evSafePlayAttempt===attempt)delete el.__evSafePlayAttempt;
  });
  el.__evSafePlayAttempt=attempt;
  return attempt;
}

function evHardStopOverlayVideosAtCut(fromSec,toSec){
  if(!Array.isArray(overlays)||!isPlaying)return;
  const from=Number(fromSec),to=Number(toSec);
  if(!Number.isFinite(from)||!Number.isFinite(to)||to<from)return;
  const lookAhead=1/Math.max(1,Number(FPS)||30);
  overlays.forEach(ov=>{
    if(!ov||ov.type!=='video')return;
    const end=Number(ov.trimEnd)||0;
    if(end>from+EV_VIDEO_CLOCK_EPS&&end<=to+lookAhead+EV_VIDEO_CLOCK_EPS){
      const el=typeof overlayMediaEls!=='undefined'?overlayMediaEls[String(ov.id)]:null;
      if(el&&el.tagName==='VIDEO'){
        evStopOverlayVideo(el,true);
        el.dataset.evCutStoppedAt=String(end);
      }
    }
  });
}

// Cache one last-good source frame shortly BEFORE each clip boundary. The canvas keeps
// the source aspect ratio and is capped at a 1280px long edge, so cuts never allocate a
// stage-size x devicePixelRatio bitmap on the frame where the decoder is already changing.
function evCapturePreCutOverlayVideoFrame(ov,el,timeSec){
  if(!ov||!el||el.tagName!=='VIDEO'||!el.isConnected||el.readyState<2||el.seeking||!el.videoWidth||!el.videoHeight)return null;
  if(ov.chromaKey&&ov.chromaKey.enabled)return null; // the keyed canvas is already the correct transparent stand-in
  if(!evOverlayVideoFrameIsPresentable(ov,el,timeSec))return null;
  const remaining=Number(ov.trimEnd)-Number(timeSec);
  if(remaining<0.15||remaining>0.25)return null;
  const snapshotKey=[String(ov.id),String(ov.src||''),Number(ov.trimStart)||0,Number(ov.trimEnd)||0,Number(ov.sourceOffset)||0,Math.max(0.05,Number(ov.speed)||1)].join('|');
  let canvas=el.__evCutHoldCanvas;
  if(canvas&&canvas.dataset.evCutSnapshotKey===snapshotKey&&canvas.dataset.evCutSnapshotReady==='1')return canvas;
  if(!canvas){
    canvas=document.createElement('canvas');
    canvas.className='overlay-media-freeze-frame';
    canvas.setAttribute('aria-hidden','true');
    el.__evCutHoldCanvas=canvas;
  }
  const sourceW=Number(el.videoWidth)||0,sourceH=Number(el.videoHeight)||0;
  const scale=Math.min(1,1280/Math.max(sourceW,sourceH));
  const width=Math.max(1,Math.round(sourceW*scale)),height=Math.max(1,Math.round(sourceH*scale));
  if(canvas.width!==width||canvas.height!==height){canvas.width=width;canvas.height=height;}
  const ctx=canvas.getContext('2d');
  if(!ctx)return null;
  try{ctx.clearRect(0,0,width,height);ctx.drawImage(el,0,0,width,height);}catch(_e){return null;}
  const cs=getComputedStyle(el);
  canvas.style.position='absolute';canvas.style.inset='0';
  canvas.style.width='100%';canvas.style.height='100%';
  canvas.style.objectFit=cs.objectFit||'contain';
  canvas.style.objectPosition=cs.objectPosition||'center';
