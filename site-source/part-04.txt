  canvas.style.pointerEvents='none';canvas.style.display='block';

function freezeOutgoingOverlayVideoFrame(id,video,layer,frameTimeSec){
  const owner=Array.isArray(overlays)?overlays.find(item=>item&&String(item.id)===String(id)):null;
  const ownerTime=Number.isFinite(Number(frameTimeSec))?Number(frameTimeSec):(Number(currentAbsoluteTimeSec)||0);
  // Chroma-key videos already have a transparent, processed stage canvas. Keep that
  // canvas rather than holding the source <video> decoder through the buffer stall.
  const chroma=overlayChromaCanvases[id];
  if(chroma&&chroma.isConnected){
    chroma.dataset.evMaskOwnerId=String(id);
    chroma.dataset.evMaskOwnerTime=String(ownerTime);
    if(owner&&typeof window.ethvelleApplyOverlayVisualStyle==='function')window.ethvelleApplyOverlayVisualStyle(chroma,owner,ownerTime,video||chroma,false);
    return chroma;
  }

  // Prefer the cached pre-cut picture. A scrub hold is also a valid last-good canvas;
  // never manufacture a fresh stage-sized bitmap at the cut itself.
  const expectedSnapshotKey=owner?[String(owner.id),String(owner.src||''),Number(owner.trimStart)||0,Number(owner.trimEnd)||0,Number(owner.sourceOffset)||0,Math.max(0.05,Number(owner.speed)||1)].join('|'):'';
  const cutHeld=video&&video.__evCutHoldCanvas&&video.__evCutHoldCanvas.dataset.evCutSnapshotReady==='1'&&
    video.__evCutHoldCanvas.dataset.evCutSnapshotKey===expectedSnapshotKey?video.__evCutHoldCanvas:null;
  const scrubHeld=video&&video.__evScrubHoldCanvas&&video.__evScrubHoldCanvas.isConnected?video.__evScrubHoldCanvas:null;
  const canvas=cutHeld||scrubHeld;
  if(!canvas||!layer)return null;
  // The canvas now belongs to the hold entry, not the pooled video element. Detaching this
  // reference prevents a later clip reusing that decoder from painting over the frozen frame.
  if(cutHeld&&video.__evCutHoldCanvas===cutHeld)delete video.__evCutHoldCanvas;
  if(scrubHeld&&video.__evScrubHoldCanvas===scrubHeld)delete video.__evScrubHoldCanvas;
  const cs=video?getComputedStyle(video):null;
  canvas.classList.add('overlay-media-freeze-frame');
  canvas.style.position='absolute';canvas.style.inset='0';
  canvas.style.width='100%';canvas.style.height='100%';
  canvas.style.objectFit=(cs&&cs.objectFit)||canvas.style.objectFit||'contain';
  canvas.style.objectPosition=(cs&&cs.objectPosition)||canvas.style.objectPosition||'center';
  canvas.style.pointerEvents='none';canvas.style.display='block';
  if(video){
    canvas.style.transform=video.style.transform||((cs&&cs.transform)||'');
    canvas.style.transformOrigin=video.style.transformOrigin||((cs&&cs.transformOrigin)||'center center');
    canvas.style.opacity=(cs&&cs.opacity)||video.style.opacity||'1';
    canvas.style.filter=video.style.filter||((cs&&cs.filter)||'none');
    canvas.style.clipPath=video.style.clipPath||((cs&&cs.clipPath)||'none');
    canvas.style.borderRadius=video.style.borderRadius||((cs&&cs.borderRadius)||'0');
  }
  canvas.dataset.evMaskOwnerId=String(id);
  canvas.dataset.evMaskOwnerTime=String(ownerTime);
  if(canvas.parentNode!==layer||(video&&video.parentNode===layer&&canvas.nextSibling!==video))layer.insertBefore(canvas,video&&video.parentNode===layer?video:null);
  if(owner&&typeof window.ethvelleApplyOverlayVisualStyle==='function')window.ethvelleApplyOverlayVisualStyle(canvas,owner,ownerTime,video||canvas,false);
  return canvas;
}

function setupOverlayVideoEl(el, ov){
  if(!el||!ov)return el;
  el.muted=true;el.defaultMuted=true;el.setAttribute('muted','');el.playsInline=true;el.loop=false;el.preload='auto';
  el.setAttribute('playsinline','');el.setAttribute('webkit-playsinline','');
  el.dataset.evV72OverlayId=String(ov.id);
  const requestedSource=window.evEffectiveVideoSrc(ov);
  const sourceChanged=String(el.dataset.evWarmSrc||'')!==requestedSource;
  if(sourceChanged)delete el.dataset.evHasDecodedTimelineFrame;
  el.dataset.evWarmSrc=requestedSource;

  const currentOwner=()=>{
    const id=String(el.dataset.evV72OverlayId||'');
    return Array.isArray(overlays)?overlays.find(item=>item&&item.type==='video'&&String(item.id)===id)||null:null;
  };
  const updateWarmReady=()=>{
    if(el.dataset.evWarmElement!=='1'||el.dataset.evWarmSeekDone!=='1'||el.readyState<3||el.seeking)return;
    const target=Number(el.dataset.evWarmTarget);
    if(Number.isFinite(target)&&Math.abs(el.currentTime-target)<=0.08)el.dataset.evWarmReady='1';
  };
  if(el.dataset.evOverlayHandlersBound!=='1'){
    el.dataset.evOverlayHandlersBound='1';
    el.addEventListener('seeked',()=>{
      const live=currentOwner();
      const requestedAt=Number(el.dataset.evSeekRequestAt)||0;
      if(requestedAt>0){
        const completedAt=performance.now();
        el.dataset.evSeekLatencyMs=String(Math.max(0,completedAt-requestedAt));
        el.dataset.evPlaySeekAt=String(completedAt);
        delete el.dataset.evSeekRequestAt;
      }
      if(el.dataset.evWarmElement==='1'){
        const target=Number(el.dataset.evWarmTarget);
        if(!el.seeking&&Number.isFinite(target)&&Math.abs(el.currentTime-target)<=0.08)el.dataset.evWarmSeekDone='1';
      }
      updateWarmReady();
      if(el.dataset.evWarmElement!=='1'&&live&&el.isConnected&&isPlaying)evSafePlayVideo(el,live);
      if(el.dataset.evWarmElement!=='1'&&live)evReleaseHeldOutgoingForIncoming(live.id,el);
    });
    el.addEventListener('loadedmetadata',()=>{
      const live=currentOwner();if(!live)return;
      const enhancedSrc=String(live.__evltRuntimeEnhancedSrc||'');
      if(enhancedSrc&&el.src===enhancedSrc&&el.dataset.evEnhPrimed!==enhancedSrc){
        if(typeof window.evPrimeEnhancedVideo==='function')window.evPrimeEnhancedVideo(el,enhancedSrc);
        return;
      }
      try{
        const speedNow=Math.max(0.05,Number(live.speed)||1);
        const maxNow=Math.max(0.05,(Number(live.sourceDuration)||el.duration||(live.trimEnd-live.trimStart))-.02);
        const elapsed=Math.max(0,(Number(currentAbsoluteTimeSec)-Number(live.trimStart))*speedNow);
        const span=Math.max(0,(Number(live.trimEnd)-Number(live.trimStart))*speedNow);
        const raw=live.reversePlayback?(Number(live.sourceOffset)||0)+span-elapsed:(Number(live.sourceOffset)||0)+elapsed;
        const warmStart=el.dataset.evWarmElement==='1'&&!live.reversePlayback;
        const requestedWarmTarget=Number(el.dataset.evWarmTarget);
        const desired=warmStart&&Number.isFinite(requestedWarmTarget)?requestedWarmTarget:(warmStart?(Number(live.sourceOffset)||0):raw);
        const localNow=Math.max(0,Math.min(maxNow,desired));
        if(Number.isFinite(localNow)){
          el.dataset.evWarmTarget=String(localNow);
          if(el.seeking||Math.abs(el.currentTime-localNow)>0.015){
            const requestedAt=performance.now();el.dataset.evSeekRequestAt=String(requestedAt);
            if(warmStart)delete el.dataset.evWarmSeekDone;else el.dataset.evPlaySeekAt=String(requestedAt);
            el.currentTime=localNow;
          }else if(warmStart)el.dataset.evWarmSeekDone='1';
        }
        if(warmStart){try{el.pause();}catch(_e){}updateWarmReady();}
        else if(el.isConnected&&isPlaying&&typeof evSafePlayOverlayVideo==='function')evSafePlayOverlayVideo(el,live);
      }catch(_e){}
    });
    el.addEventListener('canplay',()=>{const live=currentOwner();updateWarmReady();if(el.dataset.evWarmElement!=='1'&&live&&el.isConnected&&isPlaying)evSafePlayVideo(el,live);});
    el.addEventListener('loadeddata',()=>{const live=currentOwner();if(el.dataset.evWarmElement!=='1'&&live&&el.isConnected&&isPlaying)evSafePlayVideo(el,live);});
    el.addEventListener('playing',()=>{
      el.dataset.evPlayingEventSeen='1';
      if(!el.dataset.evStablePlayingSince)el.dataset.evStablePlayingSince=String(performance.now());
      const live=currentOwner();
      if(isPlaying&&live)evRampOverlayVideoAudible(el,live);
      else{try{el.muted=true;el.defaultMuted=true;el.volume=0;el.setAttribute('muted','');}catch(_e){}}
    });
    el.addEventListener('pause',()=>{delete el.dataset.evStablePlayingSince;});
  }
  if(sourceChanged){
    try{el.pause();}catch(_e){}
    try{
      // Remote videos must be CORS-enabled BEFORE the source is set, otherwise the browser
      // taints the canvas and chroma key / the canvas color picker cannot read pixels.
      // If the host refuses CORS, retry once without it so playback still works.
      if(/^https?:/i.test(requestedSource)&&el.dataset.evNoCors!=='1'){
        el.crossOrigin='anonymous';
        if(!el.__evCorsRetryBound){
          el.__evCorsRetryBound=true;
          el.addEventListener('error',()=>{
            if(el.crossOrigin&&el.dataset.evNoCors!=='1'&&/^https?:/i.test(String(el.dataset.evWarmSrc||''))){
              el.dataset.evNoCors='1';el.removeAttribute('crossorigin');
              try{el.src=String(el.dataset.evWarmSrc);el.load();}catch(_e){}
            }
          });
        }
      }else if(!/^https?:/i.test(requestedSource)){el.removeAttribute('crossorigin');delete el.dataset.evNoCors;}
      el.src=requestedSource;el.load();if(ov.__evltRuntimeEnhancedSrc===requestedSource&&typeof window.evPrimeEnhancedVideo==='function')window.evPrimeEnhancedVideo(el,requestedSource);
    }catch(_e){}
  }
  return el;
}
function takeWarmOverlayVideo(ov){
  const key=String(ov.id),el=overlayWarmEls[key];
  if(!el)return null;

  if(el.dataset.evWarmSrc!==window.evEffectiveVideoSrc(ov)){
    delete overlayWarmEls[key];
    releaseOverlayVideoEl(el);
    return null;
  }

  // Adopt even when the decoder is still seeking/buffering. Releasing this element here
  // would throw away its network and decode progress and force a brand-new cold load.
  delete overlayWarmEls[key];
  delete el.dataset.evWarmElement;
  delete el.dataset.evWarmReady;
  delete el.dataset.evWarmSeekDone;
  delete el.dataset.evWarmBoundaryWaitAt;

  el.style.opacity='';
  el.style.visibility=(evOverlayVideoFrameIsPresentable(ov,el,Number(currentAbsoluteTimeSec)||0)||evOverlayVideoHasDecodedTimelineFrame(el))?'':'hidden';
  el.style.pointerEvents='none';
  if(el.readyState>=2&&!el.seeking)delete el.dataset.evWarmAdoptedPending;
  else el.dataset.evWarmAdoptedPending='1';
  delete el.dataset.evWarmRole;
  return el;
}
function warmUpcomingOverlayVideos(tSec){
  const nowMs=performance.now();
  if(nowMs-evWarmLastCheckMs<100)return;
  evWarmLastCheckMs=nowMs;
  const layer=document.getElementById('overlayMediaLayer');
  if(!layer||!Array.isArray(overlays))return;
  // Register active/held decoders; the pool includes warm, live and retained fallback nodes.
  Object.keys(overlayMediaEls).forEach(id=>evRegisterOverlayVideoForPool(overlayMediaEls[id]));
  const held=window.renderOverlayMedia&&window.renderOverlayMedia.__evHeldOutgoing;
  if(held)Object.keys(held).forEach(id=>{const el=held[id]&&held[id].fallbackVideo;if(el)evRegisterOverlayVideoForPool(el);});
  const activeCount=Object.keys(overlayMediaEls).reduce((n,id)=>n+(overlayMediaEls[id]&&overlayMediaEls[id].tagName==='VIDEO'?1:0),0);
  const occupied=evOverlayVideoPool.filter(el=>el&&el.dataset.evPoolFree!=='1').length;
  const slots=Math.max(0,Math.min(EV_WARM_MAX,EV_OVERLAY_VIDEO_POOL_MAX-occupied));
  const candidates=overlays.filter(ov=>ov&&ov.type==='video'&&window.evEffectiveVideoSrc(ov)&&!ov.reversePlayback&&
    Number(ov.trimEnd)>Number(ov.trimStart)&&Number(ov.trimStart)>Number(tSec)+EV_VIDEO_CLOCK_EPS&&
    Number(ov.trimStart)-Number(tSec)<=EV_WARM_AHEAD_SEC&&
    !overlayMediaEls[String(ov.id)])
    .sort((a,b)=>(Number(a.trimStart)||0)-(Number(b.trimStart)||0));
  const selected=candidates.slice(0,slots);
  const wanted=new Set(selected.map(ov=>String(ov.id)));
  Object.keys(overlayWarmEls).forEach(id=>{
    const el=overlayWarmEls[id],ov=overlays.find(item=>item&&String(item.id)===id);
    if(!wanted.has(id)||!ov||window.evEffectiveVideoSrc(ov)!==String(el.dataset.evWarmSrc||'')){
      delete overlayWarmEls[id];releaseOverlayVideoEl(el);
    }
  });
  selected.forEach(ov=>{
    const key=String(ov.id),target=Math.max(0,Number(ov.sourceOffset)||0);
    let el=overlayWarmEls[key];
    if(el&&String(el.dataset.evWarmSrc||'')!==window.evEffectiveVideoSrc(ov)){
      delete overlayWarmEls[key];releaseOverlayVideoEl(el);el=null;
    }
    if(!el){
      el=evAcquireOverlayVideoEl(ov,layer,'warm',target);
      if(!el)return;
      overlayWarmEls[key]=el;
    }else if(Math.abs((Number(el.dataset.evWarmTarget)||0)-target)>.04){
      el.dataset.evWarmTarget=String(target);delete el.dataset.evWarmReady;delete el.dataset.evWarmSeekDone;
      if(el.readyState>=1&&!el.seeking){
        try{el.dataset.evSeekRequestAt=String(performance.now());el.currentTime=target;}catch(_e){}
      }
    }
  });
}

function renderOverlayMedia(tSecAbsolute){
  const layer = document.getElementById('overlayMediaLayer');
  if(!layer) return;
  // Most new projects have no overlay yet; avoid repeated filters/Set allocation for every
  // pointer-driven preview frame when there is nothing to composite.
  if(!overlays.length && !Object.keys(overlayMediaEls).length && !layer.firstChild) return;
  const active=[];
  const activeIds=new Set();
  for(let i=0;i<overlays.length;i++){
    const o=overlays[i];
    if((o.type==='video'||o.type==='image'||o.type==='text') && tSecAbsolute>=o.trimStart && tSecAbsolute<o.trimEnd){
      active.push(o);activeIds.add(o.id);
    }
  }

  const evNowMs=performance.now();
  const evHeldOutgoing=renderOverlayMedia.__evHeldOutgoing||(renderOverlayMedia.__evHeldOutgoing={});
  const evWaitingIncoming=active.find(o=>o.type==='video'&&!o.reversePlayback&&
    !evOverlayVideoFrameIsPresentable(o,evGetOverlayVideoElement(o),tSecAbsolute))||null;

  Object.keys(evHeldOutgoing).forEach(id=>{
    const hold=evHeldOutgoing[id];
    if(!hold)return;
    const activeAgainOv=active.find(o=>o.type==='video'&&String(o.id)===id)||null;
    const activeAgainEl=activeAgainOv?evGetOverlayVideoElement(activeAgainOv):null;
    // IMPORTANT for reverse/fast scrubbing: a clip can become active again while its
    // decoder is still seeking to the earlier timestamp. Its cached outgoing canvas must
    // remain visible until that exact re-entered frame is presentable; active range alone
    // is not proof that the video element has painted a usable frame.
    const activeAgainReady=!!activeAgainOv&&!!activeAgainEl&&
      evOverlayVideoFrameIsPresentable(activeAgainOv,activeAgainEl,tSecAbsolute);
    const incomingStillActive=active.some(o=>o.type==='video'&&String(o.id)===String(hold.incomingId));
    const incomingOv=Array.isArray(overlays)?overlays.find(x=>x&&x.type==='video'&&String(x.id)===String(hold.incomingId)):null;
    const incomingEl=incomingOv?evGetOverlayVideoElement(incomingOv):null;
    const incomingReady=incomingStillActive&&!!incomingOv&&!!incomingEl&&evOverlayVideoFrameIsPresentable(incomingOv,incomingEl,tSecAbsolute);
    if(activeAgainReady||incomingReady){
      evRemoveHeldOutgoingEntry(id,hold);
      delete evHeldOutgoing[id];
      return;
    }
    // A short incoming clip can end before its slow decoder presents a single frame.
    // Retarget the SAME frozen canvas to the next waiting clip instead of deleting it.
    if(!incomingStillActive){
      const nextWaiting=evWaitingIncoming&&String(evWaitingIncoming.id)!==String(hold.incomingId)?evWaitingIncoming:null;
      const nextReady=active.find(o=>o.type==='video'&&!o.reversePlayback&&
        evOverlayVideoFrameIsPresentable(o,evGetOverlayVideoElement(o),tSecAbsolute))||null;
      const nextUpcoming=(!nextWaiting&&!nextReady&&Array.isArray(overlays))
        ?overlays.filter(o=>o&&o.type==='video'&&!o.reversePlayback&&String(o.id)!==String(hold.incomingId)&&
          Number(o.trimStart)>=Number(tSecAbsolute)-0.0005&&Number(o.trimStart)<=Number(tSecAbsolute)+0.25)
          .sort((a,b)=>Number(a.trimStart)-Number(b.trimStart))[0]||null
        :null;
      if(nextWaiting||nextUpcoming){
        const next=nextWaiting||nextUpcoming;
        hold.incomingId=String(next.id);
        hold.startedAtMs=evNowMs;
        if(window.__evVideoDebug)try{console.log('[evVideo] HOLD transferred',id,'->',hold.incomingId);}catch(_e){}
      }else if(nextReady){
        evRemoveHeldOutgoingEntry(id,hold);
        delete evHeldOutgoing[id];
      }
      // With no next decoded video yet, retain the last-good canvas (including after the
      // final short clip) rather than letting a slow decoder leave the stage empty.
    }
  });

  // If a cut or scrub makes a new video active before it has a decoded frame, snapshot
  // the outgoing frame and hold it independently of the playback clock gate. This works
  // while paused and during pointer scrubbing as well as normal playback.
  Object.keys(overlayMediaEls).forEach(id=>{
    if(!activeIds.has(id)){
      const goneEl=overlayMediaEls[id];
      let keepOutgoing=false;
      if(evWaitingIncoming&&goneEl&&goneEl.tagName==='VIDEO'){
        const incomingId=String(evWaitingIncoming.id);
        let hold=evHeldOutgoing[id];
        if(!hold||String(hold.incomingId)!==incomingId){
          // If an older outgoing canvas is already waiting for this incoming clip, keep it.
          // The just-ended intermediate clip may never have decoded a frame of its own.
          const inheritedHold=Object.keys(evHeldOutgoing).some(heldId=>{
            if(String(heldId)===String(id))return false;
            const entry=evHeldOutgoing[heldId];
            return !!(entry&&String(entry.incomingId)===incomingId&&
              (entry.canvas||(entry.fallbackVideo&&evOverlayVideoHasDecodedTimelineFrame(entry.fallbackVideo))));
          });
          if(inheritedHold){
            if(window.__evVideoDebug)try{console.log('[evVideo] HOLD retained for chained cut',id,'->',incomingId);}catch(_e){}
          }else{
            const frozenCanvas=freezeOutgoingOverlayVideoFrame(id,goneEl,layer,tSecAbsolute);
            const decodedFallback=!frozenCanvas&&evOverlayVideoHasDecodedTimelineFrame(goneEl)?goneEl:null;
            if(frozenCanvas||decodedFallback){
              evStopOverlayVideo(goneEl,true);
              hold=evHeldOutgoing[id]={incomingId,canvas:frozenCanvas||null,fallbackVideo:decodedFallback,startedAtMs:evNowMs,timer:0};
              if(frozenCanvas){frozenCanvas.dataset.evMaskOwnerId=String(id);frozenCanvas.dataset.evMaskOwnerTime=String(tSecAbsolute);}
              if(hold.fallbackVideo){
                hold.fallbackVideo.dataset.evMaskOwnerId=String(id);
                hold.fallbackVideo.dataset.evMaskOwnerTime=String(tSecAbsolute);
                const outgoingClip=Array.isArray(overlays)?overlays.find(item=>item&&String(item.id)===String(id)):null;
                if(outgoingClip&&typeof window.ethvelleApplyOverlayVisualStyle==='function')window.ethvelleApplyOverlayVisualStyle(hold.fallbackVideo,outgoingClip,tSecAbsolute,hold.fallbackVideo,false);
              }
              if(window.__evVideoDebug)try{console.log('[evVideo] HOLD (outgoing frozen)',id,'->',incomingId,frozenCanvas?'canvas':'decoded-fallback-video');}catch(_e){}
              if(frozenCanvas&&overlayChromaCanvases[id]!==frozenCanvas&&overlayChromaCanvases[id]){
                overlayChromaCanvases[id].remove();delete overlayChromaCanvases[id];
              }
              delete overlayMediaEls[id];
              if(frozenCanvas)releaseOverlayVideoEl(goneEl);
              else{goneEl.style.pointerEvents='none';goneEl.style.visibility='';}
              keepOutgoing=true;
            }
          }
        }else{
          // The clip already owns a hold. Preserve its current canvas and only retarget it.
          hold.incomingId=incomingId;
          keepOutgoing=!!(hold.canvas||(hold.fallbackVideo&&evOverlayVideoHasDecodedTimelineFrame(hold.fallbackVideo)));
        }
        // No timer expires a transferred hold: the next clip may decode after several
        // 0.2-second cuts. It is released by a presentable incoming frame or a real seek away.
      }else if(evHeldOutgoing[id]){
        evRemoveHeldOutgoingEntry(id,evHeldOutgoing[id]);delete evHeldOutgoing[id];
      }
      if(keepOutgoing)return;
      if(overlayMediaEls[id]===goneEl){
        if(goneEl.tagName==='VIDEO'){evStopOverlayVideo(goneEl,true);releaseOverlayVideoEl(goneEl);}
        else goneEl.remove();
        delete overlayMediaEls[id];
      }
      if(overlayChromaCanvases[id]&&!evHeldOutgoing[id]){overlayChromaCanvases[id].remove();delete overlayChromaCanvases[id];}
    }
  });

  // Maintain overlays[] stacking order, but only reorder when the actual active sequence
  // changes. Re-appending the same nodes on every scrub frame triggered MutationObservers
  // and repeated text-effect/layout work, which was a core source of timeline stutter.
  const activeOrderKey = active.map(o => String(o.id) + (o.type === 'video' && o.chromaKey && o.chromaKey.enabled ? ':chroma' : '')).join('|');
  const needsReorder = active.length > 1 && (activeOrderKey !== overlayMediaOrderKey || active.some(o => !overlayMediaEls[o.id] || overlayMediaEls[o.id].parentNode !== layer));
  active.forEach(ov=>{
    let el = overlayMediaEls[ov.id];
    if(!el){
      // A pending warm element is adopted as-is, even when it is still loading. The clock
      // gate waits on this same decoder; it is never discarded and replaced by a cold one.
      const warmEl=ov.type==='video'?takeWarmOverlayVideo(ov):null;
      el=warmEl||document.createElement(ov.type==='video'?'video':ov.type==='image'?'img':'div');
      if(ov.type==='video'){
        if(!warmEl)setupOverlayVideoEl(el,ov);
        // A warmed clip is already parked at sourceOffset. Do not impose the old adoption
        // cooldown: the clock gate guarantees the first live frame begins at zero drift.
        if(warmEl){
          delete el.dataset.evPlayAdoptedAt;
          delete el.dataset.evPlaySeekAt;
          el.dataset.evDriftNudge='1';
          el.style.opacity='';
          el.style.visibility=((el.readyState>=2&&!el.seeking)||evOverlayVideoHasDecodedTimelineFrame(el))?'':'hidden';
          el.style.pointerEvents='none';
        }else{
          delete el.dataset.evPlayAdoptedAt;
          delete el.dataset.evPlaySeekAt;
          el.dataset.evDriftNudge='1';
        }
      }else if(ov.type==='image'){el.src=ov.src;}
      else{el.className='overlay-text-render';}
      overlayMediaEls[ov.id]=el;
      layer.appendChild(el);
    }else if(needsReorder){
      layer.appendChild(el);
    }
    // A clip sent Backward past the lowest overlapping overlay is placed behind the
    // dedicated Background image layer. Clearing this inline z-index restores normal
    // overlay stacking when the user brings the clip Forward again.
    const behindBackground=!!ov.__ethvelleBehindBackground;
    const overlayZ=behindBackground?'-1':'';
    if(el.style.zIndex!==overlayZ)el.style.zIndex=overlayZ;
    if(ov.type==='video')evWatchIncomingOverlayVideo(ov,el);
    if(ov.type==='text'){
      const requestedFont=String(ov.fontFamily||'Inter');
      if(textNodeFontMissingOrChanged(el,requestedFont)){
        ensureAdminFontLoaded(requestedFont);
        el.dataset.evLoadedTextFont=requestedFont;
      }
      el.classList.add('overlay-text-render');
      // Keep a content-sized child inside the full-stage compositor. This makes the actual
      // text clickable/selectable without turning the entire preview into a click target.
      let textNode=el.querySelector('.ethvelle-overlay-text-content');
      if(!textNode){
        el.textContent='';
        textNode=document.createElement('span');
        textNode.className='ethvelle-overlay-text-content';
        el.appendChild(textNode);
        textNode.addEventListener('pointerdown',function(ev){ev.stopPropagation();});
        textNode.addEventListener('click',function(ev){
          ev.preventDefault(); ev.stopPropagation();
          if(typeof selectClipForChrome==='function') selectClipForChrome(ov.id);
          if(typeof renderFrame==='function') renderFrame();
        });
        textNode.addEventListener('dblclick',function(ev){
          ev.preventDefault(); ev.stopPropagation();
          if(typeof openEthvelleTextComposer==='function') openEthvelleTextComposer(ov.id);
        });
      }
      const rawOverlayText=String(ov.textContent||'');
      if(textNode.dataset.evpaSourceText!==rawOverlayText){
        // Replace text only when the source string actually changes. Rebuilding plain text
        // on every frame woke the many Text/Canvas MutationObservers and caused severe lag.
        const glyph=textNode.querySelector(':scope > .evpa-text-effect-layer');
        if(glyph)glyph.textContent=rawOverlayText;else textNode.textContent=rawOverlayText;
        textNode.dataset.evpaSourceText=rawOverlayText;
        delete textNode.dataset.evpaAppliedEffectSig;
      }
      const fontFamily='"'+String(ov.fontFamily||'Inter')+'", sans-serif';
      const fontSize=Math.max(8,Number(ov.fontSize)||44)+'px';
      const fontWeight=String(ov.fontWeight||800);
      const fontStyle=ov.fontStyle==='italic'?'italic':'normal';
      const textAlign=ov.textAlign||'center';
      const textColor=ethvelleHexToRgba(ov.textColor||'#ffffff',Math.max(0,Math.min(100,Number(ov.textOpacity==null?100:ov.textOpacity)))/100);
      const sw=Math.max(0,Number(ov.strokeWidth)||0);
      const strokeValue=sw>0?(sw+'px '+ethvelleHexToRgba(ov.strokeColor||'#17172B',Math.max(0,Math.min(100,Number(ov.strokeOpacity==null?100:ov.strokeOpacity)))/100)):'0 transparent';
      const shadowValue=ov.shadowEnabled ? `${Number(ov.shadowX)||0}px ${Math.max(4,Number(ov.shadowY)||6)}px ${Math.max(0,Number(ov.shadowBlur)||0)}px ${ethvelleHexToRgba(ov.shadowColor||'#000000',Math.max(0,Math.min(100,Number(ov.shadowOpacity==null?100:ov.shadowOpacity)))/100)}` : 'none';
      const backplateValue=ov.textBackgroundEnabled ? ethvelleHexToRgba(ov.textBackgroundColor||'#182A66',Math.max(0,Math.min(100,Number(ov.textBackgroundOpacity==null?78:ov.textBackgroundOpacity)))/100) : 'transparent';
      const paddingValue=ov.textBackgroundEnabled ? `${Math.max(0,Number(ov.textBackgroundPadding==null?7:ov.textBackgroundPadding))}px` : '0';
      const radiusValue=Math.max(0,Number(ov.textBackgroundRadius==null?8:ov.textBackgroundRadius))+'px';
      // These values do not change just because the playhead moved. Writing them on every
      // preview frame invalidated text paint repeatedly on mobile WebView. Update only when
      // the actual text/effect settings change; renderOverlayMedia still updates time/transform.
      const styleSignature=JSON.stringify([rawOverlayText,fontFamily,fontSize,fontWeight,fontStyle,textAlign,textColor,strokeValue,shadowValue,backplateValue,paddingValue,radiusValue,!!ov.textBackgroundEnabled,!!ov.textHighlightEnabled]);
      if(textNode.dataset.evTextStyleSignature!==styleSignature){
        textNode.style.fontFamily=fontFamily;
        textNode.style.fontSize=fontSize;
        textNode.style.fontWeight=fontWeight;
        textNode.style.fontStyle=fontStyle;
        textNode.style.textAlign=textAlign;
        textNode.style.color=textColor;
        textNode.style.webkitTextStroke=strokeValue;
        textNode.style.paintOrder='stroke fill';
        textNode.style.textShadow=shadowValue;
        textNode.style.backgroundColor=backplateValue;
        textNode.style.padding=paddingValue;
        textNode.style.borderRadius=radiusValue;
        textNode.dataset.evTextStyleSignature=styleSignature;
      }
      const wantsSelected=String(selectedClip)===String(ov.id);
      if(textNode.classList.contains('ev-text-selected')!==wantsSelected)textNode.classList.toggle('ev-text-selected',wantsSelected);
      /* This is the outer text-clip compositor: fading it applies equally to fill, stroke,
         shadow and the independent text-background/highlight plate. */
      const overallOpacity=Math.max(0,Math.min(100,Number(ov.textOverallOpacity==null?100:ov.textOverallOpacity)))/100;
      if(el.style.opacity!==String(overallOpacity))el.style.opacity=String(overallOpacity);
    }
    // The shared helper is the sole owner of transform/mirror/animation/mask styling.
    // The fallback keeps the base renderer safe if called before the visual-toolkit script loads.
    if(typeof window.ethvelleApplyOverlayVisualStyle==='function'){
      window.ethvelleApplyOverlayVisualStyle(el,ov,tSecAbsolute,el,false);
    }else{
      const trackedTransform = window.ethvelleGetTrackedOverlayTransform ? window.ethvelleGetTrackedOverlayTransform(ov) : ov.transform;
      const t = ethvelleObjectKeyframeProperty(ov)==='transform' ? (ethvelleGetObjectKeyframeValue(ov,'transform',tSecAbsolute)||trackedTransform) : trackedTransform;
      const elementTransform = t ? `translate(${t.offsetX||0}px, ${t.offsetY||0}px) rotate(${t.rotation||0}deg) scale(${t.scale||1})` : '';
      if(el.style.transform!==elementTransform)el.style.transform = elementTransform;
    }
    if(ov.type==='video'){
      const sameVideoStall=!!evVideoBufferStall&&String(evVideoBufferStall.id)===String(ov.id);
      const framePresentable=evOverlayVideoFrameIsPresentable(ov,el,tSecAbsolute);
      if(framePresentable&&el.videoWidth>0&&el.videoHeight>0){
        el.dataset.evHasDecodedTimelineFrame='1';
        // Capture the clip's last-good picture 150-250ms BEFORE its trim end, while the
        // decoder is still presentable. freezeOutgoingOverlayVideoFrame reuses this canvas.
        try{evCapturePreCutOverlayVideoFrame(ov,el,tSecAbsolute);}catch(_e){}
      }
      // During playback as well as scrubbing, a decoder hiccup must not hide a video that
      // has already displayed a real frame. The scrub wrapper may place its retained canvas
      // above the element; otherwise leave the last decoded video surface visible.
      const keepLastDecodedFrame=evOverlayVideoHasDecodedTimelineFrame(el);
      const keepLastScrubFrame=!!window.__ethvelleTimelineScrubbing&&keepLastDecodedFrame;
      const hideIncoming=!framePresentable&&!keepLastScrubFrame&&!keepLastDecodedFrame;
      const visibilityState=hideIncoming?'hidden':'';
      if(el.dataset.evIncomingHidden!==(hideIncoming?'1':'0')){
        if(window.__evVideoDebug)try{console.log('[evVideo]',hideIncoming?'HOLD (incoming hidden)':'LIVE (incoming shown)',ov.id,'scrubbing',!!window.__ethvelleTimelineScrubbing);}catch(_e){}
        el.style.visibility=visibilityState;
        el.dataset.evIncomingHidden=hideIncoming?'1':'0';
      }

      const speed = Math.max(0.05,Number(ov.speed)||1);
      const enhancedPrimeUrl=String(ov.__evltRuntimeEnhancedSrc||'');
      const enhancedPrimeActive=!!(enhancedPrimeUrl&&el.__evEnhPrimeInProgress===enhancedPrimeUrl);
      const maxT = Math.max(0.05, (ov.sourceDuration || el.duration || (ov.trimEnd-ov.trimStart)) - 0.02);
      // localT is measured in the SOURCE video's own time, so it advances `speed`x faster
      // than timeline time elapses — at 2x, 1 second of timeline time covers 2 seconds of
      // the actual source footage; at 0.5x, half a second of it. Combined with the clip's
      // own trimEnd already being rescaled by speedPanelSetSpeed (shorter timeline
      // footprint at higher speed), this keeps the WHOLE source clip playing out exactly
      // once across the clip's new, shorter (or longer) span on the timeline.
      const elapsed=Math.max(0,(tSecAbsolute-Number(ov.trimStart))*speed);
      const span=Math.max(0,(Number(ov.trimEnd)-Number(ov.trimStart))*speed);
      const rawLocal=ov.reversePlayback?(Number(ov.sourceOffset)||0)+span-elapsed:(Number(ov.sourceOffset)||0)+elapsed;
      const localT=Math.max(0,Math.min(maxT,rawLocal));
      if(ov.reversePlayback){
        // Reverse preview is driven by timeline seeks because HTMLVideoElement does not
        // support dependable negative playbackRate on Android WebView. Keep seeks bounded
        // to ~15fps while playing; export below seeks every rendered frame for accuracy.
        if(!el.paused){try{el.pause();}catch(_e){}}
        el.muted=true;
        const now=performance.now(),last=Number(el.dataset.evReverseSeekAt)||0;
        const interval=isPlaying?66:0;
        if((!isPlaying||!last||now-last>=interval)&&Math.abs(el.currentTime-localT)>(isPlaying?0.045:0.015)){
          try{el.currentTime=localT;el.dataset.evReverseSeekAt=String(now);}catch(_e){}
        }
        el.playbackRate=1;
      }else{
        delete el.dataset.evReverseSeekAt;
      }
      // Only seek when meaningfully off (avoids fighting the video's own playback clock with
      // constant tiny seeks every frame, which stutters); while actually playing, let it run
      // and only correct drift beyond ~0.2s.
      if(!ov.reversePlayback&&window.__ethvelleTimelineScrubbing&&!enhancedPrimeActive){
        // Re-seeking an HTMLVideoElement 30+ times per second forces Android's decoder to
        // flush and restart repeatedly; this was a major source of horizontal-drag stutter.
        // Sample rough preview frames at <= ~12fps, then render the exact source time on release.
        const now=performance.now();
        const last=Number(el.dataset.evLastScrubSeekAt)||0;
        if((!last||now-last>=84) && (!last||Math.abs(el.currentTime-localT)>.24)){
          try{el.currentTime=localT;el.dataset.evLastScrubSeekAt=String(now);}catch(err){/* decode will retry on a later scrub frame */}
        }
      }else if(!ov.reversePlayback){
        delete el.dataset.evLastScrubSeekAt;
        const driftSigned=localT-el.currentTime;
        const driftNow=Math.abs(driftSigned);
        const nowMs=performance.now();
        const effectiveSyncSource=String(window.evEffectiveVideoSrc?window.evEffectiveVideoSrc(ov):(ov.src||''));
        const syncSignature=[effectiveSyncSource,Number(ov.trimStart)||0,Number(ov.trimEnd)||0,Number(ov.sourceOffset)||0,speed].join('|');
        const sourceOrTrimChanged=!enhancedPrimeActive&&el.dataset.evClipSyncSignature!==syncSignature;
        if(!enhancedPrimeActive)el.dataset.evClipSyncSignature=syncSignature;
        let doSeek=false;
        let seekTarget=localT;
        const stableSince=Number(el.dataset.evStablePlayingSince)||0;
        const stableForMs=stableSince>0?nowMs-stableSince:0;

        if(!isPlaying){
          el.dataset.evDriftNudge='1';
          const lastSeekAt=Number(el.dataset.evPlaySeekAt)||0;
          // A trim/split/sourceOffset edit bypasses the old 1000ms seek limiter.
          doSeek=!el.seeking&&el.readyState>=1&&driftNow>0.04&&
            (sourceOrTrimChanged||!lastSeekAt||nowMs-lastSeekAt>=1000);
          if(doSeek){el.dataset.evPlaySeekAt=String(nowMs);el.dataset.evSeekRequestAt=String(nowMs);}
        }else if(el.seeking||el.readyState<2){
          el.dataset.evDriftNudge='1';
        }else if(sourceOrTrimChanged&&driftNow>0.04&&el.readyState>=1){
          el.dataset.evDriftNudge='1';doSeek=true;el.dataset.evPlaySeekAt=String(nowMs);el.dataset.evSeekRequestAt=String(nowMs);
        }else if(!el.paused&&stableForMs>=300){
          // Continuous playback is corrected with a gentle proportional playback-rate
          // servo instead of repeated pause -> currentTime seek -> play cycles. Those hard
          // seeks were visibly flashing frames and breaking the audio stream on mobile.
          // Keep the correction small and enable pitch preservation below so normal drift
          // recovery does not make the soundtrack sound warped.
          const correction=Math.max(0.96,Math.min(1.04,1+(driftSigned*0.10)));
          el.dataset.evDriftNudge=(driftNow>0.025?correction:1).toFixed(4);
        }else{
          // At start/resume, keep nominal speed. Safe play performs one exact muted seek
          // before the first audible frame instead of using pitch-changing rate nudges.
          el.dataset.evDriftNudge='1';
        }

        if(doSeek&&!el.seeking&&el.readyState>=1&&!enhancedPrimeActive){
          if(isPlaying){evStopOverlayVideo(el,true);}
          try{el.currentTime=seekTarget;}
          catch(err){delete el.dataset.evSeekRequestAt;}
        }
      }
      // playbackRate lets the video element itself advance at the matching rate during
      // uninterrupted playback (so it doesn't rely purely on the periodic drift-correction
      // seeks above to keep pace), and keeps audio pitch/speed in sync with the visual rate.
      const stableSince=Number(el.dataset.evStablePlayingSince)||0;
      const stableForMs=stableSince>0?performance.now()-stableSince:0;
      const driftNudge=window.__ethvelleTimelineScrubbing||stableForMs<300?1:(Number(el.dataset.evDriftNudge)||1);
      const targetPlaybackRate=speed*driftNudge;
      if(!ov.reversePlayback){
        try{if('preservesPitch' in el)el.preservesPitch=true;if('webkitPreservesPitch' in el)el.webkitPreservesPitch=true;if('mozPreservesPitch' in el)el.mozPreservesPitch=true;}catch(_e){}
        if(Math.abs(el.playbackRate-targetPlaybackRate)>0.005)el.playbackRate=targetPlaybackRate;
      }
      // Reverse preview is video-only; the decoded image is stepped by reverse seeks.
      // Clamp to [0,1] — HTMLMediaElement.volume only accepts that range; a clip's own
      // `volume` can go up to 2 (a "boost") but anything above 1 is realized at export time
      // via the Web Audio GainNode chain instead (see mixAudioForExport), which has no such
      // upper limit. Live preview playback is correctly capped at "as loud as the source".
      const animatedVideoVolume=ethvelleGetObjectKeyframeValue(ov,'volume',tSecAbsolute);
      const vol = Math.max(0, Math.min(1, animatedVideoVolume==null?(ov.volume==null?1:ov.volume):animatedVideoVolume));
      el.dataset.evTargetVolume=String(vol);
      if(!el.dataset.evGainRampActive&&!el.muted&&el.volume!==vol)el.volume=vol;
      if(ov.reversePlayback){
        evStopOverlayVideo(el,true);
      }else if(isPlaying&&!sameVideoStall){
        evSafePlayVideo(el,ov);
      }else{
        evStopOverlayVideo(el,true);
      }
      const keyCfg = ov.chromaKey;
      if(keyCfg && keyCfg.enabled){
        let keyCanvas=overlayChromaCanvases[ov.id];
        if(!keyCanvas){ keyCanvas=document.createElement('canvas'); keyCanvas.className='overlay-chroma-canvas'; overlayChromaCanvases[ov.id]=keyCanvas; layer.appendChild(keyCanvas); }
        if(keyCanvas.style.transform!==el.style.transform)keyCanvas.style.transform=el.style.transform;
        if(needsReorder) layer.appendChild(keyCanvas);
        if(el.readyState>=2 && el.videoWidth>0){
          if(window.__ethvelleTimelineScrubbing){
            const now=performance.now(),last=Number(keyCanvas.dataset.evLastScrubPaintAt)||0;
            if(!last||now-last>=100){keyCanvas.dataset.evKeyed=paintChromaKeyFrame(el,keyCanvas,keyCfg,480)?'1':'0';keyCanvas.dataset.evLastScrubPaintAt=String(now);}
          }else{delete keyCanvas.dataset.evLastScrubPaintAt;keyCanvas.dataset.evKeyed=paintChromaKeyFrame(el,keyCanvas,keyCfg,640)?'1':'0';}
        }
        // Only hide the real video once a keyed frame exists; otherwise keep it visible so
        // the clip can never disappear from the canvas.
        const keyedReady=keyCanvas.dataset.evKeyed==='1';
        el.style.opacity=keyedReady?'0':'1';
        keyCanvas.style.visibility=keyedReady?'visible':'hidden';
      }else{
        el.style.opacity='1';
        if(overlayChromaCanvases[ov.id]){overlayChromaCanvases[ov.id].remove();delete overlayChromaCanvases[ov.id];}
      }
    }
  });
  if(window.__ethvelleTimelineScrubbing){
    // Audio doesn't need to seek on every preview position. Keep existing nodes paused and
    // defer the exact seek/sync until pointer-up to avoid repeated media decoder work.
    Object.keys(audioOverlayEls).forEach(id=>{try{if(audioOverlayEls[id]&&!audioOverlayEls[id].paused)audioOverlayEls[id].pause();}catch(_e){}});
  }else{
    renderAudioOverlayMedia(tSecAbsolute);
  }
  overlayMediaOrderKey = activeOrderKey;
  try{ warmUpcomingOverlayVideos(tSecAbsolute); }catch(_e){}
}
// FEATURE (extract audio / control clip volume): companion to renderOverlayMedia's video/
// image element lifecycle above, but for type==='audio' overlays — previously these only
// ever drew a waveform on the timeline and produced no sound at all during preview or
// export. Mirrors the exact same create/seek/play-pause/cleanup pattern (kept as a separate
// function rather than folded into the `active` filter above, since audio has no on-stage
// visual element and therefore none of the transform/thumb logic video or image need).
let audioOverlayEls = {};
let audioOverlayAudioCtx = null;
let audioOverlayNodes = {};
function ensureAudioOverlayContext(){
  if(!audioOverlayAudioCtx){
    const AC=window.AudioContext||window.webkitAudioContext;
    if(AC) audioOverlayAudioCtx=new AC({latencyHint:'playback'});
  }
  return audioOverlayAudioCtx;
}
function evAudioOverlayById(id){
  if(!Array.isArray(overlays))return null;
  return overlays.find(item=>item&&item.type==='audio'&&String(item.id)===String(id))||null;
}
function evAudioOverlayIsActive(ov,timeSec){
  if(!ov)return false;
  const t=Number(timeSec);
  return Number.isFinite(t)&&t>=Number(ov.trimStart||0)&&t<Number(ov.trimEnd||0);
}
function evAudioOverlayRate(ov){
  const cfg=(typeof getEthvelleAudioSettings==='function')?getEthvelleAudioSettings(ov):{pitch:0};
  const pitchRate=Math.pow(2,Math.max(-12,Math.min(12,Number(cfg.pitch)||0))/12);
  return {cfg,rate:Math.max(.05,Number(ov.speed)||1)*pitchRate};
}
function evAudioOverlayLocalTime(ov,timeSec,el,rate){
  const duration=Math.max(.05,Number(ov.sourceDuration)||Number(el&&el.duration)||((Number(ov.trimEnd)||0)-(Number(ov.trimStart)||0))*rate);
  const maxT=Math.max(.05,duration-.02);
  return Math.max(0,Math.min(maxT,(Number(ov.sourceOffset)||0)+(Number(timeSec)-Number(ov.trimStart||0))*rate));
}
function evAudioOverlayTryWarmSeek(ov,el){
  if(!ov||!el||el.readyState<1||el.seeking)return;
  const target=Math.max(0,Number(ov.sourceOffset)||0);
  const prior=Number(el.dataset.evAudioWarmSourceOffset);
  if(Number.isFinite(prior)&&Math.abs(prior-target)<.025)return;
  if(Math.abs((Number(el.currentTime)||0)-target)<=.05){
    el.dataset.evAudioWarmSourceOffset=String(target);
    return;
  }
  try{
    el.dataset.evAudioWarmSourceOffset=String(target);
    el.currentTime=target;
  }catch(_e){delete el.dataset.evAudioWarmSourceOffset;}
}
function evAudioOverlayEnsure(ov){
  if(!ov||!ov.id||!ov.src)return null;
  const id=String(ov.id);
  let el=audioOverlayEls[id];
  if(!el){
    el=new Audio();
    el.preload='auto';
    el.autoplay=false;
    el.loop=false;
    el.dataset.evAudioOverlayId=id;
    el.dataset.evAudioSource=String(ov.src);
    audioOverlayEls[id]=el;
    // These events also cover mobile browsers that finish loading only after a clip becomes
    // active. They always resolve the latest clip/time instead of capturing a stale playhead.
    const onReady=()=>{
      const live=evAudioOverlayById(id);
      if(!live)return;
      const now=Number(typeof currentAbsoluteTimeSec!=='undefined'?currentAbsoluteTimeSec:0)||0;
      if(evAudioOverlayIsActive(live,now)&&isPlaying)evAudioOverlayStart(live,el,now);
      else if(Number(live.trimStart)>now)evAudioOverlayTryWarmSeek(live,el);
    };
    el.addEventListener('loadedmetadata',onReady);
    el.addEventListener('canplay',onReady);
    el.addEventListener('seeked',onReady);
    el.addEventListener('playing',()=>{delete el.dataset.evAudioPlayPending;});
    el.addEventListener('pause',()=>{delete el.dataset.evAudioPlayPending;});
    try{el.src=String(ov.src);el.load();}catch(_e){}
  }else if(String(el.dataset.evAudioSource||'')!==String(ov.src)){
    // Reuse the same element/source graph if a clip's media source is replaced.
    try{el.pause();el.dataset.evAudioSource=String(ov.src);delete el.dataset.evAudioWarmSourceOffset;delete el.dataset.evAudioSyncing;el.src=String(ov.src);el.load();}catch(_e){}
  }
  // Create one persistent Web Audio source per element. Later audio-effect wrappers attach
  // their filters to these same nodes; they are not disconnected at normal clip boundaries.
  if(!audioOverlayNodes[id]){
    try{
      const ctx=ensureAudioOverlayContext();
      if(ctx){
        const source=ctx.createMediaElementSource(el),gain=ctx.createGain();
        source.connect(gain);gain.connect(ctx.destination);
        audioOverlayNodes[id]={source,gain};
        el.volume=1;
      }
    }catch(err){console.warn('Ethvelle: Web Audio gain unavailable for this clip; using native volume.',err);}
  }
  return el;
}
function evAudioOverlayStart(ov,el,timeSec){
  if(!ov||!el||!isPlaying||!evAudioOverlayIsActive(ov,timeSec))return;
  if(evVideoBufferStall)return;
  const {cfg,rate}=evAudioOverlayRate(ov);
  const target=evAudioOverlayLocalTime(ov,timeSec,el,rate);
  if(el.readyState<1||el.seeking)return;
  // On clip entry / resume / timeline jump, perform one exact source seek before playback.
  // During uninterrupted playback the media element remains its own clock; do not seek it
  // on every animation frame, which caused the audible gaps at adjacent clip boundaries.
  const needsInitialSync=el.dataset.evAudioNeedsSync==='1';
  const drift=Math.abs((Number(el.currentTime)||0)-target);
  if((needsInitialSync&&drift>.055)||(!needsInitialSync&&drift>.72)){
    if(el.dataset.evAudioSyncing!=='1'){
      el.dataset.evAudioSyncing='1';
      el.dataset.evAudioSyncTarget=String(target);
      try{el.currentTime=target;}catch(_e){delete el.dataset.evAudioSyncing;}
    }
    return;
  }
  if(needsInitialSync){delete el.dataset.evAudioNeedsSync;delete el.dataset.evAudioSyncing;}
  if(el.readyState<2||!el.paused||el.dataset.evAudioPlayPending==='1')return;
  if(audioOverlayAudioCtx&&audioOverlayAudioCtx.state==='suspended'){
    try{audioOverlayAudioCtx.resume().catch(()=>{});}catch(_e){}
  }
  const token=(Number(el.__evAudioRunToken)||0)+1;
  el.__evAudioRunToken=token;
  el.dataset.evAudioPlayPending='1';
  let p;
  try{p=el.play();}catch(err){delete el.dataset.evAudioPlayPending;return;}
  if(p&&typeof p.then==='function'){
    p.then(()=>{
      if(el.__evAudioRunToken!==token||!isPlaying||!evAudioOverlayIsActive(evAudioOverlayById(ov.id),Number(currentAbsoluteTimeSec)||0)){
        try{el.pause();}catch(_e){}
      }
      delete el.dataset.evAudioPlayPending;
    }).catch(()=>{delete el.dataset.evAudioPlayPending;});
  }else{delete el.dataset.evAudioPlayPending;}
}
function renderAudioOverlayMedia(tSecAbsolute){
  const time=Number.isFinite(Number(tSecAbsolute))?Number(tSecAbsolute):(Number(currentAbsoluteTimeSec)||0);
  const allAudio=Array.isArray(overlays)?overlays.filter(o=>o&&o.type==='audio'&&o.id&&o.src):[];
  const liveIds=new Set(allAudio.map(o=>String(o.id)));
  const activeIds=new Set(allAudio.filter(o=>evAudioOverlayIsActive(o,time)).map(o=>String(o.id)));
  const preloadAhead=10; // load the next audio clip before the playhead reaches its boundary

  // A deleted audio clip is the only normal reason to destroy its element/graph. Merely
  // crossing trimEnd pauses it but keeps it buffered and ready for later seeks/redo.
  Object.keys(audioOverlayEls).forEach(id=>{
    if(liveIds.has(String(id)))return;
    const el=audioOverlayEls[id];
    try{el.pause();}catch(_e){}
    if(el){el.__evAudioRunToken=(Number(el.__evAudioRunToken)||0)+1;try{el.removeAttribute('src');el.load();}catch(_e){}}
    const nodes=audioOverlayNodes[id];
    if(nodes){try{nodes.source.disconnect();nodes.gain.disconnect();}catch(_e){}delete audioOverlayNodes[id];}
    delete audioOverlayEls[id];
  });

  // Pre-create and buffer the next clips while the current audio is still playing.
  allAudio.forEach(ov=>{
    const start=Number(ov.trimStart)||0;
    const active=activeIds.has(String(ov.id));
    const upcoming=!active&&start>=time&&start-time<=preloadAhead;
    if(!active&&!upcoming)return;
    const el=evAudioOverlayEnsure(ov);
    if(!el)return;
    if(upcoming)evAudioOverlayTryWarmSeek(ov,el);
  });

  if(evVideoBufferStall&&isPlaying){
    Object.keys(audioOverlayEls).forEach(id=>{
      const el=audioOverlayEls[id];
      if(el&&!el.paused){try{el.pause();}catch(_e){}}
      if(el)el.dataset.evAudioNeedsSync='1';
    });
    return;
  }

  allAudio.forEach(ov=>{
    const id=String(ov.id),el=audioOverlayEls[id];
    if(!el)return;
    const active=activeIds.has(id);
    if(!active){
      if(el.dataset.evAudioTimelineActive==='1'){
        el.dataset.evAudioTimelineActive='0';
        el.dataset.evAudioNeedsSync='1';
        el.__evAudioRunToken=(Number(el.__evAudioRunToken)||0)+1;
        if(!el.paused){try{el.pause();}catch(_e){}}
      }
      return;
    }

    if(el.dataset.evAudioTimelineActive!=='1'){
      el.dataset.evAudioTimelineActive='1';
      el.dataset.evAudioNeedsSync='1';
    }
    const {cfg,rate}=evAudioOverlayRate(ov);
    if(Math.abs((Number(el.playbackRate)||1)-rate)>.001)el.playbackRate=rate;
    const pitchShiftActive=Math.abs(Number(cfg.pitch)||0)>.001;
    try{if('preservesPitch' in el)el.preservesPitch=!pitchShiftActive;if('webkitPreservesPitch' in el)el.webkitPreservesPitch=!pitchShiftActive;}catch(_e){}

    const animated=ethvelleGetObjectKeyframeValue(ov,'volume',time);
    const volume=Math.max(0,Math.min(2,animated==null?(ov.volume==null?1:Number(ov.volume)):Number(animated)));
    const nodes=audioOverlayNodes[id];
    if(nodes&&audioOverlayAudioCtx){
      try{nodes.gain.gain.setTargetAtTime(volume,audioOverlayAudioCtx.currentTime,.025);}catch(_e){nodes.gain.gain.value=volume;}
      el.volume=1;
    }else{
      const nativeVolume=Math.min(1,volume);
      if(el.volume!==nativeVolume)el.volume=nativeVolume;
    }

    if(isPlaying){
      if(Math.abs((Number(el.currentTime)||0)-evAudioOverlayLocalTime(ov,time,el,rate))>.72){
        // Large drift is a recovery case only; normal clock drift never interrupts playback.
        el.dataset.evAudioNeedsSync='1';
      }
      evAudioOverlayStart(ov,el,time);
    }else{
      el.__evAudioRunToken=(Number(el.__evAudioRunToken)||0)+1;
      if(!el.paused){try{el.pause();}catch(_e){}}
      if(el.readyState>=1&&!el.seeking){
        const target=evAudioOverlayLocalTime(ov,time,el,rate);
        if(Math.abs((Number(el.currentTime)||0)-target)>.055){
          try{el.dataset.evAudioNeedsSync='1';el.currentTime=target;}catch(_e){}
        }
      }
    }
  });
}

// Start each evaluated timeline frame from the character's authored rest pose and its
// user-chosen base placement. Pose/transform keyframes are then layered on for THIS instant.
// This keeps a dragged pose, a previous clip, or an export frame from contaminating the next
// frame's bind pose — especially important for edited Model Lab sheets whose limbs must line up
// with the same rest skeleton used when the animation was authored.
function resetCharacterForTimelineFrame(char){
  if(!char) return;
  const rigBones = char.bones || {};
  const restAngles = char.baseAngles || {};
  Object.keys(rigBones).forEach(k=>{
    if(Number.isFinite(restAngles[k]) && rigBones[k]) rigBones[k].angle = restAngles[k];
  });
  if(rigBones.hip && Number.isFinite(char.hipRestY)) rigBones.hip.y = char.hipRestY;
  if(char.modelTransform){
    if(!char.restTransform) char.restTransform = { ...char.modelTransform };
    Object.assign(char.modelTransform, char.restTransform);
  }
}

// Poses every OTHER character (all except the active one, which renderFrame already poses
// via the shared bones/activeClip/modelTransform variables) at the given absolute timeline
// second, using each character's own clip/segments — so every character with an animation
// on its own row now actually moves during playback/export, not just whichever one happens
// to be selected. Each character's rig is independent, so this never touches the active
// character's live pose or reads/writes the shared bones/baseAngles/modelTransform names.
function animateAllCharacters(tSecAbsolute){
  characters.forEach((c, idx)=>{
    if(idx === activeCharIdx) return; // the active character is handled by renderFrame
    if(!c.clip) return; // static rows need no per-frame rig reset
    resetCharacterForTimelineFrame(c);
    const pose = poseAtClipTime(tSecAbsolute, c.clip, c.segments);
    if(pose) applyPoseToBones(pose, c.bones, c.baseAngles, c.hipRestY);
    const clipTransform = transformAtClipTime(tSecAbsolute, c.clip, c.segments);
    if(clipTransform) applyTransformToModel(clipTransform, c.modelTransform);
  });
}
// Renders whatever the canonical clock (currentAbsoluteTimeSec) currently points at.
// Evaluates pose/transform against the clip that ACTUALLY COVERS that absolute time
// (via stickmanClipAt inside poseAtClipTime/transformAtClipTime), not always activeClip —
// this is what makes split-segment keyframes evaluate correctly during playback/scrub
// instead of only ever showing activeClip's animation. Also drives every OTHER character's
// own pose from its own clip (see animateAllCharacters), so multiple characters with their
// own animations all move together instead of only the active one animating while the rest
// stand frozen in their rest pose.
function renderTimelineScrubPreview(){
  // Keep the visible composition frame-accurate during a finger scrub, but bypass the
  // full editor refresh wrappers (text-effect decoration scans, palette/selection syncing)
  // and nonessential toolbar state work. Those scans were being repeated at display rate on
  // Android and were a major source of horizontal timeline stutter.
  refreshCurrentBgFromPlayhead();
  applyBackground();
  const active = activeCharacter();
  if(active) resetCharacterForTimelineFrame(active);
  if(activeClip){
    const pose = poseAtClipTime(currentAbsoluteTimeSec);
    if(pose) applyPoseToBones(pose);
    const clipTransform = transformAtClipTime(currentAbsoluteTimeSec);
    if(clipTransform) applyTransformToModel(clipTransform);
  }
  animateAllCharacters(currentAbsoluteTimeSec);
  renderOverlayMedia(currentAbsoluteTimeSec);
  drawStick();
  // Keep the timestamp responsive without running the more expensive keyframe/toolbar sync.
  updateTimeReadout();
  // scheduleTimelineScrub() has already positioned scrollLeft directly from this gesture.
  // Rewriting scrollLeft here makes Android WebView perform another timeline scroll/layout pass
  // while the finger is moving; leave that to the input path and keep this renderer preview-only.
  // The keyframe button and final centered playhead are synchronized by the release render.
}
function renderFrame(){
  // Resolve the background first; every character is then reset to the same authored rest
  // pose/base transform before the active clip at this exact time is applied. This prevents
  // stale transforms/poses from a previous frame or another clip from leaking into preview.
  refreshCurrentBgFromPlayhead();
  applyBackground();
  const active = activeCharacter();
  if(active) resetCharacterForTimelineFrame(active);
  if(activeClip){
    const pose = poseAtClipTime(currentAbsoluteTimeSec);
    if(pose) applyPoseToBones(pose);
    const clipTransform = transformAtClipTime(currentAbsoluteTimeSec);
    if(clipTransform) applyTransformToModel(clipTransform);
  }
  animateAllCharacters(currentAbsoluteTimeSec);
  renderOverlayMedia(currentAbsoluteTimeSec);
  drawStick();
  updateTimeReadout();
  movePlayhead();
  updateKeyButtonState();
}

function seekFrame(dir){
  // Tapping prev/next frame pauses FIRST (via the centralized controller, which freezes
  // the EXACT real-elapsed-time instant, not "whatever frame the last rAF tick left
  // behind"), and only then applies the seek — so it always lands exactly one step from
  // the exact position you paused at, never a moving target.
  pausePlayback();
  const stepSec = Math.round(FPS*0.25)/FPS;
  // Clamp against the FULL project's duration (every character, plus however far
  // Background/Clips/Audio extend — see totalProjectDuration()), so stepping frame-by-frame
  // can reach the end of a longer non-active character's clip, or a longer background/
  // media/audio clip, too — not just the active character's own animation.
  const dur = totalProjectDuration();
  setCurrentAbsoluteTime(Math.max(0, Math.min(dur, currentAbsoluteTimeSec + dir*stepSec)));
  renderFrame();
}
// v220: m:ss(.cc) clock so times past 60s read 1:10, 1:20 ... instead of 70s, 80s ...
function evFormatClock(t,withCs){
  t=Math.max(0,Number(t)||0);
  const cs=Math.round(t*100),m=Math.floor(cs/6000),rem=cs%6000,sec=Math.floor(rem/100),c=rem%100;
  const p=n=>String(n).padStart(2,'0');
  return withCs?`${m}:${p(sec)}.${p(c)}`:`${m}:${p(sec)}`;
}
function updateTimeReadout(){
  // The transport's total-time label is an export/content endpoint, not the padded
  // timeline width used to size the ruler and scrollable track area.
  const dur = getExportContentDurationSec();
  const nowSec = (activeClip ? currentAbsoluteTimeSec - activeClip.trimStart : currentAbsoluteTimeSec);
  document.getElementById('timeReadout').textContent = `${evFormatClock(nowSec,true)} / ${evFormatClock(dur,true)}`;
}
// CapCut-style: the playhead line is FIXED at the horizontal center of the timeline viewport.
// The track content scrolls underneath it instead. We scroll the container so that
// "current time" always lands under that fixed center line.
//
// ROOT CAUSE #3 (scroll-seek feedback loop): movePlayhead() scrolls the container
// programmatically to follow the playhead; the timeline's native 'scroll' event listener
// (initTimelineScrub, below) ALSO seeks time from scrollLeft, because that's how genuine
// user drag-scrubbing is detected. Without a way to tell those two apart, a programmatic
// scroll re-triggers the scroll listener, which re-seeks time, which re-triggers
// movePlayhead — a feedback loop, previously patched with fragile
// suppressScrollSeek/timing-window flags. FIX: an explicit ownership flag. Only the
// timeline's own pointer state machine sets timelineScrollOwner='user' while actively
// dragging; movePlayhead() (and any other programmatic scroll) leaves it alone, and the
// scroll listener only ever seeks time when it IS 'user'. This is not a timing race; it's
// a direct state check.
let timelineScrollOwner = 'programmatic'; // 'user' | 'programmatic'
function movePlayhead(){
  const scroller = document.getElementById('timelineScroll');
  if(!scroller) return;
  const pps = pxPerSecond();
  // currentAbsoluteTimeSec IS the absolute timeline position already — no trimStart
  // arithmetic needed here anymore (that used to be re-derived from clip-relative `frame`,
  // which is what caused keyframes/scrubbing to land in the wrong spot whenever trimStart
  // > 0 or the wrong clip's trimStart got used).
  const absSec = currentAbsoluteTimeSec;
  // .timeline-inner has left/right padding == scroller.clientWidth/2 (see syncTimelineInnerWidth),
  // so content at absSec sits at pixel (pad + absSec*pps) inside it. To land that pixel under the
  // sticky center line, scrollLeft must equal absSec*pps exactly (the two half-widths cancel out).
  const targetScroll = Math.max(0, absSec*pps);
  const wasOwner = timelineScrollOwner;
  timelineScrollOwner = 'programmatic';
  if(Math.abs(scroller.scrollLeft-targetScroll)>0.5) scroller.scrollLeft = targetScroll;
  timelineScrollOwner = wasOwner === 'user' ? 'user' : 'programmatic';
}
// Keeps the playhead line exactly as tall as the visible timeline viewport, so it is
// always fully visible top-to-bottom and never clipped, cut short, or left floating
// after layers are added/removed or the panel is resized.
function syncPlayheadHeight(){
  const scroller = document.getElementById('timelineScroll');
  const line = document.getElementById('playhead');
  if(!scroller || !line) return;
  line.style.height = scroller.clientHeight + 'px';
}

/* ---------------- Timeline clip rendering (drag-trim + cut) ---------------- */
const PPS_BASE = 90; // fixed px-per-second zoom level, CapCut-style (content scrolls, doesn't stretch)
// FEATURE (two-finger pinch on a clip zooms the timeline for finer editing): a second
// finger landing on a clip while a first is already down scales every clip/ruler/handle
// wider on screen (never taller — this only ever changes horizontal px-per-second), the same
// way editors like CapCut let you pinch the timeline open to get more room on a short clip.
// timelineZoom is a session-only multiplier (not saved — it resets to 1 on reload, same as
// any other "current view" state like scroll position) applied on top of PPS_BASE. Every one
// of this file's 20+ callers of pxPerSecond() automatically respects it since they all read
// through this single function — no other call site needed to change.
let timelineZoom = 1;
const TIMELINE_ZOOM_MIN = 0.2, TIMELINE_ZOOM_MAX = 6; // v279: min lowered from 1 so pinching IN can shrink the timeline (zoom out)
function pxPerSecond(){
  return PPS_BASE * timelineZoom;
}

/* ---------------- Double-tap + drag to extend a clip (all timeline clip types) ----------------
   A quick double press on the same clip arms edge-trim mode. The second press is captured here
   before a clip's normal move/scrub handler sees it, so horizontal dragging changes trimEnd and
   cannot accidentally slide the clip. A normal tap/drag and the existing visible edge handles
   keep their original behavior. */
let _doubleTapClipDown = null;
let _lastClipTap = null;
let _doubleTapTrimGesture = null;
function timelineClipTargetAt(node){
  if(!node || !node.closest) return null;
  if(node.closest('.kf-marker, .anim-clip-handle, [data-ov-handle], [data-bg-handle], [data-seg-handle]')) return null;
  const mainEl = node.closest('#animClipEl');
  if(mainEl && activeClip) return {key:'main', type:'main', model:activeClip, el:mainEl};
  const segEl = node.closest('.anim-clip[data-seg-idx]');
  if(segEl){
    const idx = Number(segEl.dataset.segIdx);
    const model = splitStickmanSegments[idx];
    if(model) return {key:'seg:'+idx, type:'seg', index:idx, model, el:segEl};
  }
  const overlayEl = node.closest('.overlay-clip[data-overlay-id]');
  if(overlayEl){
    const model = overlays.find(o=>String(o.id)===String(overlayEl.dataset.overlayId));
    if(model) return {key:'ov:'+model.id, type:'overlay', model, el:overlayEl};
  }
  const bgEl = node.closest('.track-clip[data-bg-id]');
  if(bgEl){
    const model = backgrounds.find(b=>String(b.id)===String(bgEl.dataset.bgId));
    if(model) return {key:'bg:'+model.id, type:'background', model, el:bgEl};
  }
  return null;
}
function ensureDoubleTapClipTrimWired(){
  const scroller = document.getElementById('timelineScroll');
  if(!scroller || scroller.dataset.doubleTapTrimWired) return;
  scroller.dataset.doubleTapTrimWired = '1';
  scroller.addEventListener('pointerdown', e=>{
    // A second simultaneous touch belongs to the pinch recognizer, never double-tap trim.
    if(e.pointerType === 'touch' && e.isPrimary === false){
      _lastClipTap = null; _doubleTapClipDown = null;
      if(_doubleTapTrimGesture){
        const old = _doubleTapTrimGesture;
        old.target.model.trimEnd = old.origEnd;
        old.target.el.classList.remove('trim-gesture-active');
        if(old.target.type === 'background') bgHandleDragActive = false;
        _doubleTapTrimGesture = null; disarmClipDrag(); hideTrimFrameBadge();
      }
      return;
    }
    if(_doubleTapTrimGesture){ e.preventDefault(); e.stopImmediatePropagation(); return; }
    const target = timelineClipTargetAt(e.target);
    if(!target || e.button > 0) { _doubleTapClipDown = null; _lastClipTap = null; return; }
    const now = Date.now();
    if(_lastClipTap && _lastClipTap.key===target.key && now-_lastClipTap.time < 550){
      _lastClipTap = null;
      e.preventDefault(); e.stopImmediatePropagation();
      if(isPlaying) pausePlayback();
      const liveTarget = timelineClipTargetAt(e.target) || target;
      try{ selectClipForChrome(liveTarget.type==='main'?'main':liveTarget.type==='seg'?'seg'+liveTarget.index:liveTarget.model.id); }catch(_){}
      _doubleTapTrimGesture = {
        pointerId:e.pointerId, target:liveTarget, startX:e.clientX,
        origStart:liveTarget.model.trimStart, origEnd:liveTarget.model.trimEnd,
        moved:false, startedAt:now
      };
      _doubleTapClipDown = null;
      try{ scroller.setPointerCapture(e.pointerId); }catch(_){}
      liveTarget.el.classList.add('trim-gesture-active');
      if(liveTarget.type==='background') bgHandleDragActive = true;
      armClipDrag();
      if(navigator.vibrate) navigator.vibrate(8);
      return;
    }
    _doubleTapClipDown = { key:target.key, time:now, startX:e.clientX, startY:e.clientY, pointerId:e.pointerId, moved:false };
  }, {capture:true});
  scroller.addEventListener('pointermove', e=>{
    if(!_doubleTapTrimGesture){
      if(_doubleTapClipDown && e.pointerId===_doubleTapClipDown.pointerId && Math.hypot(e.clientX-_doubleTapClipDown.startX,e.clientY-_doubleTapClipDown.startY)>9) _doubleTapClipDown.moved=true;
      return;
    }
    const g = _doubleTapTrimGesture;
    if(e.pointerId!==g.pointerId) return;
    e.preventDefault(); e.stopImmediatePropagation();
    const model = g.target.model;
    const pps = pxPerSecond();
    let newEnd = Math.max(g.origStart + MIN_CLIP_SEC, g.origEnd + (e.clientX-g.startX)/pps);
    newEnd = Math.min(newEnd, g.origStart + 120);
    if(g.target.type==='background'){
      const next = backgrounds.filter(b=>b.id!==model.id && b.trimStart>=g.origEnd-0.001).reduce((m,b)=>Math.min(m,b.trimStart), Infinity);
      if(Number.isFinite(next)) newEnd = Math.min(newEnd, next-0.03);
    }
    model.trimEnd = Math.max(g.origStart+MIN_CLIP_SEC, newEnd);
    g.moved = g.moved || Math.abs(e.clientX-g.startX)>1;
    if(model.trimEnd > timelineDuration-0.3){ timelineDuration=model.trimEnd+1; syncTimelineInnerWidth(); renderRuler(); }
    // Direct DOM width writes avoid replacing the captured element mid-gesture. Defer
    // character collision resolution, background neighbour shifts and overlay lane packing
    // until the gesture ends; these full-row passes must never run inside pointermove.
    g.target.el.style.width = Math.max(MIN_CLIP_SEC*pps,(model.trimEnd-model.trimStart)*pps)+'px';
    updateTimeReadout();
    showTrimFrameBadge(g.target.el, model.trimEnd-g.origEnd, e.clientX);
  }, {capture:true, passive:false});
  const finish = (e, cancelled=false)=>{
    if(_doubleTapTrimGesture && e.pointerId===_doubleTapTrimGesture.pointerId){
      e.preventDefault(); e.stopImmediatePropagation();
      const g = _doubleTapTrimGesture;
      _doubleTapTrimGesture = null;
      if(cancelled) g.target.model.trimEnd = g.origEnd;
      g.target.el.classList.remove('trim-gesture-active');
      if(g.target.type==='background') bgHandleDragActive = false;
      disarmClipDrag(); hideTrimFrameBadge();
      if(g.moved || cancelled){
        if(g.target.type==='main' || g.target.type==='seg'){
          applyCharClipLanes(g.target.type==='main'?'main':'seg'+g.target.index);
          renderTimelineClip();
        }
        else if(g.target.type==='background'){
          resolveBgClipOverlaps(g.target.model.id);
          renderBackgroundTrack(); syncTimelineInnerWidth();
        }
        else if(g.target.type==='overlay'){ renderOverlayLayers(); selectClipForChrome(g.target.model.id); }
        setCurrentAbsoluteTime(currentAbsoluteTimeSec);
        renderFrame();
        if(!cancelled) persistProjectState();
      }
      try{ scroller.releasePointerCapture(e.pointerId); }catch(_){}
      _lastClipTap = null; _doubleTapClipDown = null;
      return;
    }
    if(_doubleTapClipDown && e.pointerId===_doubleTapClipDown.pointerId){
      if(!_doubleTapClipDown.moved) _lastClipTap = {key:_doubleTapClipDown.key,time:Date.now()};
      else _lastClipTap = null;
      _doubleTapClipDown = null;
    }
  };
  scroller.addEventListener('pointerup', e=>finish(e,false), {capture:true});
  scroller.addEventListener('pointercancel', e=>finish(e,true), {capture:true});
}
// Stable timeline-level two-finger zoom. The old per-clip handler allowed normal scrub/drag
// handlers to keep running during a pinch and called syncTimelineInnerWidth() on every move,
// which rebuilt the background row and let scrollLeft jump as browser layout clamped the width.
// This handler owns both pointers globally for the duration of the pinch and updates clip widths
// in place. It works even when the fingers land on different child elements of the same track.
let timelineZoomGesture = null;
let timelinePinchCandidate = null;
let _timelinePinchPointers = new Map();
let _timelinePinchEnding = false;
function wireClipZoomGesture(el){
  // Retained so existing render paths need no rewriting. Pinch zoom is now handled once by

}
function dispatchTimelinePointerCancel(pointer){
  if(!pointer || !pointer.target || typeof PointerEvent === 'undefined') return;
  try{
    pointer.target.dispatchEvent(new PointerEvent('pointercancel', {
      bubbles:true, cancelable:false, pointerId:pointer.id, pointerType:'touch',
      clientX:pointer.x, clientY:pointer.y, buttons:0, isPrimary:false
    }));
  }catch(_){}
}

function ensureTimelinePinchZoomWired(){
  const scroller = document.getElementById('timelineScroll');
  if(!scroller || scroller.dataset.stablePinchWired) return;
  scroller.dataset.stablePinchWired = '1';
  // Accept both fingers anywhere inside the timeline canvas. Split clips can become too
  // narrow for two fingers to fit on the same chip, so requiring two clip-surface hits made
  // pinch zoom fail specifically on short first/second split pieces.
  const isTimelineSurface = node => !!(node && node.closest && node.closest('#timelineInner, #timelineScroll'));
  const isTrackedPair = g => g && g.pointerIds && g.pointerIds.length===2 &&
    g.pointerIds.every(id=>_timelinePinchPointers.has(id));
  let pinchMoveRaf = 0;
  const processLatestPinchMove = ()=>{
    pinchMoveRaf = 0;
    const candidate = timelinePinchCandidate;
    const g = timelineZoomGesture || candidate;
    if(!g || !isTrackedPair(g)) return;
    const a = _timelinePinchPointers.get(g.pointerIds[0]);
    const b = _timelinePinchPointers.get(g.pointerIds[1]);
    if(!a || !b) return;
    const dist = Math.max(1,Math.hypot(a.x-b.x,a.y-b.y));

    if(!timelineZoomGesture){
      if(!candidate || Math.abs(dist-candidate.startDist)<8) return;
      timelineZoomGesture = candidate;
      timelinePinchCandidate = null;
      if(navigator.vibrate) navigator.vibrate(8);
    }
    const zoomGesture = timelineZoomGesture;
    const newZoom = Math.max(TIMELINE_ZOOM_MIN,Math.min(TIMELINE_ZOOM_MAX,zoomGesture.startZoom*(dist/zoomGesture.startDist)));
    const midX = (a.x+b.x)/2;
    let zoomChanged = false;
    if(Math.abs(newZoom-timelineZoom)>=0.004){
      timelineZoom = newZoom;
      // Reposition existing nodes in place once per display frame.
      repositionAllClipsForZoom(false);
      setTimelineWidthForPinch(zoomGesture.scrollerClientWidth);
      zoomChanged = true;
    }
    const pad = zoomGesture.scrollerClientWidth / 2;
    const nextScrollLeft = Math.max(0,zoomGesture.anchorSec*pxPerSecond()+pad-(midX-zoomGesture.scrollerRect.left));
    if(zoomChanged){
      // v279 FIX (ruler/frame text stretched sideways while pinching): the old code scaled the whole
      // ruler with CSS scaleX, which also stretched every label. Instead the ruler is redrawn at the
      // new zoom with normal-width text, but only for the part of the ruler that is on screen
      // (plus a margin), so it stays cheap enough to run once per display frame.
      const ppsNow = pxPerSecond(), marginSec = (pad + 80) / ppsNow + 1;
      renderRuler({
        from: Math.max(0,(nextScrollLeft - pad) / ppsNow - marginSec),
        to: (nextScrollLeft + zoomGesture.scrollerClientWidth - pad) / ppsNow + marginSec
      });
    }
    zoomGesture.scroller.scrollLeft = nextScrollLeft;
  };
  const schedulePinchMoveFrame = ()=>{
    if(pinchMoveRaf) return;
    pinchMoveRaf = requestAnimationFrame(processLatestPinchMove);
  };

  // Any interaction in the timeline pauses playback immediately. Child clip handlers
  // still own selection, scrubbing, trimming and moving; this capture listener only ensures
  // the playhead cannot continue running underneath a touch on a badge/blank track/clip.
  scroller.addEventListener('pointerdown', e=>{
    if(isPlaying) pausePlayback();
    if(e.pointerType && e.pointerType !== 'touch' && e.pointerType !== 'pen') return;
    if(_timelinePinchEnding){
      _timelinePinchPointers.set(e.pointerId, {id:e.pointerId, x:e.clientX, y:e.clientY, target:e.target});
      e.preventDefault(); e.stopImmediatePropagation();
      return;
    }
    _timelinePinchPointers.set(e.pointerId, {id:e.pointerId, x:e.clientX, y:e.clientY, target:e.target});
    if(_timelinePinchPointers.size < 2) return;

    const points = [..._timelinePinchPointers.values()].slice(0,2);
    // Zoom belongs to a genuine two-touch gesture inside the timeline. This still excludes
    // touches outside the timeline while allowing short split clips to be pinched reliably.
    if(!isTimelineSurface(points[0].target) || !isTimelineSurface(points[1].target)){
      _timelinePinchPointers.delete(e.pointerId);
      return;
    }

    // A two-touch candidate cancels the first touch's pending scrub/move state, but zoom
    // does not begin until the distance actually changes by 8px. Thus two fingers merely
    // touching a clip do not cause a zoom; spreading/pinching them does.
    e.preventDefault(); e.stopImmediatePropagation();
    const first=points[0], second=points[1];
    try{ scroller.setPointerCapture(first.id); }catch(_){}
    try{ scroller.setPointerCapture(second.id); }catch(_){}
    _lastClipTap = null; _doubleTapClipDown = null;
    if(_doubleTapTrimGesture){
      const old=_doubleTapTrimGesture;
      old.target.model.trimEnd=old.origEnd;
      old.target.el.classList.remove('trim-gesture-active');
      if(old.target.type==='background') bgHandleDragActive=false;
      _doubleTapTrimGesture=null; disarmClipDrag(); hideTrimFrameBadge();
    }
    dispatchTimelinePointerCancel(first);
    const dist=Math.max(1,Math.hypot(first.x-second.x,first.y-second.y));
    const midX=(first.x+second.x)/2;
    const rect=scroller.getBoundingClientRect();
    const scrollerRect={left:rect.left,width:rect.width};
    const scrollerClientWidth=scroller.clientWidth;
    const startPps=pxPerSecond();
    const pad=scrollerClientWidth/2;
    const midContentX=midX-scrollerRect.left+scroller.scrollLeft;
    timelinePinchCandidate={
      pointerIds:[first.id,second.id], startDist:dist, startZoom:timelineZoom,
      anchorSec:Math.max(0,(midContentX-pad)/startPps), scroller, scrollerRect,
      scrollerClientWidth, startPps, startMidX:midX
    };
    timelineScrollOwner='programmatic';
  }, {capture:true,passive:false});

  scroller.addEventListener('pointermove', e=>{
    if(!_timelinePinchPointers.has(e.pointerId)) return;
    const tracked=_timelinePinchPointers.get(e.pointerId);
    tracked.x=e.clientX; tracked.y=e.clientY;

    const candidate=timelinePinchCandidate;
    const active=timelineZoomGesture;
    const g = active && active.pointerIds.includes(e.pointerId) ? active :
              candidate && candidate.pointerIds.includes(e.pointerId) ? candidate : null;
    if(!g){
      if(_timelinePinchEnding){ e.preventDefault(); e.stopImmediatePropagation(); }
      return;
    }
    e.preventDefault(); e.stopImmediatePropagation();
    // The pointer map is updated on every raw sample; expensive zoom work consumes only its latest values once per frame.
    schedulePinchMoveFrame();
  }, {capture:true,passive:false});

  const endPointer=e=>{
    if(!e || !_timelinePinchPointers.has(e.pointerId)) return;
    // Do not lose the final coalesced sample when a finger lifts between animation frames.
    if(pinchMoveRaf){cancelAnimationFrame(pinchMoveRaf);pinchMoveRaf=0;processLatestPinchMove();}
    const wasPinching=!!timelineZoomGesture;
    const finishedZoomGesture=timelineZoomGesture;
    const wasCandidate=!!timelinePinchCandidate;
    const shouldBlock=wasPinching || wasCandidate || _timelinePinchEnding;
    if(shouldBlock){ e.preventDefault(); e.stopImmediatePropagation(); }
    _timelinePinchPointers.delete(e.pointerId);

    if(wasPinching){
      timelineZoomGesture=null;
      timelinePinchCandidate=null;
      _timelinePinchEnding=_timelinePinchPointers.size>0;
      timelineScrollOwner='programmatic';
      // One cleanup pass after the gesture finishes; no clip DOM replacements mid-touch.
      repositionAllClipsForZoom(true);
      renderRuler();
      setTimelineWidthForPinch(finishedZoomGesture ? finishedZoomGesture.scrollerClientWidth : undefined);
    } else if(wasCandidate){
      timelinePinchCandidate=null;
      _timelinePinchEnding=_timelinePinchPointers.size>0;
      timelineScrollOwner='programmatic';
    }
    if(_timelinePinchPointers.size===0) _timelinePinchEnding=false;
  };
  scroller.addEventListener('pointerup',endPointer,{capture:true,passive:false});
  scroller.addEventListener('pointercancel',e=>{
    if(!e.isTrusted) return; // ignore synthetic cancellation used to reset a child gesture
    endPointer(e);
  },{capture:true,passive:false});
}

// Repositions every currently-rendered clip (main stickman clip, split segments, background
// clips, overlay/audio clips) and their keyframe-dot children purely by writing style.left/
// style.width on the elements that already exist in the DOM — mirrors exactly what each
// clip type's own local reposition()/applyCharClipLanes()-style helper already does during a
// normal drag, just applied to ALL clips at once instead of one being dragged. Deliberately
// does NOT touch top/height or re-run lane-packing (packIntoLanes/resolveCharClipOverlaps) —
// a zoom changes horizontal scale only, so which lane/row a clip sits in never changes, and
// skipping that work keeps this cheap enough to call on every pinch pointermove frame.
function repositionAllClipsForZoom(redrawWaveforms=true){
  const pps = pxPerSecond();
  // Main stickman clip + its keyframe dots.
  if(activeClip){
    const mainEl = document.getElementById('animClipEl');
    if(mainEl){
      mainEl.style.left = (activeClip.trimStart*pps)+'px';
      mainEl.style.width = Math.max(MIN_CLIP_SEC*pps, (activeClip.trimEnd-activeClip.trimStart)*pps)+'px';
      const kfTrack = document.getElementById('kfTrack');
      if(kfTrack){
        activeClip.keyframes.forEach((kf, idx)=>{
          const marker = kfTrack.querySelector(`[data-kf-idx="${idx}"]`);
          if(marker) marker.style.left = ((kf.t - activeClip.trimStart)*pps)+'px';
        });
      }
    }
  }
  // Split segments + their own keyframe dots.
  splitStickmanSegments.forEach((seg, idx)=>{
    const el = document.getElementById('seg'+idx+'El');
    if(!el) return;
    el.style.left = (seg.trimStart*pps)+'px';
    el.style.width = Math.max(MIN_CLIP_SEC*pps, (seg.trimEnd-seg.trimStart)*pps)+'px';
    const kfTrack = document.getElementById('kfTrack-seg'+idx);
    if(kfTrack){
      (seg.keyframes||[]).forEach((kf, kidx)=>{
        const marker = kfTrack.querySelector(`[data-kf-idx="${kidx}"]`);
        if(marker) marker.style.left = ((kf.t - seg.trimStart)*pps)+'px';
      });
    }
  });
  // Build the two clip lookup maps in one DOM pass instead of scanning the document once per model item.
  const backgroundElementsById = new Map();
  const overlayElementsById = new Map();
  document.querySelectorAll('.track-clip[data-bg-id], .overlay-clip[data-overlay-id]').forEach(el=>{
    if(el.dataset.bgId!=null) backgroundElementsById.set(String(el.dataset.bgId),el);
    if(el.dataset.overlayId!=null) overlayElementsById.set(String(el.dataset.overlayId),el);
  });
  // Background clips.
  (typeof backgrounds !== 'undefined' ? backgrounds : []).forEach(b=>{
    const el = backgroundElementsById.get(String(b.id));
    if(!el) return;
    el.style.left = (b.trimStart*pps)+'px';
    el.style.width = Math.max(MIN_CLIP_SEC*pps, (b.trimEnd-b.trimStart)*pps)+'px';
  });
  // Overlay/audio clips — redraw the waveform too, since it's drawn to a fixed canvas
  // width that needs to match the clip's new on-screen width or it'll look stretched.
  (typeof overlays !== 'undefined' ? overlays : []).forEach(ov=>{
    const el = overlayElementsById.get(String(ov.id));
    if(!el) return;
    el.style.left = (ov.trimStart*pps)+'px';
    el.style.width = Math.max(MIN_CLIP_SEC*pps, (ov.trimEnd-ov.trimStart)*pps)+'px';
    if(redrawWaveforms && ov.type==='audio' && typeof drawWaveform === 'function') drawWaveform(ov);
  });
  // Video thumbnails are recalculated after the new zoom geometry is committed.
  try{if(window.__evTimelineVideoThumbs)window.__evTimelineVideoThumbs.schedule();}catch(_e){}
}

// v279: trim handles (the wide white left/right edge bars of a selected clip) now need a LONG PRESS
// before they trim. A quick slide that starts on a handle behaves like sliding on the clip body or on
// empty timeline space: sideways = scrub the playhead, up/down = scroll the layer rows. Only holding
// the finger still for ~280ms arms trim; after that, dragging trims/extends that edge.
function ethvelleMakeHandleHoldGate(onArm){
  const HOLD_MS = 280, MOVE_TOL = 8;
  let g = null, timer = 0;
  const clearTimer = ()=>{ if(timer){ clearTimeout(timer); timer = 0; } };
  return {
    down(e){
      clearTimer();
      const sc = document.getElementById('timelineScroll');
      g = { type:'pending', pointerId:e.pointerId, startX:e.clientX, startY:e.clientY, lastX:e.clientX,
            startTime:getCurrentAbsoluteTime(), scrollTopStart:sc?sc.scrollTop:0 };
      timer = setTimeout(()=>{
        timer = 0;
        if(!g || g.type!=='pending') return;
        g.type = 'trim';
        try{ if(navigator.vibrate) navigator.vibrate(10); }catch(_e){}
        onArm(g.lastX);
      }, HOLD_MS);
    },
    // 'trim'    -> long-press already fired: caller runs its normal trim-drag code for this move
    // 'handled' -> gate consumed the move (still waiting / scrubbing / scrolling rows)
    // ''        -> this pointer is not ours
    move(e){
      if(!g || e.pointerId!==g.pointerId) return '';
      g.lastX = e.clientX;
      if(g.type==='trim') return 'trim';
      if(g.type==='scroll'){ applyClipVerticalScrollHandoff(g, e.clientY); return 'handled'; }
      if(g.type==='scrub'){ scheduleTimelineScrub(timeFromDragAnchor(g.startX,g.startTime,e.clientX)); return 'handled'; }
      const dx = Math.abs(e.clientX-g.startX), dy = Math.abs(e.clientY-g.startY);
      if(dy>=MOVE_TOL && dy>=dx*0.82){ clearTimer(); g.type='scroll'; applyClipVerticalScrollHandoff(g,e.clientY); return 'handled'; }
      if(dx>=MOVE_TOL){ clearTimer(); g.type='scrub'; scheduleTimelineScrub(timeFromDragAnchor(g.startX,g.startTime,e.clientX)); }
      return 'handled';
    },
    // Ends the gesture; returns what it was: 'pending' (plain tap), 'scrub', 'scroll' or 'trim'.
    up(e){
      clearTimer();
      const done = g; g = null;
      if(!done) return '';
      if(done.type==='scrub'){ try{ flushTimelineScrub(timeFromDragAnchor(done.startX,done.startTime,e.clientX), false); }catch(_e){} }
      return done.type;
    },
    cancel(){ clearTimer(); const done = g; g = null; return done ? done.type : ''; }
  };
}

// Additive transform/volume keyframes for Background, Image/Text, Video and Audio clips.
// The original Character pose-keyframe system above remains unchanged.
const ETHVELLE_OBJECT_KF_TOLERANCE = 0.05;
function ethvelleObjectKeyframeContactWindow(){
  // Match the visual half-width of the 15px keyframe diamond, so any overlap with the
  // fixed 2px playhead line counts as being parked on that keyframe at every timeline zoom.
  return Math.max(ETHVELLE_OBJECT_KF_TOLERANCE,10.5/Math.max(1,pxPerSecond()));
}

function ethvelleObjectKeyframeProperty(clip){
  return clip&&clip.type==='audio'?'volume':'transform';

function ethvelleCloneObjectKeyframeValue(value,property){
  if(property==='volume')return Number(value==null?1:value);
  const v=value||{};
  return {scale:Number(v.scale==null?1:v.scale),offsetX:Number(v.offsetX)||0,offsetY:Number(v.offsetY)||0,rotation:Number(v.rotation)||0};
}
function ethvelleObjectBaseValue(clip,property){
  const base=clip&&clip.ethvelleMediaKeyframeBase;
  if(base&&Object.prototype.hasOwnProperty.call(base,property))return ethvelleCloneObjectKeyframeValue(base[property],property);
  if(property==='volume')return Number(clip&&clip.volume!=null?clip.volume:1);
  return ethvelleCloneObjectKeyframeValue(clip&&clip.transform,property);
}
function ethvelleInterpolateObjectValue(a,b,p,property){
  const mix=(x,y)=>Number(x||0)+(Number(y||0)-Number(x||0))*p;
  if(property==='volume')return mix(a,b);
  const out={};
  ['scale','offsetX','offsetY'].forEach(k=>out[k]=mix(a&&a[k],b&&b[k]));
  const ar=Number(a&&a.rotation)||0,br=Number(b&&b.rotation)||0;
  const delta=((br-ar+540)%360)-180;
  out.rotation=ar+delta*p;
  return out;
}
const ethvelleObjectKeyframeCache=new WeakMap();
function ethvelleGetObjectKeyframeValue(clip,property,time){
  if(!clip)return null;
  const t=Number(time),source=Array.isArray(clip.ethvelleMediaKeyframes)?clip.ethvelleMediaKeyframes:[];
  if(!Number.isFinite(t)||!source.length)return null;
  let cache=ethvelleObjectKeyframeCache.get(clip);
  // The keyframe array is often edited in place (same array and same item count):
  // an existing key's time/property/value may change without changing array identity.
  // Validate the small key list before reusing the sorted groups, otherwise edits can
  // keep rendering stale interpolation or show a key at its old time.
  let cacheValid=!!(cache&&cache.source===source&&cache.length===source.length);
  if(cacheValid){
    for(let i=0;i<source.length;i++){
      const key=source[i],entry=cache.entries&&cache.entries[i];
      if(!entry||entry.ref!==key||entry.t!==Number(key&&key.t)||entry.property!==(key&&key.property)||entry.valueRef!==(key&&key.value)){cacheValid=false;break;}
    }
  }
  if(!cacheValid){
    const groups=Object.create(null),entries=new Array(source.length);
    for(let i=0;i<source.length;i++){
      const key=source[i];
      entries[i]={ref:key,t:Number(key&&key.t),property:key&&key.property,valueRef:key&&key.value};
      if(!key||!key.property||!Number.isFinite(Number(key.t))||key.value==null)continue;
      (groups[key.property]||(groups[key.property]=[])).push(key);
    }
    Object.keys(groups).forEach(name=>groups[name].sort((a,b)=>Number(a.t)-Number(b.t)));
    cache={source,length:source.length,entries,groups};ethvelleObjectKeyframeCache.set(clip,cache);
  }
  const keys=cache.groups[property]||[];if(!keys.length)return null;
  const first=keys[0],firstT=Number(first.t),base=ethvelleObjectBaseValue(clip,property);
  const clipStart=Number.isFinite(Number(clip.trimStart))?Number(clip.trimStart):firstT;
  if(t<firstT){
    const span=firstT-clipStart;
    if(span<=0)return ethvelleCloneObjectKeyframeValue(first.value,property);
    return ethvelleInterpolateObjectValue(base,first.value,Math.max(0,Math.min(1,(t-clipStart)/span)),property);
  }
  let prev=first;
  for(let i=1;i<keys.length;i++){
    const next=keys[i],a=Number(prev.t),b=Number(next.t);
    if(t<=b){const f=b>a?Math.max(0,Math.min(1,(t-a)/(b-a))):1;return ethvelleInterpolateObjectValue(prev.value,next.value,f,property);}
    prev=next;
  }
  return ethvelleCloneObjectKeyframeValue(prev.value,property);
}

  if(!Array.isArray(clip.ethvelleMediaKeyframes))clip.ethvelleMediaKeyframes=[];
  if(!clip.ethvelleMediaKeyframeBase||typeof clip.ethvelleMediaKeyframeBase!=='object')clip.ethvelleMediaKeyframeBase={};
  if(!Object.prototype.hasOwnProperty.call(clip.ethvelleMediaKeyframeBase,property)){
    clip.ethvelleMediaKeyframeBase[property]=ethvelleObjectBaseValue(clip,property);
  }
  const keys=clip.ethvelleMediaKeyframes;
  const existing=keys.findIndex(k=>k&&k.property===property&&Math.abs(Number(k.t)-t)<=ethvelleObjectKeyframeContactWindow());
  if(existing>=0){keys.splice(existing,1);window.__ethvelleSelectedObjectKeyframe=null;}
  else{
    // Snapshot the currently displayed/interpolated value, not merely the authored base.
    // This lets a newly inserted key preserve the motion/volume that was live at this time.
    const liveValue=ethvelleGetObjectKeyframeValue(clip,property,t);
    const value=property==='volume'
      ? Number(liveValue==null?(clip.volume==null?1:clip.volume):liveValue)
      : ethvelleCloneObjectKeyframeValue(liveValue||clip.transform,property);
    keys.push({t,property,value});
    keys.sort((a,b)=>Number(a.t)-Number(b.t));
    const idx=keys.findIndex(k=>k&&k.property===property&&Math.abs(Number(k.t)-t)<0.00001);
    window.__ethvelleSelectedObjectKeyframe={id:String(id),index:idx};
  }
  // The clip row and its badge/icon are already in the DOM. Refresh only this clip's
  // keyframe-marker track; rebuilding the complete Background/Media row caused its badge
  // to jump sideways for one frame whenever a keyframe was added or removed.
  ethvelleRefreshObjectKeyframeTracks(id);
  updateKeyButtonState();renderFrame();
  showToast(existing>=0?'Keyframe removed':'Keyframe added',{icon:toastIcons.check,duration:1200});
  persistProjectState();
}
function ethvelleScheduleObjectVolumeAutomation(param,clip,when,duration){
  if(!param||!clip||!Number.isFinite(duration)||duration<=0)return;
  const start=Number(clip.trimStart)||0,end=Math.min(Number(clip.trimEnd)||start+duration,start+duration);
  const base=ethvelleGetObjectKeyframeValue(clip,'volume',start);
  const keys=(Array.isArray(clip.ethvelleMediaKeyframes)?clip.ethvelleMediaKeyframes:[])
    .filter(k=>k&&k.property==='volume'&&Number(k.t)>start&&Number(k.t)<end)
    .slice().sort((a,b)=>Number(a.t)-Number(b.t));
  const clampVol=v=>Math.max(0,Math.min(2,Number(v)||0));
  try{
    param.cancelScheduledValues(when);
    param.setValueAtTime(clampVol(base==null?ethvelleObjectBaseValue(clip,'volume'):base),when);
    keys.forEach(k=>param.linearRampToValueAtTime(clampVol(k.value),when+(Number(k.t)-start)));
    const atEnd=ethvelleGetObjectKeyframeValue(clip,'volume',Math.max(start,end-0.0001));
    if(atEnd!=null)param.linearRampToValueAtTime(clampVol(atEnd),when+Math.max(0,duration));
  }catch(_e){try{param.value=clampVol(base==null?clip.volume:base);}catch(__e){}}
}
// Sub-lane vertical geometry for a character's own clip row — mirrors SUB_LANE_H(48)/the
// +6/+4 spacing renderOverlayLayers already uses for Clips/Audio, just applied to the
// main+segments clip lane instead. Kept as named constants so both renderTimelineClip
// (initial layout) and the live-drag repack path (wireClipInteractions/
// wireSplitSegmentInteractions) always agree on the exact same numbers.
// Real (non-pseudo-element) selection border bars, one per edge — see the .track-clip.selected/
// .anim-clip.selected/.overlay-clip.selected CSS comments for why this replaced the old
// box-shadow-ring approach. Only visually shows once the CSS `.selected` class is also on the
// clip (`.track-clip .sel-edge-top` etc select unconditionally, `.track-clip.selected` is what
// actually needs the border), but it's harmless/invisible markup to include on every clip
// unconditionally — far simpler than conditionally splicing it in only for the selected one,
// and it costs nothing since these four divs are just thin (2-3px) empty boxes.
const SEL_EDGE_HTML = '<div class="sel-edge-top"></div><div class="sel-edge-bottom"></div><div class="sel-edge-left"></div><div class="sel-edge-right"></div>';
const CHAR_SUB_LANE_H = 36;
const CHAR_LANE_GAP = 6;
function charClipLaneTop(laneIdx){ return 3 + laneIdx*(CHAR_SUB_LANE_H+CHAR_LANE_GAP); }
// Character tracks use the same overlap packing model as Clips/Text/Audio: clips remain at
// their actual timeline times, and any time-overlap is represented by a separate vertical
// sub-lane. This keeps the active clip under the pointer and never shifts another clip's time.
function charRowHeightForLanes(laneCount){
  const count=Math.max(1,Math.floor(Number(laneCount)||1));
  // Preserve the existing 42px track / 46px row in the single-lane case; each added
  // lane contributes one 36px clip plus the same 6px breathing gap.
  return CHAR_SUB_LANE_H+6+(count-1)*(CHAR_SUB_LANE_H+CHAR_LANE_GAP);
}
function characterClipLaneLayout(clip, segments){
  const entries=getAllStickmanClips(clip,segments);
  const lanes=[], laneOf=new Map();
  entries.forEach(entry=>{
    const start=Number(entry.clip.trimStart)||0;
    let laneIndex=lanes.findIndex(lane=>start >= lane.end-0.001);
    if(laneIndex<0) laneIndex=lanes.length;
    if(!lanes[laneIndex]) lanes[laneIndex]={end:-Infinity};
    lanes[laneIndex].end=Number(entry.clip.trimEnd)||start;
    laneOf.set(entry.ref,laneIndex);
  });
  const signature=entries.map(entry=>entry.ref+':'+laneOf.get(entry.ref)).join('|');
  return {laneOf,laneCount:Math.max(1,lanes.length),signature,entries};
}
// Kept as a compatibility hook for the speed panel and older call sites. Character clips no
// longer push one another sideways in time; this now only returns their vertical lane layout.
function resolveCharClipOverlaps(clip, segments, movedRef){
  return characterClipLaneLayout(clip,segments);
}
function setCharacterLaneGeometry(row,track,laneCount){
  if(!row || !track) return;
  const trackH=charRowHeightForLanes(laneCount);
  const rowH=trackH+4;
  const rowValue=rowH+'px',trackValue=trackH+'px';
  if(row.style.getPropertyValue('--ev-char-row-height')!==rowValue) row.style.setProperty('--ev-char-row-height',rowValue);
  if(track.style.getPropertyValue('--ev-char-track-height')!==trackValue) track.style.setProperty('--ev-char-track-height',trackValue);
  // Keep inline fallbacks for browsers / embedded WebViews that ignore the custom properties.
  if(row.style.height!==rowValue) row.style.height=rowValue;
  if(track.style.height!==trackValue) track.style.height=trackValue;
  return {rowH,trackH};
}
function setCharacterClipLane(el,laneIndex){
  if(!el) return;
  const top=charClipLaneTop(laneIndex)+'px';
  if(el.style.getPropertyValue('--ev-char-lane-top')!==top){
    el.style.setProperty('--ev-char-lane-top',top);
    el.style.top=top;
  }
}
// Recomputes only the light lane geometry on every pointer frame (no innerHTML rebuild, so
// pointer capture remains intact). Overlapping character clips move down/up immediately,
// while their trimStart/trimEnd and keyframe times are never rewritten by the lane solver.
function applyCharClipLanes(movedRef, preserveDraggedRef){
  const track=document.getElementById('track-char-'+activeCharIdx);
  const row=track?track.closest('.char-row'):null;
  if(!track||!row) return;
  const layout=characterClipLaneLayout(activeClip,splitStickmanSegments);
  setCharacterLaneGeometry(row,track,layout.laneCount);
  const pps=pxPerSecond();
  const allEnd=layout.entries.reduce((max,entry)=>Math.max(max,Number(entry.clip.trimEnd)||0),0);
  if(allEnd>timelineDuration-0.3){
    timelineDuration=allEnd+1;
    syncTimelineInnerWidth();
    renderRuler();
  }
  const mainEl=document.getElementById('animClipEl');
  if(mainEl){
    setCharacterClipLane(mainEl,layout.laneOf.get('main')||0);
    if(preserveDraggedRef!=='main'){
      setTimelineInlineStyle(mainEl,'left',(activeClip.trimStart*pps)+'px');
      setTimelineInlineStyle(mainEl,'width',Math.max(MIN_CLIP_SEC*pps,(activeClip.trimEnd-activeClip.trimStart)*pps)+'px');
      const kfTrack=document.getElementById('kfTrack');
      if(kfTrack){
        (activeClip.keyframes||[]).forEach((kf,idx)=>{
          const marker=kfTrack.querySelector(`[data-kf-idx="${idx}"]`);
          if(marker) setTimelineInlineStyle(marker,'left',keyframeMarkerX(activeClip,kf,pps)+'px');
        });
      }
    }
  }
  splitStickmanSegments.forEach((seg,idx)=>{
    const ref='seg'+idx,el=document.getElementById(ref+'El');
    if(!el) return;
    setCharacterClipLane(el,layout.laneOf.get(ref)||0);
    if(preserveDraggedRef!==ref){
      setTimelineInlineStyle(el,'left',(seg.trimStart*pps)+'px');
      setTimelineInlineStyle(el,'width',Math.max(MIN_CLIP_SEC*pps,(seg.trimEnd-seg.trimStart)*pps)+'px');
      const kfTrack=document.getElementById('kfTrack-'+ref);
      if(kfTrack){
        (seg.keyframes||[]).forEach((kf,kidx)=>{
          const marker=kfTrack.querySelector(`[data-kf-idx="${kidx}"]`);
          if(marker) setTimelineInlineStyle(marker,'left',keyframeMarkerX(seg,kf,pps)+'px');
        });
      }
    }
  });
}
function renderTimelineClip(options){
  // The interactive clip markup always renders into the currently ACTIVE character's own
  // row track (id="track-char-{activeCharIdx}") — there is no separate shared stickman
  // track anymore. Every other character's row shows its own read-only preview instead
  // (see readOnlyCharClipMarkup), rebuilt by renderCharacterLayers whenever the active
  // character changes.
  // BUGFIX ("timeline ruler (1s, 2s...) doesn't show until after you select+extend a
  // clip"): renderRuler()/syncTimelineInnerWidth() used to be called only AFTER this
  // function's own track-lookup guard below, so if the active character's clip row
  // (#track-char-N) didn't exist in the DOM yet — which is true for a brief moment right
  // when the editor first opens, before renderCharacterLayers() has built that row — this
  // whole function returned early and the ruler was simply never drawn. It would then only
  // ever get a first chance to draw once some other action (selecting/extending a clip)
  // triggered a later renderTimelineClip() call, by which point the row finally existed.
  // The ruler has nothing to do with any particular character's row, so it must never be
  // gated behind that row's existence — moved both calls up, before the guard.
  renderRuler();
  syncPlayheadHeight();
  const track = document.getElementById('track-char-'+activeCharIdx);
  if(!track){ syncTimelineInnerWidth(); return; }
  syncTimelineInnerWidth();
  if(!activeClip){
    track.innerHTML = '';
    splitStickmanSegments = []; // segments are meaningless without a base clip
    const emptyRow=track.closest('.char-row');
    setCharacterLaneGeometry(emptyRow,track,1);
    syncClipRefsToCharacter();
    if(!(options&&options.skipOverlayRender)) renderOverlayLayers();
    updateKeyButtonState();
    return;
  }
  const pps = pxPerSecond();
  const left = activeClip.trimStart * pps;
  const width = Math.max(MIN_CLIP_SEC*pps, (activeClip.trimEnd - activeClip.trimStart) * pps);
  const preset = animPresets[activeClip.animKey];

  // Keyframe diamonds are persistent timeline information, not selection-only controls.
  // Render them for every clip so they are already painted before the user taps a clip.
  const kfMarkers = kfMarkersHtml('main', activeClip);

  // Pack main + split segments into vertical sub-lanes exactly like the Clips/Text/Audio
  // rows. Timeline times remain unchanged; an overlapping clip occupies the next lane.
  const charLayout=characterClipLaneLayout(activeClip,splitStickmanSegments);

  // Segments created by Split() render as their own independent clip elements on the
  // same stickman track — same visuals as the main clip, but each with its own
  // trim/select/delete via data-seg-idx instead of the fixed #animClipEl id. Each segment
  // keeps its keyframe markers visible regardless of selection.
  const segMarkup = splitStickmanSegments.map((seg, segIdx)=>{
    const segId = 'seg'+segIdx;
    const segLeft = seg.trimStart * pps;
    const segWidth = Math.max(MIN_CLIP_SEC*pps, (seg.trimEnd - seg.trimStart) * pps);
    const segPreset = animPresets[seg.animKey];
    const segKfMarkers = kfMarkersHtml(segId, seg);
    const segTop=charClipLaneTop(charLayout.laneOf.get(segId)||0);
    return `
      <div class="anim-clip ${selectedClip===segId?'selected':''}" data-seg-idx="${segIdx}" id="${segId}El" style="left:${segLeft}px; width:${segWidth}px; --ev-char-lane-top:${segTop}px; top:${segTop}px; height:${CHAR_SUB_LANE_H}px;">
        <div class="anim-clip-handle left" data-seg-handle="left"></div>
        <div class="anim-clip-body" data-seg-body="${segIdx}"><span class="anim-clip-label">${segPreset ? segPreset.name : 'Animation'}</span></div>
        <div class="anim-clip-handle right" data-seg-handle="right"></div>
        <div class="keyframe-track" id="kfTrack-${segId}">${segKfMarkers}</div>
        ${SEL_EDGE_HTML}
      </div>
    `;
  }).join('');

  const mainTop=charClipLaneTop(charLayout.laneOf.get('main')||0);
  track.innerHTML = `
    <div class="anim-clip ${selectedClip==='main'?'selected':''}" id="animClipEl" style="left:${left}px; width:${width}px; --ev-char-lane-top:${mainTop}px; top:${mainTop}px; height:${CHAR_SUB_LANE_H}px;">
      <div class="anim-clip-handle left" id="clipHandleLeft"></div>
      <div class="anim-clip-body" id="clipBody"><span class="anim-clip-label">${preset.name}</span></div>
      <div class="anim-clip-handle right" id="clipHandleRight"></div>
      <div class="keyframe-track" id="kfTrack">${kfMarkers}</div>
      ${SEL_EDGE_HTML}
    </div>
    ${segMarkup}
  `;
  const row=track.closest('.char-row');
  setCharacterLaneGeometry(row,track,charLayout.laneCount);
  wireClipInteractions();
  wireSplitSegmentInteractions();
  if(!(options&&options.skipOverlayRender)) renderOverlayLayers();
  updateKeyButtonState();
}

// Wires drag/resize/select for each split-off segment clip. Mirrors wireClipInteractions
// but operates on splitStickmanSegments[idx] instead of the global activeClip, and has no
// keyframes/animation-swap of its own (v1 segments just replay the same preset pose-to-pose).
function wireSplitSegmentInteractions(){
  splitStickmanSegments.forEach((seg, segIdx)=>{
    const segId = 'seg'+segIdx;
    const clipEl = document.getElementById(segId+'El');
    if(!clipEl) return;
    const body = clipEl.querySelector(`[data-seg-body="${segIdx}"]`);
    const leftHandle = clipEl.querySelector('[data-seg-handle="left"]');
    const rightHandle = clipEl.querySelector('[data-seg-handle="right"]');
    wireClipZoomGesture(body); // FEATURE: two-finger pinch on this clip zooms the timeline
    clipEl.addEventListener('pointerdown', (e)=>{
      if(e.target !== clipEl && !e.target.closest('.sel-edge-top, .sel-edge-bottom, .sel-edge-left, .sel-edge-right')) return;
      e.stopPropagation();
      if(isPlaying) pausePlayback();
      selectClipForChrome(segId);
    });

    function reposition(){
      // Recalculate the character row's vertical sub-lanes on every trim frame. Any
      // overlapping clip moves to another lane immediately without changing timeline time.
      applyCharClipLanes(segId);
    }

    // A quick touch-and-slide on a segment body ALWAYS scrubs the playhead, never moves the
    // segment — matches the main clip body exactly. Only a LONG_PRESS_SWAP_MS hold-still,
    // done before any sliding, arms move mode; once armed, every pointermove repositions the
    // segment live in real time, tracking the finger 1:1. If that overlaps a neighbor, the
    // row immediately places the overlapping clip on a separate vertical sub-lane.
    const SEG_DRAG_TOLERANCE = 7;
    let dragState = null; // null | {type:'pending'|'move'|'scroll', ...}
    let swapPressTimer = null;
    function clearSwapPressTimer(){ if(swapPressTimer){ clearTimeout(swapPressTimer); swapPressTimer=null; } }
    function applySegmentMoveAt(clientX){
      if(!dragState||dragState.type!=='move')return;
      const pps=pxPerSecond();
      const deltaSec=timelineDragDeltaPixels(dragState,clientX)/pps;
      const len=dragState.origEnd-dragState.origStart;
      const newStart=Math.max(0,dragState.origStart+deltaSec);
      const shift=newStart-dragState.origStart;
      seg.trimStart=newStart;seg.trimEnd=newStart+len;
      // Re-pack neighbours first, then position the compositor-translated active segment
      // from its FINAL solved time. This also handles the t=0 boundary case without a
      // one-frame visual mismatch between the data and the finger-following DOM element.
      applyCharClipLanes(segId,segId);
      ensureTimelineRoomForDraggedClip(seg);
      paintTimelineDragX(clipEl,(seg.trimStart-dragState.origStart)*pps);
    }
    body.onpointerdown = (e)=>{
      e.stopPropagation();
      if(isPlaying) pausePlayback();
      selectClipForChrome(segId);
      const timelineScroller=document.getElementById('timelineScroll');
      dragState = { type:'pending', pointerId:e.pointerId, startScrollLeft:timelineScroller?timelineScroller.scrollLeft:0, startX:e.clientX, startY:e.clientY, lastX:e.clientX, startTime: getCurrentAbsoluteTime(), origStart:seg.trimStart, origEnd:seg.trimEnd, origKfTimes: (seg.keyframes||[]).map(k=>k.t) };
      body.setPointerCapture(e.pointerId);
      clearSwapPressTimer();
      swapPressTimer=setTimeout(()=>{
        swapPressTimer=null;
        if(!dragState||dragState.type!=='pending')return;
        dragState.type='move';clipEl.classList.add('dragging');if(window.ethvelleNotifyClipMoveReady)window.ethvelleNotifyClipMoveReady();armClipDrag();
        startTimelineEdgeAutoScroll(dragState.pointerId,dragState.lastX,()=>{if(dragState&&dragState.type==='move')applySegmentMoveAt(dragState.lastX);});
        if(Number.isFinite(dragState.lastX))applySegmentMoveAt(dragState.lastX);
      },240);
    };
    body.onpointermove = (e)=>{
      if(!dragState) return;
      dragState.lastX=e.clientX;
      if(dragState.type==='scroll'){ applyClipVerticalScrollHandoff(dragState, e.clientY); return; }
      const movedPx = Math.abs(e.clientX - dragState.startX);
      if(dragState.type==='pending'){
        // A dominantly-vertical slide before the long-press fires means "scroll the row
        // list", not "interact with this segment" — hand off to the shared scroller, same
        // as sliding on empty timeline space (see applyClipVerticalScrollHandoff above).
        const movedPy = Math.abs(e.clientY - dragState.startY);
        if(movedPy >= SEG_DRAG_TOLERANCE && movedPy >= movedPx*0.82){
          clearSwapPressTimer();
          dragState.type = 'scroll';
          const scroller = document.getElementById('timelineScroll');
          dragState.scrollTopStart = scroller ? scroller.scrollTop : 0;
          applyClipVerticalScrollHandoff(dragState, e.clientY);
          return;
        }
        // A quick horizontal slide is a scrub. Moving the clip requires holding still for
        // LONG_PRESS_MS; after that timer arms move mode, the clip follows the finger.
        if(movedPx > 46){clearSwapPressTimer();dragState.type='scrub';}
        return;
      }
      if(dragState.type==='scrub')return;
      updateTimelineEdgeAutoScroll(e.pointerId,e.clientX,()=>{if(dragState&&dragState.type==='move')applySegmentMoveAt(dragState.lastX);});
      scheduleTimelineDragFrame(body,e,point=>applySegmentMoveAt(point.clientX));
      return;
    };
    body.onpointerup = (e)=>{
      if(!dragState) return;
      stopTimelineEdgeAutoScroll(e.pointerId);
      clearSwapPressTimer();
      const wasMove = dragState.type==='move';
      const wasScroll = dragState.type==='scroll';
      const wasScrub = dragState.type==='scrub';
      const anchorX = dragState.startX, anchorTime = dragState.startTime;
      if(wasMove)flushTimelineDragFrame(body,e,point=>applySegmentMoveAt(point.clientX));else cancelTimelineDragFrame(body);
      if(wasMove){const movedBy=seg.trimStart-dragState.origStart;shiftTimelineKeyframesOnce(seg.keyframes,dragState.origKfTimes,movedBy);if(window.ethvelleNotifyClipMoved)window.ethvelleNotifyClipMoved(movedBy);clearTimelineDragX(clipEl);}
      dragState=null;body.releasePointerCapture(e.pointerId);
      clipEl.classList.remove('dragging');
      if(wasMove){
        applyCharClipLanes(segId); // one final neighbor/layout pass, not once per pointer frame
        disarmClipDrag();
        // FIX: no snap-to-neighbor — the segment stays exactly where it was dropped, on
        // whichever lane it now packs into (already applied live — see reposition above).
        renderTimelineClip();
      } else if(wasScrub&&!wasScroll){
        const t = timeFromDragAnchor(anchorX, anchorTime, e.clientX);
        flushTimelineScrub(t);
      }
      persistProjectState({skipHistory:true});
    };
    body.onpointercancel = ()=>{
      if(!dragState) return;
      stopTimelineEdgeAutoScroll(dragState.pointerId);
      clearSwapPressTimer();
      const wasMove=dragState.type==='move';
      cancelTimelineDragFrame(body);
      if(wasMove){const movedBy=seg.trimStart-dragState.origStart;shiftTimelineKeyframesOnce(seg.keyframes,dragState.origKfTimes,movedBy);clearTimelineDragX(clipEl);}
      dragState=null;
      clipEl.classList.remove('dragging');
      if(wasMove){
        applyCharClipLanes(segId);
        disarmClipDrag();
        renderTimelineClip();
      }
    };

    function wireHandle(handle, side){
      let hs=null;
      function applySegmentHandleAt(clientX){
        if(!hs)return;
        const pps=pxPerSecond();
        const deltaSec=(clientX-hs.startX)/pps;
        // Trimming is not clamped against a neighboring segment. If the new time range
        // overlaps another clip, the live lane pass below puts it on a separate row lane.
        if(side==='left'){
          let newStart = Math.min(hs.origEnd-MIN_CLIP_SEC, Math.max(0, hs.origStart+deltaSec));
          seg.trimStart = newStart;
        } else {
          let newEnd = Math.max(hs.origStart+MIN_CLIP_SEC, hs.origEnd+deltaSec);
          newEnd = Math.min(newEnd, timelineDuration + 30);
          seg.trimEnd = newEnd;
          if(seg.trimEnd > timelineDuration - 0.3){ timelineDuration = seg.trimEnd+1; syncTimelineInnerWidth(); renderRuler(); }
        }
        // Reconcile all character sub-lanes immediately, before pointerup, so no two
        // overlapping segments are drawn on top of one another during the resize.
        applyCharClipLanes(segId);
        // ADDED: same CapCut-style "Nf" live frame-count readout as the main clip's handles.
        showTrimFrameBadge(handle,side==='left'?(seg.trimStart-hs.origStart):(seg.trimEnd-hs.origEnd));
      }
      // v279: trimming starts only after a long press on the handle (see ethvelleMakeHandleHoldGate).
      const gate=ethvelleMakeHandleHoldGate((x)=>{
        if(!handle.isConnected)return;
        hs={startX:x,origStart:seg.trimStart,origEnd:seg.trimEnd};
        clipEl.dataset.evTimelineDragActive='1';armClipDrag();
      });
      handle.onpointerdown=(e)=>{
        e.stopPropagation();selectClipForChrome(segId);
        gate.down(e);
        try{handle.setPointerCapture(e.pointerId);}catch(_e){}
      };
      handle.onpointermove=(e)=>{if(gate.move(e)==='trim'&&hs)scheduleTimelineDragFrame(handle,e,point=>applySegmentHandleAt(point.clientX));};
      handle.onpointerup=(e)=>{
        gate.up(e);
        if(hs===null){
          // Plain tap, quick slide (scrub) or row scroll: nothing was trimmed.
          cancelTimelineDragFrame(handle);
          try{handle.releasePointerCapture(e.pointerId);}catch(_e){}
          return;
        }
        flushTimelineDragFrame(handle,e,point=>applySegmentHandleAt(point.clientX));
        hs=null;delete clipEl.dataset.evTimelineDragActive;
        try{handle.releasePointerCapture(e.pointerId);}catch(_e){}
        disarmClipDrag();
        hideTrimFrameBadge();
        renderTimelineClip();
        persistProjectState();
      };
      handle.onpointercancel=()=>{
        gate.cancel();
        if(hs===null)return;
        cancelTimelineDragFrame(handle);hs=null;delete clipEl.dataset.evTimelineDragActive;
        hideTrimFrameBadge();
        disarmClipDrag();
      };
    }
    wireHandle(leftHandle,'left');
    wireHandle(rightHandle,'right');
  });
}

// Total scrollable width of the timeline, matching timelineDuration at the fixed zoom level.
// BUGFIX (animation clip looked "cut off" / rows stopped short of the clip): every track row is
// sized from `timelineDuration` (track width = timelineDuration * pps), but a freshly added
// character arrives with a 5s 'default' clip (DEFAULT_CLIP_LEN) while timelineDuration still
// held its 2s starting value — addCharacter()/addCharacterFromModelAsset()/the custom builder/
// auto-rig never raised it (only setActiveAnimation() did). So the track was ~207px wide with
// a 450px clip inside it, and the track's own overflow:hidden sliced the clip's right end off
// flat, while the Background/"Add character" rows stopped at the same short width. Instead of
// patching every creation path one by one, this is the single choke point every one of them
// already goes through (renderTimelineClip -> syncTimelineInnerWidth): before any width is
// computed, make sure the timeline is at least as long as the furthest end of ANY clip on ANY
// row — characters (incl. split segments), backgrounds and overlays. It only ever GROWS the
// timeline (never shrinks it), so it can't fight code that intentionally sets it larger.
function ensureTimelineCoversContent(){
  let need = 0;
  try{
    if(typeof combinedAllCharactersDuration === 'function' && characters.length) need = Math.max(need, combinedAllCharactersDuration());
    (backgrounds||[]).forEach(b=>{ if(b && isFinite(b.trimEnd)) need = Math.max(need, b.trimEnd); });
    (overlays||[]).forEach(o=>{ if(o && isFinite(o.trimEnd)) need = Math.max(need, o.trimEnd); });
  }catch(e){ return false; }
  if(need > timelineDuration + 1e-6){ timelineDuration = need; return true; }
  return false;
}
function syncTimelineInnerWidth(){
  const inner = document.getElementById('timelineInner');
  const scroller = document.getElementById('timelineScroll');
  if(!inner || !scroller) return;
  if(ensureTimelineCoversContent()){
    // The ruler's tick count is also derived from timelineDuration — keep it in step.
    // (renderRuler doesn't call back into this function, so there's no recursion.)
    renderRuler();
  }
  const pps = pxPerSecond();
  const pad = scroller.clientWidth/2; // let earliest/last frame reach the fixed center line
  inner.style.width = (timelineDuration*pps + pad*2) + 'px';
  // Left padding is reduced by TRACK_LEAD_PX so that, combined with the ruler's matching
  // margin-left and every layer-track's badge-column indent (both +TRACK_LEAD_PX), t=0
  // still lands exactly `pad` px in — i.e. exactly under the fixed center playhead when
  // scrollLeft is 0 — for the ruler AND the tracks alike, instead of the tracks landing
  // 48px further right than the ruler (which was the source of the keyframe/clip vs.
  // playhead misalignment).
  inner.style.paddingLeft = Math.max(0, pad - TRACK_LEAD_PX) + 'px';
  inner.style.paddingRight = pad + 'px';
  // BUGFIX (dragging a Background clip's edge to extend it "sometimes" did nothing):
  // renderTimelineClip() calls this function on practically every timeline change (selecting
  // a clip, toggling a keyframe, playback ticking, etc. — dozens of call sites), and this used
  // to unconditionally call renderBackgroundTrack() every single time, which rebuilds
  // the ENTIRE background row's innerHTML from scratch — including whichever handle element
  // the user might be mid-drag on right now. Replacing that element out from under an active
  // pointer-capture drag silently kills the gesture (same failure mode the comment in
  // wireBackgroundTrackInteractions already called out for ITS OWN calls into this function —
  // the bug was simply reachable from this OTHER direction too, any time something unrelated
  // triggered a re-render while the user's finger was still down on the handle). Skipping the
  // rebuild while bgHandleDragActive is true (set for the duration of that one drag — see
  // wireBackgroundTrackInteractions) leaves the in-progress handle/drag state completely alone;
  // the width/padding updates above still apply immediately either way, and a full
  // renderBackgroundTrack() still runs once on pointerup, after the gesture has actually ended.
  if(!bgHandleDragActive) renderBackgroundTrack();
}

// FEATURE (multiple sequential backgrounds, non-overlapping): keeps `backgrounds` sorted by
// trimStart and clamps the just-moved/resized clip against its immediate left/right
// neighbors — exactly the same "hard wall, never push a neighbor" contract as
// resolveCharClipOverlaps (see that function's own comment for the full rationale). Kept as
// a separate function (rather than sharing one with the character-clip version) since the
// two operate on different arrays with different call shapes, but the clamping logic itself
// is intentionally identical.
function resolveBgClipOverlaps(movedId){
  const moved=backgrounds.find(b=>String(b.id)===String(movedId));
  if(!moved)return;
  const gap=0.015; // less than one frame; preserves a clear seam without a visible dead zone
  const ordered=[...backgrounds].sort((a,b)=>a.trimStart-b.trimStart || (String(a.id)===String(movedId)?-1:String(b.id)===String(movedId)?1:0));
  const movedIdx=ordered.findIndex(b=>String(b.id)===String(movedId));
  const shiftEntry=(clip,newStart)=>{
    const oldStart=Number(clip.trimStart)||0;
    const duration=Math.max(MIN_CLIP_SEC,(Number(clip.trimEnd)||oldStart+MIN_CLIP_SEC)-oldStart);
    const nextStart=Math.max(0,Number(newStart)||0);
    const delta=nextStart-oldStart;
    if(!Number.isFinite(delta)||Math.abs(delta)<0.000001)return;
    clip.trimStart=nextStart;
    clip.trimEnd=nextStart+duration;
    (Array.isArray(clip.ethvelleMediaKeyframes)?clip.ethvelleMediaKeyframes:[]).forEach(kf=>{
      const t=Number(kf.t);if(Number.isFinite(t))kf.t=t+delta;
    });
  };
  // Keep the clip under the finger. Push earlier neighbours backwards instead of
  // clamping the selected clip against an invisible collision wall.
  let nextStart=moved.trimStart;
  for(let i=movedIdx-1;i>=0;i--){
    const clip=ordered[i];
    if(clip.trimEnd>nextStart-gap){
      const duration=Math.max(MIN_CLIP_SEC,clip.trimEnd-clip.trimStart);
      shiftEntry(clip,Math.max(0,nextStart-gap-duration));
    }
    nextStart=clip.trimStart;
  }
  let previousEnd=moved.trimEnd;
  for(let i=movedIdx+1;i<ordered.length;i++){
    const clip=ordered[i];
    if(clip.trimStart<previousEnd+gap)shiftEntry(clip,previousEnd+gap);
    previousEnd=clip.trimEnd;
  }
  backgrounds.sort((a,b)=>a.trimStart-b.trimStart);
  const last=backgrounds.reduce((max,b)=>Math.max(max,b.trimEnd),0);
  if(last>timelineDuration-0.3){timelineDuration=last+1;syncTimelineInnerWidth();renderRuler();}
}

const BG_ROW_LANE_H = 36; // slightly taller than the compact character/overlay clips

// FEATURE (multiple sequential backgrounds): replaces the old single-clip background
// renderer, which
// only ever built ONE fixed #bgTrackClip element. This now renders the ENTIRE `backgrounds`
// array as a row of clips sitting side by side along the timeline — same visual language as
// a Clips row (real image thumbnail strip, drag to move, edge-handles to trim), but clips on
// this row can never overlap (see resolveBgClipOverlaps) since only one background can ever
// be "playing" at a given instant, unlike Clips/Audio which intentionally allow overlap.
function captureEthvelleRowScroll(){
  const timeline=document.getElementById('timelineScroll');
  const page=document.scrollingElement||document.documentElement;
  return {timeline, page, timelineTop:timeline?timeline.scrollTop:0, pageTop:page?page.scrollTop:0};
}
function restoreEthvelleRowScroll(snapshot){
  if(!snapshot)return;
  const apply=()=>{
    if(snapshot.timeline && snapshot.timeline.isConnected) snapshot.timeline.scrollTop=snapshot.timelineTop;
    if(snapshot.page) snapshot.page.scrollTop=snapshot.pageTop;
  };
  // Restore immediately after the render, then only repeat on the next frame if no newer
  // scroll has happened. The old unconditional RAF restore could overwrite a finger-driven
  // vertical scroll that started in the same frame as a layer selection, snapping the whole
  // timeline back to the stale scrollTop captured before the gesture.
  apply();
  const appliedTimelineTop=snapshot.timeline&&snapshot.timeline.isConnected?snapshot.timeline.scrollTop:null;
  const appliedPageTop=snapshot.page?snapshot.page.scrollTop:null;
  requestAnimationFrame(()=>{
    if(snapshot.timeline&&snapshot.timeline.isConnected&&appliedTimelineTop!==null&&Math.abs(snapshot.timeline.scrollTop-appliedTimelineTop)<0.5){
      snapshot.timeline.scrollTop=snapshot.timelineTop;
    }
    if(snapshot.page&&appliedPageTop!==null&&Math.abs(snapshot.page.scrollTop-appliedPageTop)<0.5){
      snapshot.page.scrollTop=snapshot.pageTop;
    }
  });
}
function renderBackgroundTrack(){

  if(badge) badge.innerHTML = BG_FILLED_ICON;

  // BG_ROW_LANE_H (60px) is sized to fit exactly inside that, matching every other track
  // row's fixed 60px-track/72px-row convention, so no JS height is set here at all.
  if(!backgrounds.length){
    track.style.height = BG_ROW_LANE_H+'px';

    const isUrl = b.src && String(b.src).startsWith('url(');

    if(isUrl){
      const url = String(b.src).slice(4,-1).replace(/^["']|["']$/g,'');
      styleBg = '';

    } else {
      styleBg = b.src || '#ffffff';
      fillHtml = `<div class="clip-media-label clip-media-label-bg" unselectable="on" onselectstart="return false;">Background</div>`;
    }

}
// Drag-to-move / edge-handle-to-trim for EACH background clip — mirrors
// wireOverlayInteractions almost exactly, except every move/resize is clamped via
// resolveBgClipOverlaps (background clips must never overlap) instead of being free to
// overlap into their own sub-lane the way Clips/Audio can.
function wireBackgroundTrackInteractions(){
  backgrounds.forEach(b=>{
    const clipEl = document.querySelector(`.track-clip[data-bg-id="${b.id}"]`);
    if(!clipEl) return;
    wireClipZoomGesture(clipEl); // FEATURE: two-finger pinch on this clip zooms the timeline
    function reposition(){
      const pps = pxPerSecond();
      setTimelineInlineStyle(clipEl,'left',(b.trimStart*pps)+'px');
      setTimelineInlineStyle(clipEl,'width',Math.max(MIN_CLIP_SEC*pps,(b.trimEnd-b.trimStart)*pps)+'px');
    }
    const BG_DRAG_TOLERANCE = 7;
    const BG_SCRUB_TOLERANCE = 18;
    const LONG_PRESS_MS = 240;
    let dragState = null; // null | {type:'pending'|'move'|'scroll', ...}
    let longPressTimer = null;
    function clearLongPressTimer(){ if(longPressTimer){ clearTimeout(longPressTimer); longPressTimer=null; } }
    // Background clips use direct, RAF-batched `left` writes while moving. Some Android
    // WebViews defer repainting the CSS-variable transform until pointerup, even though the
    // model and custom property are updated during the gesture. Writing the absolute layout
    // position keeps the background clip visibly under the finger on every frame. This is
    // intentionally local to the Background row; overlay/character drag paths stay unchanged.
    function paintBackgroundClipLeft(el,startTime,pps){
      if(!el)return;
      const left=Math.max(0,Number(startTime)||0)*pps;
      const value=left.toFixed(3)+'px';
      if(el.style.left!==value)el.style.left=value;
    }
    function applyBackgroundMoveAt(clientX){
      if(!dragState||dragState.type!=='move')return;
      const pps=pxPerSecond();
      let desiredStart=Math.max(0,dragState.origStart+timelineDragDeltaPixels(dragState,clientX)/pps);
      const duration=Math.max(MIN_CLIP_SEC,dragState.origEnd-dragState.origStart);
      const others=dragState.layoutSnapshot||[]; // sorted once on pointerdown
      const gap=0.015;
      // Binary-search the dragged clip's current slot among the pre-sorted neighbours.
      // If there is not enough timeline space to push every earlier neighbour left, clamp
      // to the first collision-free position instead of allowing a temporary overlap at t=0.
      const minStartByCount=dragState.minStartByCount||[0];
      let low=0,high=others.length,rounds=0;
      while(rounds++<=others.length){
        low=0;high=others.length;
        while(low<high){
          const mid=(low+high)>>>1;
          if(others[mid].start<desiredStart-0.000001)low=mid+1;else high=mid;
        }
        const minAllowed=minStartByCount[low]||0;
        if(desiredStart>=minAllowed-0.000001)break;
        desiredStart=minAllowed;
      }
      let nextStart=desiredStart;
      for(let i=low-1;i>=0;i--){
        const item=others[i];
        let start=item.start;
        if(start+item.duration>nextStart-gap)start=Math.max(0,nextStart-gap-item.duration);
        item.clip.trimStart=start;item.clip.trimEnd=start+item.duration;
        if(item.el)paintBackgroundClipLeft(item.el,start,pps);
        nextStart=start;
      }
      let previousEnd=desiredStart+duration;
      for(let i=low;i<others.length;i++){
        const item=others[i];
        let start=item.start;
        if(start<previousEnd+gap)start=previousEnd+gap;
        item.clip.trimStart=start;item.clip.trimEnd=start+item.duration;
        if(item.el)paintBackgroundClipLeft(item.el,start,pps);
        previousEnd=start+item.duration;
      }
      b.trimStart=desiredStart;b.trimEnd=desiredStart+duration;
      const furthestBackgroundEnd=backgrounds.reduce((max,clip)=>Math.max(max,Number(clip&&clip.trimEnd)||0),0);
      ensureTimelineRoomForDraggedClip({trimEnd:furthestBackgroundEnd});
      paintBackgroundClipLeft(clipEl,desiredStart,pps);
    }
    function commitBackgroundDragLayout(snapshot){
      (snapshot||[]).forEach(item=>{
        const delta=(Number(item.clip.trimStart)||0)-item.start;
        if(Math.abs(delta)>0.000001)shiftTimelineKeyframesOnce(item.clip.ethvelleMediaKeyframes,item.keyframeTimes,delta);
        if(item.el)clearTimelineDragX(item.el);
      });
    }
    clipEl.addEventListener('pointerdown', (e)=>{
      // Keyframe diamonds own their tap; do not let the clip-body drag/scrub handler
      // capture that pointer or reset the selected key before its click is delivered.
      if((e.target.closest&&e.target.closest('.kf-marker'))||e.target.hasAttribute('data-bg-handle')) return;
      e.stopPropagation();
      // Protect the captured clip node immediately: selectClipForChrome can synchronize
      // the timeline and otherwise rebuild this row before the long-press timer is armed.
      bgHandleDragActive=true;
      if(isPlaying) pausePlayback();
      selectClipForChrome(b.id);
      const bgTrack=document.getElementById('track-background');
      const timelineScroller=document.getElementById('timelineScroll');
      const bgElements=new Map();
      if(bgTrack)bgTrack.querySelectorAll('.track-clip[data-bg-id]').forEach(el=>bgElements.set(String(el.dataset.bgId),el));
      const layoutSnapshot=backgrounds.filter(clip=>clip!==b).map(clip=>{
        const start=Number(clip.trimStart)||0;
        const end=Math.max(start+MIN_CLIP_SEC,Number(clip.trimEnd)||start+MIN_CLIP_SEC);
        return {clip,start,duration:end-start,keyframeTimes:(Array.isArray(clip.ethvelleMediaKeyframes)?clip.ethvelleMediaKeyframes:[]).map(k=>Number(k.t)||0),el:bgElements.get(String(clip.id))||null};
      }).sort((a,bEntry)=>a.start-bEntry.start);
      const minStartByCount=[0];let requiredStart=0;
      layoutSnapshot.forEach(item=>{requiredStart+=item.duration+0.015;minStartByCount.push(requiredStart);});
      dragState = { type:'pending', pointerId:e.pointerId, startScrollLeft:timelineScroller?timelineScroller.scrollLeft:0, startX:e.clientX, startY:e.clientY, lastX:e.clientX, startTime:getCurrentAbsoluteTime(), origStart:b.trimStart, origEnd:b.trimEnd, origKfTimes:(Array.isArray(b.ethvelleMediaKeyframes)?b.ethvelleMediaKeyframes:[]).map(k=>Number(k.t)||0), layoutSnapshot, minStartByCount };
      clipEl.setPointerCapture(e.pointerId);
      clearLongPressTimer();
      longPressTimer=setTimeout(()=>{
        longPressTimer=null;
        if(!dragState||dragState.type!=='pending')return;
        dragState.type='move';clipEl.classList.add('dragging');if(window.ethvelleNotifyClipMoveReady)window.ethvelleNotifyClipMoveReady();armClipDrag();
        startTimelineEdgeAutoScroll(dragState.pointerId,dragState.lastX,()=>{if(dragState&&dragState.type==='move')applyBackgroundMoveAt(dragState.lastX);});
        if(Number.isFinite(dragState.lastX))applyBackgroundMoveAt(dragState.lastX);
      },LONG_PRESS_MS);
    });
    clipEl.addEventListener('pointermove', (e)=>{
      if(!dragState) return;
      dragState.lastX=e.clientX;
      if(dragState.type==='scroll'){ applyClipVerticalScrollHandoff(dragState, e.clientY); return; }
      const movedPx = Math.abs(e.clientX - dragState.startX);
      if(dragState.type==='pending'){
        // A dominantly-vertical slide before the long-press fires means "scroll the row
        // list", not "interact with this background clip" — hand off to the shared scroller,
        // same as sliding on empty timeline space (see applyClipVerticalScrollHandoff above).
        const movedPy = Math.abs(e.clientY - dragState.startY);
        if(movedPy >= BG_DRAG_TOLERANCE && movedPy >= movedPx*0.82){
          clearLongPressTimer();
          dragState.type = 'scroll';
          const scroller = document.getElementById('timelineScroll');
          dragState.scrollTopStart = scroller ? scroller.scrollTop : 0;
          applyClipVerticalScrollHandoff(dragState, e.clientY);
          return;
        }
        if(movedPx > BG_SCRUB_TOLERANCE && movedPx >= Math.max(1, movedPy*0.72)){
          clearLongPressTimer();dragState.type='scrub';
          scheduleTimelineScrub(timeFromDragAnchor(dragState.startX,dragState.startTime,e.clientX));
        }
        return;
      }
      if(dragState.type==='scrub'){
        scheduleTimelineScrub(timeFromDragAnchor(dragState.startX,dragState.startTime,e.clientX));
        return;
      }
      updateTimelineEdgeAutoScroll(e.pointerId,e.clientX,()=>{if(dragState&&dragState.type==='move')applyBackgroundMoveAt(dragState.lastX);});
      scheduleTimelineDragFrame(clipEl,e,point=>applyBackgroundMoveAt(point.clientX));
      return;
    });
    clipEl.addEventListener('pointerup', (e)=>{
      if(!dragState) return;
      stopTimelineEdgeAutoScroll(e.pointerId);
      clearLongPressTimer();
      const wasMove = dragState.type==='move';
      const wasScroll = dragState.type==='scroll';
      const wasScrub = dragState.type==='scrub';
      const anchorX=dragState.startX,anchorTime=dragState.startTime;
      if(wasMove)flushTimelineDragFrame(clipEl,e,point=>applyBackgroundMoveAt(point.clientX));else cancelTimelineDragFrame(clipEl);
      if(wasMove){
        const movedBy=b.trimStart-dragState.origStart;
        shiftTimelineKeyframesOnce(b.ethvelleMediaKeyframes,dragState.origKfTimes,movedBy);
        if(window.ethvelleNotifyClipMoved)window.ethvelleNotifyClipMoved(movedBy);
        commitBackgroundDragLayout(dragState.layoutSnapshot||[]);
        clearTimelineDragX(clipEl);
      }
      dragState=null;clipEl.releasePointerCapture(e.pointerId);
      clipEl.classList.remove('dragging');
      bgHandleDragActive = false;
      if(wasMove) disarmClipDrag();
      if(wasMove){
        resolveBgClipOverlaps(b.id);
        renderBackgroundTrack();
        selectClipForChrome(b.id);
        refreshCurrentBgFromPlayhead();applyBackground();
      } else if(wasScrub&&!wasScroll){
        const t = timeFromDragAnchor(anchorX, anchorTime, e.clientX);
        flushTimelineScrub(t);
      }
      persistProjectState({skipHistory:true});
    });
    clipEl.addEventListener('pointercancel', ()=>{
      const wasMove=dragState&&dragState.type==='move';
      if(dragState)stopTimelineEdgeAutoScroll(dragState.pointerId);
      clearLongPressTimer();cancelTimelineDragFrame(clipEl);
      if(wasMove){
        const movedBy=b.trimStart-dragState.origStart;
        shiftTimelineKeyframesOnce(b.ethvelleMediaKeyframes,dragState.origKfTimes,movedBy);
        commitBackgroundDragLayout(dragState.layoutSnapshot||[]);
        clearTimelineDragX(clipEl);
      }
      dragState=null;
      clipEl.classList.remove('dragging');
      bgHandleDragActive = false;
      if(wasMove){
        disarmClipDrag();
        resolveBgClipOverlaps(b.id);
        renderBackgroundTrack();
        selectClipForChrome(b.id);
        refreshCurrentBgFromPlayhead();applyBackground();
      }
    });
    [['left'],['right']].forEach(([side])=>{
      const handle = clipEl.querySelector(`[data-bg-handle="${side}"]`);
      if(!handle) return;
      let hs=null;
      function applyBackgroundTrimAt(clientX){
        if(!hs)return;
        const pps=pxPerSecond();
        const deltaSec=(clientX-hs.startX)/pps;
        if(side==='left'){
          b.trimStart=Math.min(hs.origEnd-MIN_CLIP_SEC,Math.max(0,hs.origStart+deltaSec));
        }else{
          b.trimEnd=Math.max(hs.origStart+MIN_CLIP_SEC,hs.origEnd+deltaSec);
          if(b.trimEnd>timelineDuration-0.3){timelineDuration=b.trimEnd+1;syncTimelineInnerWidth();renderRuler();}
        }
        // Defer neighbour resolution until pointerup; keep the live trim to two style writes.
        reposition();
      }
      // v279: trimming starts only after a long press on the handle (see ethvelleMakeHandleHoldGate).
      const gate = ethvelleMakeHandleHoldGate((x)=>{
        if(!handle.isConnected) return;
        hs = { startX:x, origStart:b.trimStart, origEnd:b.trimEnd };
        clipEl.classList.add('dragging');
        armClipDrag();
      });
      handle.addEventListener('pointerdown', (e)=>{
        e.stopPropagation();
        // The trim handle is also pointer-captured; guard before selection sync so neither
        // the clip nor the handle can be detached by a background-row render.
        bgHandleDragActive = true;
        if(isPlaying) pausePlayback();
        selectClipForChrome(b.id);
        gate.down(e);
        try{handle.setPointerCapture(e.pointerId);}catch(_e){}
      });
      handle.addEventListener('pointermove',(e)=>{if(gate.move(e)==='trim'&&hs)scheduleTimelineDragFrame(handle,e,point=>applyBackgroundTrimAt(point.clientX));});
      handle.addEventListener('pointerup',(e)=>{
        gate.up(e);
        if(hs===null){
          // Plain tap, quick slide (scrub) or row scroll: nothing was trimmed.
          cancelTimelineDragFrame(handle);
          try{handle.releasePointerCapture(e.pointerId);}catch(_e){}
          bgHandleDragActive = false;
          return;
        }
        const wasDragging=true;
        flushTimelineDragFrame(handle,e,point=>applyBackgroundTrimAt(point.clientX));
        hs=null;try{handle.releasePointerCapture(e.pointerId);}catch(_e){}
        clipEl.classList.remove('dragging');
        bgHandleDragActive = false;
        if(wasDragging) disarmClipDrag();
        if(wasDragging)resolveBgClipOverlaps(b.id);
        renderBackgroundTrack();
        refreshCurrentBgFromPlayhead();applyBackground();
        persistProjectState();
      });
      handle.addEventListener('pointercancel',()=>{
        gate.cancel();
        const wasDragging=hs!==null;cancelTimelineDragFrame(handle);hs=null;
        clipEl.classList.remove('dragging');
        bgHandleDragActive = false;
        if(wasDragging) disarmClipDrag();
      });
    });
  });
}

// Ruler ticks: one label every second, matched to the fixed px-per-second zoom.
// v279: optional `range` ({from,to} in seconds) rebuilds only the visible part of the ruler (used every
// frame while pinch-zooming); with no argument the whole ruler is rebuilt. When the timeline is zoomed OUT
// only every Nth second gets a tick + label so the text never overlaps.
function renderRuler(range){
  const ruler=document.getElementById('timelineRuler');
  if(!ruler) return;
  // A full ruler rebuild is the gesture-end commit; clear any leftover CSS transform from older builds.
  ruler.style.transform='';
  ruler.style.transformOrigin='';
  const pps=pxPerSecond();
  const totalSec=Math.max(2,Math.ceil(timelineDuration)+1);
  // Seconds stay visible at all zoom levels. Frame marks are added progressively as the
  // user zooms in, rather than swapping 20f directly for 10f (which skipped 15f).
  // Visible countdown labels are cumulative:
  //   slight zoom -> 20f
  //   medium zoom -> 20f, 15f
  //   high zoom   -> 20f, 15f, 10f
  //   very high   -> 20f, 15f, 10f, 5f
  // Each label is anchored to its actual frame position relative to the NEXT second.
  const zoomNow = Number.isFinite(timelineZoom) ? timelineZoom : 1;
  const showFrameLabels = zoomNow > 1.04;
  const fps=Math.max(1,Number(FPS)||24);
  let visibleRemainingFrames=[];
  if(showFrameLabels){
    visibleRemainingFrames=[20];
    if(zoomNow>=1.55) visibleRemainingFrames.push(15);
    if(zoomNow>=2.05) visibleRemainingFrames.push(10);
    if(zoomNow>=2.75) visibleRemainingFrames.push(5);
    visibleRemainingFrames=visibleRemainingFrames.filter(f=>f<fps);
  }
  // Seconds between labelled ticks (1 at normal/zoomed-in levels; larger when zoomed out).
  let secStep=1;
  if(!showFrameLabels){
    const steps=[1,2,5,10,15,30,60,120,300,600];
    secStep=steps[steps.length-1];
    for(const st of steps){ if(st*pps>=44){ secStep=st; break; } }
  }
  let firstSec=0,lastSec=totalSec;
  if(range && Number.isFinite(range.from) && Number.isFinite(range.to)){
    firstSec=Math.max(0,Math.floor(range.from/secStep)*secStep);
    lastSec=Math.min(totalSec,Math.ceil(range.to));
  }
  let html='';
  // At normal zoom, keep the familiar half-second dots between second labels.
  // Once frame labels take over, hide the normal time-scale dots so the frame separators
  // remain a single clean, CapCut-style dotted sequence.
  if(!showFrameLabels && secStep===1){
    const hEnd=Math.min(totalSec*2,lastSec*2+2);
    for(let h=firstSec*2+1;h<hEnd;h+=2){
      const x=(h/2)*pps;
      html+=`<div class="ruler-tick-half" style="left:${x}px;"></div>`;
    }
  }
  for(let sec=firstSec;sec<=lastSec;sec+=secStep){
    const x=sec*pps;
    html+=`<div class="ruler-tick" style="left:${x}px;"></div>`;
  }
  const marks=[];
  for(let sec=firstSec;sec<=lastSec;sec+=secStep) marks.push({x:sec*pps,label:(sec>=60?evFormatClock(sec,false):`${sec}s`),type:'second'});
  if(showFrameLabels){
    const frameEnd=Math.min(lastSec,totalSec);
    for(let sec=firstSec;sec<frameEnd;sec++){
      // CapCut-style adaptive ruler: frame labels form a centered group inside each
      // one-second interval rather than being anchored to the start/end corners. With one
      // visible label (20f), it sits at the midpoint; with two labels they sit at 1/3 and
      // 2/3; as pinch zoom reveals more labels, the whole group redistributes evenly while
      // staying centered. Keep the semantic countdown order left-to-right: 20f, 15f, 10f, 5f.
      const frameCount = visibleRemainingFrames.length;
      visibleRemainingFrames.forEach((remaining, index)=>{
        const centeredFraction = (index + 1) / (frameCount + 1);
        marks.push({x:(sec+centeredFraction)*pps,label:`${remaining}f`,type:'frame'});
      });
    }
  }
  marks.sort((a,b)=>a.x-b.x);
  // Dot separators join the seconds and frame marks in one compact ruler row.
  if(showFrameLabels){
    for(let i=0;i<marks.length-1;i++){
      const mid=(marks[i].x+marks[i+1].x)/2;
      html+=`<span class="ruler-separator-dot" style="left:${mid.toFixed(2)}px;"></span>`;
    }
  }
  for(const mark of marks){
    const cls=mark.type==='second'?'ruler-label':'ruler-frame-label';
    html+=`<span class="${cls}" style="left:${mark.x.toFixed(2)}px;">${mark.label}</span>`;
  }
  ruler.innerHTML=html;
  ruler.style.width=(totalSec*pps)+'px';
  const oldReadout=document.getElementById('timelineFrameReadout');
  if(oldReadout){oldReadout.innerHTML='';oldReadout.style.display='none';oldReadout.style.visibility='hidden';}
}
function renderFrameRuler(){
  // Frame marks are now rendered directly into #timelineRuler by renderRuler().
  const host=document.getElementById('timelineFrameReadout');
  if(host){host.innerHTML='';host.style.display='none';host.style.visibility='hidden';}
}

/* ---------------- Selection state machine (CapCut-style contextual bottom bar) ----------------
   Exactly one of these is ever "active" at a time:
     - no selection            -> default bottom-tools row (Media/Bones/Transform/...)
     - a clip is selected      -> bottomToolsClip row (Split / Key / Animations / Done / Delete)
     - a keyframe is selected  -> bottomToolsKeyframe row (Done / Delete key), nested under 'main' clip
   Selecting a clip does NOT auto-open a panel anymore (that used to jump straight into a
   sheet); it just arms the contextual toolbar underneath, exactly like tapping a clip in
   CapCut arms Split/Delete without opening anything else first. */
function selectClipForChrome(which){
  // A first media/background clip selection can switch the contextual bottom row and
  // populate keyframe markers. Preserve the editor's vertical viewport through that one
  // synchronous DOM update so Android's scroll anchoring cannot jump the whole editor.
  const rowScrollSnapshot=captureEthvelleRowScroll();
  // Selection is a single immediate tap. Keep the current playhead position and pause
  // playback once; clear stale clip borders and paint only the newly selected clip.
  const previousSelection=selectedClip;
  pausePlayback();
  const findClipElement=(ref)=>{
    if(ref==null)return null;
    const key=String(ref);
    if(key==='main')return document.getElementById('animClipEl');
    if(/^seg\d+$/.test(key))return document.getElementById(key+'El');
    const safe=key.replace(/\\/g,'\\\\').replace(/"/g,'\\"');
    const ov=(typeof overlays!=='undefined'&&Array.isArray(overlays))?overlays.find(x=>String(x.id)===key):null;
    if(ov)return document.querySelector(`.overlay-clip[data-overlay-id="${safe}"]`);
    const bg=(typeof backgrounds!=='undefined'&&Array.isArray(backgrounds))?backgrounds.find(x=>String(x.id)===key):null;
    if(bg)return document.querySelector(`.track-clip[data-bg-id="${safe}"]`);
    return null;
  };
  const nextElement=findClipElement(which);
  // A previous render can replace the old clip node before we resolve it here. Clearing
  // every timeline clip's selection class prevents an orphaned white border from surviving
  // on the old DOM element/state when IDs arrive as mixed strings and numbers.
  document.querySelectorAll('#timelineInner .anim-clip.selected, #timelineInner .overlay-clip.selected, #timelineInner .track-clip.selected')
    .forEach(el=>el.classList.remove('selected'));
  if(nextElement)nextElement.classList.add('selected');

  const updateCharacterRow=(ref,active)=>{
    if(typeof ref!=='string'||!ref.startsWith('char'))return;
    const index=ref.slice(4);
    const row=document.querySelector(`.char-row[data-char-idx="${index}"]`);
    if(row)row.classList.toggle('chrome-selected',!!active);
  };
  updateCharacterRow(previousSelection,false);
  updateCharacterRow(which,true);

  selectedClip=which;
  selectedKfIdx=null;
  window.__ethvelleSelectedObjectKeyframe=null;

  // Clear only the previously active character marker rail and populate only the new one.
  const charTrackForRef=(ref)=>{
    if(ref==='main')return document.getElementById('kfTrack');
    if(typeof ref==='string'&&/^seg\d+$/.test(ref))return document.getElementById('kfTrack-'+ref);
    return null;
  };
  const oldCharTrack=charTrackForRef(previousSelection);
  const newCharTrack=charTrackForRef(which);
  if(oldCharTrack&&oldCharTrack!==newCharTrack){
    const previousCharClip=getClipByRef(previousSelection);
    oldCharTrack.innerHTML=previousCharClip?kfMarkersHtml(previousSelection,previousCharClip):'';
  }
  if(previousSelection&&String(previousSelection)!==String(which)&&ethvelleObjectClipById(previousSelection)){
    ethvelleRefreshObjectKeyframeTracks(previousSelection);
  }
  if(newCharTrack){
    if(which==='main')newCharTrack.innerHTML=kfMarkersHtml('main',activeClip);
    else{const seg=getClipByRef(which);newCharTrack.innerHTML=seg?kfMarkersHtml(which,seg):'';}
  }
  ethvelleRefreshObjectKeyframeTracks(which);
  updateKeyButtonState();
  // Route through the current toolbar wrapper once so later contextual controls stay in sync.
  // The old v66 select wrapper called this a second time after this base function had already
  // synchronized it, duplicating toolbar DOM/style work on every layer selection.
  if(typeof window.syncBottomToolsMode==='function')window.syncBottomToolsMode();
  else syncBottomToolsMode();

  // Avoid re-decoding every media element on selection. Only create a missing preview for
  // the selected visual clip when it is actually under the playhead.
  // Only overlay media belongs in overlayMediaEls. ethvelleObjectClipById() also returns
  // background-row clips; treating those as overlay media forced a full renderOverlayMedia()
  // pass whenever a background clip was selected, even though its preview is rendered by
  // the background pipeline. On projects with media layers this could block the UI while
  // the user was switching rows.
  const selectedVisual=overlays.find(ov=>String(ov.id)===String(which))||null;
  const now=getCurrentAbsoluteTime();
  if(selectedVisual&&selectedVisual.type!=='audio'&&now>=selectedVisual.trimStart&&now<selectedVisual.trimEnd){
    const existingPreview=(typeof overlayMediaEls!=='undefined'&&overlayMediaEls)?overlayMediaEls[selectedVisual.id]:null;
    if(!existingPreview&&typeof renderOverlayMedia==='function')renderOverlayMedia(now);
  }
  updateSelBox();
  restoreEthvelleRowScroll(rowScrollSnapshot);
  onClipSelected(which,'select');
}
// One policy point for clip selection, insertion, playhead-dependent tools, and deletion.
// `select` is deliberately side-effect-free: the existing clip renderer above has already
// selected the clip, and a tap must never seek or scroll. `add` selects the new clip and only
// seeks/centres the timeline when the current playhead is outside it.
function onClipSelected(clip, source){
  const action=String(source||'select');
  const toRef=value=>{
    if(value==null)return null;
    if(typeof value==='string'||typeof value==='number')return String(value);
    if(value===activeClip)return 'main';
    const segmentIndex=Array.isArray(splitStickmanSegments)?splitStickmanSegments.indexOf(value):-1;
    if(segmentIndex>=0)return 'seg'+segmentIndex;
    if(value.id!=null)return String(value.id);
    return null;
  };
  const ref=toRef(clip);
  const resolveModel=value=>{
    if(value&&typeof value==='object'&&Number.isFinite(Number(value.trimStart))&&Number.isFinite(Number(value.trimEnd)))return value;
    const key=toRef(value);
    if(!key)return null;
    if(key==='main')return activeClip||null;
    if(/^seg\d+$/.test(key)){
      const index=Number(key.slice(3));
      return Array.isArray(splitStickmanSegments)?splitStickmanSegments[index]||null:null;
    }
    const charMatch=key.match(/^char(\d+)$/);
    if(charMatch){
      const character=Array.isArray(characters)?characters[Number(charMatch[1])]:null;
      return character&&(character.clip||(Array.isArray(character.segments)?character.segments[0]:null))||null;
    }
    return (Array.isArray(backgrounds)?backgrounds.find(item=>String(item.id)===key):null)||
      (Array.isArray(overlays)?overlays.find(item=>String(item.id)===key):null);
  };

  if(action==='select')return true;

  if(action==='add'){
    const model=resolveModel(clip);
    if(ref!=null)selectClipForChrome(ref);
    if(!model)return true;
    const start=Math.max(0,Number(model.trimStart)||0);
    const end=Math.max(start,Number(model.trimEnd)||start);
    const now=Number(typeof getCurrentAbsoluteTime==='function'?getCurrentAbsoluteTime():currentAbsoluteTimeSec)||0;
    if(now<start-0.000001||now>=end-0.000001){
      pausePlayback();
      seekAbsoluteTimelineSeconds(start);
      if(typeof movePlayhead==='function')movePlayhead();
    }
    return true;
  }

  if(action==='split'||action==='keyframe'||action==='freeze'||action==='require-inside'){
    const model=resolveModel(clip!=null?clip:selectedClip);
    const now=Number(typeof getCurrentAbsoluteTime==='function'?getCurrentAbsoluteTime():currentAbsoluteTimeSec);
    const start=model?Number(model.trimStart):NaN;
    const end=model?Number(model.trimEnd):NaN;
    if(!model||!Number.isFinite(now)||!Number.isFinite(start)||!Number.isFinite(end)||now<start-0.000001||now>=end-0.000001){
      showToast('Move playhead inside the clip',{icon:toastIcons.info,duration:1600});
      return false;
    }
    return true;
  }

  if(action==='delete'){
    const now=Number(typeof getCurrentAbsoluteTime==='function'?getCurrentAbsoluteTime():currentAbsoluteTimeSec)||0;
    const duration=Math.max(0,Number(typeof totalProjectDuration==='function'?totalProjectDuration():timelineDuration)||0);
    const clamped=Math.max(0,Math.min(duration,now));
    if(Math.abs(clamped-now)>0.000001){
      pausePlayback();
      seekAbsoluteTimelineSeconds(clamped);
      if(typeof movePlayhead==='function')movePlayhead();
    }else if(typeof setCurrentAbsoluteTime==='function'){
      setCurrentAbsoluteTime(clamped);
    }
    return true;
  }

  return true;
}
// Moves the playhead (and, for the main stickman clip/segments, the scrub position) to the
// very start of whichever clip was just selected. Background/overlay clips don't have their
// own local `frame` counter, so for those we just move the shared timeline playhead itself.
function seekPlayheadToClipStart(which){
  pausePlayback();
  if(which === 'main' && activeClip){
    seekAbsoluteTimelineSeconds(activeClip.trimStart);
  } else if(String(which).startsWith('seg')){
    const segIdx = parseInt(String(which).slice(3), 10);
    const seg = splitStickmanSegments[segIdx];
    if(seg) seekAbsoluteTimelineSeconds(seg.trimStart);
  } else if(String(which).startsWith('bg')){
    const bg = backgrounds.find(b=>b.id===which);
    if(bg) seekAbsoluteTimelineSeconds(bg.trimStart);
  } else if(which){
    const ov = overlays.find(o=>o.id===which);
    if(ov) seekAbsoluteTimelineSeconds(ov.trimStart);
  }
}
// Seeks the shared timeline playhead to an absolute timeline second via the canonical
// clock (also keeps the legacy `frame` counter in sync — see setCurrentAbsoluteTime).
function seekAbsoluteTimelineSeconds(absSec){
  setCurrentAbsoluteTime(absSec);
  renderFrame();
}
function deselectAllClips(){
  const previousSelection=selectedClip;
  const key=previousSelection==null?'':String(previousSelection);
  let previousElement=null;
  if(key==='main')previousElement=document.getElementById('animClipEl');
  else if(/^seg\d+$/.test(key))previousElement=document.getElementById(key+'El');
  else if(key){
    const safe=key.replace(/\\/g,'\\\\').replace(/"/g,'\\"');
    const ov=(typeof overlays!=='undefined'&&Array.isArray(overlays))?overlays.find(x=>String(x.id)===key):null;
    const bg=(typeof backgrounds!=='undefined'&&Array.isArray(backgrounds))?backgrounds.find(x=>String(x.id)===key):null;
    if(ov)previousElement=document.querySelector(`.overlay-clip[data-overlay-id="${safe}"]`);
    else if(bg)previousElement=document.querySelector(`.track-clip[data-bg-id="${safe}"]`);
  }
  if(previousElement)previousElement.classList.remove('selected');
  if(key.startsWith('char')){
    const row=document.querySelector(`.char-row[data-char-idx="${key.slice(4)}"]`);
    if(row)row.classList.remove('chrome-selected');
  }
  const oldCharTrack=key==='main'?document.getElementById('kfTrack'):(/^seg\d+$/.test(key)?document.getElementById('kfTrack-'+key):null);
  const previousOverlay=(typeof overlays!=='undefined'&&Array.isArray(overlays))?overlays.find(x=>String(x.id)===key):null;
  const previousBackground=(typeof backgrounds!=='undefined'&&Array.isArray(backgrounds))?backgrounds.find(x=>String(x.id)===key):null;
  if(previousBackground){const row=document.getElementById('backgroundRow');if(row)row.classList.remove('selected');}
  if(previousOverlay){
    const group=previousOverlay.type==='audio'?'audio':(previousOverlay.type==='text'?'text':'clips');
    const row=document.querySelector(`.overlay-track-row[data-group=\"${group}\"]`);
    if(row)row.classList.remove('selected');
  }
  selectedClip=null;
  selectedKfIdx=null;
  window.__ethvelleSelectedObjectKeyframe=null;
  if(oldCharTrack){
    const previousCharClip=getClipByRef(key);
    oldCharTrack.innerHTML=previousCharClip?kfMarkersHtml(key,previousCharClip):'';
  }
  if(key&&ethvelleObjectClipById(key))ethvelleRefreshObjectKeyframeTracks(key);
  ethvelleRefreshObjectKeyframeTracks(null);
  updateKeyButtonState();
  syncBottomToolsMode();
  drawStick();
}
// Swap which bottom-tools row is visible based on current selection. Only one is ever shown.
function syncBottomToolsMode(){
  const rowDefault = document.getElementById('bottomToolsDefault');
  const rowClip = document.getElementById('bottomToolsClip');
  const rowKf = document.getElementById('bottomToolsKeyframe');
  if(!rowDefault) return;
  if(selectedKfIdx !== null){
    rowDefault.style.display = 'none';
    rowClip.style.display = 'none';
    rowKf.style.display = 'flex';
  } else if(selectedClip){
    rowDefault.style.display = 'none';
    rowKf.style.display = 'none';
    rowClip.style.display = 'flex';
    // Which tools actually apply depends on clip type — Bones/keyframe-posing and preset
    // Animations only mean anything for the stickman rig ('main' or a split segment 'segN').
    // Background/overlay/media clips instead get Trim (numeric start/end adjustment) and,
    // for overlay rows specifically, an Add button to drop more media into that same layer.
    const isStickman = selectedClip==='main' || String(selectedClip).startsWith('seg');
    const isOverlay = !isStickman && !!overlays.find(o=>o.id===selectedClip);
    // A whole character LAYER (not one of its clips) selected via its row badge/track — see
    // onCharTrackClick. Only Delete (and Done) apply to "remove this entire layer"; every
    // other tool here targets a specific clip's content/timing, which a character-layer
    // selection isn't.
    const isCharLayer = typeof selectedClip==='string' && selectedClip.startsWith('char') && !isStickman;
    const keyTool = document.getElementById('clipKeyTool');
    const animTool = document.getElementById('clipAnimTool');
    const faceGestureTool = document.getElementById('clipCharacterActionsTool');
    const trimTool = document.getElementById('clipTrimTool');
    const speedTool = document.getElementById('clipSpeedTool');
    const volumeTool = document.getElementById('clipVolumeTool');
    const chromaTool = document.getElementById('clipChromaTool');
    const extractAudioTool = document.getElementById('clipExtractAudioTool');
    const selectedOverlayForTools = overlays.find(o=>o.id===selectedClip);
    if(selectedOverlayForTools&&(selectedOverlayForTools.type==='image'||selectedOverlayForTools.type==='video')&&!selectedOverlayForTools.transform){
      selectedOverlayForTools.transform={scale:1,offsetX:0,offsetY:0,rotation:0};
    }
    const isVideoClip = !!selectedOverlayForTools && selectedOverlayForTools.type==='video';
    if(chromaTool) chromaTool.style.display = isVideoClip ? 'flex' : 'none';
    if(extractAudioTool) extractAudioTool.style.display = isVideoClip ? 'flex' : 'none';
    const addTool = document.getElementById('clipAddMediaTool');
    const transformTool = document.getElementById('clipTransformTool');
    const splitTool = document.getElementById('clipSplitTool');
    if(splitTool) splitTool.style.display = isCharLayer ? 'none' : 'flex';
    if(keyTool) keyTool.style.display = isStickman ? 'flex' : 'none';
    if(animTool) animTool.style.display = isStickman ? 'flex' : 'none';
    if(faceGestureTool) faceGestureTool.style.display = isStickman && !!getClipByRef(selectedClip) ? 'flex' : 'none';
    // BUGFIX (Trim removed, Replace added per request): Replace now covers every real clip
    // type — Background (reopens the image picker), a media/overlay clip (swaps its content
    // in place), and a stickman animation clip (swaps its preset via the Animate panel) — see
    // replaceSelectedClip(). Only a whole character-LAYER selection (the row itself, not one
    // of its clips) has nothing to replace, same as every other clip-specific tool here.
    if(trimTool) trimTool.style.display = isCharLayer ? 'none' : 'flex';
    // Speed applies to the stickman animation clip and to real playing media (video/image/
    // audio overlays) — not to the Background row (a static image spanning the whole
    // project, not a clip with its own playback rate) or plain shape overlays (solid color,
    // nothing to play back). See speedTargetClip() for the single source of truth this
    // mirrors.
    const speedClip = isCharLayer ? null : speedTargetClip();
    if(speedTool) speedTool.style.display = speedClip ? 'flex' : 'none';
    const speedLabelEl = document.getElementById('clipSpeedLabel');
    if(speedLabelEl && speedClip){
      const sp = speedClip.speed || 1;
      speedLabelEl.textContent = (Math.abs(sp-1)<0.01) ? 'Speed' : (sp.toFixed(2).replace(/0+$/,'').replace(/\.$/,'')+'x');
    }
    // FEATURE (control any clip's own audio volume): Volume applies only to clips that can
    // actually carry sound — video and audio overlays — see volumeTargetClip() for the
    // single source of truth this mirrors (same isCharLayer guard as every other
    // clip-specific tool here, since a character-LAYER selection has no one clip's volume
    // to speak of).
    const volumeClip = isCharLayer ? null : volumeTargetClip();
    if(volumeTool) volumeTool.style.display = volumeClip ? 'flex' : 'none';
    const volumeLabelEl = document.getElementById('clipVolumeLabel');
    if(volumeLabelEl && volumeClip){
      const vp = Math.round((volumeClip.volume==null?1:volumeClip.volume)*100);
      volumeLabelEl.textContent = (vp===100) ? 'Volume' : (vp===0 ? 'Muted' : vp+'%');
    }
    if(addTool) addTool.style.display = isOverlay ? 'flex' : 'none';
    // BUGFIX: Transform now genuinely edits Background/Clip position+size+rotation (see
    // transformTarget()), but only image/video clips actually have a picture on the canvas
    // to move — audio waveforms and plain-color shape layers have nothing visual of their
    // own to transform, so the button (which would otherwise silently edit the STICKMAN's
    // transform instead, confusingly) is hidden for those two clip types specifically.
    const ovForTransform = overlays.find(o=>o.id===selectedClip);
    const hasVisualTransform = !!backgroundClipForSelection() || isStickman || (ovForTransform && !!ovForTransform.transform);
    if(transformTool) transformTool.style.display = (isCharLayer || !hasVisualTransform) ? 'none' : 'flex';
    // FEATURE ("move any layer or animation character or clip forward or backward... bring
    // one clip above another"): Forward/Backward apply to anything with a real paint-order
    // concept — an overlay clip, a background clip, or a whole character layer — using
    // reorderSelectedLayer()'s per-type logic. They do NOT apply to a specific stickman
    // ANIMATION clip ('main'/'segN'): a character's z-order is a property of the character
    // itself (see characters[] array order / activeCharIdx), not of any one clip on its
    // timeline, so reordering only makes sense from the character-layer selection, same
    // scope Delete already uses for "remove this whole layer" versus "trim this one clip".
    const bgForReorder = backgroundClipForSelection();
    const canReorder = isCharLayer || isOverlay || !!bgForReorder;
    const forwardTool = document.getElementById('clipForwardTool');
    const backwardTool = document.getElementById('clipBackwardTool');
    if(forwardTool) forwardTool.style.display = canReorder ? 'flex' : 'none';
    if(backwardTool) backwardTool.style.display = canReorder ? 'flex' : 'none';
    // Duplicate applies to anything Split already applies to (a real clip or a whole
    // character layer) — same scope, see splitSelectedClipAtPlayhead's own isCharLayer gate,
    // except Duplicate additionally makes sense for a character LAYER too (split doesn't,
    // since "splitting a character" isn't a meaningful operation the way splitting one of
    // its clips is) — duplicating a whole character is exactly "add another one like this".
    const dupTool = document.getElementById('clipDuplicateTool');
    if(dupTool) dupTool.style.display = 'flex';
    const deleteCharacterTool = document.getElementById('clipDeleteCharacterTool');
    if(deleteCharacterTool) deleteCharacterTool.style.display = isStickman ? 'flex' : 'none';
  } else {
    rowClip.style.display = 'none';
    rowKf.style.display = 'none';
    rowDefault.style.display = 'flex';
  }
}
// Deletes whichever clip is currently selected — background clears its image/color,
// an overlay is removed entirely, and the main stickman clip clears its animation.
function deleteSelectedClip(){
  if(!selectedClip) return;
  if(selectedClip === 'main'){
    // The main clip is just "whichever clip happens to be in the activeClip slot" now that
    // several animation clips can coexist — deleting it should NOT wipe out other clips
    // still sitting in splitStickmanSegments. If any remain, promote the earliest one into
    // the activeClip slot instead of resetting the whole track; only reset fully when this
    // really was the last clip left.
    if(splitStickmanSegments.length){
      splitStickmanSegments.sort((a,b)=>a.trimStart-b.trimStart);
      activeClip = splitStickmanSegments.shift();
      syncClipRefsToCharacter();
      currentAnimKey = activeClip.animKey;
      renderTimelineClip();
      renderAnimList();
      updateTimeReadout();
      showToast('Clip deleted', {icon: toastIcons.check, duration:1200});
      persistProjectState();
      deselectAllClips();
    } else {
      // BUGFIX (deleting a lone character's last clip left a blank row instead of "Add
      // character"): this used to only remove the character when at least one OTHER
      // character still existed, and fell back to resetPose() otherwise — because
      // deleteCharacter() itself refuses to ever drop the roster to zero (that guard exists
      // for its OTHER call site, the character row's own explicit delete button). But that
      // meant deleting the only remaining character's last clip left it in `characters[]`
      // with no clip and no segments: the row still rendered (its badge, an empty-looking
      // track) instead of disappearing back into the dashed "Add character" placeholder a
      // brand-new project starts with. Zero characters is already a fully-supported state
      // (see initStickman's own comment), so this now reaches it deliberately instead of
      // stopping one short of it — same steps deleteCharacter() itself takes, just without
      // its "keep at least one" guard, since removing the LAST clip is a clearer signal of
      // intent to remove the character than tapping its row's delete button is.
      const charIdx = activeCharIdx;
      if(activeCharacter()){
        characters.splice(charIdx, 1);
        if(activeCharIdx >= characters.length) activeCharIdx = characters.length-1;
        syncActiveCharacterRefs();
        selectedJoint = null;
        drawStick();
        renderCharacterLayers();
        syncTransformSliders();
        persistProjectState();
        showToast('Character deleted', {icon: toastIcons.check, duration:1200});
        deselectAllClips();
        onClipSelected(null,'delete');
      } else {
        resetPose();
      }
      onClipSelected(null,'delete');
      return; // both branches above already call deselectAllClips() themselves
    }
  } else if(backgroundClipForSelection()){
    // FEATURE (multiple sequential backgrounds): deletes ONLY the specific selected
    // background clip, not the whole row — other background clips before/after it are
    // untouched, matching how deleting one overlay clip never affects its neighbors.
    const bgClip = backgroundClipForSelection();
    backgrounds = backgrounds.filter(b=>b.id!==bgClip.id);
    deselectAllClips();
    refreshCurrentBgFromPlayhead();
    applyBackground();
    renderBackgroundTrack();
    showToast('Background deleted', {icon: toastIcons.check, duration:1200});
    persistProjectState();
  } else if(String(selectedClip).startsWith('seg')){
    const segIdx = parseInt(String(selectedClip).slice(3), 10);
    splitStickmanSegments.splice(segIdx, 1);
    renderTimelineClip();
    showToast('Clip deleted', {icon: toastIcons.check, duration:1200});
    persistProjectState();
  } else if(String(selectedClip).startsWith('char')){
    // Deletes the whole character layer — same operation the old inline .char-row-delete
    // button used to perform, now reached from the bottom toolbar like every other delete
    // (see the .char-row-delete removal note in renderCharacterLayers). deleteCharacter()
    // already refuses to remove the last remaining character and shows its own toast for
    // that case, so nothing extra is needed here for that guard.
    const idx = parseInt(String(selectedClip).slice(4), 10);
    deleteCharacter(idx);
    showToast('Character deleted', {icon: toastIcons.check, duration:1200});
  } else {
    overlays = overlays.filter(o=>o.id!==selectedClip);
    renderOverlayLayers();
    showToast('Layer deleted', {icon: toastIcons.check, duration:1200});
    persistProjectState();
  }
  deselectAllClips();
  onClipSelected(null,'delete');
}
// Removes the entire active character row while an animation clip (main or split segment) is selected.
// This is deliberately separate from the normal Delete action, which deletes only the selected clip.
function deleteSelectedCharacterFromAnimation(){
  const isAnimationClip = selectedClip === 'main' || (typeof selectedClip === 'string' && selectedClip.startsWith('seg'));
  if(!isAnimationClip){ showToast('Select an animation clip first', {icon:toastIcons.info, duration:1400}); return; }
  const idx = activeCharIdx;
  if(idx < 0 || idx >= characters.length || !characters[idx]) return;
  pausePlayback();
  characters.splice(idx, 1);
  activeCharIdx = characters.length ? Math.min(idx, characters.length - 1) : 0;
  selectedClip = null;
  selectedKfIdx = null;
  selectedJoint = null;
  if(characters.length){
    syncActiveCharacterRefs();
  } else {
    // Empty projects are supported; clear stale clip references so the transport cannot keep
    // targeting the deleted character after its row has disappeared.
    activeClip = null;
    splitStickmanSegments = [];
  }
  drawStick();
  renderCharacterLayers();
  syncTransformSliders();
  syncBottomToolsMode();
  updateKeyButtonState();
  persistProjectState();
  showToast('Character row deleted', {icon:toastIcons.check, duration:1400});
  onClipSelected(null,'delete');
}

// Split (CapCut-style): cuts whichever clip is selected at the current playhead position,
// producing two independent clips that can each be trimmed/dragged/deleted separately.
function splitSelectedClipAtPlayhead(){
  if(!selectedClip){
    showToast('Select a clip first', {icon: toastIcons.info, duration:1400});
    return;
  }
  // Reject a cut outside the selected clip before pausing or changing timeline state.
  if(!onClipSelected(selectedClip,'split'))return;
  // Pause exactly at the current time first (PART 22), then read the canonical clock
  // directly rather than back-deriving it from scroller.scrollLeft — scrollLeft can be
  // mid-animation/mid-snap on some browsers, whereas currentAbsoluteTimeSec is always the
  // exact committed time.
  const splitTimeBeforePause=typeof getCurrentAbsoluteTime==='function'?getCurrentAbsoluteTime():currentAbsoluteTimeSec;
  pausePlayback();
  const absPlayheadSec = Number.isFinite(Number(splitTimeBeforePause))?Number(splitTimeBeforePause):currentAbsoluteTimeSec;

  if(selectedClip === 'main'){
    if(!activeClip) return;
    if(absPlayheadSec-activeClip.trimStart < MIN_CLIP_SEC-1e-6 ||
       activeClip.trimEnd-absPlayheadSec < MIN_CLIP_SEC-1e-6){
      showToast('Move playhead inside the clip to split', {icon: toastIcons.info, duration:1600});
      return;
    }
    // Second half becomes its own independent segment (see splitStickmanSegments) sharing the
    // same animation, so both halves are independently trimmable/deletable — exactly like
    // CapCut splitting one clip into two adjacent clips on the same track.
    const secondHalf = {
      animKey: activeClip.animKey,
      trimStart: absPlayheadSec,
      trimEnd: activeClip.trimEnd,
      keyframes: activeClip.keyframes.filter(kf=>kf.t >= absPlayheadSec).map(k=>({...k})),
      speed: activeClip.speed || 1
    };
    activeClip.trimEnd = absPlayheadSec;
    activeClip.keyframes = activeClip.keyframes.filter(kf=>kf.t < absPlayheadSec);
    splitStickmanSegments.push(secondHalf);
    // BUGFIX ("cutting the same clip twice kept producing extra orphaned pieces"):
    // selectedClip used to stay pointed at 'main' (the FIRST half) after a split. If the
    // person then slid the playhead and hit Split again without re-tapping a clip, this
    // function split the already-finished first half YET AGAIN instead of acting on the
    // piece they'd actually moved past — so every repeated cut kept adding one more
    // leftover fragment to the timeline rather than cleanly stopping at two. CapCut always
    // leaves the clip AFTER the cut point selected once a split finishes, so the very next
    // Split naturally acts on new, not-yet-cut footage. The new second half is always the
    // last entry just pushed onto splitStickmanSegments.
    selectedClip = 'seg' + (splitStickmanSegments.length - 1);
    renderTimelineClip();
    selectClipForChrome(selectedClip);
  } else if(backgroundClipForSelection()){
    // FEATURE (multiple sequential backgrounds — split now works, since a background clip
    // is a real timeline clip like an overlay now, not the single project-wide value it used
    // to be): same "cut in half at the playhead" pattern as the overlay branch below —
    // trimEnd of the original clip becomes the cut point, a new clip with a fresh id picks
    // up from there to the original trimEnd. resolveBgClipOverlaps isn't needed here since
    // the two halves exactly share the cut edge by construction (never overlapping).
    const bg = backgroundClipForSelection();
    if(absPlayheadSec-bg.trimStart < MIN_CLIP_SEC-1e-6 ||
       bg.trimEnd-absPlayheadSec < MIN_CLIP_SEC-1e-6){
      showToast('Move playhead inside the clip to split', {icon: toastIcons.info, duration:1600});
      return;
    }
    const bgId2 = 'bg'+Date.now();
    const bgCutTransform=ethvelleGetObjectKeyframeValue(bg,'transform',absPlayheadSec)||ethvelleCloneObjectKeyframeValue(bg.transform,'transform');
    const bgKeys=Array.isArray(bg.ethvelleMediaKeyframes)?bg.ethvelleMediaKeyframes:[];
    const secondHalf = { ...bg, id:bgId2, transform: {...bg.transform}, trimStart: absPlayheadSec,
      ethvelleMediaKeyframeBase:{...(bg.ethvelleMediaKeyframeBase||{}),transform:bgCutTransform},
      ethvelleMediaKeyframes:bgKeys.filter(k=>Number(k.t)>=absPlayheadSec).map(k=>({...k,value:ethvelleCloneObjectKeyframeValue(k.value,k.property)})) };
    bg.trimEnd = absPlayheadSec;
    bg.ethvelleMediaKeyframes=bgKeys.filter(k=>Number(k.t)<absPlayheadSec);
    backgrounds.push(secondHalf);
    backgrounds.sort((a,b)=>a.trimStart-b.trimStart);
    // BUGFIX: same fix as the main-clip branch above — move selection onto the new second
    // half so a repeated Split acts on fresh footage instead of re-cutting the same first
    // half over and over and piling up orphaned background fragments.
    selectedClip = bgId2;
    renderBackgroundTrack();
    selectClipForChrome(selectedClip);
  } else if(String(selectedClip).startsWith('seg')){
    const segIdx = parseInt(String(selectedClip).slice(3), 10);
    const seg = splitStickmanSegments[segIdx];
    if(!seg) return;
    if(absPlayheadSec-seg.trimStart < MIN_CLIP_SEC-1e-6 ||
       seg.trimEnd-absPlayheadSec < MIN_CLIP_SEC-1e-6){
      showToast('Move playhead inside the clip to split', {icon: toastIcons.info, duration:1600});
      return;
    }
    const secondHalf = {
      animKey: seg.animKey,
      trimStart: absPlayheadSec,
      trimEnd: seg.trimEnd,
      keyframes: (seg.keyframes||[]).filter(kf=>kf.t >= absPlayheadSec).map(k=>({...k})),
      speed: seg.speed || 1
    };
    seg.trimEnd = absPlayheadSec;
    seg.keyframes = (seg.keyframes||[]).filter(kf=>kf.t < absPlayheadSec);
    splitStickmanSegments.push(secondHalf);
    // BUGFIX: same fix as the main-clip branch above.
    selectedClip = 'seg' + (splitStickmanSegments.length - 1);
    renderTimelineClip();
    selectClipForChrome(selectedClip);
  } else {
    const ov = overlays.find(o=>o.id===selectedClip);
    if(!ov) return;
    if(absPlayheadSec-ov.trimStart < MIN_CLIP_SEC-1e-6 ||
       ov.trimEnd-absPlayheadSec < MIN_CLIP_SEC-1e-6){
      showToast('Move playhead inside the clip to split', {icon: toastIcons.info, duration:1600});
      return;
    }
    let id2 = 'ov'+Date.now()+Math.random().toString(36).slice(2,7);
    while(overlays.some(o=>String(o.id)===String(id2))) id2='ov'+Math.random().toString(36).slice(2,10);
    // BUGFIX ("editing one half of a split clip also changed the other half"): this used to
    // be a shallow `{...ov}` spread, which copies `transform` (and any other nested object
    // field) by REFERENCE, not by value — so the new second-half clip and the original first
    // half ended up pointing at the exact same transform object. updateTransform() edits
    // scale/offsetX/offsetY/rotation IN PLACE on that object (t.scale = ...), so adjusting
    // size/position/rotation on either half silently moved both at once, since there was
    // really only ever one shared transform behind the two clips. Every split-off clip must
    // get its OWN independent copy of every nested/object field from here on, exactly like
    // the background-clip split branch already correctly does for its own `transform`.
    const secondHalf = {
      ...ov, id:id2, name: ov.name, trimStart: absPlayheadSec, splitPart:true,
      // A split must continue at the exact source-media position of the cut, not restart
      // the duplicated file at 0. This offset is measured in source seconds (speed-aware).
      sourceOffset: (Number(ov.sourceOffset)||0) + Math.max(0,(absPlayheadSec-ov.trimStart)*(Number(ov.speed)||1)),
      transform: ov.transform ? {...ov.transform} : ov.transform,
      audioSettings: ov.audioSettings ? {...ov.audioSettings} : ov.audioSettings,
      ethvelleMediaKeyframeBase:{...(ov.ethvelleMediaKeyframeBase||{}),[ethvelleObjectKeyframeProperty(ov)]:ethvelleGetObjectKeyframeValue(ov,ethvelleObjectKeyframeProperty(ov),absPlayheadSec)??ethvelleObjectBaseValue(ov,ethvelleObjectKeyframeProperty(ov))},
      ethvelleMediaKeyframes:(Array.isArray(ov.ethvelleMediaKeyframes)?ov.ethvelleMediaKeyframes:[]).filter(k=>Number(k.t)>=absPlayheadSec).map(k=>({...k,value:ethvelleCloneObjectKeyframeValue(k.value,k.property)}))
    };
    ov.splitPart = true;
    ov.trimEnd = absPlayheadSec;
    ov.ethvelleMediaKeyframes=(Array.isArray(ov.ethvelleMediaKeyframes)?ov.ethvelleMediaKeyframes:[]).filter(k=>Number(k.t)<absPlayheadSec);
    overlays.push(secondHalf);
    // BUGFIX: same fix as the main-clip branch above — select the new second half so
    // repeated splits don't keep re-cutting the same already-finished first half.
    selectedClip = id2;
    renderOverlayLayers();
    selectClipForChrome(selectedClip);
  }
  // Keep the playhead exactly where the cut just happened instead of snapping back to 0 —
  // `frame` is clip-relative to activeClip.trimStart, which the split above does NOT change
  // for the first half, so re-deriving frame from the same absolute second keeps the visual
  // playhead glued to the cut point (matches CapCut: splitting never moves your position).
  seekAbsoluteTimelineSeconds(absPlayheadSec);
  showToast('Clip split', {icon: toastIcons.check, duration:1200});
  persistProjectState();
}

// Tapping a layer's badge/icon (Stickman / Background) highlights that row in the
// timeline and arms the contextual clip toolbar underneath. It no longer force-opens a
// tool panel — that's a separate, explicit action now (matching CapCut: selecting a clip
// and opening its properties are two different taps).
// BUGFIX: this used to be wired to onclick on the ENTIRE .layer-row, not just the badge.
// Because .layer-track is flex:1, that row spans the full (often very wide/scrollable)
// timeline width — so clicking ANY empty timeline space in that row, not just the actual
// clip, was bubbling up to this and selecting+pausing. That's what looked like "playback
// pauses when I click anywhere on screen". The clip body itself already selects on its
// own onpointerdown (see wireClipInteractions/wireOverlayInteractions) and stops the event
// from bubbling here, so this handler only needs to live on the small badge icon now —
// empty track space no longer does anything, and playback only pauses when you actually
// tap a clip/layer, exactly like CapCut.
let selectedLayer = null;
function selectLayer(which){
  selectedLayer = which;
  document.querySelectorAll('.layer-row').forEach(row=>{
    row.classList.toggle('selected', row.dataset.layer === which);
  });
  if(which === 'stickman'){
    // There's no separate "stickman" row anymore — each character has its own row and own
    // clip. Selecting "Stickman" from the Media panel now means "the active character's
    // clip", so re-highlight that character's row (data-char-idx, not data-layer) and select
    // its main clip exactly like before.
    const activeRow = document.querySelector(`.layer-row[data-char-idx="${activeCharIdx}"]`);
    if(activeRow) activeRow.classList.add('selected');
    selectClipForChrome('main');
  } else if(which === 'background'){
    // FEATURE (multiple sequential backgrounds): "Background" here means "whichever
    // background clip is currently under the playhead" if one exists, otherwise there's
    // nothing yet to select — fall through to the normal empty-row "Add background" tap
    // behavior instead of selecting a nonexistent clip.
    const bg = currentBackgroundClip(getCurrentAbsoluteTime());
    if(bg) selectClipForChrome(bg.id);
    else openPanel('bg');
  }
}
function selectOverlayLayer(id){
  selectedLayer = null;
  document.querySelectorAll('.layer-row').forEach(row=>row.classList.remove('selected'));
  const ov = overlays.find(o=>o.id===id);
  const groupKey = ov && ov.type==='audio' ? 'audio' : (ov && ov.type==='text' ? 'text' : 'clips');
  const row = document.querySelector(`.overlay-track-row[data-group="${groupKey}"]`);
  if(row) row.classList.add('selected');
  selectClipForChrome(id);
  // BUGFIX (thin white line on select): selectClipForChrome only toggles .selected classes,
  // it never recomputes left/width/background. If anything upstream left stale geometry on
  // this clip, a plain class toggle would just keep showing it collapsed. Re-running the
  // overlay render here makes selecting a layer always paint fresh geometry, not just fresh
  // selection chrome.
  renderOverlayLayers();
}

/* ---------------- Overlays (CapCut-style extra layer tracks, add / drag / trim / delete) ---------------- */
const OVERLAY_COLORS = ['#F5A623','#FF6B9D','#7C9CFF','#C08CFF','#FFD166'];
function addOverlayLayer(){
  const id='ov'+Date.now(),color=OVERLAY_COLORS[overlays.length%OVERLAY_COLORS.length],len=2;
  const group=overlays.filter(o=>o&&o.type!=='audio'&&o.type!=='text').sort((a,b)=>(Number(a.trimStart)||0)-(Number(b.trimStart)||0));
  let playhead=0;try{playhead=Math.max(0,Number(typeof getCurrentAbsoluteTime==='function'?getCurrentAbsoluteTime():currentAbsoluteTimeSec)||0);}catch(_e){}
  let at=group.length;
  for(let i=0;i<group.length;i++){
    const c=group[i],st=Number(c.trimStart)||0,l=Math.max(MIN_CLIP_SEC,(Number(c.trimEnd)||st+MIN_CLIP_SEC)-st),en=st+l;
    if(playhead<=st+.001){at=i;break;}
    if(playhead<en-.001){at=playhead<(st+en)/2?i:i+1;break;}
  }
  const newClip={id,name:'Shape '+(overlays.length+1),color,type:'shape',trimStart:0,trimEnd:len};
  const ordered=[...group.slice(0,at),newClip,...group.slice(at)];let cursor=0;
  ordered.forEach(c=>{
    const oldStart=Number(c.trimStart)||0;
    const oldKeyframeTimes=Array.isArray(c.ethvelleMediaKeyframes)?c.ethvelleMediaKeyframes.map(k=>Number(k&&k.t)||0):[];
    const l=c===newClip?len:Math.max(MIN_CLIP_SEC,(Number(c.trimEnd)||0)-(Number(c.trimStart)||0));
    c.trimStart=cursor;c.trimEnd=cursor+l;
    const movedBy=c.trimStart-oldStart;
    if(c!==newClip&&Math.abs(movedBy)>0.000001)shiftTimelineKeyframesOnce(c.ethvelleMediaKeyframes,oldKeyframeTimes,movedBy);
    cursor=c.trimEnd;
  });
  overlays.push(newClip);timelineDuration=Math.max(Number(timelineDuration)||0,cursor);
  renderOverlayLayers();syncTimelineInnerWidth();renderRuler();selectClipForChrome(id);renderOverlayListPanel();
  showToast('Shape layer added',{icon:toastIcons.check,duration:1200});persistProjectState();closePanel();
}
function renderOverlayListPanel(){
  const container = document.getElementById('overlayListBody');
  if(!container) return;
  if(!overlays.length){container.innerHTML='';return;}
  container.innerHTML = overlays.map(ov=>{
    const selected = String(selectedClip) === String(ov.id);
    const isVisual=ov.type==='image'||ov.type==='video';
    const thumb=(Array.isArray(ov.thumbs)&&ov.thumbs.length?ov.thumbs[0]:(ov.type==='image'?ov.src:''));
    // Video previews are filled by the V8 src-keyed filmstrip job. Do not start a
    // second, per-clip one-frame decoder here (split halves share the same source).
    const thumbMarkup=isVisual&&thumb
      ? `<div class="anim-play-mini ev-media-list-thumb"><img src="${escapeHtml(thumb)}" alt="${escapeHtml(ov.name||'Media thumbnail')}" loading="lazy" decoding="async"></div>`
      : `<div class="anim-play-mini ev-media-list-thumb ev-media-list-placeholder" aria-hidden="true">${layerBadgeIcon(ov.type)}</div>`;
    return `
      <div class="anim-item ${selected?'selected':''}" onclick="selectClipForChrome('${ov.id}');renderOverlayListPanel();">
        ${thumbMarkup}
        <div class="anim-item-name">${escapeHtml(ov.name)}</div>
        ${selected ? '<span class="anim-item-badge">SELECTED</span>' : ''}
      </div>`;
  }).join('');
}
// ---------------- Trim panel: shown for the currently-selected overlay/background/media
// clip via the contextual bottom toolbar's "Trim" button (see syncBottomToolsMode). Lets
// the person nudge trimStart/trimEnd numerically instead of only via the drag handles on
// the timeline — handy for fine adjustments the finger can't hit precisely on a small
// screen. Not shown for the main stickman clip, which already has its own Split-based flow.
function trimPanelTargetClip(){
  if(!selectedClip || selectedClip==='main' || String(selectedClip).startsWith('seg')) return null;
  const bg = backgroundClipForSelection();
  if(bg) return bg;
  return overlays.find(o=>o.id===selectedClip) || null;
}
function renderTrimPanel(){
  const body = document.getElementById('trimPanelBody');
  if(!body) return;
  const ov = trimPanelTargetClip();
  if(!ov){
    body.innerHTML = `<p class="field-label">Select a media or overlay clip on the timeline to trim it.</p>`;
    return;
  }
  const isBg = !!backgrounds.find(b=>b.id===ov.id);
  const len = (ov.trimEnd - ov.trimStart).toFixed(2);
  const label = isBg ? 'Background' : ov.name;
  // Background images have no fixed source duration to cap End at (unlike video/audio,
  // which can only be trimmed up to how much footage actually exists) — allow a generous
  // fixed ceiling instead of ov.sourceDuration, which background clips don't have.
  const endMax = isBg ? (ov.trimStart + 60) : (ov.trimStart + (ov.sourceDuration||30));
  body.innerHTML = `
    <p class="field-label">${escapeHtml(label)} · ${len}s</p>
    <div class="trim-row">
      <label>Start</label>
      <input type="range" min="0" max="${Math.max(MIN_CLIP_SEC, ov.trimEnd-MIN_CLIP_SEC).toFixed(2)}" step="0.05" value="${ov.trimStart.toFixed(2)}" oninput="trimPanelSetStart(this.value)">
      <span class="trim-val" id="trimStartVal">${ov.trimStart.toFixed(2)}s</span>
    </div>
    <div class="trim-row">
      <label>End</label>
      <input type="range" min="${(ov.trimStart+MIN_CLIP_SEC).toFixed(2)}" max="${endMax.toFixed(2)}" step="0.05" value="${ov.trimEnd.toFixed(2)}" oninput="trimPanelSetEnd(this.value)">
      <span class="trim-val" id="trimEndVal">${ov.trimEnd.toFixed(2)}s</span>
    </div>
    <button class="btn-primary-block" style="margin-top:14px;" onclick="closePanel()">Done</button>
  `;
}
function trimPanelSetStart(val){
  const ov = trimPanelTargetClip();
  if(!ov) return;
  const isBg = !!backgrounds.find(b=>b.id===ov.id);
  const oldStart=ov.trimStart;
  ov.trimStart = Math.min(ov.trimEnd-MIN_CLIP_SEC, Math.max(0, parseFloat(val)));
  if(!isBg&&(ov.type==='audio'||ov.type==='video')) ov.sourceOffset=Math.max(0,(Number(ov.sourceOffset)||0)+(ov.trimStart-oldStart)*(Number(ov.speed)||1));
  if(isBg) resolveBgClipOverlaps(ov.id); // hard wall — numeric entry can't overlap a neighbor either
  document.getElementById('trimStartVal').textContent = ov.trimStart>=60?evFormatClock(ov.trimStart,true):ov.trimStart.toFixed(2)+'s';
  if(isBg){ renderBackgroundTrack(); refreshCurrentBgFromPlayhead(); applyBackground(); }
  else renderOverlayLayers();
  persistProjectState({skipHistory:true});
}
function trimPanelSetEnd(val){
  const ov = trimPanelTargetClip();
  if(!ov) return;
  const isBg = !!backgrounds.find(b=>b.id===ov.id);
  ov.trimEnd = Math.max(ov.trimStart+MIN_CLIP_SEC, parseFloat(val));
  if(isBg) resolveBgClipOverlaps(ov.id); // hard wall — numeric entry can't overlap a neighbor either
  if(ov.trimEnd > timelineDuration - 0.3){ timelineDuration = ov.trimEnd+1; syncTimelineInnerWidth(); renderRuler(); }
  document.getElementById('trimEndVal').textContent = ov.trimEnd>=60?evFormatClock(ov.trimEnd,true):ov.trimEnd.toFixed(2)+'s';
  if(isBg){ renderBackgroundTrack(); refreshCurrentBgFromPlayhead(); applyBackground(); }
  else renderOverlayLayers();
  persistProjectState({skipHistory:true});
}

// ---------------- Speed panel: shown for the currently-selected clip via the contextual
// bottom toolbar's "Speed" button (see syncBottomToolsMode). Works across every clip type
// that actually plays back content over time — the stickman animation clip (main/segN),
// and video/image/audio overlays — CapCut-style: raising speed shrinks the clip's footprint
// on the timeline (it now takes less time to play through the same content), lowering speed
// grows it. Not offered for the Background row (background clips can be trimmed/split/moved
// but have no play-rate of their own — see renderBackgroundTrack/resolveBgClipOverlaps for
// the actual "adjust how much of the timeline a background covers" controls) or for plain
// shape overlays (solid color, nothing to play back).
function speedTargetClip(){
  if(!selectedClip) return null;
  if(selectedClip==='main') return activeClip;
  if(String(selectedClip).startsWith('seg')) return getClipByRef(selectedClip);
  const ov = overlays.find(o=>o.id===selectedClip);
  if(ov && ov.type!=='shape') return ov;
  return null;
}
// Whether the currently-selected clip is a stickman animation clip (main or segN) — these
// use preset.duration as their "1x length" reference instead of sourceDuration.
function isStickmanSpeedTarget(clip){
  return selectedClip==='main' || String(selectedClip).startsWith('seg');
}
// The clip's own "natural", 1x-speed length in seconds — for stickman clips this is however
// long the clip currently spans (there's no fixed source length to fall back to, since a
// looped preset can be stretched to any length); for real media (video/audio) it's the
// uploaded file's own sourceDuration, so raising speed always shrinks toward however much
// of the ACTUAL content plays, and lowering speed stretches it back out, exactly like CapCut.
function speedTargetBaseLen(clip){
  if(isStickmanSpeedTarget(clip)){
    // Lazily remember the length this clip had the very first time its speed is ever
    // touched, so repeated speed changes always scale from the same 1x reference instead of
    // compounding off whatever the PREVIOUS speed setting happened to leave behind.
    if(clip._speedBaseLen == null) clip._speedBaseLen = clip.trimEnd - clip.trimStart;
    return clip._speedBaseLen;
  }
  return clip.sourceDuration || (clip.trimEnd - clip.trimStart);
}

  return null;
}

// FEATURE (extract audio from any clip): pulls the audio track out of the selected VIDEO
// clip into a brand new, independent audio-only overlay clip — same timeline position and
// length as the source video, so the two start out perfectly in sync and can then be
// trimmed/moved/volume-adjusted completely independently of one another (e.g. muting the
// video's own sound via its Volume panel while keeping the extracted copy playing, or
// sliding just the audio a few frames to fix sync). Re-uses the exact same source file — no
// server-side transcoding exists in this app, so the "extraction" is really "treat this
// video's own file as an audio source too", which works because <audio>/decodeAudioData
// both happily read the audio track straight out of a video file's container; the ORIGINAL
// video clip is left exactly as it was (still has its own sound, still visible) so this is
// non-destructive and can be undone by just deleting the new audio clip.
function extractAudioFromClip(){
  const clip = volumeTargetClip();
  if(!clip || clip.type !== 'video'){ showToast('Select a video clip to extract its audio', {icon: toastIcons.info}); return; }
  const newClip = {
    id: 'ov'+Date.now()+Math.random().toString(36).slice(2,5),
    name: (clip.name||'Video') + ' (audio)',
    color: clip.color,
    type: 'audio', src: clip.src, sourceDuration: clip.sourceDuration, sourceOffset: Number(clip.sourceOffset)||0,
    trimStart: clip.trimStart, trimEnd: clip.trimEnd,
    thumbs: [], peaks: clip.peaks && clip.peaks.length ? clip.peaks : null,
    speed: clip.speed || 1, volume: 1, transform: null
  };
  overlays.push(newClip);
  timelineDuration = Math.max(timelineDuration, newClip.trimEnd);
  // A freshly extracted clip with no waveform peaks yet (peaks are only generated at
  // upload time, see generateAudioWaveform/triggerUpload — this clip never went through an
  // upload) gets them generated now, same function, same async fill-in-when-ready pattern
  // every other audio clip's waveform already uses.
  if(!newClip.peaks && typeof generateAudioWaveform === 'function') generateAudioWaveform(newClip.src).then(result=>{ newClip.peaks = result.peaks; renderOverlayLayers(); }).catch(()=>{});
  renderOverlayLayers();
  syncTimelineInnerWidth();
  renderRuler();
  onClipSelected(newClip,'add');
  renderOverlayListPanel();
  showToast('Audio extracted as its own clip', {icon: toastIcons.check, duration:1600});
  persistProjectState();
}

function speedPanelSetSpeed(val){
  const clip = speedTargetClip();
  if(!clip) return;
  const newSpeed = Math.max(0.25, Math.min(3, parseFloat(val)));
  const baseLen = speedTargetBaseLen(clip);
  clip.speed = newSpeed;
  // CapCut-style: the clip's own length ON THE TIMELINE rescales with speed (2x -> half as
  // long, 0.5x -> twice as long), anchored at its current trimStart so it never jumps to a
  // different position while you're mid-adjustment. sourceDuration/anim-preset content keeps
  // playing at the new rate (see poseAtClipTime's speed-scaled clipLocalT, and
  // renderOverlayMedia's speed-scaled localT / playbackRate for video & audio).
  const newLen = Math.max(MIN_CLIP_SEC, baseLen / newSpeed);
  clip.trimEnd = clip.trimStart + newLen;
  if(clip.trimEnd > timelineDuration - 0.3){ timelineDuration = clip.trimEnd + 1; syncTimelineInnerWidth(); renderRuler(); }
  // A speed change may make this animation overlap another character clip. That is
  // represented as a separate vertical sub-lane by renderTimelineClip below; neither clip's
  // timeline position is clamped or silently shifted.
  const label = document.getElementById('speedVal');
  if(label) label.textContent = newSpeed.toFixed(2)+'x';
  const slider = document.querySelector('#speedPanelBody input[type="range"]');
  if(slider) slider.value = newSpeed.toFixed(2);
  const header = document.getElementById('speedPanelHeader');
  if(header) header.textContent = `${clip.name || (isStickmanSpeedTarget(clip) ? 'Animation' : 'Clip')} · ${(clip.trimEnd-clip.trimStart).toFixed(2)}s at ${newSpeed.toFixed(2)}x`;
  document.querySelectorAll('.speed-preset-btn').forEach(btn=>{
    const btnVal = parseFloat(btn.dataset.speed);
    btn.classList.toggle('active', Math.abs(btnVal-newSpeed) < 0.01);
  });
  const clipSpeedLabel = document.getElementById('clipSpeedLabel');
  if(clipSpeedLabel) clipSpeedLabel.textContent = (Math.abs(newSpeed-1)<0.01) ? 'Speed' : newSpeed.toFixed(2).replace(/0+$/,'').replace(/\.$/,'')+'x';
  if(isStickmanSpeedTarget(clip)){
    renderTimelineClip();
  } else {
    renderOverlayLayers();
  }
  renderFrame();
  persistProjectState({skipHistory:true});
}

// Builds the inner fill of a clip body based on its media type: a real repeating
// filmstrip for video, a single stretched thumbnail for images, a canvas-drawn
// waveform for audio, or a flat tint for plain shape overlays.
function clipFillMarkup(ov, widthPx){
  const label = '';
  // Videos have exactly one filmstrip owner: the lazy, time-positioned tile renderer.
  // The empty strip is a neutral placeholder until visible source-time frames arrive.
  // Saved legacy `thumbs` arrays are deliberately ignored; they are not a time map.
  if(ov.type==='video'){
    return `<div class="clip-thumb-strip ev-video-thumbs" data-ev-video-thumb-clip="${escapeHtml(String(ov.id))}" aria-hidden="true"></div>${label}`;
  }
  if(ov.type==='image'){
    const imageSrc=(ov.thumbs&&ov.thumbs.length)?ov.thumbs[0]:ov.src;
    if(imageSrc)return `<div class="clip-thumb-strip" style="background-image:url(&quot;${escapeHtml(String(imageSrc))}&quot;); background-size:auto 100%; background-repeat:repeat-x;"></div>${label}`;
  }
  if(ov.type==='audio'){
    const audioLabel = `<div class="clip-media-label clip-media-label-audio">${escapeHtml(ov.name)}</div>`;
    return `<canvas class="clip-waveform" data-waveform-for="${ov.id}"></canvas>${audioLabel}`;
  }
  if(ov.type==='text') return `<div class="overlay-clip-label">T · ${escapeHtml(ov.name||'Text')}</div>`;
  if(ov.type==='video'||ov.type==='image') return '';
  return `<div class="overlay-clip-label">${escapeHtml(ov.name)}</div>`;
}
// Draws ov.peaks (min/max pairs from generateAudioWaveform) onto its <canvas>, scaled to
// however wide the clip currently is. Re-run after any resize/trim so it never looks stretched.
function drawWaveform(ov){
  const canvas = document.querySelector(`canvas[data-waveform-for="${ov.id}"]`);
  if(!canvas || !ov.peaks || !ov.peaks.length) return;
  const rect=canvas.getBoundingClientRect();
  const w=Math.max(1,Math.round(rect.width));
  const h=Math.max(1,Math.round(rect.height));
  const dpr=Math.max(1,Math.min(2,window.devicePixelRatio||1));
  const pw=Math.round(w*dpr), ph=Math.round(h*dpr);
  // v189: a clip's waveform only changes when its size, peaks or trim change. Selecting a clip
  // rebuilds the timeline rows and used to re-stroke every waveform bar-by-bar each time; now an
  // unchanged waveform is blitted from a cached bitmap.
  const cache=window.__evWaveCache||(window.__evWaveCache=new Map());
  const key=[pw,ph,ov.peaks.length,ov.trimStart,ov.trimEnd,ov.speed,ov.sourceOffset,ov.sourceDuration].join('|');
  const hit=cache.get(String(ov.id));
  canvas.width=pw;
  canvas.height=ph;
  const cctx=canvas.getContext('2d');
  if(hit && hit.key===key && hit.peaks===ov.peaks && hit.bmp){
    cctx.setTransform(1,0,0,1,0,0);
    cctx.drawImage(hit.bmp,0,0);
    return;
  }
  cctx.setTransform(dpr,0,0,dpr,0,0);
  cctx.clearRect(0,0,w,h);
  // Ethvelle indigo waveform: thin, crisp sample strokes (not chunky filled bars).
  cctx.fillStyle='rgba(102, 95, 255, 0.98)';
  cctx.shadowColor='rgba(91, 82, 255, 0.22)';
  cctx.shadowBlur=0.55;
  const mid=h/2, scale=Math.max(1,(h-5)/2);
  const sourceDuration=Math.max(0.01,Number(ov.sourceDuration)||((ov.trimEnd-ov.trimStart)*(Number(ov.speed)||1)));
  const sourceStart=Math.max(0,Number(ov.sourceOffset)||0);
  const sourceSpan=Math.max(0.01,(ov.trimEnd-ov.trimStart)*(Number(ov.speed)||1));
  const startIndex=Math.max(0,Math.min(ov.peaks.length-1,Math.floor(sourceStart/sourceDuration*ov.peaks.length)));
  const endIndex=Math.max(startIndex+1,Math.min(ov.peaks.length,Math.ceil((sourceStart+sourceSpan)/sourceDuration*ov.peaks.length)));
  const visiblePeaks=ov.peaks.slice(startIndex,endIndex);
  const barW=w/Math.max(1,visiblePeaks.length);
  visiblePeaks.forEach((pk,i)=>{
    const min=Math.max(-1,Math.min(0,Number(pk[0])||0));
    const max=Math.max(0,Math.min(1,Number(pk[1])||0));
    const strokeW=Math.min(1.25,Math.max(0.5,barW*0.34));
    const x=i*barW+(barW-strokeW)/2;
    const yTop=mid-max*scale;
    const yBottom=mid-min*scale;
    const bh=Math.max(1.8,yBottom-yTop);
    cctx.fillRect(x,Math.max(1,yTop),strokeW,Math.min(h-2,bh));
  });
  cctx.shadowBlur=0;
  try{
    const bmp=document.createElement('canvas');bmp.width=pw;bmp.height=ph;
    bmp.getContext('2d').drawImage(canvas,0,0);
    cache.set(String(ov.id),{key,peaks:ov.peaks,bmp});
    if(cache.size>60){cache.delete(cache.keys().next().value);}
  }catch(_e){}
}
function redrawAllWaveforms(){
  overlays.filter(o=>o.type==='audio').forEach(drawWaveform);
}
// ---------------- Groups: fixed "Clips" and "Audio" tracks ----------------
// Per redesign: the timeline only ever shows 4 permanent rows total — Background, the
// stickman/Animation track, one shared "Clips" row for every shape/image/video overlay, and
// one "Audio" row for every audio overlay. A brand new row is never created per media item.
// Within a group, clips that don't overlap in time render on the SAME visual line (packed
// left-to-right like a normal single track). Clips only split onto a second/third sub-lane
// INSIDE that one row when their time ranges actually overlap — the row's height grows just
// enough to fit however many sub-lanes are concurrently in use, instead of ever spawning a
// separate top-level row. Packing is fully automatic from each clip's trimStart/trimEnd —
// dragging a clip so it overlaps another splits them onto two lines; dragging them apart in
// time merges them back onto one, with no separate "assign to lane" step needed.
// Generic sub-lane packer shared by the Clips/Text/Audio rows. Takes an array of
// {trimStart, trimEnd} objects and returns the minimum number of non-overlapping horizontal
// lanes, sorted left-to-right within each lane. Character animation clips use the equivalent
// stable-ref packer in characterClipLaneLayout so their main/split elements can be updated live.
function packIntoLanes(clips){
  if(!clips.length) return [];
  const sorted = [...clips].sort((a,b)=> a.trimStart-b.trimStart);
  const subLanes = []; // each entry: array of clips already placed in that sub-lane, in order
  sorted.forEach(ov=>{
    let placed = false;
    for(const lane of subLanes){
      const last = lane[lane.length-1];
      if(ov.trimStart >= last.trimEnd - 0.001){ lane.push(ov); placed = true; break; }
    }
    if(!placed) subLanes.push([ov]);
  });
  return subLanes; // array of arrays — outer index is sub-lane, 0 = top
}
function packOverlayGroup(clips){ return packIntoLanes(clips); }
// Compatibility helper retained for older call sites that expect a laneOf/laneCount pair.
// The active character renderer uses characterClipLaneLayout so lane assignments retain each
// clip's stable main/segment reference while its DOM is updated live during a gesture.
function packClipsIntoLanes(clip, segments){
  const all = getAllStickmanClips(clip, segments); // [{ref, clip}], already time-sorted
  const lanes = packIntoLanes(all.map(a=>a.clip));
  const laneOf = new Map();
  lanes.forEach((lane, laneIdx)=>{
    lane.forEach(c=>{
      const found = all.find(a=>a.clip===c);
      if(found) laneOf.set(found.ref, laneIdx);
    });
  });
  return { laneOf, laneCount: Math.max(1, lanes.length) };
}

          <div class="keyframe-track object-keyframe-track" data-object-kf-track="${ov.id}">${ethvelleObjectKeyframeMarkersHtml(ov.id,ov)}</div>
          ${SEL_EDGE_HTML}
        </div>
      `;
    }).join('')).join('');
    return `
      <div class="${rowClass}" data-row-id="${group.key}" data-group="${group.key}" style="height:${rowH+4}px;">
        <div class="layer-badge" title="${group.label}" ${group.key==='text'?'role="button" tabindex="0" aria-label="Add a new text layer" onclick="'+addOnclick+'"':''}>${group.filledIcon}</div>
        <div class="layer-track" style="height:${rowH}px; min-height:${rowH}px; max-height:${rowH}px;">
          ${clipsHtml}
        </div>
      </div>
    `;
  }).join('');
  // Force a synchronous layout flush right after the innerHTML write (reading a layout
  // property forces the browser to commit final geometry immediately) instead of letting it
  // happen lazily before whatever paint happens to come next — see BUGFIX note above.
  void container.offsetHeight;
  // One container layout flush above is sufficient. Reading offsetHeight once per clip
  // caused N additional synchronous style/layout checks on every row rebuild, which became
  // expensive on projects with many media/text clips on low-memory Android devices.
  wireOverlayInteractions();
  syncPlayheadHeight();
  restoreEthvelleRowScroll(rowScrollSnapshot);
  requestAnimationFrame(redrawAllWaveforms);
}
// Overlay-group sub-lane vertical geometry — matches renderOverlayLayers' compact 30px clip
// / 3px top inset / 4px lane gap exactly, so the live-drag repack below (applyOverlayLanes)
// always agrees with a full rebuild. Named here (not just a local inside renderOverlayLayers)
// so both places share one source of truth for the numbers.
const OVERLAY_SUB_LANE_H = 30;
const OVERLAY_LANE_GAP = 4;
function overlayLaneTop(laneIdx){ return 3 + laneIdx*(OVERLAY_SUB_LANE_H+OVERLAY_LANE_GAP); }
function overlayRowHeightForLanes(laneCount){ return laneCount*OVERLAY_SUB_LANE_H + (laneCount-1)*OVERLAY_LANE_GAP + 4; }
// FIX (dragging one overlay clip onto another only slid it into its own lane AFTER the
// finger lifted, not during the drag — see wireOverlayInteractions' reposition()): re-packs
// and re-applies every clip's `top` (and this row's own height) for the given group LIVE, on
// every pointermove, exactly like applyCharClipLanes already does for character rows. Only
// touches top/height — left/width for the actively-dragged clip is handled separately by its
// own reposition() call right before this runs.
function applyOverlayLanes(groupKey){
  const row = document.querySelector(`.overlay-track-row[data-group="${groupKey}"]`);
  if(!row) return;
  const track = row.querySelector('.layer-track');
  if(!track) return;
  const groupClips = overlays.filter(o=> groupKey==='audio' ? o.type==='audio' : (groupKey==='text' ? o.type==='text' : o.type!=='audio'&&o.type!=='text'));
  const subLanes = packIntoLanes(groupClips);
  const laneCount = Math.max(1, subLanes.length);
  const rowH = overlayRowHeightForLanes(laneCount);
  // Match renderOverlayLayers(): the row uses a compact 4px outer gutter around the track.
  // Keeping these numbers shared prevents the row height from jumping during a live drag.
  setTimelineInlineStyle(row,'height',(rowH+4)+'px');
  setTimelineInlineStyle(row,'minHeight',(rowH+4)+'px');
  setTimelineInlineStyle(row,'maxHeight','none');
  setTimelineInlineStyle(track,'height',rowH+'px');
  setTimelineInlineStyle(track,'minHeight',rowH+'px');
  setTimelineInlineStyle(track,'maxHeight',rowH+'px');
  subLanes.forEach((lane, laneIdx)=>{
    lane.forEach(ov=>{
      const el = track.querySelector(`.overlay-clip[data-overlay-id="${ov.id}"]`);
      if(el)setTimelineInlineStyle(el,'top',overlayLaneTop(laneIdx)+'px');
    });
  });
}
// Selecting the group badge itself (tapping the icon, not a specific clip) just highlights
// the row — actual clip selection/chrome still comes from tapping an individual clip.
function selectOverlayGroup(groupKey){
  document.querySelectorAll('.layer-row').forEach(row=>row.classList.remove('selected'));
  const row = document.querySelector(`.overlay-track-row[data-group="${groupKey}"]`);
  if(row) row.classList.add('selected');
}
function wireOverlayInteractions(){
  overlays.forEach(ov=>{
    const clipEl = document.querySelector(`.overlay-clip[data-overlay-id="${ov.id}"]`);

    if(!clipEl) return;
    const groupKey = ov.type==='audio' ? 'audio' : (ov.type==='text' ? 'text' : 'clips');
    wireClipZoomGesture(clipEl); // FEATURE: two-finger pinch on this clip zooms the timeline

    function reposition(){
      // FIX ("doesn't slide down in real time dragging, only slides after"): this used to
      // only ever move THIS element's own left/width, and left the sub-lane repacking (which
      // is what actually pushes a NEIGHBOR up/down when they start overlapping) to a full
      // renderOverlayLayers() rebuild that only ran once, on pointerup. That's exactly why a
      // neighbor visually "jumped" into place only after the finger lifted instead of
      // sliding live under it. applyOverlayLanes() below re-packs every clip in this group
      // and re-applies each one's `top` (and the row's own height) on every single
      // pointermove — cheap (just style writes, no innerHTML rebuild) so it can't lose
      // pointer capture mid-gesture the way a full DOM rebuild would.
      const pps = pxPerSecond();
      setTimelineInlineStyle(clipEl,'left',(ov.trimStart*pps)+'px');
      setTimelineInlineStyle(clipEl,'width',Math.max(MIN_CLIP_SEC*pps,(ov.trimEnd-ov.trimStart)*pps)+'px');
      // Repack the group now so a neighboring clip changes sub-lane on the same drag frame
      // that creates/removes the overlap. This writes only top/row geometry; it does not
      // rebuild innerHTML or replace the active pointer-capture element.
      if(typeof window.applyOverlayLanes==='function') window.applyOverlayLanes(groupKey);
    }
    function selectRowForGroup(){
      document.querySelectorAll('.layer-row:not(.char-row)').forEach(row=>row.classList.remove('selected'));
      const groupKey = ov.type==='audio' ? 'audio' : (ov.type==='text' ? 'text' : 'clips');
      const row = document.querySelector(`.overlay-track-row[data-group="${groupKey}"]`);
      if(row) row.classList.add('selected');
    }

    // A quick touch-and-slide on an overlay/background clip ALWAYS scrubs the playhead,
    // never moves the clip — matches the stickman clip/segments exactly. Only holding still
    // for LONG_PRESS_MS, before any sliding starts, arms move mode; once armed, every
    // pointermove repositions the clip live in real time, tracking the finger 1:1. Dragging
    // the held clip so it overlaps another clip in the same group (Clips or Audio)
    // automatically packs them onto separate sub-lanes so BOTH stay fully visible (see
    // packOverlayGroup) — no separate drop step needed.
    const OV_DRAG_TOLERANCE = 7;
    const LONG_PRESS_MS = 240;
    let dragState = null; // null | {type:'pending'|'move'|'scroll', ...}
    let longPressTimer = null;
    function clearLongPressTimer(){ if(longPressTimer){ clearTimeout(longPressTimer); longPressTimer=null; } }
    function applyOverlayMoveAt(clientX){
      if(!dragState||dragState.type!=='move')return;
      const pps=pxPerSecond();
      const deltaSec=timelineDragDeltaPixels(dragState,clientX)/pps;
      const len=dragState.origEnd-dragState.origStart;
      const newStart=Math.max(0,dragState.origStart+deltaSec);
      const shift=newStart-dragState.origStart;
      ov.trimStart=newStart;ov.trimEnd=newStart+len;
      // Compositor-only X movement during drag; keyframe timestamps are committed once
      // at release so this path does not walk all keyframes on every touch frame.
      paintTimelineDragX(clipEl,shift*pps);
      // Update sub-lanes on this same animation frame. applyOverlayLanes keeps its lane
      // signature cache, so unchanged lane assignments cost no DOM writes.
      if(typeof window.applyOverlayLanes==='function') window.applyOverlayLanes(groupKey);
    }
    clipEl.addEventListener('pointerdown', (e)=>{
      // Preserve marker selection: a keyframe tap is not a clip-drag gesture.
      if((e.target.closest&&e.target.closest('.kf-marker'))||e.target.hasAttribute('data-ov-handle')) return;
      e.stopPropagation();
      if(isPlaying) pausePlayback();
      /* v231: do not rerun the full selection stack when this clip is already selected. */
      if(String(selectedClip)!==String(ov.id))selectClipForChrome(ov.id);
      selectRowForGroup();
      const timelineScroller=document.getElementById('timelineScroll');
      dragState = { type:'pending', pointerId:e.pointerId, startScrollLeft:timelineScroller?timelineScroller.scrollLeft:0, startX:e.clientX, startY:e.clientY, lastX:e.clientX, startTime: getCurrentAbsoluteTime(), origStart:ov.trimStart, origEnd:ov.trimEnd, origKfTimes:(Array.isArray(ov.ethvelleMediaKeyframes)?ov.ethvelleMediaKeyframes:[]).map(k=>Number(k.t)||0), scrollTopStart:timelineScroller?timelineScroller.scrollTop:0 };
      clipEl.setPointerCapture(e.pointerId);
      clearLongPressTimer();
      longPressTimer=setTimeout(()=>{
        longPressTimer=null;
        if(!dragState||dragState.type!=='pending')return;
        dragState.type='move';clipEl.classList.add('dragging');if(window.ethvelleNotifyClipMoveReady)window.ethvelleNotifyClipMoveReady();armClipDrag();
        startTimelineEdgeAutoScroll(dragState.pointerId,dragState.lastX,()=>{if(dragState&&dragState.type==='move')applyOverlayMoveAt(dragState.lastX);});
        if(Number.isFinite(dragState.lastX))applyOverlayMoveAt(dragState.lastX);
      },LONG_PRESS_MS);
    });
    clipEl.addEventListener('pointermove', (e)=>{
      if(!dragState) return;
      dragState.lastX=e.clientX;
      if(dragState.type==='scroll'){ applyClipVerticalScrollHandoff(dragState, e.clientY); return; }
      const movedPx = Math.abs(e.clientX - dragState.startX);
      if(dragState.type==='pending'){
        // A dominantly-vertical slide before the long-press fires means "scroll the row
        // list", not "interact with this clip" — hand off to the shared scroller, same as
        // sliding on empty timeline space (see applyClipVerticalScrollHandoff above).
        const movedPy = Math.abs(e.clientY - dragState.startY);
        // Treat a clearly vertical (including slightly diagonal) finger gesture as a
        // request to scroll the layer list. The old strict movedPy>movedPx test rejected
        // ordinary thumb movement on narrow audio lanes, making vertical scrolling seem
        // intermittent depending on a few pixels of horizontal drift.
        if(movedPy >= OV_DRAG_TOLERANCE && movedPy >= Math.max(1,movedPx*0.82)){
          clearLongPressTimer();
          dragState.type = 'scroll';
          applyClipVerticalScrollHandoff(dragState, e.clientY);
          return;
        }
        // Quick horizontal movement scrubs; clip movement is armed only by a stationary
        // long press. After arming, translation follows the finger in real time.
        if(movedPx > 6 && movedPx >= Math.max(1, movedPy*0.72)){
          clearLongPressTimer();dragState.type='scrub';
          scheduleTimelineScrub(timeFromDragAnchor(dragState.startX,dragState.startTime,e.clientX));
        }
        return;
      }
      if(dragState.type==='scrub'){
        scheduleTimelineScrub(timeFromDragAnchor(dragState.startX,dragState.startTime,e.clientX));
        return;
      }
      updateTimelineEdgeAutoScroll(e.pointerId,e.clientX,()=>{if(dragState&&dragState.type==='move')applyOverlayMoveAt(dragState.lastX);});
      scheduleTimelineDragFrame(clipEl,e,point=>applyOverlayMoveAt(point.clientX));
      return;
    });
    clipEl.addEventListener('pointerup', (e)=>{
      if(!dragState) return;
      stopTimelineEdgeAutoScroll(e.pointerId);
      clearLongPressTimer();
      const wasMove = dragState.type==='move';
      const wasScroll = dragState.type==='scroll';
      const wasScrub = dragState.type==='scrub';
      const anchorX=dragState.startX,anchorTime=dragState.startTime;
      if(wasMove)flushTimelineDragFrame(clipEl,e,point=>applyOverlayMoveAt(point.clientX));else cancelTimelineDragFrame(clipEl);
      if(wasMove){const movedBy=ov.trimStart-dragState.origStart;shiftTimelineKeyframesOnce(ov.ethvelleMediaKeyframes,dragState.origKfTimes,movedBy);if(window.ethvelleNotifyClipMoved)window.ethvelleNotifyClipMoved(movedBy);clearTimelineDragX(clipEl);}
      dragState=null;clipEl.releasePointerCapture(e.pointerId);
      clipEl.classList.remove('dragging');
      if(wasMove) disarmClipDrag();
      if(wasMove){
        ensureTimelineRoomForDraggedClip(ov);
        if(ov.trimEnd > timelineDuration - 0.3){ timelineDuration = ov.trimEnd+1; syncTimelineInnerWidth(); renderRuler(); }
        renderOverlayLayers();
        selectClipForChrome(ov.id);
      } else if(wasScrub&&!wasScroll){
        const t = timeFromDragAnchor(anchorX, anchorTime, e.clientX);
        flushTimelineScrub(t);
      }
      persistProjectState({skipHistory:true});
    });
    clipEl.addEventListener('pointercancel', ()=>{
      const wasMove=dragState&&dragState.type==='move';
      if(dragState)stopTimelineEdgeAutoScroll(dragState.pointerId);
      clearLongPressTimer();cancelTimelineDragFrame(clipEl);
      if(wasMove){const movedBy=ov.trimStart-dragState.origStart;shiftTimelineKeyframesOnce(ov.ethvelleMediaKeyframes,dragState.origKfTimes,movedBy);clearTimelineDragX(clipEl);}
      dragState=null;
      clipEl.classList.remove('dragging');
      if(wasMove){
        disarmClipDrag();
        renderOverlayLayers();
        selectClipForChrome(ov.id);
        renderFrame();
      }
    });


  });
}

/* ==================================================================================
   CENTRALIZED TIMELINE POINTER STATE MACHINE (Scrub the timeline by dragging tracks
   under the fixed playhead, CapCut-style)
   ==================================================================================
   Previously, whether a touch anywhere in the timeline region paused/seeked correctly
   depended on WHICH element it landed on: clip bodies had their own onpointerdown that
   stopPropagation()'d past this listener, empty space only ever deselected (never
   scrubbed), and the scroll event doubled as a seek source with its own separate
   suppression timing. That's the "different parts of the timeline behave differently"
   bug. FIX: one state machine, on the scroller itself, using capture-phase listeners so
   it always sees every pointerdown in the region FIRST — even ones that land on a child
   clip/keyframe/handle — before that child's own handler runs and (possibly) stops
   propagation. Every gesture is classified into exactly one of TAP / SCRUB / CHILD:
     - CHILD: pointerdown target is a clip/handle/keyframe -> this state machine gets out
       of the way entirely and lets that element's own drag logic run (still paused first,
       see below), matching PART 32: only stickman clips use bone/transform keyframes, but
       ALL clip types (background/overlay/segments) get correct pause+select behavior.
     - TAP: pointerdown on empty timeline space, pointerup without exceeding the move
       threshold -> seek exactly to the tapped time (PART 20: never deselects, resets, or
       leaves playback running).
     - SCRUB: pointerdown on empty timeline space, then moved past the threshold -> live
       drag-seek every pointermove, matching finger position exactly (PART 25: canvas
       redraws immediately on every move, doesn't wait for a render tick).
   ==================================================================================== */
// Converts a pointer's clientX into an absolute timeline second, using the timeline
// scroller's OWN geometry (its scrollLeft + the pointer's offset from the fixed center
// line) — never estimated from DOM child positions (PART 7). This is the exact inverse of
// movePlayhead()'s own math, so "time under the finger" always equals "playhead time".
// Standalone top-level helper (not just an initTimelineScrub closure) so any future
// pointer-driven seek path can reuse the exact same math instead of re-deriving it.
function getAbsoluteTimeFromPointer(clientX){
  const scroller = document.getElementById('timelineScroll');
  if(!scroller) return currentAbsoluteTimeSec;
  const rect = scroller.getBoundingClientRect();
  const pps = pxPerSecond();
  const deltaPx = clientX - (rect.left + rect.width/2);
  const targetPx = scroller.scrollLeft + deltaPx;
  // Scrub range spans the FULL project — every character's own clips AND however far the
  // Background/Clips/Audio rows extend (see totalProjectDuration()) — so you can always
  // reach the end of a longer non-active character's animation, or of a background/media/
  // audio clip, even before any animation has been added at all.
  const cachedScrubDuration = window.__ethvelleTimelineScrubbing ? Number(window.__ethvelleTimelineScrubMaxDuration) : NaN;
  const dur = Number.isFinite(cachedScrubDuration) && cachedScrubDuration>0 ? cachedScrubDuration : totalProjectDuration();
  return Math.max(0, Math.min(dur, targetPx/pps));
}
// BUGFIX (scrub over-speed / wrong-direction / snap-back): getAbsoluteTimeFromPointer() is
// an ABSOLUTE screen-position-to-time conversion — correct for a single tap (content hasn't
// moved, so "distance from the center playhead" directly tells you the tapped time). It is
// WRONG for a live drag: every pointermove re-reads scroller.scrollLeft, which the very last
// movePlayhead() call (triggered by the previous pointermove) just changed to match the
// previous move's result. Re-adding "current distance from the fixed center line" on top of
// that on every single event — instead of the finger's actual movement since the drag
// started — turns each move into a compounding jump instead of a 1:1 follow, which is
// exactly the "slides way faster than my finger" / "jumps then snaps back" behavior. FIX:
// every live-drag call site now anchors to the exact time+pointer-position recorded once at
// gesture start (dragState.startX/startTime, gesture.startX/startTime, etc.) and computes
// time purely from how far the finger has moved since THEN — a direct, stable 1:1 mapping,
// matching natural touch-scroll direction (drag left -> time moves forward/later, drag right
// -> time moves backward/earlier, exactly like content sliding under a fixed playhead).
function timeFromDragAnchor(anchorX, anchorTimeSec, clientX){
  const pps = pxPerSecond();
  // BUGFIX: was combinedStickmanDuration() — same 2s-hard-cap issue as setCurrentAbsoluteTime
  // above (see totalProjectDuration()'s comment) whenever no animation was selected/added yet.
  const cachedScrubDuration = window.__ethvelleTimelineScrubbing ? Number(window.__ethvelleTimelineScrubMaxDuration) : NaN;
  const dur = Number.isFinite(cachedScrubDuration) && cachedScrubDuration>0 ? cachedScrubDuration : totalProjectDuration();
  const deltaPx = clientX - anchorX;
  return Math.max(0, Math.min(dur, anchorTimeSec - deltaPx/pps));
}

// Smooth timeline scrubbing: pointer events can arrive faster than Android WebView can
// redraw the canvas and every overlay. Keep the newest requested time and render at most
// once per display frame. Pointer-up flushes the exact final position synchronously.
let timelineScrubRenderRaf = 0;
let timelineScrubPendingTime = null;
// The old 2.5-4x idle backing surface could exceed 18 million pixels on a phone.
// That made every character-joint repaint expensive even though drawStick was RAF-limited.
// Keep the live preview crisp at up to 2.2x density and use 1.5x while scrubbing; export
// uses its own full-resolution renderer and is not affected by this display-only budget.
function setTimelineScrubCanvasQuality(scrubbing){
  const canvas=document.getElementById('stickCanvas');
  if(!canvas || !(stageW>0) || !(stageH>0)) return;
  const dpr=Math.max(1,Number(window.devicePixelRatio)||1);
  const zoomPreview=!!document.querySelector('#screen-editor .canvas-wrap.evca-inline-zoom');
  const requested=scrubbing?Math.min(1.5,dpr):(zoomPreview?Math.min(1.8,dpr):Math.min(2.2,Math.max(1.75,dpr)));
  const scale=Math.max(1,Math.min(requested,Math.sqrt(12000000/Math.max(1,stageW*stageH))));
  const width=Math.max(1,Math.round(stageW*scale));
  const height=Math.max(1,Math.round(stageH*scale));
  if(canvas.width===width && canvas.height===height) return;
  canvas.width=width;canvas.height=height;
  ctx=canvas.getContext('2d',{alpha:true});
  ctx.setTransform(canvas.width/Math.max(1,stageW),0,0,canvas.height/Math.max(1,stageH),0,0);
  ctx.imageSmoothingEnabled=true;
  try{ctx.imageSmoothingQuality='high';}catch(_e){}
  ctx.lineCap='round';ctx.lineJoin='round';ctx.miterLimit=2;
}
let timelineScrubLastPreviewAt = 0;
function scheduleTimelineScrub(tSec){
  const nextTime=Number(tSec);
  if(!Number.isFinite(nextTime))return;
  if(!window.__ethvelleTimelineScrubbing){
    window.__ethvelleTimelineScrubMaxDuration=totalProjectDuration();
    setTimelineScrubCanvasQuality(true);
    timelineScrubLastPreviewAt=0;
  }
  window.__ethvelleTimelineScrubbing=true;
  timelineScrubPendingTime=nextTime;
  // One write per display frame. Writing scrollLeft for every high-frequency pointer event
  // forced Android to repeatedly update the timeline's scroll/paint tree between frames.
  if(timelineScrubRenderRaf)return;
  timelineScrubRenderRaf=requestAnimationFrame(()=>{
    timelineScrubRenderRaf=0;
    const latestTime=timelineScrubPendingTime;
    timelineScrubPendingTime=null;
    if(latestTime===null||!Number.isFinite(latestTime))return;
    const scrubScroller=document.getElementById('timelineScroll');
    if(scrubScroller){
      const priorOwner=timelineScrollOwner;
      timelineScrollOwner='programmatic';
      const wantedScroll=Math.max(0,latestTime*pxPerSecond());
      if(Math.abs(scrubScroller.scrollLeft-wantedScroll)>.35)scrubScroller.scrollLeft=wantedScroll;
      timelineScrollOwner=priorOwner;
    }
    setCurrentAbsoluteTime(latestTime);
    const now=performance.now();
    const hasActiveVideo=overlays.some(o=>o&&o.type==='video'&&latestTime>=Number(o.trimStart||0)&&latestTime<Number(o.trimEnd||0));
    // The playhead and scroll are updated every display frame; the heavier canvas preview below
    // is budgeted separately so it cannot block the finger-driven timeline movement.
    // The playhead/time/scroll continue to follow the finger on every display frame.
    // Only the expensive stage composition is budgeted separately: ~33fps for still/rig
    // content and ~20fps while a video decoder is active on compact phones. This keeps the
    // timeline interaction responsive without forcing a full canvas+overlay redraw at 60Hz.
    const compactAndroid=/(Android)/i.test(navigator.userAgent||'')&&Math.min(window.innerWidth||0,window.innerHeight||0)<=480;
    const minInterval=hasActiveVideo?(compactAndroid?54:48):(compactAndroid?30:24);
    if(timelineScrubLastPreviewAt&&now-timelineScrubLastPreviewAt<minInterval){
      timelineScrubPendingTime=latestTime;
      return;
    }
    timelineScrubLastPreviewAt=now;
    renderTimelineScrubPreview();
  });
}
function cancelScheduledTimelineScrub(){
  if(timelineScrubRenderRaf) cancelAnimationFrame(timelineScrubRenderRaf);
  timelineScrubRenderRaf = 0;
  timelineScrubPendingTime = null;
  window.__ethvelleTimelineScrubbing = false;
  delete window.__ethvelleTimelineScrubMaxDuration;
  setTimelineScrubCanvasQuality(false);
}
function setTimelineInlineStyle(el, property, value){
  if(el && el.style[property] !== value) el.style[property] = value;
}
// During clip-body movement update only the active clip's compositor translation, once per
// animation frame. The timeline data and layout position are left untouched until release,
// avoiding repeated track reflow and leaving markers/thumbnails/canvas out of the drag loop.
function paintTimelineDragX(el, deltaPx){
  if(!el)return;
  const rawX=Number.isFinite(Number(deltaPx))?Number(deltaPx):0;
  const dpr=Math.max(1,Number(window.devicePixelRatio)||1);
  const x=Math.round(rawX*dpr)/dpr; // physical-pixel alignment keeps clip text/edges crisp
  if(!el.dataset.evV87DragBaseLeft){
    const base=parseFloat(el.style.left);
    el.dataset.evV87DragBaseLeft=String(Number.isFinite(base)?base:0);
  }
  // Keep the timeline's layout position untouched while dragging. A translate3d on
  // just the active clip is composited by the GPU and avoids relaying out the track
  // on every pointer frame (especially important on Android Chrome/WebView).
  el.classList.add('evv-timeline-transform-dragging');
  const value=x.toFixed(3)+'px';
  if(el.style.getPropertyValue('--evv-timeline-drag-x')!==value)el.style.setProperty('--evv-timeline-drag-x',value);
}
function clearTimelineDragX(el){
  if(!el)return;
  delete el.dataset.evV87DragBaseLeft;
  el.classList.remove('evv-timeline-transform-dragging');
  el.style.removeProperty('--evv-timeline-drag-x');
}
function shiftTimelineKeyframesOnce(list,originalTimes,deltaSec){
  if(!Array.isArray(list)||!Array.isArray(originalTimes))return;
  const shift=Number(deltaSec)||0;
  list.forEach((kf,i)=>{const t=Number(originalTimes[i]);if(kf&&Number.isFinite(t))kf.t=t+shift;});
}
// Pointermove can arrive faster than the display refreshes. Keep only the latest coordinates
// and do one timeline/layout update per animation frame; flush the final pointerup position.
const timelineDragFrameJobs = new WeakMap();
function scheduleTimelineDragFrame(owner, event, callback){
  if(!owner || !event || typeof callback!=='function') return;
  let job=timelineDragFrameJobs.get(owner);
  if(!job){job={raf:0,x:0,y:0,callback:null};timelineDragFrameJobs.set(owner,job);}
  job.x=event.clientX;job.y=event.clientY;job.callback=callback;
  if(job.raf)return;
  job.raf=requestAnimationFrame(()=>{
    job.raf=0;
    const cb=job.callback;job.callback=null;
    if(cb)cb({clientX:job.x,clientY:job.y});
  });
}
function flushTimelineDragFrame(owner, event, callback){
  const job=owner?timelineDragFrameJobs.get(owner):null;
  if(job&&job.raf)cancelAnimationFrame(job.raf);
  if(job){job.raf=0;job.callback=null;}
  if(typeof callback==='function')callback({clientX:event.clientX,clientY:event.clientY});
}
function cancelTimelineDragFrame(owner){
  const job=owner?timelineDragFrameJobs.get(owner):null;
  if(!job)return;
  if(job.raf)cancelAnimationFrame(job.raf);
  job.raf=0;job.callback=null;
}

function flushTimelineScrub(tSec, render = true){
  cancelScheduledTimelineScrub();
  setCurrentAbsoluteTime(tSec);
  // A scrub is previewed with throttled decoder seeks. Give the release render a unique
  // generation so every active video can perform one exact seek to the FINAL finger position,
  // even if its previous scrub seek completed less than one second ago.
  const nextGeneration=(Number(window.__ethvelleTimelineScrubReleaseSeq)||0)+1;
  window.__ethvelleTimelineScrubReleaseSeq=nextGeneration;
  window.__ethvelleTimelineScrubReleaseId=String(nextGeneration);
  window.__ethvelleTimelineScrubReleaseAt=performance.now();
  if(render) renderFrame();
}

// FEATURE ("a normal click+slide up/down on any clip should scroll the whole row list, same
// as sliding on empty timeline space — only a long-press-then-drag should move just that one
// clip"): every per-clip drag handler below (wireSplitSegmentInteractions/
// wireBackgroundTrackInteractions/wireOverlayInteractions/wireClipInteractions) starts a
// gesture in a 'pending' state and only commits to moving/trimming that ONE clip once its own
// long-press timer fires. Before that timer fires, a dominantly-vertical slide now flips
// dragState.type to 'scroll' (see each handler's pointermove) and this helper takes over —
// it drives the shared #timelineScroll container's scrollTop by hand, anchored to where the
// gesture started, the exact same anchor-relative approach timeFromDragAnchor already uses
// for horizontal scrubbing above.
// WHY NOT JUST LET THE BROWSER SCROLL IT NATIVELY (like empty timeline space already does via
// touch-action:pan-y): every clip element (.anim-clip/.overlay-clip/.track-clip and their
// body/handle children) sets touch-action:none, which is required so a long-press-then-drag
// starting anywhere over a clip is never hijacked by the browser's own touch handling (see the
// .track-clip touch-action BUGFIX comment above). touch-action is decided once, at the very
// start of the touch — so a touch that begins on a clip can never fall back to native
// scrolling later in the same gesture, unlike one that begins on empty track space. Driving
// scrollTop directly here is what makes the two cases feel identical to the user.
let clipVerticalScrollRaf=0,clipVerticalScrollPending=null;
function applyClipVerticalScrollHandoff(dragState, clientY){
  cancelScheduledTimelineScrub();
  const scroller=document.getElementById('timelineScroll');
  if(!scroller)return;
  const nowMs=(window.performance&&performance.now)?performance.now():Date.now();
  let motion=window.__ethvelleV110ClipScrollMotion;
  if(!motion||motion.scroller!==scroller||!motion.active){
    motion={scroller,active:true,lastY:Number(dragState.startY)||clientY,lastAt:nowMs-16,velocity:0,targetTop:null};
    window.__ethvelleV110ClipScrollMotion=motion;
  }
  const dt=Math.max(1,nowMs-motion.lastAt);
  const dy=clientY-motion.lastY;
  const sample=Math.max(-2.8,Math.min(2.8,-dy/dt));
  motion.velocity=Math.max(-2.8,Math.min(2.8,motion.velocity*0.38+sample*0.62));
  motion.lastY=clientY;motion.lastAt=nowMs;
  const targetTop=Math.max(0,dragState.scrollTopStart+(dragState.startY-clientY));
  motion.targetTop=targetTop;
  clipVerticalScrollPending={scroller,top:targetTop};
  if(clipVerticalScrollRaf)return;
  clipVerticalScrollRaf=requestAnimationFrame(()=>{
    clipVerticalScrollRaf=0;
    const pending=clipVerticalScrollPending;clipVerticalScrollPending=null;
    if(pending&&pending.scroller?.isConnected){
      const top=Math.max(0,pending.top);
      if(Math.abs(pending.scroller.scrollTop-top)>0.5)pending.scroller.scrollTop=top;
    }
  });
}

function initTimelineScrub(){
  const scroller = document.getElementById('timelineScroll');
  if(!scroller || scroller.dataset.wired) return;
  scroller.dataset.wired = '1';
  scroller.style.touchAction = 'pan-y'; // let vertical layer-list scroll pass through untouched, we own horizontal
  const TAP_MOVE_TOLERANCE = 6; // px — matches the tolerance used for canvas/stage taps
  const absTimeFromPointerX = getAbsoluteTimeFromPointer; // local alias, same function

  let gesture = null; // null | {type:'pending'|'scrub', startX, startY, pointerId}
  // v234 CapCut-style momentum: after a finger-scrub is released with speed, the timeline keeps
  // gliding under the fixed playhead and decays to rest. Runs through the same
  // scheduleTimelineScrub() pipeline as a live drag, so no extra rendering path exists and the
  // clip-row frames (painted from the stored source frames) never reload during the glide.
  let flingRaf = 0, flingT = 0, flingSamples = [];
  function stopFling(){
    if(!flingRaf) return;
    cancelAnimationFrame(flingRaf); flingRaf = 0;
    flushTimelineScrub(flingT);
    timelineScrollOwner = 'programmatic';
  }
  function startFling(velocityPxPerSec){
    const pps = pxPerSecond();
    if(!(pps>0)) return;
    const dur = totalProjectDuration();
    let vel = Math.max(-7000, Math.min(7000, velocityPxPerSec));
    flingT = currentAbsoluteTimeSec;
    let last = performance.now();
    const step = (now)=>{
      const dt = Math.min(0.05, Math.max(0.001, (now-last)/1000)); last = now;
      if(isPlaying){ flingRaf = 0; flushTimelineScrub(flingT); timelineScrollOwner='programmatic'; return; }
      flingT = Math.max(0, Math.min(dur, flingT - vel*dt/pps));
      vel *= Math.exp(-dt/0.32);
      const atEdge = (flingT<=0 && vel>0) || (flingT>=dur && vel<0);
      if(Math.abs(vel)<24 || atEdge){
        flingRaf = 0;
        flushTimelineScrub(flingT);
        timelineScrollOwner = 'programmatic';
        persistProjectState({skipHistory:true});
        return;
      }
      scheduleTimelineScrub(flingT);
      flingRaf = requestAnimationFrame(step);
    };
    flingRaf = requestAnimationFrame(step);
  }

  scroller.addEventListener('pointerdown', (e)=>{
    // CHILD: a clip body, resize handle, keyframe diamond, or layer badge owns this
    // gesture instead. Those elements each call selectClipForChrome()/selectKeyframe()/
    // selectLayer() from their own pointerdown/click, which already calls pausePlayback()
    // centrally — so we simply step aside rather than also trying to interpret the tap
    // ourselves (PART 6: "do not allow the generic timeline handler to fight the child
    // interaction").
    //
    // BUGFIX (timeline drag "sometimes does nothing" while paused): this used to be an
    // ALLOW-list — only e.target===scroller, ===timelineInner, or classList.contains
    // ('layer-track') counted as scrub territory, and anything else (including a touch
    // landing on .layer-row's own padding/gap area, the .timeline-ruler, or any element
    // not explicitly enumerated) was misclassified as CHILD and the gesture was nulled
    // out — i.e. silently did nothing. That's exactly "sometimes does nothing" depending
    // on exactly which pixel the finger lands on. Fixed to a DENY-list instead: only the
    // actual known interactive children (clip body/handle, keyframe marker, layer badge,
    // track-clip) opt out via closest(); every other point anywhere in the timeline
    // region (row padding, ruler, empty track background) now correctly starts a scrub.
    // BUGFIX (dragging a background clip pans/scrubs the whole timeline instead of moving
    // just that clip): '.track-clip' used to be deliberately left OUT of this deny-list
    // (see old comment, preserved in git history) back when it was only the static
    // #bgTrackClip placeholder with no drag of its own. Since wireBackgroundTrackInteractions
    // gave background clips their own real long-press-to-move + edge-handle-to-trim gesture
    // (mirroring .overlay-clip, which IS in this deny-list), leaving '.track-clip' out meant
    // BOTH this outer scrub handler and the clip's own handler were tracking the same
    // pointerdown. The clip's own handler already calls e.stopPropagation() on pointerdown so
    // this listener's pointerDOWN branch never saw it directly — but pointerdown fires in the
    // CAPTURE phase (see the `true` below), which runs BEFORE that stopPropagation happens, so
    // `gesture` still got armed here first. Once the finger moved past TAP_MOVE_TOLERANCE
    // (6px) — almost immediately on a real drag, often before the clip's own 380ms long-press
    // even fired — this handler called scroller.setPointerCapture(), which STEALS pointer
    // capture away from the clip (fires lostpointercapture on it) and redirects every further
    // pointermove to the scroller instead. From that point the gesture becomes a full
    // timeline scrub/pan — which, since every track sits in this one shared scroller, LOOKS
    // like every clip in the project sliding, and the background clip itself stops
    // responding after just its first pixel or two of movement. Adding '.track-clip' back to
    // the deny-list (its trim handles are descendants of it, so closest() catches those too)
    // makes this listener step aside for background clips exactly like it already does for
    // overlay clips, leaving wireBackgroundTrackInteractions as the sole owner of that drag.
    stopFling(); flingSamples = [];
    const child = e.target.closest('.anim-clip, .anim-clip-body, .anim-clip-handle, .kf-marker, .layer-badge, .overlay-clip, .track-clip, [data-ov-handle], [data-seg-handle], [data-seg-body], [data-bg-handle]');
    if(child){
      const animSurface=child.closest('.anim-clip');
      const innerControl=e.target.closest('.anim-clip-body, .anim-clip-handle, .kf-marker, .layer-badge, .overlay-clip, .track-clip, [data-ov-handle], [data-seg-handle], [data-seg-body], [data-bg-handle]');
      if(animSurface && !innerControl){
        // A tap on the bare clip surface still selects the animation clip, while a left/right
        // swipe there now moves the playhead exactly like a swipe on empty timeline space.
        if(isPlaying)pausePlayback();
        gesture={type:'pending',startX:e.clientX,startY:e.clientY,pointerId:e.pointerId,startTime:getCurrentAbsoluteTime()};
        try{scroller.setPointerCapture(e.pointerId);}catch(_e){}
        return;
      }
      gesture=null;return;
    }
    // If playback is running, freeze it immediately at the exact instant of this touch —
    // BEFORE any drag-threshold logic, so play/pause never races a scrub gesture that's
    // still deciding whether it's a tap or a drag.
    if(isPlaying) pausePlayback();
    gesture = { type:'pending', startX:e.clientX, startY:e.clientY, pointerId:e.pointerId, startTime: getCurrentAbsoluteTime() };
  }, true); // capture phase: see this before any child stopPropagation()

  scroller.addEventListener('pointermove', (e)=>{
    if(!gesture || e.pointerId !== gesture.pointerId) return;
    const dx = e.clientX-gesture.startX, dy = e.clientY-gesture.startY;
    const movedPx = Math.hypot(dx, dy);
    if(gesture.type==='pending'){
      if(movedPx <= TAP_MOVE_TOLERANCE) return; // still could be a tap — do nothing yet
      // BUGFIX (couldn't vertically scroll the row list at all unless a clip happened to be
      // under the finger): this used to treat ANY movement past the tolerance — including a
      // drag that was almost entirely vertical — as the start of a horizontal timeline scrub,
      // then immediately called setPointerCapture(), which hands the whole gesture to this
      // listener and stops the browser's own native pan-y vertical scroll from ever taking
      // over for that touch. On a project with more character rows than fit on screen (or
      // just the empty Background/Add-character rows before anything's added), any attempt
      // to slide the row list up/down that moved more than 6px got silently swallowed the
      // same way — it just never scrolled, which is exactly the "stuck" feeling reported.
      // Touches landing directly on a clip/handle/badge were never affected (see the
      // deny-list above, which steps aside for those before gesture tracking even starts) —
      // that's why selecting a clip first, then dragging over ITS row, "worked": the finger
      // was on territory this listener never claimed in the first place.
      // FIX: classify by which axis actually dominates once the tolerance is crossed. A
      // mostly-vertical drag releases the gesture entirely (no pointer capture taken, no
      // preventDefault fired) so the browser's native pan-y scroll handles it exactly like
      // any other scrollable list. Only a drag that's actually more horizontal than vertical
      // becomes a scrub. This check happens once, at the moment the gesture leaves 'pending'
      // — a real horizontal scrub that later drifts a little vertically (a finger is never
      // perfectly straight) is unaffected, since once type is 'scrub' this branch no longer runs.
      if(Math.abs(dy) > Math.abs(dx)){
        gesture = null;
        return;
      }
      gesture.type = 'scrub';
      timelineScrollOwner = 'user';
      try{ scroller.setPointerCapture(e.pointerId); }catch(err){}
    }
    if(gesture.type==='scrub'){
      // Compute time from the finger's movement since this gesture began (anchor-relative),
      // then coalesce expensive canvas/overlay redraws to one latest-value update per frame.
      const t = timeFromDragAnchor(gesture.startX, gesture.startTime, e.clientX);
      scheduleTimelineScrub(t);
      flingSamples.push({x:e.clientX, ts:e.timeStamp});
      while(flingSamples.length>2 && e.timeStamp-flingSamples[0].ts>100) flingSamples.shift();
    }
  });

  function endGesture(e){
    if(!gesture || (e && e.pointerId!==undefined && e.pointerId!==gesture.pointerId)) return;
    const wasScrub = gesture.type==='scrub';
    if(wasScrub){
      const finalX = (e && Number.isFinite(e.clientX)) ? e.clientX : gesture.startX;
      flushTimelineScrub(timeFromDragAnchor(gesture.startX, gesture.startTime, finalX));
      timelineScrollOwner = 'programmatic';
      try{ scroller.releasePointerCapture(gesture.pointerId); }catch(err){}
      persistProjectState({skipHistory:true});
      try{
        const s = flingSamples; flingSamples = [];
        if(e && s.length>=2){
          const a=s[0], z=s[s.length-1], span=z.ts-a.ts;
          if(span>8 && e.timeStamp-z.ts<70){
            const v=(z.x-a.x)/(span/1000);
            if(Math.abs(v)>160) startFling(v);
          }
        }
      }catch(_e){}
    } else if(e && e.clientX!==undefined){
      // TAP: seek exactly to the tapped absolute time. Deliberately does NOT deselect —
      // a tap on empty timeline space must not reset selection, frame, or clip choice
      // (PART 20) — selection is left exactly as it was; only the playhead moves. Movement
      // never crossed the drag threshold here, so anchor-relative and absolute agree; using
      // the same anchor-relative helper keeps every seek path on one consistent formula.
      const t = timeFromDragAnchor(gesture.startX, gesture.startTime, e.clientX);
      flushTimelineScrub(t);
      // v234: CapCut drops the selection frame instantly when empty timeline space is tapped.
      try{
        if(selectedClip!=null && e.target && e.target.closest && !e.target.closest('.overlay-clip, .track-clip, .anim-clip, .layer-badge, .kf-marker')) deselectAllClips();
      }catch(_e){}
    }
    gesture = null;
  }
  scroller.addEventListener('pointerup', endGesture);
  scroller.addEventListener('pointercancel', ()=>{
    cancelScheduledTimelineScrub();
    timelineScrollOwner = 'programmatic';
    gesture = null;
  });

  // The native 'scroll' event now exists purely as a passive mirror for browsers that
  // rubber-band/inertia-scroll the container on its own — it must NEVER independently
  // seek time. Ownership is checked directly rather than inferred from timing windows:
  // only an active user drag (tracked above) sets timelineScrollOwner='user'; every
  // programmatic scroll (movePlayhead, seekPlayheadToClipStart, etc.) sets it back to
  // 'programmatic' around its own assignment. This is what makes PART 24's requirement
  // exact: programmatic playhead movement can never recursively modify current time.
  scroller.addEventListener('scroll', ()=>{
    if(timelineScrollOwner !== 'user') return;
    const t = absTimeFromPointerX(scroller.getBoundingClientRect().left + scroller.clientWidth/2);
    // movePlayhead() can emit a scroll event during a scrub; skip duplicate frame redraws.
    if(Math.abs(t-currentAbsoluteTimeSec) < 0.5/Math.max(1,Number(FPS)||24)) return;
    scheduleTimelineScrub(t);
  }, {passive:true});
}

// ---------------- Free drag-to-reposition for stickman-track clips (CapCut-style) ----------------
// A quick/ordinary touch on a clip body NEVER moves it — it only scrubs the playhead (see
// wireClipInteractions below), exactly like touching empty timeline space. Only holding the
// clip down for LONG_PRESS_SWAP_MS, before any sliding starts, arms "move mode": from that
// instant every pointermove repositions the clip in real time, tracking the finger 1:1.
const LONG_PRESS_SWAP_MS = 240;
// REPLACED the old findSwapTarget/insertStickmanClipNextTo mechanic: that forcibly snapped a
// held clip flush against whichever neighbor its center crossed, so it was never actually
// possible to just drop a clip wherever you wanted. A held clip now goes exactly where the
// finger takes it (only floor-clamped at 0, same as before); if that means it now overlaps a
// neighbor, that neighbor is pushed forward in time by exactly the overlap amount instead
// (see resolveCharClipOverlaps/applyCharClipLanes) — the row always stays exactly one lane
// tall, and two of this character's own clips can never actually end up time-overlapping.

// Non-active characters' clips can't change from a gesture on the ACTIVE character's own
// clip (each character's clip data is independent), so this only needs to run after
// something that could change which characters HAVE clips at all or how many — it's kept
// separate from the active clip's own repositionClipEl() so a trim/drag on the active
// clip never has to rebuild every other row's DOM mid-gesture (see handle onpointerup
// bugfix above).
function refreshOtherCharacterClipPreviews(){
  characters.forEach((c, idx)=>{
    if(idx === activeCharIdx) return;
    const track = document.getElementById('track-char-'+idx);
    if(!track) return;
    if(!c.clip){ track.innerHTML = `<span class="char-row-name">${c.name}</span>`; return; }
    // FIX (rows only extended all together): re-apply this row's OWN lane-based height too,
    // not just its clip markup — otherwise a non-active row that just gained an overlap
    // (e.g. another split segment dragged on top of one of its clips) would show two
    // stacked clips inside a row that never actually grew to fit them.
    const readOnly = readOnlyCharClipMarkup(c, idx);
    track.innerHTML = readOnly.html;
    const row = track.closest('.char-row');
    setCharacterLaneGeometry(row,track,readOnly.laneCount||1);
  });
}
function wireClipInteractions(){
  const clipEl = document.getElementById('animClipEl');
  const body = document.getElementById('clipBody');
  const leftHandle = document.getElementById('clipHandleLeft');
  const rightHandle = document.getElementById('clipHandleRight');
  if(!clipEl) return;
  wireClipZoomGesture(body); // FEATURE: two-finger pinch on this clip zooms the timeline
  clipEl.addEventListener('pointerdown', (e)=>{
    // Body and handles already have dedicated gesture handlers. Capture only clicks on the
    // outer clip surface/selection bars so the very first tap selects the clip itself.
    if(e.target !== clipEl && !e.target.closest('.sel-edge-top, .sel-edge-bottom, .sel-edge-left, .sel-edge-right')) return;
    e.stopPropagation();
    if(isPlaying) pausePlayback();
    selectClipForChrome('main');
  });

  // Apply current trimStart/trimEnd and vertical lane placement directly to existing DOM
  // nodes (no rebuild), so drag/resize stays smooth and never loses pointer capture mid-gesture.
  // When a character clip overlaps another, applyCharClipLanes moves the colliding clip to a
  // separate vertical sub-lane immediately while preserving both clips' actual timeline times.
  function repositionClipEl(){
    // Resize is reconciled on every drag frame: neighbouring clips move sideways in real
    // time while the selected clip keeps its current trim geometry.
    applyCharClipLanes('main');
  }
  function repositionDraggedMainClip(){
    paintTimelineDragX(clipEl,(activeClip.trimStart-dragState.origStart)*pxPerSecond());
  }

  // Drag the whole clip to reposition it on the timeline. A quick touch-and-slide on the
  // clip body ALWAYS scrubs the playhead, exactly like touching empty timeline space — it
  // never moves the clip, no matter how far the finger travels. Only a LONG_PRESS_SWAP_MS
  // hold-still, done BEFORE any sliding starts, arms "move mode"; once armed, every
  // subsequent pointermove repositions the clip in real time, tracking the finger 1:1 (no
  // deferred "snap on release" — the clip's DOM position is updated on every move event,
  // same frame). This is the "must long-press first, then it should move" behavior
  // requested — sliding without holding first is reserved for scrubbing.
  const DRAG_MOVE_TOLERANCE = 7; // px — small vertical gestures hand off to row scrolling immediately
  let dragState = null; // null | {type:'pending'|'move'|'scroll', startX, origStart, origEnd, origKfTimes}
  let swapPressTimer = null;
  function clearSwapPressTimer(){ if(swapPressTimer){ clearTimeout(swapPressTimer); swapPressTimer=null; } }
  function applyBodyMoveAt(clientX){
    if(!dragState||dragState.type!=='move')return;
    const pps=pxPerSecond();
    const deltaSec=timelineDragDeltaPixels(dragState,clientX)/pps;
    const len=dragState.origEnd-dragState.origStart;
    const newStart=Math.max(0,dragState.origStart+deltaSec);
    activeClip.trimStart=newStart;activeClip.trimEnd=newStart+len;
    // Keep keyframe data untouched until release. Repack the other clips into vertical
    // lanes now, then paint the active transform so the clip stays under the finger.
    applyCharClipLanes('main','main');
    ensureTimelineRoomForDraggedClip(activeClip);
    repositionDraggedMainClip();
  }
  body.onpointerdown = (e)=>{
    e.stopPropagation();
    if(isPlaying) pausePlayback();
    selectClipForChrome('main');
    const timelineScroller=document.getElementById('timelineScroll');
    // The clip being dragged always belongs to the currently-active character's own row —
    // that row is already showing as selected (see renderCharacterLayers) and stays that
    // way; there is no separate "stickman" row to re-highlight anymore now that each
    // character's clips live directly on its own row.
    dragState = { type:'pending', pointerId:e.pointerId, startScrollLeft:timelineScroller?timelineScroller.scrollLeft:0, startX:e.clientX, startY:e.clientY, lastX:e.clientX, startTime: getCurrentAbsoluteTime(), origStart:activeClip.trimStart, origEnd:activeClip.trimEnd, origKfTimes: activeClip.keyframes.map(k=>k.t) };
    body.setPointerCapture(e.pointerId);
    clearSwapPressTimer();
    swapPressTimer=setTimeout(()=>{
      swapPressTimer=null;
      if(!dragState||dragState.type!=='pending')return;
      dragState.type='move';clipEl.classList.add('dragging');if(window.ethvelleNotifyClipMoveReady)window.ethvelleNotifyClipMoveReady();armClipDrag();
      startTimelineEdgeAutoScroll(dragState.pointerId,dragState.lastX,()=>{if(dragState&&dragState.type==='move')applyBodyMoveAt(dragState.lastX);});
      if(Number.isFinite(dragState.lastX))applyBodyMoveAt(dragState.lastX);
    },LONG_PRESS_SWAP_MS);
  };
  body.onpointermove = (e)=>{
    if(!dragState) return;
    dragState.lastX=e.clientX;
    if(dragState.type==='scroll'){ applyClipVerticalScrollHandoff(dragState, e.clientY); return; }
    const movedPx = Math.abs(e.clientX - dragState.startX);
    if(dragState.type==='pending'){
      // A dominantly-vertical slide before the long-press fires means "scroll the row list",
      // not "interact with this clip" — hand off to the shared scroller, same as sliding on
      // empty timeline space (see applyClipVerticalScrollHandoff above).
      const movedPy = Math.abs(e.clientY - dragState.startY);
      if(movedPy >= DRAG_MOVE_TOLERANCE && movedPy >= movedPx*0.82){
        clearSwapPressTimer();
        dragState.type = 'scroll';
        const scroller = document.getElementById('timelineScroll');
        dragState.scrollTopStart = scroller ? scroller.scrollTop : 0;
        applyClipVerticalScrollHandoff(dragState, e.clientY);
        return;
      }
      // A normal horizontal touch-and-drag scrubs immediately; only a stationary hold
      // arms clip-move mode, so timeline navigation never needs a long press.
      if(movedPx > 6 && movedPx >= Math.max(1, movedPy*0.72)){
        clearSwapPressTimer();dragState.type='scrub';
        scheduleTimelineScrub(timeFromDragAnchor(dragState.startX,dragState.startTime,e.clientX));
      }
      return;
    }
    if(dragState.type==='scrub'){
      scheduleTimelineScrub(timeFromDragAnchor(dragState.startX,dragState.startTime,e.clientX));
      return;
    }
    // Only one compositor translation is painted per display frame while a clip follows the finger.
    updateTimelineEdgeAutoScroll(e.pointerId,e.clientX,()=>{if(dragState&&dragState.type==='move')applyBodyMoveAt(dragState.lastX);});
    scheduleTimelineDragFrame(body,e,point=>applyBodyMoveAt(point.clientX));
      return;
  };
  body.onpointerup = (e)=>{
    if(!dragState) return;
    stopTimelineEdgeAutoScroll(e.pointerId);
    clearSwapPressTimer();
    const wasMove = dragState.type==='move';
    const wasScroll = dragState.type==='scroll';
    const wasScrub = dragState.type==='scrub';
    const anchorX = dragState.startX, anchorTime = dragState.startTime;
    if(wasMove) flushTimelineDragFrame(body,e,point=>applyBodyMoveAt(point.clientX));
    else cancelTimelineDragFrame(body);
    if(wasMove){const movedBy=activeClip.trimStart-dragState.origStart;shiftTimelineKeyframesOnce(activeClip.keyframes,dragState.origKfTimes,movedBy);if(window.ethvelleNotifyClipMoved)window.ethvelleNotifyClipMoved(movedBy);clearTimelineDragX(clipEl);}
    dragState=null; body.releasePointerCapture(e.pointerId);
    clipEl.classList.remove('dragging');
    if(wasMove) disarmClipDrag();
    if(wasMove){
      applyCharClipLanes('main'); // apply neighbour spacing once, after the fast drag completes
      // FIX: no more snap-to-neighbor on release — the clip stays exactly wherever it was
      // dropped. If it's still overlapping a neighbor, that's fine now: the row already has
      // both on separate lanes (see applyCharClipLanes, called live throughout the drag) and
      // a full renderTimelineClip() below re-confirms that packing from scratch.
      // Dragging a clip must not cause an accidental playback seek (PART 31) — keep the
      // playhead at whatever absolute time it already was rather than snapping to 0.
      setCurrentAbsoluteTime(currentAbsoluteTimeSec);
      renderTimelineClip();
    } else if(wasScrub&&!wasScroll){
      // Only a deliberate horizontal slide is a scrub; a stationary tap selects only.
      const t = timeFromDragAnchor(anchorX, anchorTime, e.clientX);
      flushTimelineScrub(t, false);
    }
    renderFrame();
    persistProjectState({skipHistory:true});
  };
  // Safety net if the gesture is interrupted (OS call/notification, app losing focus, etc.)
  // rather than ending in a normal pointerup — without this, a mid-drag interruption could
  // leave clip-drag-active permanently on (disabling row-edge cropping for good) since
  // disarmClipDrag() would never get called otherwise. Mirrors the same safety-net pattern
  // wireBackgroundTrackInteractions already uses for bgHandleDragActive.
  body.onpointercancel = ()=>{
    if(!dragState) return;
    stopTimelineEdgeAutoScroll(dragState.pointerId);
    clearSwapPressTimer();
    const wasMove = dragState.type==='move';
    cancelTimelineDragFrame(body);
    if(wasMove){const movedBy=activeClip.trimStart-dragState.origStart;shiftTimelineKeyframesOnce(activeClip.keyframes,dragState.origKfTimes,movedBy);clearTimelineDragX(clipEl);}
    dragState = null;
    clipEl.classList.remove('dragging');
    if(wasMove){
      applyCharClipLanes('main');
      disarmClipDrag();
      renderTimelineClip();
      renderFrame();
    }
  };

  // Resize handles: extend/trim clip length
  function wireHandle(handle, side){
    let hs=null;
    function applyHandleAt(clientX){
      if(!hs)return;
      const pps=pxPerSecond();
      const deltaSec=(clientX-hs.startX)/pps;
      // Trimming is not clamped against neighbors. If the new length overlaps another clip,
      // repositionClipEl's live lane pass moves the collision to a separate vertical lane.
      if(side==='left'){
        let newStart = Math.min(hs.origEnd - MIN_CLIP_SEC, Math.max(0, hs.origStart + deltaSec));
        activeClip.trimStart = newStart;
      } else {
        let newEnd = Math.max(hs.origStart + MIN_CLIP_SEC, hs.origEnd + deltaSec);
        newEnd = Math.min(newEnd, timelineDuration + 30); // allow extending well past view; view grows
        activeClip.trimEnd = newEnd;
        // BUGFIX (extending an animation clip's handle could drag it visually off/under the
        // end of the row): timelineDuration grew here but #timelineInner's own scrollable
        // width (and the ruler drawn across it) previously only got resynced AFTER the
        // finger lifted (see onpointerup's renderTimelineClip below) — so mid-drag, once the
        // clip's right edge passed the container's last-synced width, it kept extending
        // into space the row/track didn't actually occupy yet, reading as the clip running
        // out past its own row. Resyncing live, every move, keeps the track/ruler width
        // always at least as wide as the clip currently being stretched, exactly like the
        // Speed panel path (speedPanelSetSpeed) already does for the same growth case.
        if(newEnd > timelineDuration - 0.3){
          timelineDuration = newEnd + 1;
          syncTimelineInnerWidth();
          renderRuler();
        }
      }
      repositionClipEl();
      updateTimeReadout();
      // ADDED: live "Nf" frame-count readout while trimming, same gesture that already
      // drives the resize above — just reflects how far THIS handle has moved so far.
      showTrimFrameBadge(handle, side==='left' ? (activeClip.trimStart-hs.origStart) : (activeClip.trimEnd-hs.origEnd));
    }
    // v279: trimming starts only after a long press on the handle (see ethvelleMakeHandleHoldGate).
    const gate = ethvelleMakeHandleHoldGate((x)=>{
      if(!handle.isConnected) return;
      hs={startX:x,origStart:activeClip.trimStart,origEnd:activeClip.trimEnd};
      clipEl.dataset.evTimelineDragActive='1';
      armClipDrag();
    });
    handle.onpointerdown = (e)=>{
      e.stopPropagation();
      gate.down(e);
      try{handle.setPointerCapture(e.pointerId);}catch(_e){}
    };
    handle.onpointermove = (e)=>{
      if(gate.move(e)==='trim' && hs) scheduleTimelineDragFrame(handle,e,point=>applyHandleAt(point.clientX));
    };
    handle.onpointerup = (e)=>{
      gate.up(e);
      const wasDragging=hs!==null;
      if(!wasDragging){
        // Plain tap, quick slide (scrub) or row scroll: nothing was trimmed, so nothing to commit.
        cancelTimelineDragFrame(handle);
        try{handle.releasePointerCapture(e.pointerId);}catch(_e){}
        return;
      }
      flushTimelineDragFrame(handle,e,point=>applyHandleAt(point.clientX));
      hs=null;delete clipEl.dataset.evTimelineDragActive;
      try{handle.releasePointerCapture(e.pointerId);}catch(_e){}
      disarmClipDrag();
      hideTrimFrameBadge();
      // Same as the clip-body drag above: trimming must not cause an accidental seek.
      setCurrentAbsoluteTime(currentAbsoluteTimeSec);
      renderFrame();
      // Rebuild once after the gesture ends, when it is safe to replace the handle DOM.
      // The live applyCharClipLanes('main') call already kept overlapping clips in separate
      // vertical lanes throughout the drag; this confirms the same layout from scratch.
      renderTimelineClip();
      refreshOtherCharacterClipPreviews();
      persistProjectState();
    };
    handle.onpointercancel = ()=>{
      gate.cancel();
      if(!hs) return;
      cancelTimelineDragFrame(handle);hs=null;delete clipEl.dataset.evTimelineDragActive;
      hideTrimFrameBadge();
      disarmClipDrag();
    };
  }
  wireHandle(leftHandle, 'left');
  wireHandle(rightHandle, 'right');
}

/* ---------------- Keyframes ---------------- */
// How close (in seconds) the playhead must be to an existing keyframe to count as
// "sitting on" it, for both the red-highlight check and the toggle-button logic.
const KF_HIT_WINDOW = 0.05;
function keyframeContactWindow(){
  // The playhead counts as touching the diamond once its 2px line reaches the diamond's
  // outer white edge (15px rotated square = about 11px wide), not only when centered.
  return Math.max(KF_HIT_WINDOW, 10.5 / Math.max(1, pxPerSecond()));
}

// Finds a keyframe whenever the playhead line touches its visible diamond, for this clip.
function keyframeAtPlayhead(clip, tSec){
  if(!clip) return -1;
  const hitWindow = keyframeContactWindow();
  return clip.keyframes.findIndex(kf => Math.abs(kf.t - tSec) <= hitWindow);
}
