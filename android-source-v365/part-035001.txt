  function focusFrameOnSelected(id){
    // Selection chrome is visual only. Selecting a clip must never seek the playhead,
    // render a new frame, or move timelineScroll; taps have their own explicit seek path.

  function wireSlideSelection(){
    const scroller=$('timelineScroll');if(!scroller||scroller.dataset.evV32SlideSelection)return;
    scroller.dataset.evV32SlideSelection='1';let tracking=null;
    const targetAt=(x,y)=>{
      const el=document.elementFromPoint(x,y);if(!el)return null;
      let target=null;try{target=typeof timelineClipTargetAt==='function'?timelineClipTargetAt(el):null;}catch(_e){}
      if(!target){
        const readonly=el.closest&&el.closest('.anim-clip-readonly[data-readonly-clip-ref]');
        if(readonly)return String(readonly.dataset.readonlyClipRef);
      }
      if(!target)return null;
      return target.type==='main'?'main':target.type==='seg'?'seg'+target.index:String(target.model?.id??'');
    };
    scroller.addEventListener('pointerdown',e=>{
      if(e.button>0)return;
      const id=targetAt(e.clientX,e.clientY);if(!id)return;
      tracking={pointerId:e.pointerId,startX:e.clientX,startY:e.clientY,lastId:id,moved:false};
      const startRow=(document.elementFromPoint(e.clientX,e.clientY)||e.target).closest?.('.overlay-track-row,#backgroundRow,.char-row')||null;
      window.__ethvelleTimelineSelectionPointer={pointerId:e.pointerId,x:e.clientX,y:e.clientY,active:true,moved:false,startRow};
    },{capture:true,passive:true});
    // Track only the live finger coordinates here. v76 owns the single RAF-coalesced
    // trimStart/trimEnd lookup and light selection update, triggered by the scrub's
    // setCurrentAbsoluteTime wrapper. This avoids doing the same model scan twice per frame.
    scroller.addEventListener('pointermove',e=>{
      const d=tracking;if(!d||d.pointerId!==e.pointerId)return;
      if(Math.hypot(e.clientX-d.startX,e.clientY-d.startY)>=7)d.moved=true;
      if(!d.moved)return;
      const prevRow=window.__ethvelleTimelineSelectionPointer&&window.__ethvelleTimelineSelectionPointer.startRow;
      window.__ethvelleTimelineSelectionPointer={pointerId:e.pointerId,x:e.clientX,y:e.clientY,active:true,moved:true,startRow:prevRow};
    },{passive:true});
    const finish=e=>{
      if(!tracking||tracking.pointerId!==e.pointerId)return;
      const d=tracking;
      tracking=null;
      // Keep the last row/pointer coordinates available until v76's single pointer-up
      // RAF resolves the final clip and commits its heavy selection synchronization.
      if(window.__ethvelleTimelineSelectionPointer&&window.__ethvelleTimelineSelectionPointer.pointerId===e.pointerId)window.__ethvelleTimelineSelectionPointer.ending=true;
    };
    scroller.addEventListener('pointerup',finish,{capture:true,passive:true});
    scroller.addEventListener('pointercancel',e=>{
      if(tracking&&tracking.pointerId===e.pointerId)tracking=null;
      if(window.__ethvelleTimelineSelectionPointer&&window.__ethvelleTimelineSelectionPointer.pointerId===e.pointerId)window.__ethvelleTimelineSelectionPointer.active=false;
    },{capture:true,passive:true});
  }

  function wireClipTapSeek(){
    const scroller=$('timelineScroll');if(!scroller||scroller.dataset.evV32ClipTapSeek)return;
    scroller.dataset.evV32ClipTapSeek='1';
    let tap=null;
    const clipSelector='.anim-clip,.overlay-clip,.track-clip[data-bg-id]';
    const excludedSelector='.kf-marker,.anim-clip-handle,[data-ov-handle],[data-seg-handle],[data-bg-handle],.sel-edge-left,.sel-edge-right,.sel-edge-top,.sel-edge-bottom';
    scroller.addEventListener('pointerdown',e=>{
      if(e.button>0){tap=null;return;}
      const clip=e.target&&e.target.closest?e.target.closest(clipSelector):null;
      if(!clip||!clip.closest('#timelineInner')||(e.target.closest&&e.target.closest(excludedSelector))){tap=null;return;}
      tap={pointerId:e.pointerId,startX:e.clientX,startY:e.clientY,clientX:e.clientX,startedAt:performance.now(),moved:false,clip};
    },{capture:true,passive:true});
    scroller.addEventListener('pointermove',e=>{
      if(!tap||tap.pointerId!==e.pointerId)return;
      tap.clientX=e.clientX;
      if(Math.hypot(e.clientX-tap.startX,e.clientY-tap.startY)>=7)tap.moved=true;
    },{capture:true,passive:true});
    document.addEventListener('pointerup',e=>{
      if(!tap||tap.pointerId!==e.pointerId)return;
      const d=tap;tap=null;
      if(d.moved||performance.now()-d.startedAt>=240||!d.clip.isConnected)return;
      const duration=typeof totalProjectDuration==='function'?totalProjectDuration():0;
      const requested=typeof getAbsoluteTimeFromPointer==='function'?getAbsoluteTimeFromPointer(e.clientX):Number(currentAbsoluteTimeSec)||0;
      const target=Math.max(0,Math.min(Number(duration)||0,Number(requested)||0));
      try{
        const current=typeof getCurrentAbsoluteTime==='function'?Number(getCurrentAbsoluteTime()):Number(currentAbsoluteTimeSec)||0;
        // A tap directly under the fixed playhead may select a different layer without
        // changing time. In that case there is no new preview frame to render; forcing a
        // full renderOverlayMedia pass here caused a visible pause on media-heavy projects.
        if(!Number.isFinite(current)||Math.abs(current-target)>0.000001){
          if(typeof setCurrentAbsoluteTime==='function')setCurrentAbsoluteTime(target);
          if(typeof renderFrame==='function')renderFrame();
        }
      }catch(_e){}
    },false);
    document.addEventListener('pointercancel',e=>{if(tap&&tap.pointerId===e.pointerId)tap=null;},false);
  }

  function boot(){
    queuePaletteLayout();syncGlyphEffects();syncTimelineSelection();wireSlideSelection();
    ['renderFrame','renderOverlayMedia','renderOverlayLayers','renderTimelineClip','renderBackgroundTrack'].forEach(wrapRender);
    const dock=$('ethvellePicsartTextDock');
    if(dock&&!dock.dataset.evV32PaletteObserver){dock.dataset.evV32PaletteObserver='1';new MutationObserver(queuePaletteLayout).observe(dock,{subtree:true,childList:true,attributes:true,attributeFilter:['class']});}
    const layer=$('overlayMediaLayer');
    if(layer&&!layer.dataset.evV32EffectsObserver){layer.dataset.evV32EffectsObserver='1';new MutationObserver(()=>raf(syncGlyphEffects)).observe(layer,{subtree:true,childList:true});}
    const scroller=$('timelineScroll');if(!scroller){const bodyObserver=new MutationObserver(()=>{wireSlideSelection();});bodyObserver.observe(document.body,{childList:true,subtree:true});}
    document.addEventListener('click',e=>{const tab=e.target?.closest?.('#evpaTabs .evpa-tab');if(!tab)return;const dock=$('ethvellePicsartTextDock');if(tab.dataset.tab!=='font'){restoreFontPalette($('evpaPalette')?.querySelector('.evpa-swatches'));dock?.classList.remove('evpa-font-palette-mode');}queuePaletteLayout();},true);
  }
  if(document.readyState==='loading')document.addEventListener('DOMContentLoaded',boot,{once:true});else boot();
  /* Ensure if the editor/timeline is assembled after this layer loads, it still receives the handlers. */
  const readyObserver=new MutationObserver(()=>{wireSlideSelection();queuePaletteLayout();});
  if(document.body)readyObserver.observe(document.body,{childList:true,subtree:true});
})();
