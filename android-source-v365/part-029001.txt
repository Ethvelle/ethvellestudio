function ethvelleIs21Pose(p){
  return !!p && ['pelvis','spine_lower','spine_upper','chest','neck','clavicleL','clavicleR','upperArmL','upperArmR','forearmL','forearmR','thighL','thighR','shinL','shinR','footL','footR'].some(k=>Object.prototype.hasOwnProperty.call(p,k));
}
function ethvelleDiscrete21(preset,tFrac,field){
  const kfs=preset&&preset.keyframes;if(!Array.isArray(kfs))return undefined;
  let val;
  for(let i=0;i<kfs.length;i++)if(kfs[i].t<=tFrac&&kfs[i][field]!==undefined)val=kfs[i][field];
  return val;
}
function ethvelleComplete21Pose(preset,tFrac,pose){
  const p=ethvelleNormalizeLegacyPoseTo21(pose||{});
  if(preset&&preset._ethvelleAnimKey)p.__ethvelleAnimKey=preset._ethvelleAnimKey;
  p.__ethvelleTFrac=((Number(tFrac)||0)%1+1)%1; /* v326: loop position, used for the walk knee bend */
  const fist=ethvelleDiscrete21(preset,tFrac,'fist');
  if(fist!==undefined)p.gestureFist=fist?1:0;
  else p.gestureFist=Number(p.gestureFist)||0;
  p.gesturePoint=Number(p.gesturePoint)||0;
  p.gesturePeace=Number(p.gesturePeace)||0;
  p.gestureThumbsUp=Number(p.gestureThumbsUp)||0;
  const mouth=ethvelleDiscrete21(preset,tFrac,'mouth');
  const blink=ethvelleDiscrete21(preset,tFrac,'blink');
  p.__mouth=mouth===undefined?'closed':mouth;
  p.__blink=blink===undefined?false:!!blink;
  return p;
}

// Updated presets authored for a parent-relative 21-bone rig also need a compatibility pose
// for the original stickman and saved legacy rigs. The sheet renderer uses the full pose below.
function ethvelle21PoseToLegacy(p){
  const out={}, skeleton=new Set(['pelvis','spine_lower','spine_upper','chest','neck','head','clavicleL','clavicleR','upperArmL','upperArmR','forearmL','forearmR','handL','handR','thighL','thighR','shinL','shinR','footL','footR','hipY']);
  Object.keys(p||{}).forEach(k=>{if(!skeleton.has(k)&&!k.startsWith('__ethvelle'))out[k]=p[k];});
  const spine=(Number(p.spine_lower)||0)+(Number(p.spine_upper)||0)+(Number(p.chest)||0);
  out.spine=spine;
  out.head=spine+(Number(p.neck)||0)+(Number(p.head)||0);
  ['L','R'].forEach(S=>{
    const arm=(Number(p['clavicle'+S])||0)+(Number(p['upperArm'+S])||0)+spine;
    out['arm'+S]=arm;
    out['arm'+S+'_lo']=arm+(Number(p['forearm'+S])||0);
    out['hand'+S]=out['arm'+S+'_lo']+(Number(p['hand'+S])||0);
    const leg=(Number(p.pelvis)||0)+(Number(p['thigh'+S])||0);
    out['leg'+S]=leg;
    out['leg'+S+'_lo']=leg+(Number(p['shin'+S])||0)+(Number(p['foot'+S])||0);
  });
  if(p.hipY!==undefined)out.hipY=p.hipY;
  return out;
}

function ethvelleModernPoseForClipAtTime(tSecAbsolute,clipOverride,segmentsOverride){
  const useClip=clipOverride!==undefined?clipOverride:activeClip;
  const useSegments=segmentsOverride!==undefined?segmentsOverride:splitStickmanSegments;
  if(!useClip)return null;
  const hit=stickmanClipAt(tSecAbsolute,useClip,useSegments);if(!hit)return null;
  const clip=hit.clip,preset=animPresets[clip.animKey];
  if(!preset||preset._ethvelleRigVersion!==21)return null;
  const speed=clip.speed||1, localT=Math.max(0,(tSecAbsolute-clip.trimStart)*speed);
  const loopT=(localT%preset.duration)/preset.duration;
  let pose=ethvelleComplete21Pose(preset,loopT,ethvelleOriginalEvalPreset21(preset,loopT));
  if(Array.isArray(clip.keyframes)&&clip.keyframes.length){
    const kfs=[...clip.keyframes].sort((a,b)=>a.t-b.t), WINDOW=.36;
    const holdAndBlend=(kf,base)=>{
      const dist=Math.abs(tSecAbsolute-kf.t);if(dist>=WINDOW)return null;
      const weight=.5*(1+Math.cos(Math.PI*(dist/WINDOW)));
      const kp=ethvelleNormalizeLegacyPoseTo21(kf.pose||{}), blended={};
      const keys=new Set([...Object.keys(base),...Object.keys(kp)]);
      keys.forEach(k=>{
        if(k.startsWith('__'))return;
        const bv=base[k]??0;
        blended[k]=lerp(bv,kp[k]??bv,weight);
      });
      return {...base,...blended};
    };
    if(tSecAbsolute<=kfs[0].t)pose=holdAndBlend(kfs[0],pose)||pose;
    else if(tSecAbsolute>=kfs[kfs.length-1].t)pose=holdAndBlend(kfs[kfs.length-1],pose)||pose;
    else{
      let i=0;while(i<kfs.length-1&&tSecAbsolute>kfs[i+1].t)i++;
      const a=kfs[i],b=kfs[i+1],near=Math.abs(tSecAbsolute-a.t)<=Math.abs(tSecAbsolute-b.t)?a:b,far=near===a?b:a;
      pose=holdAndBlend(near,pose)||holdAndBlend(far,pose)||pose;
    }
    // Preserve preset-level discrete expression fields through the user keyframe blend.
    const mouth=ethvelleDiscrete21(preset,loopT,'mouth'),blink=ethvelleDiscrete21(preset,loopT,'blink'),fist=ethvelleDiscrete21(preset,loopT,'fist');
    pose.__mouth=mouth===undefined?'closed':mouth;pose.__blink=blink===undefined?false:!!blink;
    pose.gestureFist=fist===undefined?(Number(pose.gestureFist)||0):(fist?1:0);
  }
  return pose;
}

/* v326: knee bend (degrees, backward) for the LEFT-bone leg over one walk loop (0..1).
   The right-bone leg uses the same curve shifted by half a loop. Peak = foot in the air (leg lifting
   and swinging forward), small bend at toe-off, almost straight when the foot is planted. */
const ETHVELLE_WALK_KNEE_KEYS=[[0,12],[0.12,26],[0.25,44],[0.36,46],[0.46,20],[0.5,6],[0.75,3],[1,12]];
function ethvelleWalkKneeFlexDeg(t){
  t=((Number(t)||0)%1+1)%1;
  const k=ETHVELLE_WALK_KNEE_KEYS;
  for(let i=0;i<k.length-1;i++){
    if(t>=k[i][0]&&t<=k[i+1][0]){
      const u=(t-k[i][0])/((k[i+1][0]-k[i][0])||1),e=.5-.5*Math.cos(Math.PI*u);
      return k[i][1]+(k[i+1][1]-k[i][1])*e;
    }
  }
  return k[k.length-1][1];
}
/* v327: hip joint. The two thighs used to pivot too far apart / too low, so a gap opened under the
   pelvis triangle when the legs swung wide. Pull both hip joints a little toward the centre and up so
   the thigh tops stay overlapped with the pelvis piece. The pull grows a bit with the thigh swing.
   Tune with the three numbers below (bigger = closer / higher). */
const ETHVELLE_HIP_IN_REST=0.12, ETHVELLE_HIP_IN_SWING=0.18, ETHVELLE_HIP_UP_REST=0.008, ETHVELLE_HIP_UP_SWING=0.012;
function ethvelleTightenHips(rig,char,p){
  if(!rig||!rig.bones)return;
  const type=(char&&(char.modelType||(char.sheet&&char.sheet.modelType)))||'humanoid';
  if(type!=='humanoid')return;
  const ov=(char&&char.sheet&&char.sheet.rigOverride)||{};
  const scale=stageH*(Number(ov.figFrac)||.5), deg=Math.PI/180;
  ['L','R'].forEach(S=>{
    const b=rig.bones['thigh'+S]; if(!b||!b.off)return;
    const sw=p?Math.min(1,Math.abs(Number(p['thigh'+S])||0)/(24*deg)):0;
    b.off={x:b.off.x*(1-(ETHVELLE_HIP_IN_REST+ETHVELLE_HIP_IN_SWING*sw)),
           y:b.off.y-(ETHVELLE_HIP_UP_REST+ETHVELLE_HIP_UP_SWING*sw)*scale};
  });
}
// Apply the full parent-relative pose directly to the 21-bone runtime used by 14-part sheets.
const ethvelleOriginalSheetRigForCharacter21=sheetRigForCharacter;
sheetRigForCharacter=function(char){
  const rig=ethvelleOriginalSheetRigForCharacter21(char);
  const p=char&&char._ethvelleModernPose;
  if(!p||!rig||!rig.bones||!rig.baseAngles){ethvelleTightenHips(rig,char,null);return rig;}
  ethvelleTightenHips(rig,char,p);
  const clamp=(v,lo,hi)=>Math.max(lo,Math.min(hi,v));
  const deg=Math.PI/180;
  const limits={pelvis:28,spine_lower:28,spine_upper:28,chest:30,neck:35,head:42,clavicleL:28,clavicleR:28,upperArmL:165,upperArmR:165,forearmL:145,forearmR:145,handL:95,handR:95,thighL:125,thighR:125,shinL:145,shinR:145,footL:85,footR:85};
  // v326 WALK KNEES: while a leg is lifted (swing phase) the knee must fold BACKWARD (shin goes behind
  // the thigh), never forward. The old preset values made one leg bend forward and used a one-leg
  // sign flip. For the walk presets (all views except back) the shin angle is now driven by a
  // swing-phase curve: both knees only ever fold backward, the most when the foot is in the air,
  // and are almost straight when the foot is planted. Thigh, foot placement, arms and every other
  // animation stay exactly as they were.
  const walkKneeFix=String((char&&char.sheet&&char.sheet.view)||'front')!=='back' &&
    (char.modelType||(char.sheet&&char.sheet.modelType)||'humanoid')==='humanoid' &&
    ['walk','walkHold','frontWalk'].includes(p.__ethvelleAnimKey) && typeof p.__ethvelleTFrac==='number';
  const kneeScale=p.__ethvelleAnimKey==='frontWalk'?0.8:1;
  const kneeFlex={
    shinL:walkKneeFix?ethvelleWalkKneeFlexDeg(p.__ethvelleTFrac)*kneeScale*deg:0,
    shinR:walkKneeFix?ethvelleWalkKneeFlexDeg(p.__ethvelleTFrac+0.5)*kneeScale*deg:0
  };
  Object.keys(rig.bones).forEach(k=>{
    if(k==='root'||rig.baseAngles[k]===undefined)return;
    const raw=Number(p[k])||0;
    let poseDelta=raw;
    if(walkKneeFix){
      if(k==='shinL'||k==='shinR')poseDelta=kneeFlex[k];                       // knee folds back only
      else if(k==='footL'||k==='footR')poseDelta=raw*0.4-0.45*kneeFlex['shin'+k.slice(-1)]; // keep the shoe from flipping up
      else if(k==='forearmL'||k==='forearmR')poseDelta=-(5*deg+0.9*Math.abs(raw));      // v327: elbow folds FORWARD only, never back
    }
    const limit=(limits[k]||165)*deg,delta=clamp(poseDelta,-limit,limit);
    rig.bones[k].angle=rig.baseAngles[k]+delta;
  });
  return rig;
};

      el.style.setProperty('filter','none',pr);

    }
  };

  function syncLayout(){
    if(!root||!root.classList.contains('active'))return;
    const input=document.getElementById('evtcStageText');const stage=document.getElementById('canvasStage');
    const vv=window.visualViewport;
    // Once the user starts typing, Android may shrink the visual viewport for the keyboard.
    // Keep the full-screen composer and the writing field at their original geometry instead
    // of repeatedly recalculating the canvas/text size on every keyboard resize/input event.
    const activelyTyping=!!input&&document.activeElement===input;
    // Track the visible viewport even while typing: Android shrinks visualViewport when
    // the soft keyboard opens, so the bottom text palette remains floating above it.
    if(vv){root.style.top=vv.offsetTop+'px';root.style.height=vv.height+'px';root.style.bottom='auto';}
    else {root.style.top='0px';root.style.height='100dvh';root.style.bottom='auto';}
    if(stage&&input&&!activelyTyping){
      const r=stage.getBoundingClientRect();
      // Normally this is the live canvas stage. If the editor is still completing its first
      // layout (or mobile keyboard resize briefly reports a zero-size stage), keep the input
      // centered and fully usable instead of collapsing it to a tiny single-character field.
      const wrap=document.querySelector('.canvas-wrap');const wr=wrap?wrap.getBoundingClientRect():{width:0,height:0};
      const fallbackW=Math.min(Math.max(240,wr.width||0),Math.max(240,window.innerWidth*.92));
      const w=r.width>80?r.width:fallbackW;
      const h=r.height>80?r.height:Math.min(window.innerHeight*.44,w*9/16);
      const left=r.width>80?r.left:(window.innerWidth-w)/2;
      const top=r.height>80?r.top:Math.max(82,Math.min(window.innerHeight*.34,window.innerHeight*.28));
      input.style.left=Math.max(0,left+w*.045)+'px';
      input.style.top=(top+h*.20)+'px';
      input.style.width=Math.max(150,w*.91)+'px';
      input.style.height=Math.max(76,h*.60)+'px';
      input.style.fontSize=Math.max(22,Math.min(56,Number(draft&&draft.fontSize)||44,w*.095))+'px';
    }
  }
