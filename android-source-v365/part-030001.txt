     black-to-transparent vertically), so convert with HSV rather than HSL. This makes the
     marker colour exactly match the pixel under it instead of showing white/pastel at the wrong point. */
  function hsvHex(h,s,v){
    h=((Number(h)||0)%360+360)%360;s=Math.max(0,Math.min(100,Number(s)||0))/100;v=Math.max(0,Math.min(100,Number(v)||0))/100;
    const chroma=v*s,x=chroma*(1-Math.abs((h/60)%2-1)),m=v-chroma;let r=0,g=0,b=0;
    if(h<60){r=chroma;g=x;}else if(h<120){r=x;g=chroma;}else if(h<180){g=chroma;b=x;}else if(h<240){g=x;b=chroma;}else if(h<300){r=x;b=chroma;}else{r=chroma;b=x;}
    return '#'+[r+m,g+m,b+m].map(n=>Math.round(n*255).toString(16).padStart(2,'0')).join('').toUpperCase();
  }

    const s=max===0?0:d/max;return {h,s:s*100,l:max*100};
  }
