    let c=String(value??'').trim();
    if(/^#[0-9a-f]{3}$/i.test(c))c='#'+c.slice(1).split('').map(ch=>ch+ch).join('');
    return /^#[0-9a-f]{6}$/i.test(c)?c.toUpperCase():'';
  }
  function rgba(value,opacity){
    const c=normaliseHex(value)||'#FFFFFF';
    const n=[1,3,5].map(i=>parseInt(c.slice(i,i+2),16));
    const a=Math.max(0,Math.min(100,Number(opacity==null?100:opacity)))/100;
    return 'rgba('+n[0]+','+n[1]+','+n[2]+','+a+')';
  }
  const priorSetColor=window.ethvellePicsartSetColor;
  if(typeof priorSetColor==='function'&&!priorSetColor.__evV57ShadowIsolation){
    const isolatedSetColor=function(key,color){
      const activeTab=dock()?.querySelector('#evpaTabs .evpa-tab.active')?.dataset.tab||'';
      const routedKey=activeTab==='shadow'?'shadowColor':key;
      if(routedKey!=='shadowColor')return priorSetColor.apply(this,[routedKey,color]);

      const target=typeof window.ethvellePicsartGetTarget==='function'?window.ethvellePicsartGetTarget():null;
      const hadFill=!!target&&Object.prototype.hasOwnProperty.call(target,'textColor');
      const hadFillOpacity=!!target&&Object.prototype.hasOwnProperty.call(target,'textOpacity');
      const oldFill=target?.textColor;
      const oldFillOpacity=target?.textOpacity;
      const picked=normaliseHex(color)||color;

      const result=priorSetColor.apply(this,['shadowColor',picked]);

      if(target){
        if(hadFill)target.textColor=oldFill;else delete target.textColor;
        if(hadFillOpacity)target.textOpacity=oldFillOpacity;else delete target.textOpacity;
        target.shadowColor=picked;
      }

      /* The composer preview is live DOM; restore only its fill colour and leave its shadow intact. */
      const composer=$('ethvelleTextComposer');
      if(composer?.classList.contains('active')&&target){
        const input=$('evtcStageText');
        if(input){
          input.style.setProperty('color',rgba(target.textColor||'#FFFFFF',target.textOpacity),'important');
          window.ethvelleApplyTextShadow(input,target.shadowEnabled
            ?(Number(target.shadowX)||0)+'px '+Math.max(4,Number(target.shadowY)||6)+'px '+Math.max(0,Number(target.shadowBlur)||0)+'px '+rgba(target.shadowColor||'#000000',target.shadowOpacity)
            :'none',input.dataset.evFill==='1',true);
        }
      }else{
        try{if(typeof window.renderFrame==='function')window.renderFrame();}catch(_e){}
        try{if(typeof window.renderOverlayLayers==='function')window.renderOverlayLayers();}catch(_e){}
      }
      try{if(typeof window.persistProjectState==='function'&&!composer?.classList.contains('active'))window.persistProjectState({skipHistory:true});}catch(_e){}
      return result;
    };
    isolatedSetColor.__evV57ShadowIsolation=true;
    window.ethvellePicsartSetColor=isolatedSetColor;
  }
})();
