// animationAccess[key] is a list of plan ids an admin checked for this animation (see the
// admin panel's multi-select Animation access rules). Older data saved before multi-plan
// support stored a single plan-id (or name, or 'free'/empty) as a bare string — normalized
// here into a one-item array so both shapes read identically.
function getAnimationAccessPlans(key){
  const map=siteContent.animationAccess||{}; const raw=map[key];
  const plans=typeof getPlans==='function'?getPlans():[];
  const free=plans.find(p=>Number(p.price)===0)||{id:'p_free',name:'Free',price:0};
  const ids=Array.isArray(raw)?raw.map(String).filter(Boolean):(raw==null||raw==='')?[]:[String(raw)];
  if(!ids.length)return [free];
  const resolved=ids.map(id=>{
    const normalized=id.toLowerCase();
    return plans.find(p=>String(p.id)===id) ||
      plans.find(p=>String(p.name||'').toLowerCase()===normalized) ||
      ((normalized==='free'||normalized==='p_free')?free:null);
  }).filter(Boolean);
  return resolved.length?resolved:[free];
}
// The badge/lock shown to a visitor is driven by the CHEAPEST plan an admin checked for this
// animation — e.g. if both Pro and Studio are checked, a Pro subscriber should already be able
// to use it rather than being told they need Studio.
function getAnimationRequiredPlan(key){
  const options=getAnimationAccessPlans(key);
  return options.slice().sort((a,b)=>Number(a.price||0)-Number(b.price||0))[0];
}
function isAnimationLocked(key){
  const required=getAnimationRequiredPlan(key);
  if(!required||Number(required.price)<=0)return false;
  const current=typeof getMyPlanData==='function'?getMyPlanData():{id:'p_free',name:'Free',price:0};
  // Treat the plans as tiers by price, so a higher tier (or an exact plan match) can use the
  // animation even if its id isn't literally in the checked list — the cheapest checked plan
  // sets the floor price a user's current plan must meet.
  const options=getAnimationAccessPlans(key);
  const directMatch=!!current && options.some(p=>String(p.id)===String(current.id));
  if(directMatch) return false;
  return !current || Number(current.price||0) < Number(required.price||0);
}

async function downloadPromptImage(button){
  if(!button||button.disabled)return;
  const url=button.dataset.src||'';
  if(!url){showToast('Image is unavailable',{icon:toastIcons.info});return;}
  const kind=button.dataset.kind==='template'?'template':'reference';
  const title=String(button.dataset.title||'Ethvelle').normalize('NFKD').replace(/[\u0300-\u036f]/g,'').replace(/[^a-zA-Z0-9_-]+/g,'-').replace(/^-+|-+$/g,'').slice(0,55)||'Ethvelle';
  const baseName=`${title}-${kind}-image`;
  const originalText=button.querySelector('span')?.textContent||'Download';
  button.disabled=true;
  if(button.querySelector('span'))button.querySelector('span').textContent='Downloading…';
  try{
    const response=await fetch(url,{mode:'cors',credentials:'omit'});
    if(!response.ok)throw new Error('Image request failed');
    const blob=await response.blob();
    if(!blob.size)throw new Error('Empty image');
    const mime=(blob.type||'').split(';')[0].toLowerCase();
    const mimeExt={'image/jpeg':'jpg','image/jpg':'jpg','image/png':'png','image/webp':'webp','image/gif':'gif','image/avif':'avif','image/bmp':'bmp','image/svg+xml':'svg'};
    let ext=mimeExt[mime]||'';
    if(!ext){try{const path=new URL(url,location.href).pathname;const match=path.match(/\.([a-z0-9]{2,5})$/i);ext=match?match[1].toLowerCase():'jpg';}catch(e){ext='jpg';}}
    const objectUrl=URL.createObjectURL(blob);
    const link=document.createElement('a');link.href=objectUrl;link.download=`${baseName}.${ext}`;link.style.display='none';
    document.body.appendChild(link);link.click();link.remove();
    setTimeout(()=>URL.revokeObjectURL(objectUrl),1500);
    showToast('Image downloaded',{icon:toastIcons.check});
  }catch(error){
    // Some image hosts block cross-origin downloads. Open the source directly as a fallback.
    const link=document.createElement('a');link.href=url;link.target='_blank';link.rel='noopener noreferrer';link.download=`${baseName}.jpg`;link.style.display='none';
    document.body.appendChild(link);link.click();link.remove();
    showToast('Opening image — use Save image if download does not start',{icon:toastIcons.info});
  }finally{
    button.disabled=false;
    if(button.querySelector('span'))button.querySelector('span').textContent=originalText;
  }
}

function normalizeNotificationScreen(value){
  const v=String(value==null?'':value).trim().toLowerCase().replace(/[^a-z]/g,'');
  if(!v) return '';
  if(v==='all'||v==='everywhere'||v==='allscreens') return 'all';
  if(v.startsWith('home')||v.includes('homepage')||v.includes('homescreen')) return 'home';
  if(v.startsWith('project')||v.includes('projectscreen')) return 'projects';
  if(v.startsWith('profile')||v.includes('profilescreen')) return 'profile';
  if(v.startsWith('payment')||v.includes('paymentscreen')) return 'payment';
  return v.replace(/screen$/,'');
}
function notificationShowTargets(value){
  if(Array.isArray(value)) return value.map(normalizeNotificationScreen).filter(Boolean);
  if(typeof value==='string' && value.trim()) return value.split(/[;,|]/).map(normalizeNotificationScreen).filter(Boolean);
  if(value && typeof value==='object') return Object.keys(value).filter(k=>value[k]===true || value[k]===1 || value[k]==='true').map(normalizeNotificationScreen).filter(Boolean);
  return [];
}
function maybeShowLocationNotification(screenName){
  if(!currentUser||!screenName||!Array.isArray(myNotifications))return;
  const active=normalizeNotificationScreen(screenName);
  const backdrop=document.getElementById('notificationPopupBackdrop');if(!backdrop||backdrop.classList.contains('active'))return;
  const titleEl=document.getElementById('notificationPopupTitle'), bodyEl=document.getElementById('notificationPopupBody');
  if(!titleEl||!bodyEl)return;
  const seen=getNotifPopupSeen();
  const candidate=visibleNotifications().find(n=>{
    if(seen.includes(n.id)) return false;
    const rawTarget = n.showOn ?? n.showOnScreens ?? n.popupOn ?? n.displayOn ?? n.screen ?? n.location ?? n.showLocation ?? n.popupLocation ?? n.displayLocation;
    const targets=notificationShowTargets(rawTarget);
    // Older admin notifications omit their placement field; they are treated as Home notifications.
    // "All screens" should be eligible everywhere, regardless of its original casing/spacing.
    return targets.length ? (targets.includes('all') || targets.includes(active)) : active==='home';
  });
  if(!candidate)return;
  locationPopupNotificationId=candidate.id;
  titleEl.textContent=candidate.title||candidate.subject||'Notification';
  bodyEl.textContent=candidate.body||candidate.message||candidate.description||'';

  seen.push(candidate.id);writeIdList(notifPopupSeenKey(currentUser.uid),seen);
}

function calculateExportCreditCost(quality,durationSeconds){
  const settings=getCreditSettings();
  if(!settings.enabled) return 0;
  const seconds=Math.max(0,Number(durationSeconds)||0);

// FEATURE (admin-authored Custom Animations, published from the admin panel's new
// "Custom animations" page): merges every admin-published preset into the live animPresets
// object so they show up in the Animate picker exactly like idle/walk/wave/jump do, for
// every user. Only ever ADDS/overwrites admin-authored keys (each one prefixed 'custom_' by
// the admin panel when it creates them, so a custom preset can never collide with or
// shadow a built-in key) — never deletes from animPresets, so removing a custom preset in
// the admin panel just means its key stops being re-applied on the next sync; a project
// that already has a clip using a deleted custom preset keeps working until the user picks
// something else (same "orphaned animKey" tolerance evalPreset/animPresets[clip.animKey]
// already has to handle anyway if a key is ever missing).
function applyCustomAnimPresets(){
  // Shape published by the admin panel's Custom Animations page: a flat array of
  // { id, key, name, duration, keyframes }, one per preset — key is always 'custom_'-prefixed
  // (see the admin panel's slugifyAnimName) and keyframes are already in radians (the admin
  // panel converts from the degrees an admin types in before publishing).
  const list = siteContent.customAnimPresets;
  if(!Array.isArray(list)) return;
  list.forEach(entry=>{
    if(!entry || !entry.key || !entry.name) return;
    if(!String(entry.key).startsWith('custom_')) return; // defensive: admin panel always sets this prefix
    if(!Array.isArray(entry.keyframes) || !entry.keyframes.length) return; // malformed — skip, don't crash the picker
    animPresets[entry.key] = { name:String(entry.name), duration: Number(entry.duration)>0 ? Number(entry.duration) : 1, keyframes: entry.keyframes };
  });
}

/* ================= PROMO POPUP =================
   Shows at most one promo popup per app session (not per visit to Home — once dismissed or
   acted on, it won't reappear until the page is fully reloaded), picked from whichever active
   popups exist, after its own configured delay. If several are active, the one with the
   shortest delay fires first; the rest are skipped for this session rather than queued, so the
   person is never hit with a second popup right after closing the first.
   window.activePromoDiscount is read by computePlanDiscount() (Payment screen) once a
   "Give a discount" popup is tapped — { planId, type, value, label } or null. */
window.activePromoDiscount = null;
const ETHVELLE_PROMO_ATTRIBUTION_KEY = 'ethvelle.promoAttribution.v1';
function setPromoAttribution(pp){
  if(!pp || !pp.id) return;
  try{ sessionStorage.setItem(ETHVELLE_PROMO_ATTRIBUTION_KEY, JSON.stringify({popupId:String(pp.id), clickedAt:Date.now()})); }catch(e){}
}
function getPromoAttribution(){
  try{
    const raw=sessionStorage.getItem(ETHVELLE_PROMO_ATTRIBUTION_KEY);
    if(!raw) return null;
    const data=JSON.parse(raw);
    if(!data || !data.popupId) return null;
    // Attribution is tab/session scoped and expires after four hours to avoid stale cross-journey credit.
    if(data.clickedAt && Date.now()-Number(data.clickedAt)>4*60*60*1000){ sessionStorage.removeItem(ETHVELLE_PROMO_ATTRIBUTION_KEY); return null; }
    return {popupId:String(data.popupId)};
  }catch(e){ return null; }
}
function clearPromoAttribution(){ try{ sessionStorage.removeItem(ETHVELLE_PROMO_ATTRIBUTION_KEY); }catch(e){} }
function withPromoAttribution(details){
  const base=details && typeof details==='object' ? details : {};
  const attribution=getPromoAttribution();
  return attribution ? {...base, popupId:attribution.popupId, source:'promo_popup'} : base;
}
function trackPromoAttributedEvent(kind, details){
  const attribution=getPromoAttribution();
  if(!attribution) return;
  trackEvent(kind, {...(details && typeof details==='object' ? details : {}), popupId:attribution.popupId, source:'promo_popup'});
}
let promoPopupShownThisSession = false;
let promoPopupTimer = null;
let promoPopupCurrent = null;
// When Firestore is available, do not schedule a popup from the stale local cache. Wait for
// the first server-confirmed snapshot, then schedule from the newest published data. This
// prevents the old popup/offer appearing on the first visit after an admin update.
let promoPopupContentServerConfirmed = !(firebaseReady && fbDb);
function promoPopupVisibleOnScreen(pp,screenName){
  const targets=Array.isArray(pp&&pp.showOn)?pp.showOn:(typeof (pp&&pp.showOn)==='string'?[pp.showOn]:['home']);
  return targets.includes('all') || targets.includes(screenName);
}
// Audience scope is independent from the action/discount target. Legacy popups that predate
// audiencePlans remain visible to everyone; newly configured popups can target guests, Free
// users, or one/more paid plan IDs.
function promoPopupVisibleToCurrentUser(pp){
  const targets = Array.isArray(pp && pp.audiencePlans) ? pp.audiencePlans : ['__all__'];
  if(targets.includes('__all__')) return true;
  if(!currentUser) return targets.includes('guest');
  const plans = getPlans();
  const activePlan = plans.find(p=>p.id===myActivePlanId) || plans.find(p=>Number(p.price)===0);
  if(!activePlan || Number(activePlan.price)<=0) return targets.includes('free');
  return targets.includes(activePlan.id);
}
function wirePromoPopups(forceReschedule=false, screenOverride=null){
  if(promoPopupShownThisSession || !promoPopupContentServerConfirmed) return;
  if(promoPopupTimer){
    if(!forceReschedule) return;
    clearTimeout(promoPopupTimer); promoPopupTimer=null;
  }
  const screenName=screenOverride || activeScreenId() || 'home';
  const eligible=getPromoPopups().filter(pp=>pp&&pp.active!==false&&safeImageUrl(pp.image)&&promoPopupVisibleOnScreen(pp,screenName)&&promoPopupVisibleToCurrentUser(pp));
  if(!eligible.length)return;
  const soonest=eligible.slice().sort((a,b)=>(Number(a.delaySeconds)||0)-(Number(b.delaySeconds)||0))[0];
  promoPopupTimer=setTimeout(()=>{
    promoPopupTimer=null;
    if(promoPopupShownThisSession)return;
    const currentScreen=activeScreenId();
    const latest=getPromoPopups().filter(pp=>pp&&pp.active!==false&&safeImageUrl(pp.image)&&promoPopupVisibleOnScreen(pp,currentScreen)&&promoPopupVisibleToCurrentUser(pp))
      .slice().sort((a,b)=>(Number(a.delaySeconds)||0)-(Number(b.delaySeconds)||0))[0];
    if(latest) showPromoPopup(latest);
    else wirePromoPopups(true,currentScreen); // audience/screen may have changed while the delay was running
  },Math.max(0,Number(soonest.delaySeconds)||0)*1000);
}
function showPromoPopup(pp){
  promoPopupCurrent = pp;
  promoPopupShownThisSession = true;
  trackEvent('promo_popup_shown', { popupId:pp && pp.id, screen:trackingActiveScreen() });
  document.getElementById('promoPopupImg').src = safeImageUrl(pp.image);
  document.getElementById('promoPopupBackdrop').classList.add('active');
}
function closePromoPopup(){
  document.getElementById('promoPopupBackdrop').classList.remove('active');

function handlePromoPopupTap(){
  const pp = promoPopupCurrent;
  closePromoPopup();
  if(!pp) return;
  setPromoAttribution(pp);
  trackEvent('promo_popup_click', { popupId:pp.id, screen:trackingActiveScreen(), source:'promo_popup' });
  if(pp.actionType === 'discount' && pp.discountPlanId){
    // A promo can target one paid plan or every paid plan (__all_paid__). The promo is
    // intentionally kept for the rest of this page session so the prices stay consistent
    // while the user browses the Payment screen and proceeds to checkout.
    // Older saved popups predate discountScope; treat them as all-plan promos so a
    // previously configured home popup no longer discounts only its legacy target plan.
    const applyToAll = pp.discountPlanId === '__all_paid__' || pp.discountScope !== 'specific';
    const selectionPlanId = pp.discountPlanId !== '__all_paid__' ? pp.discountPlanId : '';
    window.activePromoDiscount = {
      planId: applyToAll ? '__all_paid__' : pp.discountPlanId,
      applyToAll,
      type: pp.discountType || 'percent',
      value: Number(pp.discountValue) || 0,
      label: 'Promo discount'
    };
    _plansSig = null;
    goTo('payment');
    // Navigation alone does not redraw the plans; rebuild them now so the promo prices are
    // visible immediately instead of waiting for an unrelated Firestore snapshot.
    renderPaymentPlans();
    if(selectionPlanId){
      const card = document.querySelector('#screen-payment .plan-card[data-plan="'+selectionPlanId+'"]');
      if(card) selectPlan(card);
    }
  } else {
    runAdminAction(pp.actionType, pp.actionValue);
  }
}

/* ---------------- SAFE HELPERS ---------------- */
// Only ever put http(s) URLs or inline image data URLs into an <img src>.
function safeImageUrl(u){
  return (typeof u === 'string' && /^(https?:\/\/|data:image\/)/i.test(u)) ? u : '';
}
// Re-rendering replaces the whole strip, which snaps the swipe position back to slide 1 and
// makes every picture flash. Firestore snapshots fire more often than the content actually
// changes, so each renderer remembers what it last drew and skips identical data.
let _bannerSig = null, _animSig = null, _plansSig = null, _tutorialsSig = null;

// Banner / example-card taps go through ONE delegated listener reading data-* attributes.
// (The old inline onclick="…runAdminAction('screen', "home")" put JSON's double quotes inside a
// double-quoted attribute, which cut the attribute short — so taps never fired.)
function wireContentTaps(){
  ['homeBannerScroll','homeVideoScroll','homeTileRow','tutorialGrid'].forEach(id=>{
    const el = document.getElementById(id);
    if(!el || el.dataset.tapsWired) return;
    el.dataset.tapsWired = '1';
    el.addEventListener('click', e=>{
      const hit = e.target.closest('[data-atype]');
      if(!hit || !el.contains(hit)) return;
      runAdminAction(hit.dataset.atype, hit.dataset.avalue || '');
    });
  });
}

// Image load / error handling for banner <img>s. A failed load is retried once (a CDN that is
// briefly slow shouldn't leave a blank banner) before showing the fallback panel.
function bannerImgLoaded(img){
  const box = img.closest('.home-banner');
  if(box){ box.classList.remove('failed'); box.classList.add('loaded'); }
}
function bannerImgFailed(img){
  const box = img.closest('.home-banner');
  const url = img.getAttribute('src') || '';
  if(!img.dataset.retried && /^https?:/i.test(url)){
    img.dataset.retried = '1';
    setTimeout(()=>{ img.src = url + (url.indexOf('?') > -1 ? '&' : '?') + 'r=' + Date.now(); }, 2500);
    return;
  }
  if(box){ box.classList.remove('loaded'); box.classList.add('failed'); }
}

/* ---- Tile image persistence ----
   BUGFIX (tile pictures vanished after a few hours / never loaded): a tile's picture was only
   ever kept as a plain imgBB URL. The cache above stores that URL, not the picture itself, so
   any slow connection, a CDN hiccup or a link that later stops working left a blank tile. Two
   things were wrong:
     1) the failure fallback was an inline onerror="…innerHTML=${JSON.stringify(svg)}" — JSON
        output is full of double quotes, which end a double-quoted HTML attribute early, so
        that handler was a syntax error and NEVER ran: a failed picture left an empty box
        instead of the icon (exactly the blank "Templates" tile);
     2) nothing retried or remembered the picture.
   Now: every tile picture is fetched once, shrunk to a small JPEG/PNG data URL and kept in
   localStorage keyed by its URL, so it keeps showing even if imgBB is slow, blocked or the
   link has expired. The live URL is still tried first (it may be newer), with one retry, and
   the icon fallback is attached with a real listener instead of an inline attribute. */
const TILE_IMG_CACHE_KEY = 'riggy.tileImg.cache.v2'; // bumped: v1 entries were cached before high-quality smoothing was added
const TILE_IMG_MAX_ENTRIES = 12;
const TILE_IMG_MAX_DIM = 320; // tiles are ~80-100 css px square; 320 keeps them sharp on 3x screens
let tileImgCache = {};
try{ tileImgCache = JSON.parse(localStorage.getItem(TILE_IMG_CACHE_KEY) || '{}') || {}; }catch(e){ tileImgCache = {}; }
function saveTileImgCache(){
  try{
    localStorage.setItem(TILE_IMG_CACHE_KEY, JSON.stringify(tileImgCache));
  }catch(e){
    // Storage full: drop the oldest half and try once more, never throw into the UI.
    try{
      const keys = Object.keys(tileImgCache);
      keys.slice(0, Math.ceil(keys.length/2)).forEach(k=>delete tileImgCache[k]);
      localStorage.setItem(TILE_IMG_CACHE_KEY, JSON.stringify(tileImgCache));
    }catch(e2){}
  }
}
// Downloads the picture, shrinks it on a canvas and stores it as a data URL. Best-effort:
// a cross-origin host that doesn't send CORS headers simply can't be snapshotted (canvas
// would be tainted) — in that case nothing is stored and the live URL keeps being used.
function cacheTileImage(url){
  if(!/^https?:/i.test(url) || tileImgCache[url] || cacheTileImage._busy[url]) return;
  cacheTileImage._busy[url] = true;
  const probe = new Image();
  probe.crossOrigin = 'anonymous';
  probe.onload = ()=>{
    try{
      const scale = Math.min(1, TILE_IMG_MAX_DIM / Math.max(probe.naturalWidth, probe.naturalHeight));
      const w = Math.max(1, Math.round(probe.naturalWidth * scale));
      const h = Math.max(1, Math.round(probe.naturalHeight * scale));
      const cv = document.createElement('canvas'); cv.width = w; cv.height = h;
      const tctx = cv.getContext('2d');
      tctx.imageSmoothingEnabled = true;
      tctx.imageSmoothingQuality = 'high';
      tctx.drawImage(probe, 0, 0, w, h);
      // PNG keeps transparency for cut-out tile art; fall back to JPEG if it's huge.
      let data = cv.toDataURL('image/png');
      if(data.length > 90000) data = cv.toDataURL('image/jpeg', 0.82);
      if(data.length <= 140000){
        const keys = Object.keys(tileImgCache);
        if(keys.length >= TILE_IMG_MAX_ENTRIES) delete tileImgCache[keys[0]];
        tileImgCache[url] = data;
        saveTileImgCache();
      }
    }catch(e){ /* tainted canvas / quota — live URL still works */ }
    delete cacheTileImage._busy[url];
  };
  probe.onerror = ()=>{ delete cacheTileImage._busy[url]; };
  probe.src = url;
}
cacheTileImage._busy = {};

function tileShowFallback(thumb, iconKey){
  if(!thumb) return;
  thumb.classList.remove('tile-photo');
  thumb.innerHTML = TILE_ICONS[iconKey] || TILE_ICONS.spark;
}
// Wires load/error handling onto one tile <img>. Order tried on failure:
//   live URL -> live URL once more (cache-busted) -> locally saved copy -> icon fallback.
function wireTileImage(img, url, iconKey){
  const thumb = img.closest('.home-tile-thumb');
  let stage = 0;
  img.addEventListener('load', ()=>{
    img.classList.add('loaded');
    if(img.src === url || img.dataset.live === '1') cacheTileImage(url);
  });
  img.addEventListener('error', ()=>{
    stage++;
    if(stage === 1 && /^https?:/i.test(url)){
      setTimeout(()=>{ img.dataset.live = '1'; img.src = url + (url.indexOf('?') > -1 ? '&' : '?') + 'r=' + Date.now(); }, 1500);
    } else if(stage <= 2 && tileImgCache[url]){
      stage = 3; img.dataset.live = '0'; img.src = tileImgCache[url];
    } else {
      tileShowFallback(thumb, iconKey);
    }
  });
  img.dataset.live = '1';
  // Paint the saved copy instantly when we have one (no blank flash on a cold/slow network),
  // then let the live URL replace it if it loads.
  if(tileImgCache[url]){
    img.src = tileImgCache[url];
    img.dataset.live = '0';
    const live = new Image();
    live.onload = ()=>{ img.dataset.live = '1'; img.src = url; };
    live.onerror = ()=>{ /* keep the saved copy showing */ };
    if(/^https?:/i.test(url)) live.src = url;
  } else {
    img.src = url;
  }
  // A picture that already finished (or failed) before the listeners existed would otherwise
  // sit invisible — settle it explicitly.
  if(img.complete){
    if(img.naturalWidth > 0) img.classList.add('loaded');
    else img.dispatchEvent(new Event('error'));
  }
}
function renderHomeTiles(){
  const wrap = document.getElementById('homeTileRow');
  if(!wrap) return;
  const tiles = getTiles();
  const sig = JSON.stringify(tiles);
  if(sig === _tilesSig) return;
  _tilesSig = sig;
  wrap.innerHTML = tiles.map((t, i)=>{
    const img = safeImageUrl(t.image);
    const isPromptTile = String(t.actionValue||'') === 'keyframeInfo' || String(t.label||'').trim().toLowerCase() === 'keyframe';
    const tileLabel = isPromptTile ? 'Prompts' : (t.label||'');
    const tileIcon = isPromptTile ? 'templates' : (t.icon||'spark');
    const attrs = ` data-atype="${escapeHtml(t.actionType||'ratio')}" data-avalue="${escapeHtml(t.actionValue||'')}"`;
    const thumb = img
      ? `<div class="home-tile-thumb tile-photo" data-tile-i="${i}"><div class="home-tile-photo-card">
           <img alt="${escapeHtml(tileLabel)}" decoding="async">
         </div></div>`
      : `<div class="home-tile-thumb">${TILE_ICONS[tileIcon] || TILE_ICONS.spark}</div>`;
    return `<div class="home-tile"${attrs}>${thumb}<span>${escapeHtml(tileLabel)}</span></div>`;
  }).join('');
  // Attach picture handling AFTER the markup exists (no inline handlers, so no quoting traps).
  tiles.forEach((t, i)=>{
    const img = safeImageUrl(t.image);
    if(!img) return;
    const el = wrap.querySelector('.home-tile-thumb[data-tile-i="'+i+'"] img');
    if(el) wireTileImage(el, img, t.icon);
  });
}
