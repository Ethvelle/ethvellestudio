<script id="ethvelle-v110-instant-touch-js">
(function(){
  'use strict';
  const $=id=>document.getElementById(id);
  const clamp=(v,a,b)=>Math.max(a,Math.min(b,v));
  const clock=()=>((window.performance&&performance.now)?performance.now():Date.now());

  /* 1. Native-feeling bottom toolbar swipe: begin after a normal 6px move, never a long press. */
  const toolbarStates=new WeakMap();
  function bindToolbar(row){
    if(!row||row.dataset.evv110TouchScroll==='1')return;
    row.dataset.evv110TouchScroll='1';
    row.style.setProperty('touch-action','pan-y','important');
    const state={id:null,mode:'idle',x:0,y:0,lastX:0,lastAt:0,startScroll:0,targetScroll:0,maxScroll:0,velocity:0,moveRaf:0,glideRaf:0,suppressUntil:0};
    toolbarStates.set(row,state);
    const stopGlide=()=>{if(state.glideRaf){cancelAnimationFrame(state.glideRaf);state.glideRaf=0;}state.velocity=0;};
    const flush=()=>{
      state.moveRaf=0;
      if(state.mode==='drag')row.scrollLeft=clamp(state.targetScroll,0,state.maxScroll);
    };
    const glide=()=>{
      if(Math.abs(state.velocity)<0.07)return;
      let previous=clock();
      const tick=()=>{
        const t=clock(),dt=Math.min(32,Math.max(1,t-previous));previous=t;
        const next=clamp(row.scrollLeft+state.velocity*dt,0,state.maxScroll);
        row.scrollLeft=next;
        state.velocity*=Math.pow(0.925,dt/16.667);
        if(next<=0||next>=state.maxScroll||Math.abs(state.velocity)<0.035){state.glideRaf=0;state.velocity=0;return;}
        state.glideRaf=requestAnimationFrame(tick);
      };
      state.glideRaf=requestAnimationFrame(tick);
    };
    row.addEventListener('pointerdown',e=>{
      if(e.isPrimary===false||(e.pointerType==='mouse'&&e.button!==0))return;
      // Geometry is stable for the duration of one swipe. Read the scroll range once here,
      // instead of asking layout for scrollWidth/clientWidth on every RAF flush and glide tick.
      state.maxScroll=Math.max(0,row.scrollWidth-row.clientWidth);
      if(state.maxScroll<=2)return;
      stopGlide();
      state.id=e.pointerId;state.mode='pending';state.x=state.lastX=e.clientX;state.y=e.clientY;
      state.lastAt=clock();state.startScroll=row.scrollLeft;state.targetScroll=row.scrollLeft;state.velocity=0;
    },{capture:true,passive:true});
    row.addEventListener('pointermove',e=>{
      if(state.id!==e.pointerId)return;
      const dx=e.clientX-state.x,dy=e.clientY-state.y;
      if(state.mode==='pending'){
        if(Math.max(Math.abs(dx),Math.abs(dy))<6)return;
        if(Math.abs(dy)>Math.abs(dx)*1.12){
          // Keep the pointer identity until pointerup so the generated click from a vertical
          // swipe that starts on a tool cannot activate that tool. Clearing state.id here
          // made pointerup invisible to this controller and let the synthetic click through.
          state.mode='cancelled';
          state.suppressUntil=clock()+500;
          return;
        }
        if(Math.abs(dx)<6)return;
        state.mode='drag';
        state.suppressUntil=clock()+260;
        try{row.setPointerCapture(e.pointerId);}catch(_e){}
      }
      if(state.mode!=='drag')return;
      const t=clock(),dt=Math.max(1,t-state.lastAt),deltaX=e.clientX-state.lastX;
      const instant=-deltaX/dt;
      state.velocity=clamp(state.velocity*.32+instant*.68,-2.4,2.4);
      state.lastX=e.clientX;state.lastAt=t;
      state.targetScroll=state.startScroll-dx;
      if(!state.moveRaf)state.moveRaf=requestAnimationFrame(flush);
      if(e.cancelable)e.preventDefault();
      e.stopPropagation();
    },{passive:false});
    row.addEventListener('pointerup',e=>{
      if(state.id!==e.pointerId)return;
      const dragged=state.mode==='drag';
      if(state.moveRaf){cancelAnimationFrame(state.moveRaf);state.moveRaf=0;flush();}
      state.id=null;state.mode='idle';
      if(dragged){state.suppressUntil=clock()+260;try{row.releasePointerCapture(e.pointerId);}catch(_e){}glide();}
    },{passive:true});
    row.addEventListener('pointercancel',e=>{
      if(state.id!==e.pointerId)return;
      state.id=null;state.mode='idle';state.velocity=0;state.suppressUntil=0;
      if(state.moveRaf){cancelAnimationFrame(state.moveRaf);state.moveRaf=0;}
    },{passive:true});
    row.addEventListener('click',e=>{
      if(!state.suppressUntil)return;
      if(clock()>state.suppressUntil){state.suppressUntil=0;return;}
      state.suppressUntil=0;e.preventDefault();e.stopPropagation();e.stopImmediatePropagation();
    },true);
  }
  ['bottomToolsDefault','bottomToolsClip','bottomToolsKeyframe'].forEach(id=>bindToolbar($(id)));

  /* 2. Hidden Transform ruler: require a grab on its centre marker, not a random rail tap. */
  document.addEventListener('pointerdown',e=>{
    const target=e.target;
    const ruler=target&&target.closest?target.closest('#screen-editor #toolPanel .evtr-ruler'):null;
    if(ruler){
      const r=ruler.getBoundingClientRect();
      // The transform ruler has a fixed blue centre marker; only that marker can start a drag.
      if(Math.abs(e.clientX-(r.left+r.width/2))>13||Math.abs(e.clientY-(r.top+r.height/2))>Math.min(24,r.height/2+1)){
        if(e.cancelable)e.preventDefault();e.stopPropagation();e.stopImmediatePropagation();
      }
    }
  },{capture:true,passive:false});

  /* 3. Finish clip-row scrolling with gentle inertia after the direct RAF handoff. */
  let clipGlideRaf=0,clipGlideGeneration=0;
  function stopClipGlide(){clipGlideGeneration++;if(clipGlideRaf){cancelAnimationFrame(clipGlideRaf);clipGlideRaf=0;}}
  function startClipGlide(){
    const generation=++clipGlideGeneration;
    const motion=window.__ethvelleV110ClipScrollMotion;
    if(!motion||!motion.scroller||!motion.scroller.isConnected)return;
    motion.active=false;
    let velocity=clamp(Number(motion.velocity)||0,-2.4,2.4);
    const scroller=motion.scroller;
    // Flush the last finger position first; the original helper may have one RAF queued.
    requestAnimationFrame(()=>{
      if(generation!==clipGlideGeneration)return;
      if(Number.isFinite(motion.targetTop))scroller.scrollTop=clamp(motion.targetTop,0,Math.max(0,scroller.scrollHeight-scroller.clientHeight));
      if(Math.abs(velocity)<0.07)return;
      let previous=clock();
      const tick=()=>{
        const t=clock(),dt=Math.min(32,Math.max(1,t-previous));previous=t;
        const max=Math.max(0,scroller.scrollHeight-scroller.clientHeight);
        const next=clamp(scroller.scrollTop+velocity*dt,0,max);
        scroller.scrollTop=next;
        velocity*=Math.pow(0.93,dt/16.667);
        if(next<=0||next>=max||Math.abs(velocity)<0.035){clipGlideRaf=0;return;}
        clipGlideRaf=requestAnimationFrame(tick);
      };
      clipGlideRaf=requestAnimationFrame(tick);
    });
  }
  document.addEventListener('pointerdown',e=>{
    if(e.target?.closest?.('#screen-editor #timelineScroll')){
      stopClipGlide();
      const m=window.__ethvelleV110ClipScrollMotion;if(m)m.active=false;
    }
  },{capture:true,passive:true});
  document.addEventListener('pointerup',()=>{
    const m=window.__ethvelleV110ClipScrollMotion;
    if(m&&m.active)startClipGlide();
  },{capture:true,passive:true});
  document.addEventListener('pointercancel',()=>{
    const m=window.__ethvelleV110ClipScrollMotion;if(m)m.active=false;
    stopClipGlide();
  },{capture:true,passive:true});

  /* 4. Selection-outline X: block the legacy pointer/click delete handlers, then deselect. */
  const closeButton=$('sbDelete');
  if(closeButton){
    closeButton.title='Deselect layer';closeButton.setAttribute('aria-label','Deselect layer');
    closeButton.setAttribute('role','button');closeButton.tabIndex=0;
    closeButton.addEventListener('pointerdown',e=>{e.stopPropagation();e.stopImmediatePropagation();},true);
    const deselect=e=>{
      e.preventDefault();e.stopPropagation();e.stopImmediatePropagation();
      if(typeof window.deselectAllClips==='function')window.deselectAllClips();
      else if(typeof deselectAllClips==='function')deselectAllClips();
    };
    closeButton.addEventListener('click',deselect,true);
    closeButton.addEventListener('keydown',e=>{if(e.key==='Enter'||e.key===' '){deselect(e);}},true);
  }
})();
</script>

<script id="ethvelle-v110-keyframe-toolbar-options-js">
(function(){
  'use strict';
  const clipRow=document.getElementById('bottomToolsClip');
  const keyRow=document.getElementById('bottomToolsKeyframe');
  if(!clipRow||!keyRow)return;
  const duplicateControls=new Set([
    'clipMediaAccessTool','evcaFrameCaptureTool-clip','evv72SaveFrame-clip','evv72AddFrame-clip'
  ]);
  let lastSignature='';

  function removeMirrors(){
    keyRow.querySelectorAll('.evv-kf-mirror').forEach(el=>el.remove());
    lastSignature='';
  }
  function sourceTools(){
    return Array.from(clipRow.children).filter(el=>{
      if(!el.classList?.contains('tool-item'))return false;
      if(duplicateControls.has(el.id))return false;
      if(el.id==='clipDeleteCharacterTool')return false; // No whole-character Delete in keyframe mode.
      const label=(el.querySelector('span:last-child')?.textContent||'').trim().toLowerCase();
      if(label==='delete')return false;                 // No normal selected-layer/clip Delete in keyframe mode.
      if(label==='done')return false;
      return getComputedStyle(el).display!=='none';
    });
  }
  function buildMirrors(){
    const selected=typeof selectedKfIdx!=='undefined'&&selectedKfIdx!==null;
    const ref=typeof selectedClip==='undefined'?'':String(selectedClip||'');
    const isCharacterKeyframe=selected&&(ref==='main'||/^seg\d+$/.test(ref));
    if(!isCharacterKeyframe){removeMirrors();return;}

    // Recompute the normal character-row visibility using the app's existing sync logic,
    // then immediately restore the selected-keyframe row before the browser paints.
    if(typeof oldSync==='function'){
      const savedKfIndex=selectedKfIdx;
      selectedKfIdx=null;
      try{oldSync.apply(window,[]);}finally{selectedKfIdx=savedKfIndex;}
      const defaultRow=document.getElementById('bottomToolsDefault');
      [defaultRow,clipRow].forEach(row=>row?.style.setProperty('display','none','important'));
      keyRow.style.setProperty('display','flex','important');
    }

    const sources=sourceTools();
    // Keep the existing Add model action available while the two Delete actions are hidden.
    const addCharacterTool=document.querySelector('#bottomToolsDefault [data-tool="model"]');
    if(addCharacterTool)sources.unshift(addCharacterTool);
    const signature=sources.map(el=>(el.id||'')+':'+(el.textContent||'').trim()).join('|');
    if(signature===lastSignature&&keyRow.querySelectorAll('.evv-kf-mirror').length===sources.length)return;
    removeMirrors();
    sources.forEach(source=>{
      const proxy=source.cloneNode(true);
      proxy.removeAttribute('id');
      proxy.classList.add('evv-kf-mirror');
      proxy.setAttribute('role','button');
      proxy.setAttribute('tabindex','0');
      proxy.setAttribute('aria-label',(source.getAttribute('aria-label')||(source.textContent||'').trim()||'Character tool').trim());
      Array.from(proxy.attributes||[]).forEach(attr=>{if(/^on/i.test(attr.name))proxy.removeAttribute(attr.name);});
      proxy.querySelectorAll('[id]').forEach(node=>node.removeAttribute('id'));
      proxy.querySelectorAll('*').forEach(node=>{
        Array.from(node.attributes||[]).forEach(attr=>{if(/^on/i.test(attr.name))node.removeAttribute(attr.name);});
      });
      proxy.style.setProperty('display','flex','important');
      const activate=e=>{
        if(e){e.preventDefault();e.stopPropagation();}
        source.click();
      };
      proxy.addEventListener('click',activate);
      proxy.addEventListener('keydown',e=>{if(e.key==='Enter'||e.key===' '){activate(e);}});
      keyRow.appendChild(proxy);
    });
    lastSignature=signature;
  }

  const oldSync=window.syncBottomToolsMode;
  if(typeof oldSync==='function'&&!oldSync.__ethvelleV110KeyframeTools){
    const wrapped=function(){
      const result=oldSync.apply(this,arguments);
      buildMirrors();
      return result;
    };
    wrapped.__ethvelleV110KeyframeTools=true;
    window.syncBottomToolsMode=wrapped;
  }
  // This is an initialization-only fallback; later keyframe selections flow through the wrapper above.
  buildMirrors();
})();
</script>

<script id="ethvelle-v114-requested-ui-updates-js">
(function(){
  'use strict';
  const panel=document.getElementById('toolPanel');
  const editor=document.getElementById('screen-editor');
  if(!panel||!editor)return;

  function fitAssetsPanel(){
    if(!panel.classList.contains('open')||!panel.classList.contains('ev-v114-assets-fit'))return;
    const canvas=editor.querySelector('.canvas-wrap');
    if(!canvas)return;
    const screenRect=editor.getBoundingClientRect();
    const canvasRect=canvas.getBoundingClientRect();
    const top=Math.max(0,Math.min(screenRect.height-150,canvasRect.bottom-screenRect.top+1));
    const height=Math.max(150,screenRect.height-top);
    panel.style.setProperty('--ev-v114-assets-top',top+'px');
    panel.style.setProperty('--ev-v114-assets-height',height+'px');
  }

  const originalOpen=window.openPanel;
  if(typeof originalOpen==='function'&&!originalOpen.__evV114AssetsFit){
    const wrappedOpen=function(key){
      const result=originalOpen.apply(this,arguments);
      panel.classList.toggle('ev-v114-assets-fit',key==='assets');
      if(key==='assets'){
        requestAnimationFrame(()=>requestAnimationFrame(fitAssetsPanel));
      }else{
        panel.style.removeProperty('--ev-v114-assets-top');
        panel.style.removeProperty('--ev-v114-assets-height');
      }
      return result;
    };
    wrappedOpen.__evV114AssetsFit=true;
    window.openPanel=wrappedOpen;
  }

  const originalClose=window.closePanel;
  if(typeof originalClose==='function'&&!originalClose.__evV114AssetsFit){
    const wrappedClose=function(){
      const result=originalClose.apply(this,arguments);
      panel.classList.remove('ev-v114-assets-fit');
      panel.style.removeProperty('--ev-v114-assets-top');
      panel.style.removeProperty('--ev-v114-assets-height');
      return result;
    };
    wrappedClose.__evV114AssetsFit=true;
    window.closePanel=wrappedClose;
  }

  window.addEventListener('resize',()=>requestAnimationFrame(fitAssetsPanel),{passive:true});
  if(window.visualViewport){
    window.visualViewport.addEventListener('resize',()=>requestAnimationFrame(fitAssetsPanel),{passive:true});
  }
})();
</script>
