    if(clip.type==='video'){
      // Use the cached frame at the source timestamp shown FIRST by this timeline clip.
      // The video may be reversed or trimmed, so the first timeline frame is not always t=0.
      const sourceFrames=window.__evSourceFrames;
      if(sourceFrames&&typeof sourceFrames.setFor==='function'&&typeof sourceFrames.nearest==='function'){
        try{
          const set=sourceFrames.setFor(clip);
          if(set&&set.times&&set.times.length){
            const span=Math.max(0,(Number(clip.trimEnd)-Number(clip.trimStart))*(Number(clip.speed)||1));
            let firstSourceTime=Math.max(0,Number(clip.sourceOffset)||0);
            if(clip.reversePlayback)firstSourceTime+=span;
            const duration=Math.max(0,Number(clip.sourceDuration)||Number(set.duration)||0);
            if(duration>0)firstSourceTime=Math.min(firstSourceTime,Math.max(0,duration-.04));
            const cached=sourceFrames.nearest(set,firstSourceTime);
            if(cached)return String(cached);
          }
        }catch(_e){}
      }
      const id=String(clip.id);
      const node=Array.from(document.querySelectorAll('#timelineInner .overlay-clip')).find(el=>String(el.dataset.overlayId||'')===id);
      // The strip can be virtualized while scrolled. Accept only tile zero, never an arbitrary
      // visible middle frame, so the tracking card always represents the clip's first frame.
      const tile=node&&node.querySelector('.ev-video-thumb-tile[data-ev-thumb-index="0"].ev-video-thumb-ready');
      if(tile)return String(tile.dataset.evThumbUrl||unwrapTrackingThumb(getComputedStyle(tile).backgroundImage)||'');
      // A supplied poster is only a fallback; prefer a real cached first frame whenever present.
      return String(clip.poster||clip.thumbnail||'');
    }
    if(clip.type==='text'){
      const label=String(clip.textContent||clip.name||'Text').slice(0,36);
      const svg='<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 320 180"><rect width="320" height="180" fill="#242735"/><text x="160" y="98" text-anchor="middle" font-family="Arial" font-size="24" font-weight="700" fill="#fff">'+label.replace(/[&<>"]/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c]))+'</text></svg>';
      return 'data:image/svg+xml;charset=UTF-8,'+encodeURIComponent(svg);
    }
    return '';
  }

  function trackingAspectRatio(clip){
    if(!clip)return 16/9;
    let el=null;
    try{
      if(clip.id!=null&&typeof overlayMediaEls!=='undefined'&&overlayMediaEls)el=overlayMediaEls[clip.id]||null;
      if(!el&&String(clip.id||'').startsWith('bg'))el=document.getElementById('bgImgLayer');
    }catch(_e){}
    const w=Number(clip.sourceWidth)||Number(el&&(el.videoWidth||el.naturalWidth||el.width))||Number(clip.naturalWidth)||Number(clip.width)||0;
    const h=Number(clip.sourceHeight)||Number(el&&(el.videoHeight||el.naturalHeight||el.height))||Number(clip.naturalHeight)||Number(clip.height)||0;
    const explicit=Number(clip.aspectRatio||clip.ratio)||0;
    const ratio=(w>0&&h>0)?w/h:explicit>0?explicit:16/9;
    return Number.isFinite(ratio)&&ratio>0?Math.max(.18,Math.min(5.5,ratio)):16/9;
  }
  function trackingThumbBox(ratio){
    const r=Math.max(.18,Math.min(5.5,Number(ratio)||16/9));
    const h=Math.min(54,118/r),w=h*r;
    return {width:Math.round(w),height:Math.round(h)};

  function getTrackingCandidates(follower){
    /* v339: a clip can follow a clip from ANY section — Background, Layer/Media, Text, Effects and
       the Animation (character) rows — as long as the two overlap in time. */
    const followerKey=trackingFollowerKey(follower);
    const anims=allAnimClips();
    const followerAnim=anims.find(e=>e.clip===follower)||null;
    const chars=Array.isArray(characters)?characters:[];
    const fxChar=o=>(o&&o.ethvelleEffectClip&&o.ethvelleEffectKind==='character')?(chars.find(c=>c&&String(c.id)===String(o.ethvelleEffectCharId))||null):null;
    const followerFxChar=fxChar(follower);
    const backgroundsList=Array.isArray(backgrounds)?backgrounds:[];
    const overlaysList=Array.isArray(overlays)?overlays:[];
    const byTimeline=(a,b)=>(Number(a.trimStart)||0)-(Number(b.trimStart)||0)||String(a.id).localeCompare(String(b.id));
    const isFx=o=>!!(o&&o.ethvelleEffectClip);
    const notSelf=o=>o&&o.id!=null&&('ov:'+String(o.id))!==followerKey;
    const bgItems=backgroundsList
      .filter(b=>b&&b.id!=null&&b!==follower&&trackingRangesOverlap(b,follower))
      .sort(byTimeline)
      .map((b,i)=>({value:'bg:'+String(b.id),kind:'background',thumb:trackingThumbForBackground(b),ratio:trackingAspectRatio(b),label:'Background row clip '+(i+1),clipId:String(b.id)}));
    const overlayClips=overlaysList
      .filter(o=>notSelf(o)&&['image','video'].includes(o.type)&&!isFx(o)&&trackingRangesOverlap(o,follower))
      .sort(byTimeline);
    const textClips=overlaysList
      .filter(o=>notSelf(o)&&o.type==='text'&&trackingRangesOverlap(o,follower))
      .sort(byTimeline);
    const fxClips=overlaysList
      .filter(o=>notSelf(o)&&isFx(o)&&trackingRangesOverlap(o,follower)&&!(followerAnim&&fxChar(o)===followerAnim.ch))
      .sort(byTimeline);
    const animClips=anims
      .filter(e=>e.clip!==follower&&!(followerAnim&&e.ch===followerAnim.ch)&&!(followerFxChar&&e.ch===followerFxChar)&&trackingRangesOverlap(e.clip,follower))
      .sort((a,b)=>(Number(a.clip.trimStart)||0)-(Number(b.clip.trimStart)||0)||a.key.localeCompare(b.key));
    const overlayItems=overlayClips.map((o,i)=>({value:'ov:'+String(o.id),kind:o.type,thumb:trackingThumbForOverlay(o),ratio:trackingAspectRatio(o),label:'Layer row clip '+(i+1),clipId:String(o.id)}));
    const textItems=textClips.map((o,i)=>({value:'ov:'+String(o.id),kind:o.type,thumb:trackingThumbForOverlay(o),ratio:trackingAspectRatio(o),label:'Text row clip '+(i+1),clipId:String(o.id)}));
    const fxItems=fxClips.map((o,i)=>({value:'ov:'+String(o.id),kind:'effect',thumb:trackingThumbForOverlay(o),ratio:trackingAspectRatio(o),label:(o.ethvelleEffectKind==='character'?'Character effect ':'Effect ')+(i+1),clipId:String(o.id)}));
    const animItems=animClips.map((e,i)=>({value:e.key,kind:'animation',thumb:trackingThumbForAnim(e.ch,e.clip),ratio:16/9,label:'Animation row clip '+(i+1),clipId:String(e.clip.evTrackId)}));
    return [
      {label:'Background',items:bgItems},
      {label:'Layer',items:overlayItems},
      {label:'Text',items:textItems},
      {label:'Animation',items:animItems},
      {label:'Effects',items:fxItems}
    ].map(group=>({...group,items:group.items.filter(item=>!!item.thumb&&!trackingDependsOn(item.value,followerKey))})).filter(group=>group.items.length);
  }
  function getTrackingSourceTransform(ref,time,seen){
    const key=String(ref||''),t=Number.isFinite(Number(time))?Number(time):trackingNow();
    const visited=seen||new Set();
    if(key.indexOf('bg:')===0){
      const id=key.slice(3),clip=(Array.isArray(backgrounds)?backgrounds:[]).find(b=>b&&String(b.id)===id);
      if(!clip)return null;
      const keyed=typeof window.ethvelleGetObjectKeyframeValue==='function'?window.ethvelleGetObjectKeyframeValue(clip,'transform',t):null;
      return trackingSnapshot(keyed||clip.transform||{});
    }
    if(key.indexOf('char:')===0){
      const id=key.slice(5),idx=(Array.isArray(characters)?characters:[]).findIndex((ch,i)=>charKey(ch,i)===id);
      const ch=idx>=0?characters[idx]:null;
      if(!ch)return null;
      // modelTransform is the live evaluated transform: renderFrame/animateAllCharacters
      // update it from the character's own clip keyframes before overlay media is painted.
      return trackingSnapshot(ch.modelTransform||ch.restTransform||{});
    }
    if(key.indexOf('anim:')===0){
      // An animation clip drives its character's live transform (already tracked, if that clip
      // itself follows something) — read it exactly like the legacy 'char:' source below.
      const hit=allAnimClips().find(e=>e.key===key);
      if(!hit)return null;
      return trackingSnapshot(hit.ch.modelTransform||hit.ch.restTransform||{});
    }
    if(key.indexOf('ov:')===0){
      const id=key.slice(3),clip=(Array.isArray(overlays)?overlays:[]).find(o=>o&&String(o.id)===id&&['image','video','text','shape','color'].includes(o.type));
      if(!clip)return null;
      const marker='ov:'+id;
      const keyed=typeof window.ethvelleGetObjectKeyframeValue==='function'?window.ethvelleGetObjectKeyframeValue(clip,'transform',t):null;
      let tr=trackingSnapshot(keyed||clip.transform||{});
      if(visited.has(marker))return tr; // break accidental A→B→A tracking loops safely
      visited.add(marker);
      const cfg=clip.clipTracking;
      if(cfg&&cfg.enabled&&cfg.targetId&&cfg.targetBaseTransform){
        const source=getTrackingSourceTransform(cfg.targetId,t,visited);
        if(source)tr=combineTrackingDelta(tr,source,cfg.targetBaseTransform,cfg);
      }
      visited.delete(marker);
      const motion=typeof window.getEthvelleClipAnimationState==='function'?window.getEthvelleClipAnimationState(clip,t):null;
      if(motion){
        tr.offsetX+=safeNum(motion.x,0);tr.offsetY+=safeNum(motion.y,0);
        tr.rotation+=safeNum(motion.rotation,0);tr.scale*=Math.max(.0001,safeNum(motion.scale,1));
      }
      // v339: a character-effect clip is drawn relative to its character, so its on-screen motion
      // is the character's own transform plus the effect's offset/scale/rotation.
      if(clip.ethvelleEffectClip&&clip.ethvelleEffectKind==='character'){
        const fxCh=(Array.isArray(characters)?characters:[]).find(c=>c&&String(c.id)===String(clip.ethvelleEffectCharId));
        if(fxCh&&fxCh.modelTransform){
          const cm=trackingSnapshot(fxCh.modelTransform);
          tr={scale:tr.scale*cm.scale,offsetX:tr.offsetX+cm.offsetX,offsetY:tr.offsetY+cm.offsetY,rotation:tr.rotation+cm.rotation};
        }
      }
      return tr;
    }
    return null;
  }
  function ensureTrackingConfig(ov){
    if(!ov)return null;
    if(!ov.clipTracking||typeof ov.clipTracking!=='object')ov.clipTracking={enabled:false,targetId:'',targetBaseTransform:null,offsetX:0,offsetY:0,followScale:true,followRotation:true};
    const cfg=ov.clipTracking;
    if(typeof cfg.enabled!=='boolean')cfg.enabled=false;
    if(typeof cfg.targetId!=='string')cfg.targetId='';
    if(!Number.isFinite(Number(cfg.offsetX)))cfg.offsetX=0;
    if(!Number.isFinite(Number(cfg.offsetY)))cfg.offsetY=0;
    if(typeof cfg.followScale!=='boolean')cfg.followScale=true;
    if(typeof cfg.followRotation!=='boolean')cfg.followRotation=true;
    if(cfg.targetBaseTransform&&typeof cfg.targetBaseTransform==='object')cfg.targetBaseTransform=trackingSnapshot(cfg.targetBaseTransform);
    return cfg;
  }

    const base=trackingSnapshot(keyed||ov&&ov.transform||{});

    const hipY=ch.bones&&ch.bones.hip?safeNum(ch.bones.hip.y,safeNum(ch.hipRestY)):safeNum(ch.hipRestY);
    const hipShift=Number.isFinite(Number(ch.hipRestY))?hipY-safeNum(ch.hipRestY):0;
    const dr=safeNum(mt.rotation)-safeNum(rest.rotation);
    return {
      scale:base.scale*(legacy.followScale===false?1:currentScale/baseScale),
      offsetX:base.offsetX+dx+safeNum(legacy.offsetX),
      offsetY:base.offsetY+dy+hipShift+safeNum(legacy.offsetY),
      rotation:base.rotation+(legacy.followRotation?dr:0)
    };
  };

  window.ethvelleApplyCharacterTracking=function(time){
    const chars=Array.isArray(characters)?characters:[];
    const t=Number.isFinite(Number(time))?Number(time):trackingNow();
    const wants=c=>!!(c&&c.clipTracking&&c.clipTracking.enabled&&c.clipTracking.targetId);
    const jobs=[];
    chars.forEach((ch,idx)=>{
      if(!ch||!ch.modelTransform)return;
      ch.__evTrackDelta=null;ch.__evTrackJob=null;
      const segs=Array.isArray(ch.segments)?ch.segments:[];
      if(!(wants(ch.clip)||segs.some(wants)))return;
      let entry=null;
      try{entry=stickmanClipAt(t,ch.clip||null,segs);}catch(_e){}
      const clip=entry&&entry.clip,cfg=clip&&clip.clipTracking;
      if(!wants(clip)||!cfg.targetBaseTransform)return;
      if(!(t>=Number(clip.trimStart)&&t<Number(clip.trimEnd)))return;
      const job={ch,idx,cfg,clip,base:trackingSnapshot(ch.modelTransform)};
      ch.__evTrackJob=job;jobs.push(job);
    });
    if(!jobs.length)return;
    // Several followers can depend on one another (A follows B follows C): re-evaluate from the
    // untouched base a few times so every link settles. A single follower needs only one pass.
    const rounds=jobs.length>1?3:1;
    for(let r=0;r<rounds;r++){
      jobs.forEach(job=>{
        const seen=new Set(['anim:'+charKey(job.ch,job.idx)]);
        const source=getTrackingSourceTransform(job.cfg.targetId,t,seen);
        if(!source)return;
        applyCharJob(job,source);
      });
    }
  };
  function applyCharJob(job,source){
    const next=combineTrackingDelta(job.base,source,job.cfg.targetBaseTransform,job.cfg);

    job.ch.__evTrackDelta={ds:next.scale/Math.max(.0001,job.base.scale),dx:next.offsetX-job.base.offsetX,dy:next.offsetY-job.base.offsetY,dr:next.rotation-job.base.rotation};
  }
