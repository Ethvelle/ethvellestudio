  // V285: MediaRecorder writes WebM/Matroska without a Duration or a seek index (Cues). Chrome then reports
  // duration=Infinity and the app's "prime" seek (currentTime=1e7) can wedge the <video> in a never-ending seek:
  // the clip stays hidden / frozen while its audio is fine. This rewrites only the container header
  // (adds Duration + Cues, gives Segment/Cluster real sizes) and reuses the encoded frames untouched.
  // Any unexpected structure returns null so the caller keeps the original blob.
  function evltRemuxSeekable(d){
    const L1=[0x1F43B675,0x1C53BB6B,0x1549A966,0x1654AE6B,0x114D9B74,0x1254C367,0x1043A770,0x1941A469];
    const rd=(p,marker)=>{
      if(p>=d.length)throw new Error('eof');
      const b=d[p];let n=1,m=0x80;while(n<=8&&!(b&m)){n++;m>>=1;}
      if(n>8||p+n>d.length)throw new Error('vint');
      let v=marker?b:(b&(m-1));let all=(b&(m-1))===(m-1);
      for(let i=1;i<n;i++){v=v*256+d[p+i];if(d[p+i]!==0xFF)all=false;}
      return {n:n,v:v,unk:!marker&&all};
    };
    const uint=(s,e)=>{let v=0;for(let i=s;i<e;i++)v=v*256+d[i];return v;};
    const vsize=n=>n<127?1:n<16383?2:n<2097151?3:n<268435455?4:5;
    const wsize=(n,len)=>{const a=new Uint8Array(len);let v=n;for(let i=len-1;i>=0;i--){a[i]=v%256;v=Math.floor(v/256);}a[0]|=(0x80>>(len-1));return a;};
    const concat=arrs=>{let t=0;arrs.forEach(a=>t+=a.length);const o=new Uint8Array(t);let q=0;arrs.forEach(a=>{o.set(a,q);q+=a.length;});return o;};
    const idBytes=id=>{const a=[];let v=id;while(v>0){a.unshift(v%256);v=Math.floor(v/256);}return new Uint8Array(a);};
    const el=(id,payload)=>concat([idBytes(id),wsize(payload.length,vsize(payload.length)),payload]);
    // EBML header
    const h=rd(0,true);if(h.v!==0x1A45DFA3)return null;
    const hs=rd(h.n,false);const ebmlEnd=h.n+hs.n+hs.v;
    const sg=rd(ebmlEnd,true);if(sg.v!==0x18538067)return null;
    const sgs=rd(ebmlEnd+sg.n,false);
    const S=ebmlEnd+sg.n+sgs.n;const segEnd=sgs.unk?d.length:Math.min(d.length,S+sgs.v);
    let info=null,tracks=null;const others=[];const clusters=[];
    let pos=S;
    while(pos<segEnd){
      const id=rd(pos,true),sz=rd(pos+id.n,false);const ds=pos+id.n+sz.n;
      if(id.v===0x1F43B675){
        let end;
        if(!sz.unk)end=ds+sz.v;
        else{let p=ds;while(p<segEnd){const c=rd(p,true);if(L1.indexOf(c.v)>=0)break;const cs=rd(p+c.n,false);if(cs.unk)throw new Error('unk child');p+=c.n+cs.n+cs.v;}end=p;}
        if(end>segEnd)end=segEnd;
        clusters.push({hs:pos,ds:ds,end:end,unk:sz.unk,hdrLen:id.n+sz.n,szLen:sz.n});
        pos=end;continue;
      }
      if(sz.unk)throw new Error('unk l1');
      const end=ds+sz.v;if(end>segEnd)return null;
      if(id.v===0x1C53BB6B)return null; // already has Cues
      if(id.v===0x1549A966)info={ds:ds,end:end};
      else if(id.v===0x1654AE6B)tracks={ds:ds,end:end,hs:pos};
      else if(id.v!==0x114D9B74&&id.v!==0xEC)others.push({hs:pos,end:end});
      pos=end;
    }
    if(!info||!tracks||!clusters.length)return null;
    // Info
    let scale=1000000;const infoKids=[];
    {let p=info.ds;while(p<info.end){const c=rd(p,true),cs=rd(p+c.n,false);const cds=p+c.n+cs.n,cend=cds+cs.v;

      if(c.v!==0x4489&&c.v!==0xEC)infoKids.push(d.subarray(p,cend));
      p=cend;}}

    {let p=tracks.ds;while(p<tracks.end){const c=rd(p,true),cs=rd(p+c.n,false);const cds=p+c.n+cs.n,cend=cds+cs.v;

      while(p<cl.end){const c=rd(p,true),cs=rd(p+c.n,false);const cds=p+c.n+cs.n,cend=cds+cs.v;

    cues.sort((a,b)=>a.t-b.t);
    const frameDur=vCount>1?Math.round((vLast-vFirst)/(vCount-1)):Math.round(33*1000000/scale);
    const durUnits=Math.max(maxTs+Math.round(20*1000000/scale),vLast+frameDur);
    // Duration element (float64)
    const dur=new Uint8Array(8);new DataView(dur.buffer).setFloat64(0,durUnits,false);
    infoKids.push(el(0x4489,dur));
    const infoEl=el(0x1549A966,concat(infoKids));
    const tracksEl=d.subarray(tracks.hs,tracks.end);
    const otherEls=others.map(o=>d.subarray(o.hs,o.end));
    // Cues (fixed-width fields => size known before offsets)
    const cueLen=19,cuesContent=cues.length*cueLen;
    const cuesHdr=4+vsize(cuesContent);
    let pre=infoEl.length+tracksEl.length+cuesHdr+cuesContent;otherEls.forEach(o=>pre+=o.length);
    // Cluster layout
    const offs=[];let cur=pre;
    clusters.forEach(cl=>{offs.push(cur);cur+=cl.hdrLen+(cl.end-cl.ds);});
    const segSize=cur;
    const cueBytes=new Uint8Array(cuesContent);let q=0;
    cues.forEach(c=>{
      cueBytes[q++]=0xBB;cueBytes[q++]=0x80|17;
      cueBytes[q++]=0xB3;cueBytes[q++]=0x84;cueBytes[q++]=(c.t>>>24)&255;cueBytes[q++]=(c.t>>>16)&255;cueBytes[q++]=(c.t>>>8)&255;cueBytes[q++]=c.t&255;
      cueBytes[q++]=0xB7;cueBytes[q++]=0x80|9;
      cueBytes[q++]=0xF7;cueBytes[q++]=0x81;cueBytes[q++]=vtrack&255;
      const o=offs[c.ci];
      cueBytes[q++]=0xF1;cueBytes[q++]=0x84;cueBytes[q++]=(o>>>24)&255;cueBytes[q++]=(o>>>16)&255;cueBytes[q++]=(o>>>8)&255;cueBytes[q++]=o&255;
    });
    const cuesEl=concat([idBytes(0x1C53BB6B),wsize(cuesContent,vsize(cuesContent)),cueBytes]);
    // Output parts: small typed arrays + original byte ranges (copied by the Blob constructor, not by us)
    const parts=[d.subarray(0,ebmlEnd),concat([idBytes(0x18538067),wsize(segSize,8)]),infoEl,tracksEl].concat(otherEls).concat([cuesEl]);
    const ranges=[];
    clusters.forEach(cl=>{
      const hdr=concat([d.subarray(cl.hs,cl.hs+(cl.hdrLen-cl.szLen)),wsize(cl.end-cl.ds,8)]);
      // header keeps its 12-byte layout only when the original size field was 8 bytes; otherwise grow it and rebase offsets
      ranges.push({hdr:hdr,from:cl.ds,to:cl.end});
    });
    return {parts:parts,ranges:ranges,segSize:segSize,expectedHdr:clusters.map(cl=>cl.hdrLen)};
  }

  function createEnhancePipeline(sourceWidth,sourceHeight,outWidth,outHeight,pipeOpts){
    const sw=Math.max(2,Math.round(sourceWidth||2)),sh=Math.max(2,Math.round(sourceHeight||2));
    const ow=Math.max(2,Math.round(outWidth||sw)),oh=Math.max(2,Math.round(outHeight||sh));

    const stageCtx=stage.getContext('2d',{alpha:false,desynchronized:true});

    const vertexSource='attribute vec2 a_position; varying vec2 v_uv; void main(){v_uv=(a_position+1.0)*0.5;gl_Position=vec4(a_position,0.0,1.0);}';
    const fragmentSource=`
      precision highp float;

      uniform vec2 u_sourceSize;
