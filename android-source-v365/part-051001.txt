  const $=id=>document.getElementById(id);

  /* ----- Transform sheet fit ----- */
  function fitTransformPanelV98(){
    const editor=$('screen-editor'),panel=$('toolPanel');
    const transport=editor&&editor.querySelector('.transport-row');
    if(!editor||!panel||!transport||!panel.classList.contains('open'))return;
    const er=editor.getBoundingClientRect(),rr=transport.getBoundingClientRect();
    const vv=window.visualViewport;
    const visibleBottom=vv?Math.min(er.bottom,vv.offsetTop+vv.height):Math.min(er.bottom,window.innerHeight||er.bottom);
    const bottomGap=8;
    const available=Math.max(150,Math.floor(visibleBottom-bottomGap-(rr.bottom+5)));
    panel.classList.add('ev-v89-fit-panel');
    panel.style.setProperty('--ev-v98-panel-max-height',available+'px');
    const header=panel.querySelector('.tool-panel-header');
    const headerHeight=header?header.getBoundingClientRect().height:48;
    const body=$('panelBody');
    if(body)body.style.setProperty('--ev-v98-panel-body-max-height',Math.max(90,available-headerHeight-2)+'px');
  }
  const baseOpen=window.openPanel;
  if(typeof baseOpen==='function'&&!baseOpen.__evv98Fit){
    const wrapped=function(key){
      const result=baseOpen.apply(this,arguments);
      if(key==='transform')requestAnimationFrame(()=>requestAnimationFrame(fitTransformPanelV98));
      return result;
    };
    wrapped.__evv98Fit=true;window.openPanel=wrapped;
  }
  const resizeFit=()=>{if($('toolPanel')?.classList.contains('ev-v89-fit-panel'))requestAnimationFrame(fitTransformPanelV98);};
  window.addEventListener('resize',resizeFit,{passive:true});
  window.addEventListener('orientationchange',resizeFit,{passive:true});
  if(window.visualViewport)window.visualViewport.addEventListener('resize',resizeFit,{passive:true});

  /* Bottom toolbars use the browser's native horizontal scroller. A custom pointer-capture
     loop here competed with Android's own touch/momentum pipeline and caused stutter. */

  /* ----- Crop and selection frame use identical source geometry ----- */
  function cropFractions(clip){
    let c={left:0,right:0,top:0,bottom:0};
    try{c=typeof window.ethvelleCropFractions==='function'?window.ethvelleCropFractions(clip.crop):(clip.crop||c);}catch(_e){}
    const clamp=v=>Math.max(0,Math.min(.92,Number(v)||0));
    let left=clamp(c.left),right=clamp(c.right),top=clamp(c.top),bottom=clamp(c.bottom);
    if(left+right>.92){const f=.92/(left+right);left*=f;right*=f;}
    if(top+bottom>.92){const f=.92/(top+bottom);top*=f;bottom*=f;}
    return {left,right,top,bottom};
  }
  function sourceGeometry(clip,element,kind,target){
    const stage=$('canvasStage');if(!stage||!clip||!element)return;
    const sw=Math.max(1,stage.clientWidth||window.stageW||1),sh=Math.max(1,stage.clientHeight||window.stageH||1);
    const iw=Math.max(1,Number(element.videoWidth||element.naturalWidth||clip.sourceWidth)||sw);
    const ih=Math.max(1,Number(element.videoHeight||element.naturalHeight||clip.sourceHeight)||sh);
    const base=Math.min(sw/iw,sh/ih),crop=cropFractions(clip);
    const cropW=Math.max(1,iw*(1-crop.left-crop.right));
    const cropH=Math.max(1,ih*(1-crop.top-crop.bottom));
    const hasCrop=crop.left+crop.right+crop.top+crop.bottom>.0001;
    const desired=hasCrop?Math.max(sw/cropW,sh/cropH):base;
    const factor=desired/Math.max(.0001,base);
    const dx=hasCrop?(crop.right-crop.left)*iw*desired/2:0;
    const dy=hasCrop?(crop.bottom-crop.top)*ih*desired/2:0;
    target.w0=iw*base*factor;
    target.h0=ih*base*factor;
    const baseTransform=target.obj||clip.transform||{scale:1,offsetX:0,offsetY:0,rotation:0};
    const now=typeof window.getCurrentAbsoluteTime==='function'?window.getCurrentAbsoluteTime():(typeof currentAbsoluteTimeSec!=='undefined'?Number(currentAbsoluteTimeSec):0);
    const tracked=kind==='overlay'&&typeof window.ethvelleGetTrackedOverlayTransform==='function'
      ?window.ethvelleGetTrackedOverlayTransform(clip):baseTransform;
    const objectKeyframe=kind==='overlay'&&typeof window.ethvelleGetObjectKeyframeValue==='function'
      ?window.ethvelleGetObjectKeyframeValue(clip,'transform',now):null;
    // Match the renderer's active transform source. During a cropped video preview the
    // committed crop renderer uses the tracked transform; images continue to use their
    // object keyframe when one exists. Without crop, use the regular renderer's keyframe
    // fallback to the tracked/base transform.
    const keyed=kind==='overlay'
      ?(hasCrop&&clip.type==='video'?tracked:(objectKeyframe||tracked))
      :baseTransform;
    target.obj={...keyed};
    if(hasCrop){
      const motion=kind==='overlay'&&typeof window.getEthvelleClipAnimationState==='function'
        ?window.getEthvelleClipAnimationState(clip,now):{x:0,y:0,rotation:0,scale:1};
      target.obj.offsetX=(Number(keyed.offsetX)||0)+dx+(Number(motion.x)||0);
      target.obj.offsetY=(Number(keyed.offsetY)||0)+dy+(Number(motion.y)||0);
      target.obj.rotation=(Number(keyed.rotation)||0)+(Number(motion.rotation)||0);
      target.obj.scale=(Number(keyed.scale)||1)*(Number(motion.scale)||1);
    }
  }
  const baseTarget=window.selBoxTarget;
  if(typeof baseTarget==='function'&&!baseTarget.__evv98CropBounds){
    const wrapped=function(){
      const target=baseTarget.apply(this,arguments);if(!target)return target;
      try{
        const now=typeof window.getCurrentAbsoluteTime==='function'?window.getCurrentAbsoluteTime():Number(window.currentAbsoluteTimeSec)||0;
        if(target.kind==='overlay'){
          const id=String(typeof selectedClip!=='undefined'&&selectedClip!=null?selectedClip:(window.selectedClip??''));
          const list=typeof overlays!=='undefined'&&Array.isArray(overlays)?overlays:(Array.isArray(window.overlays)?window.overlays:[]);
          const clip=list.find(o=>o&&String(o.id)===id&&['image','video'].includes(o.type));
          const mediaMap=typeof overlayMediaEls!=='undefined'&&overlayMediaEls?overlayMediaEls:window.overlayMediaEls;
          const el=clip&&mediaMap?mediaMap[id]:null;
          if(clip&&el)sourceGeometry(clip,el,'overlay',target);
        }else if(target.kind==='background'){
          const clip=typeof backgroundClipForSelection==='function'?backgroundClipForSelection():null;
          const el=$('bgImgLayer');
          if(clip&&el)sourceGeometry(clip,el,'background',target);
        }
      }catch(err){console.warn('Ethvelle crop selection geometry:',err);}
      return target;
    };
    wrapped.__evv98CropBounds=true;window.selBoxTarget=wrapped;
  }
  if(typeof window.updateSelBox==='function'){
    const oldUpdate=window.updateSelBox;
    if(!oldUpdate.__evv98CropRefresh){
      const wrapped=function(){const result=oldUpdate.apply(this,arguments);return result;};
      wrapped.__evv98CropRefresh=true;window.updateSelBox=wrapped;
    }
  }
})();

<script id="ethvelle-v99-interaction-fixes-js">
(function(){
  'use strict';
  const $=id=>document.getElementById(id);
  const raf=()=>new Promise(resolve=>requestAnimationFrame(()=>resolve()));

  /* The earlier fit patch measured max-height but left the sheet bottom-anchored to an
     editor box that can extend below Android's visible viewport. Use a top+bottom constrained
     absolute sheet instead: both edges are guaranteed to remain within the visible viewport. */
  function fitTransformPanelV99(){
    const editor=$('screen-editor'),panel=$('toolPanel'),transport=editor?.querySelector('.transport-row');
    if(!editor||!panel||!transport||!panel.classList.contains('open'))return;
    const er=editor.getBoundingClientRect(),rr=transport.getBoundingClientRect(),vv=window.visualViewport;
    const visibleTop=Math.max(er.top,vv?vv.offsetTop:0);
    const visibleBottom=Math.min(er.bottom,vv?(vv.offsetTop+vv.height):document.documentElement.clientHeight);
    let topViewport=Math.max(visibleTop+8,rr.bottom+6);
    const bottomGap=Math.max(8,er.bottom-visibleBottom+8);
    if(visibleBottom-topViewport-bottomGap<164)topViewport=Math.max(visibleTop+8,visibleBottom-bottomGap-164);
    const available=Math.max(150,Math.floor(visibleBottom-topViewport-bottomGap));
    panel.classList.add('evv99-panel-fit','ev-v89-fit-panel');
    const header=panel.querySelector('.tool-panel-header'),body=$('panelBody');
    const naturalHeight=(header?header.getBoundingClientRect().height:48)+(body?body.scrollHeight:300)+2;
    const fittedHeight=Math.min(available,Math.max(160,naturalHeight));
    panel.style.setProperty('--evv99-panel-top',Math.max(0,topViewport-er.top)+'px');
    panel.style.setProperty('--evv99-panel-height',fittedHeight+'px');
    panel.style.setProperty('--evv99-panel-max-height',available+'px');
    if(body){body.style.removeProperty('max-height');body.style.removeProperty('height');}
  }
  function clearTransformPanelFit(){
    const panel=$('toolPanel');if(!panel)return;
    panel.classList.remove('evv99-panel-fit');
    panel.style.removeProperty('--evv99-panel-top');panel.style.removeProperty('--evv99-panel-height');panel.style.removeProperty('--evv99-panel-max-height');
  }
  const originalOpen=window.openPanel;
  if(typeof originalOpen==='function'&&!originalOpen.__evv99Fit){
    const wrapped=function(key){
      const result=originalOpen.apply(this,arguments);
      if(key==='transform')requestAnimationFrame(()=>requestAnimationFrame(fitTransformPanelV99));
      else clearTransformPanelFit();
      return result;
    };
    wrapped.__evv99Fit=true;window.openPanel=wrapped;
  }
  const resizeFit=()=>{if($('toolPanel')?.classList.contains('open')&&$('toolPanel')?.classList.contains('evv99-panel-fit'))requestAnimationFrame(fitTransformPanelV99);};
  window.addEventListener('resize',resizeFit,{passive:true});
  window.addEventListener('orientationchange',resizeFit,{passive:true});
  if(window.visualViewport){
    window.visualViewport.addEventListener('resize',resizeFit,{passive:true});
    window.visualViewport.addEventListener('scroll',resizeFit,{passive:true});
  }
  const transport= $('screen-editor')?.querySelector('.transport-row');
  if(transport&&typeof ResizeObserver==='function'){const ro=new ResizeObserver(resizeFit);ro.observe(transport);}

  /* Only move a layer icon when the white left trim tab actually overlaps its normal icon
     rectangle. Reconcile only the affected row, and only write row classes when their state
     really changes. The old MutationObserver was passed its records array directly to an
     event-style handler; records have no .target, so every mutation forced a full-row scan. */
  const timeline=$('timelineInner');
  let badgeRaf=0,badgeNeedsFullScan=false;
  const pendingBadgeRows=new Set();
  const badgeOwnedClasses=new Set(['evv99-icon-shift','evv104-bg-selected']);
  function setBadgeRowClass(row,name,enabled){
    if(!row||!row.classList)return;
    const has=row.classList.contains(name),want=!!enabled;
    if(has!==want)row.classList.toggle(name,want);
  }
  function updateLayerBadgeClearance(){
    badgeRaf=0;if(!timeline)return;
    const rows=badgeNeedsFullScan?Array.from(timeline.querySelectorAll('.layer-row')):Array.from(pendingBadgeRows);
    pendingBadgeRows.clear();badgeNeedsFullScan=false;
    rows.forEach(row=>{
      if(!row||!row.isConnected)return;
      // Background keeps the same left-clearance movement as the other clip rows even
      // when its selected clip has been moved away from timeline time zero.
      const bgSelected=!!row.querySelector('#track-background .track-clip.selected');
      setBadgeRowClass(row,'evv104-bg-selected',bgSelected);
      const badge=row.querySelector('.layer-badge');
      const clip=row.querySelector('.anim-clip.selected,.track-clip.selected,.overlay-clip.selected');
      if(!badge||!clip){setBadgeRowClass(row,'evv99-icon-shift',false);return;}
      const track=clip.closest('.layer-track');if(!track){setBadgeRowClass(row,'evv99-icon-shift',false);return;}
      const b=badge.getBoundingClientRect(),c=clip.getBoundingClientRect();
      const wasShifted=row.classList.contains('evv99-icon-shift');
      const badgeLeft=b.left+(wasShifted?21:0),badgeRight=b.right+(wasShifted?21:0);
      const tabLeft=c.left-21,tabRight=c.left-1;
      const overlap=Math.min(tabRight,badgeRight)-Math.max(tabLeft,badgeLeft);
      setBadgeRowClass(row,'evv99-icon-shift',overlap>2);
    });
  }
  function queueBadgeClearance(node,fullScan=false){
    const element=node&&node.nodeType===1?node:null;
    if(fullScan||!element){badgeNeedsFullScan=true;}
    else{
      const row=element.closest&&element.closest('.layer-row');
      if(row)pendingBadgeRows.add(row);
      else badgeNeedsFullScan=true;
    }
    if(!badgeRaf)badgeRaf=requestAnimationFrame(updateLayerBadgeClearance);
  }
  function scheduleBadgeClearance(event){
    queueBadgeClearance(event&&event.target?event.target:null);
  }
  function classChangeIsOnlyBadgeState(record){
    if(!record||record.attributeName!=='class'||record.oldValue==null)return false;
    const before=new Set(String(record.oldValue||'').split(/\s+/).filter(Boolean));
    const after=new Set(String(record.target&&record.target.getAttribute('class')||'').split(/\s+/).filter(Boolean));
    const changed=new Set([...before,...after]);
    for(const token of changed){if(before.has(token)!==after.has(token)&&!badgeOwnedClasses.has(token))return false;}
    return true;
  }
  function nodeContainsLayerRowOrClip(node){
    if(!node||node.nodeType!==1)return false;
    if(node.matches&&node.matches('.layer-row,.anim-clip,.track-clip,.overlay-clip'))return true;
    return !!(node.querySelector&&node.querySelector('.layer-row,.anim-clip,.track-clip,.overlay-clip'));
  }
  function handleTimelineMutations(records){
    let fullScan=false,hasQueuedRow=false;
    for(const record of records){
      if(record.type==='attributes'&&record.attributeName==='class'){
        // Our own clearance marker changes are final output, not new layout inputs. Also
        // ignore unrelated class changes; only selection changes on a timeline clip matter.
        if(classChangeIsOnlyBadgeState(record))continue;
        const target=record.target;
        if(!target||!target.matches||!target.matches('.anim-clip,.track-clip,.overlay-clip'))continue;
        const before=new Set(String(record.oldValue||'').split(/\s+/).filter(Boolean));
        const after=target.classList;
        if(before.has('selected')===after.contains('selected'))continue;
        const row=target.closest('.layer-row');
        if(row){pendingBadgeRows.add(row);hasQueuedRow=true;}
        continue;
      }
      if(record.type==='childList'){
        const target=record.target;
        const row=target&&target.nodeType===1&&target.closest?target.closest('.layer-row'):null;
        const hasRelevantNodes=[...(record.addedNodes||[]),...(record.removedNodes||[])].some(nodeContainsLayerRowOrClip);
        if(!hasRelevantNodes)continue;
        if(row){pendingBadgeRows.add(row);hasQueuedRow=true;}
        else fullScan=true;
