<script id="ethvelle-v189-observer-budget-js">
/* The editor registers ~100 MutationObservers (many on document.body with subtree:true).
   Selecting a text/audio layer toggles classes in several places, so every one of them ran in
   the same task and froze the UI. This wrapper keeps their behaviour, but:
     - merges all records that arrive within a frame into ONE callback per observer,
     - delivers them just before the next paint (so fix-ups never flash on screen),
     - spends at most ~6 ms per frame on observer callbacks; the rest roll to the next frame.  */
(function(){
  'use strict';
  var Native=window.MutationObserver;
  if(!Native||Native.__ev189)return;
  var queue=[],scheduled=false,BUDGET=6;
  function now(){return (window.performance&&performance.now)?performance.now():Date.now();}
  function schedule(){
    if(scheduled)return;scheduled=true;
    var run=function(){if(!scheduled)return;scheduled=false;pump();};
    if(window.requestAnimationFrame)requestAnimationFrame(run);
    setTimeout(run,100); // background tabs don't fire rAF
  }
  /* v220: while the playhead is being scrubbed, every DOM change the scrub itself causes used to
     wake all text-panel observers each frame (slider/palette/toolbar re-scans), which made sliding
     laggy with a text layer selected. During a scrub each observer now gets at most one merged
     delivery per SCRUB_GAP ms; whatever is still held is delivered right after the scrub ends. */
  var SCRUB_GAP=260,held=[],holdTimer=0;
  function releaseHeld(){
    holdTimer=0;
    var list=held;held=[];
    for(var i=0;i<list.length;i++){var h=list[i];h._held=false;if(!h._off&&h._pending.length&&!h._q){h._q=true;queue.push(h);}}
    if(queue.length)schedule();
  }
  function pump(){
    var t0=now();
    while(queue.length){
      var w=queue.shift();w._q=false;
      if(window.__ethvelleTimelineScrubbing&&now()-(w._last||0)<SCRUB_GAP){
        if(w._pending.length>2000)w._pending=w._pending.slice(-2000);
        if(!w._held){w._held=true;held.push(w);}
        if(!holdTimer)holdTimer=setTimeout(releaseHeld,SCRUB_GAP/2);
        continue;
      }
      w._last=now();w._deliver();
      if(queue.length&&now()-t0>BUDGET){schedule();return;}
    }
  }
  function Wrapped(cb){
    var self=this;
    this._cb=cb;this._pending=[];this._q=false;this._off=false;
    this._inner=new Native(function(records){
      if(self._off)return;
      for(var i=0;i<records.length;i++)self._pending.push(records[i]);
      if(!self._q&&!self._held){self._q=true;queue.push(self);schedule();}
    });
  }
  Wrapped.prototype._deliver=function(){
    if(this._off||!this._pending.length)return;
    var recs=this._pending;this._pending=[];
    try{this._cb.call(this,recs,this);}catch(e){try{console.error(e);}catch(_e){}}
  };
  Wrapped.prototype.observe=function(target,options){this._off=false;return this._inner.observe(target,options);};
  Wrapped.prototype.disconnect=function(){this._off=true;this._pending=[];return this._inner.disconnect();};
  Wrapped.prototype.takeRecords=function(){var r=this._pending.concat(this._inner.takeRecords());this._pending=[];return r;};
  Wrapped.__ev189=true;
  try{window.MutationObserver=Wrapped;}catch(_e){}
})();
</script>
<!-- ===== Ethvelle v191: text-panel sliders. Registered before every other window listener so a
     swipe that starts anywhere on a slider row (not only on the 14px thumb) drags the value in ONE
     touch; a vertical swipe on the same row scrolls the panel natively. ===== -->
<script id="ethvelle-v191-text-slider-js">
(function(){
  'use strict';
  if(window.__ethvelleV191TextSliders)return;
  window.__ethvelleV191TextSliders=true;

  /* Text-editing panels only. Every other slider in the app keeps its existing behaviour. */
  var SCOPE='#ethvellePicsartTextDock,#ethvelleTextStylePanel,#ethvelleTextComposer,#evpaCustomEffectEditor,#ethvelleColorPicker,#textPanelBody';
  /* One "row" = caption + slider. The whole row is the grab area, not just the 14px thumb. */
  var ROWS='.evpa-range,.evts-field,.evtc-opacity-line,.evce-field,.evcp-field,.evpa-picker-range,.evce-stop,.text-range-line';
  /* Things that own their own touches (buttons, scrolling chip rows, colour wheel...). */
  var BLOCK='button,select,textarea,a,input:not([type="range"]),[contenteditable="true"],#evpaTabs,'+
            '.evpa-font-gallery,.evpa-style-presets,.evpa-gradient-presets,.evpa-choice-row,.evpa-swatches,'+
            '.evcp-wheel,.evpa-hue-ring,.evpa-sat-square,canvas';
  var THUMB=14;          // visible thumb width in the text-panel CSS
  var SLOP_X=5;          // px of sideways travel that starts a value drag
  var SLOP_Y=8;          // px of vertical travel that hands the gesture to native scrolling
  var st=null;           // active gesture

  function num(v,d){var n=parseFloat(v);return isFinite(n)?n:d;}
  function usable(input){
    var r=input.getBoundingClientRect();
    return {r:r,w:Math.max(24,r.width-THUMB)};
  }
  function okInput(i){
    if(!i||i.disabled)return false;
    var r=i.getBoundingClientRect();
    return r.width>24&&r.height>0;
  }
  /* Which slider does a touch at (x,y) on element t belong to? */
  function pickRange(t,x,y){
    if(t.matches&&t.matches('input[type="range"]'))return okInput(t)?t:null;
    var scope=t.closest(SCOPE);if(!scope)return null;
    var list=[],row=t.closest(ROWS);
    if(row&&scope.contains(row))list=[].slice.call(row.querySelectorAll('input[type="range"]'));
    if(!list.length){
      /* No known row wrapper: fall back to a finger-sized band around each slider. */
      list=[].slice.call(scope.querySelectorAll('input[type="range"]')).filter(function(i){
        var r=i.getBoundingClientRect();
        return x>=r.left-10&&x<=r.right+10&&Math.abs(y-(r.top+r.height/2))<=Math.max(18,r.height/2+10);
      });
    }
    var best=null,bd=1e9;
    for(var k=0;k<list.length;k++){
      var i=list[k];if(!okInput(i))continue;
      var r=i.getBoundingClientRect(),d=Math.abs(y-(r.top+r.height/2));
      if(d<bd){bd=d;best=i;}
    }
    return best;
  }
  function valueAt(s,x){
    var min=s.min,max=s.max,v=s.startVal+((x-s.x0)/s.w)*(max-min);
    if(s.step>0)v=min+Math.round((v-min)/s.step)*s.step;
    v=Math.max(min,Math.min(max,v));
    var dec=(String(s.input.step||'1').split('.')[1]||'').length;
    return String(Number(v.toFixed(Math.min(7,dec))));
  }
  function push(s,x,type){
    var inp=s.input;if(!inp.isConnected)return;
    var next=valueAt(s,x);
    if(inp.value!==next){inp.value=next;inp.dispatchEvent(new Event(type||'input',{bubbles:true}));}
  }
  function finish(s,x){
    if(s.raf){cancelAnimationFrame(s.raf);s.raf=0;}
    if(s.mode!=='value')return;
    push(s,x==null?s.lastX:x,'input');
    if(s.input.isConnected)s.input.dispatchEvent(new Event('change',{bubbles:true}));
  }
  function stopAll(e){
    if(e.stopImmediatePropagation)e.stopImmediatePropagation();
    else if(e.stopPropagation)e.stopPropagation();
  }

  function onDown(e){
    if(st){
      if(e.pointerId!==st.id)return;      // ignore a second finger
      st=null;                             // stale gesture with the same id
    }
    if(e.pointerType==='mouse'&&e.button!==0)return;
    var t=e.target;if(!t||t.nodeType!==1||!t.closest)return;
    if(!t.closest(SCOPE))return;
    if(t.closest(BLOCK))return;
    var input=pickRange(t,e.clientX,e.clientY);
    if(!input)return;
    var g=usable(input);
    var min=num(input.min,0),max=num(input.max,100);
    if(!(max>min))return;
    var stepAttr=input.step,step=stepAttr==='any'?0:(num(stepAttr,1)||1);
    /* Self-heal: if some other script re-enabled native input on this slider, switch it off again. */
    try{input.style.setProperty('pointer-events','none','important');input.style.setProperty('touch-action','pan-y','important');}catch(_e){}
    st={id:e.pointerId,input:input,x0:e.clientX,y0:e.clientY,lastX:e.clientX,
        startVal:num(input.value,min),min:min,max:max,step:step,w:g.w,
        mode:'pending',raf:0,down:e};
    /* Keep the canvas / sheet gesture engines out of a slider touch (same isolation as before). */
    stopAll(e);
  }
  function onMove(e){
    var s=st;if(!s||e.pointerId!==s.id)return;
    var dx=e.clientX-s.x0,dy=e.clientY-s.y0,ax=Math.abs(dx),ay=Math.abs(dy);
    if(s.mode==='pending'){
      if(ax>=SLOP_X&&ax>ay*1.25){
        s.mode='value';
        /* Legacy "slider started" hook (e.g. inline onpointerdown on some sliders). */
        try{if(typeof s.input.onpointerdown==='function')s.input.onpointerdown.call(s.input,s.down);}catch(err){}
      }else if(ay>=SLOP_Y&&ay>=ax*0.8){
        st=null;                            // vertical swipe: let the panel scroll, value untouched
        return;
      }else return;
    }
    s.lastX=e.clientX;
    stopAll(e);
    if(e.cancelable)e.preventDefault();
    if(!s.raf)s.raf=requestAnimationFrame(function(){s.raf=0;if(st===s)push(s,s.lastX,'input');});
  }
  function onUp(e){
    var s=st;if(!s||e.pointerId!==s.id)return;
    st=null;
    if(s.mode==='value'){
      if(isFinite(e.clientX))s.lastX=e.clientX;
      stopAll(e);
      finish(s,s.lastX);
    }
    /* mode==='pending' is a plain tap: nothing changes (no track-jump). */
  }
  function onCancel(e){
    var s=st;if(!s||e.pointerId!==s.id)return;
    st=null;
    if(s.mode==='value')finish(s,s.lastX);   // keep what the user dragged to
  }

  window.addEventListener('pointerdown',onDown,{capture:true,passive:false});
  window.addEventListener('pointermove',onMove,{capture:true,passive:false});
  window.addEventListener('pointerup',onUp,{capture:true,passive:false});
  window.addEventListener('pointercancel',onCancel,{capture:true,passive:false});
  window.addEventListener('blur',function(){var s=st;st=null;if(s&&s.mode==='value')finish(s,s.lastX);});
  window.__ethvelleV191TextSlidersDebug=function(){return st;};
})();
</script>

    const p=fromStage(M,(e.clientX-sr.left)/sx,(e.clientY-sr.top)/sy);
    if(!p){if(e.cancelable)e.preventDefault();return true;}
    const startCrop=cropValues(ctx.clip);
    const handle=pickHandle(p,frameRect(ctx,startCrop),k);
    if(e.cancelable)e.preventDefault();
    if(!handle)return true;                       // touch outside the frame is swallowed, nothing moves
    const ratio=(document.querySelector('.ev-v89-ratio.active')||{dataset:{}}).dataset.ratio||'free';
    cdrag={id:e.pointerId,ctx,M,sx,sy,k,handle,ratio,startCrop,x0:e.clientX,y0:e.clientY,cx:e.clientX,cy:e.clientY,raf:0,moved:false};
    window.__ev176CropBusy=true;                 // pause the heavy stage renderer while dragging
    root.classList.add('evpc-active');
    if(handle!=='move'){try{const hot=root.querySelector('.ev-v89-crop-handle[data-handle="'+handle+'"]');if(hot)hot.classList.add('evpc-hot');}catch(_e){}}
    if(layer)layer.classList.add('evpc-drag');
    try{ctx.stage.setPointerCapture(e.pointerId);}catch(_e){}
    return true;
  }
  // Remove every "touched" highlight (frame border + the touched handle). Called on release and
  // from several safety nets, so the blue can never get stuck on after the finger is lifted.
  function clearHot(){
    const root=$('evV89CropRoot');
    if(root){root.classList.remove('evpc-active');root.querySelectorAll('.evpc-hot').forEach(n=>n.classList.remove('evpc-hot'));}
    if(layer)layer.classList.remove('evpc-drag');
  }
  function cropCompute(d){
    const ctx=d.ctx,M=d.M;
    const ux=(d.cx-d.x0)/d.sx,uy=(d.cy-d.y0)/d.sy;                 // finger delta in stage px
    const lx=(M.d*ux-M.c*uy)/M.det,ly=(-M.b*ux+M.a*uy)/M.det;      // → crop-root local px
    const dx=lx/Math.max(1,ctx.m.w),dy=ly/Math.max(1,ctx.m.h);
    const s=d.startCrop,c={left:s.left,right:s.right,top:s.top,bottom:s.bottom},h=d.handle;
    const w0=1-s.left-s.right,hh0=1-s.top-s.bottom,min=.04;
    if(h==='move'){
      c.left=clamp(s.left+dx,0,1-w0);c.right=1-w0-c.left;
      c.top=clamp(s.top+dy,0,1-hh0);c.bottom=1-hh0-c.top;
    }else if(d.ratio==='free'){
      if(h.includes('w'))c.left=clamp(s.left+dx,0,1-s.right-min);
      if(h.includes('e'))c.right=clamp(s.right-dx,0,1-s.left-min);
      if(h.includes('n'))c.top=clamp(s.top+dy,0,1-s.bottom-min);
      if(h.includes('s'))c.bottom=clamp(s.bottom-dy,0,1-s.top-min);
    }else{
      const parts=String(d.ratio).split(':').map(Number);
      const wanted=(parts[0]/parts[1])/ctx.m.aspect;
      if(!Number.isFinite(wanted)||wanted<=0)return s;
      if(h==='e'||h==='w'){
        let w=clamp(w0+(h==='e'?dx:-dx),min,1),height=clamp(w/wanted,min,1);w=height*wanted;
        if(h==='e')c.right=1-s.left-w;else c.left=1-s.right-w;
        c.top=(1-height)/2;c.bottom=1-height-c.top;
      }else if(h==='n'||h==='s'){
        let height=clamp(hh0+(h==='s'?dy:-dy),min,1),w=clamp(height*wanted,min,1);height=w/wanted;
        if(h==='s')c.bottom=1-s.top-height;else c.top=1-s.bottom-height;
        c.left=(1-w)/2;c.right=1-w-c.left;
      }else{
        const dirX=h.includes('w')?-1:1,dirY=h.includes('n')?-1:1;
        const wC=clamp(w0+dirX*dx,min,1),hC=clamp(hh0+dirY*dy,min,1);
        let w=Math.abs(wC-w0)/Math.max(min,w0)>=Math.abs(hC-hh0)/Math.max(min,hh0)?wC:hC*wanted;
        let height=w/wanted;
        if(height>1){height=1;w=height*wanted;}if(w>1){w=1;height=w/wanted;}
        if(dirX<0)c.left=1-s.right-w;else c.right=1-s.left-w;
        if(dirY<0)c.top=1-s.bottom-height;else c.bottom=1-s.top-height;
      }
    }
    return normalizeCrop(c);
  }
  function cropPaint(){
    const d=cdrag;if(!d)return;
    d.raf=0;
    const next=cropCompute(d);
    d.ctx.clip.crop=next;
    const r=frameRect(d.ctx,next);
    writeFrame(d.ctx,r);
    paintOverlay(d.ctx,r,next,d.M,d.k,false,d.ratio);
  }
  function cropEnd(e){
    const d=cdrag;if(!d)return;
    cdrag=null;
    if(d.raf){cancelAnimationFrame(d.raf);d.raf=0;}
    if(e&&e.type==='pointerup'&&Number.isFinite(e.clientX)){d.cx=e.clientX;d.cy=e.clientY;}
    try{
      const next=cropCompute(d);d.ctx.clip.crop=next;
      const r=frameRect(d.ctx,next);writeFrame(d.ctx,r);
      paintOverlay(d.ctx,r,next,d.M,d.k,false,d.ratio);
    }catch(_e){}
    clearHot();
    window.__ev176CropBusy=false;
    lastCropEnd=now();
    try{d.ctx.stage.releasePointerCapture(d.id);}catch(_e){}
    try{if(typeof window.persistProjectState==='function')window.persistProjectState({skipHistory:true});}catch(_e){}
  }
  function cropPointer(e){
    const t=e.type;
    if(cdrag){
      if(e.pointerId!==cdrag.id)return true;                      // ignore extra fingers during a crop drag
      if(t==='pointermove'){
        cdrag.cx=e.clientX;cdrag.cy=e.clientY;
        if(!cdrag.moved){if(Math.hypot(cdrag.cx-cdrag.x0,cdrag.cy-cdrag.y0)<1)return true;cdrag.moved=true;}
        if(!cdrag.raf)cdrag.raf=requestAnimationFrame(cropPaint);
        if(e.cancelable)e.preventDefault();
        return true;
      }
      if(t==='pointerup'||t==='pointercancel'||t==='lostpointercapture'){cropEnd(e);return true;}
      return true;
    }
    if(t!=='pointerdown'||!cropActive())return false;
    if(e.pointerType==='mouse'&&e.button!==0)return false;
    const stage=$('canvasStage');if(!stage)return false;
    const sr=stage.getBoundingClientRect();
    if(e.clientX<sr.left||e.clientX>sr.right||e.clientY<sr.top||e.clientY>sr.bottom)return false;
    if(interactive(e.target))return false;
    return cropStart(e);
  }
  function cropTouch(e){
    const type=e.type;
    if(cdrag){if(type==='touchmove'||type==='touchstart'){if(e.cancelable)e.preventDefault();return true;}return false;}
    if(type!=='touchstart'||!cropActive())return false;
    const stage=$('canvasStage');if(!stage||!stage.contains(e.target)||interactive(e.target))return false;
    if(e.cancelable)e.preventDefault();
    return true;
  }
  function cropClick(e){
    if(!lastCropEnd||now()-lastCropEnd>400)return false;
    const stage=$('canvasStage');
    return !!(stage&&stage.contains(e.target)&&!interactive(e.target));
  }
  // Safety: never leave the renderer paused.
  window.addEventListener('blur',()=>{if(cdrag)cropEnd(null);if(drag)finishSlider();});
  document.addEventListener('visibilitychange',()=>{if(document.hidden){if(cdrag)cropEnd(null);if(drag)finishSlider();}});

  /* =====================================================================
     C. ONE REGISTRATION POINT (runs before every older handler)
     ===================================================================== */
  const opt={capture:true,passive:false};
  function claim(fn,e){let r=false;try{r=fn(e);}catch(err){console.warn('Ethvelle v176 gate:',err);}if(r)e.stopImmediatePropagation();return r;}
  ['pointerdown','pointermove','pointerup','pointercancel','lostpointercapture'].forEach(type=>{
    window.addEventListener(type,e=>{if(!claim(cropPointer,e))claim(sliderPointer,e);},opt);
  });
  ['touchstart','touchmove','touchend','touchcancel'].forEach(type=>{
    window.addEventListener(type,e=>{if(!claim(cropTouch,e))claim(sliderTouch,e);},opt);
  });
  window.addEventListener('mousedown',e=>{claim(sliderMouse,e);},opt);
  window.addEventListener('click',e=>{if(!claim(cropClick,e))claim(sliderMouse,e);},opt);
  window.addEventListener('input',e=>{claim(sliderValue,e);},opt);
  window.addEventListener('change',e=>{claim(sliderValue,e);},opt);
  // Safety net: whenever a touch ends / is cancelled and no crop drag is running, nothing may stay blue.
  ['pointerup','pointercancel','touchend','touchcancel','lostpointercapture'].forEach(type=>{
    window.addEventListener(type,()=>{if(!cdrag)clearHot();},{capture:true,passive:true});
  });

  /* =====================================================================
     D. HOUSEKEEPING (needs the DOM / the app's globals, so it runs later)
     ===================================================================== */
  function installRenderGuard(){
    const f=window.renderFrame;
    if(typeof f!=='function'||f.__ev176Guard)return;
    const guarded=function(){if(window.__ev176CropBusy)return;return f.apply(this,arguments);};
    try{Object.keys(f).forEach(key=>{guarded[key]=f[key];});}catch(_e){}
    guarded.__ev176Guard=true;
    window.renderFrame=guarded;
  }
  function watchCrop(){
    const scr=$('screen-editor');if(!scr)return;
    let frameObs=null,sizeObs=null,active=false,tries=0;
    const attach=()=>{
      const frame=$('evV89CropFrame'),stage=$('canvasStage');
      if(!frame){if(++tries<40)requestAnimationFrame(attach);return;}
      if(frameObs)frameObs.disconnect();
      frameObs=new MutationObserver(scheduleRefresh);
      frameObs.observe(frame,{attributes:true,attributeFilter:['style']});
      if(sizeObs)sizeObs.disconnect();
      if(stage&&typeof ResizeObserver!=='undefined'){sizeObs=new ResizeObserver(scheduleRefresh);sizeObs.observe(stage);}
      scheduleRefresh();
    };
    new MutationObserver(()=>{
      const on=scr.classList.contains('ev-v89-crop-active');
      if(on===active)return;
      active=on;
      if(on){tries=0;requestAnimationFrame(attach);}
      else{
        if(frameObs){frameObs.disconnect();frameObs=null;}
        if(sizeObs){sizeObs.disconnect();sizeObs=null;}
        clearHot();removeLayer();window.__ev176CropBusy=false;cdrag=null;
      }
    }).observe(scr,{attributes:true,attributeFilter:['class']});
  }
  function boot(){installRenderGuard();watchCrop();}
  if(document.readyState==='loading')document.addEventListener('DOMContentLoaded',boot,{once:true});else boot();
  window.addEventListener('load',installRenderGuard,{once:true});
  setTimeout(installRenderGuard,1500);

  G.eligible=eligible;
})();
