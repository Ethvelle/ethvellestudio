      if(el&&ov&&el.tagName==='VIDEO'&&el.paused&&el.readyState>=1&&!el.seeking){
        const t=Math.max(Number(currentAbsoluteTimeSec)||0,Number(ov.trimStart)||0);
        let expected=Number(ov.sourceOffset)||0;
        try{if(typeof evOverlayVideoExpectedLocalTime==='function')expected=evOverlayVideoExpectedLocalTime(ov,t,el);}catch(_e){}
        if(Number.isFinite(expected)&&Math.abs((Number(el.currentTime)||0)-expected)>0.035){
          const now=performance.now();
          el.dataset.evSeekRequestAt=String(now);
          el.dataset.evPlaySeekAt=String(now);
          try{el.currentTime=expected;}catch(err){console.warn('Ethvelle incoming-video alignment:',err&&err.name||'Error',err);}
        }
      }
      return el;
    };
    wrappedTakeWarm.__ethvelleV141BoundarySync=true;
    window.takeWarmOverlayVideo=wrappedTakeWarm;
  }

  /* ------------------------------------------------------------------------
     BUG 2 — Exclude ONLY the four directional In/Out slide presets from the
     history signature and payload. They stay in the live project and saved
     project, but Undo/Redo never changes them. Other animation presets,
     durations, filters and real edits remain normal history entries.
     ------------------------------------------------------------------------ */
  const DIRECTIONAL_SLIDES=new Set(['slide-left','slide-right','slide-up','slide-down']);
  const HISTORY_PREFIX='__ethvelle_v140__';
  const historyPayloads=window.__ethvelleV140HistoryPayloads;
  const slideState=window.__ethvelleV141SlideHistoryState||
    (window.__ethvelleV141SlideHistoryState=new Map());

  function slideKey(ov){return String(ov&&ov.id!=null?ov.id:'');}
  function captureCurrentSlidePresets(){
    if(!Array.isArray(overlays))return;
    overlays.forEach(ov=>{
      if(!ov||ov.id==null)return;
      const key=slideKey(ov),cfg=ov.clipAnimations&&typeof ov.clipAnimations==='object'?ov.clipAnimations:{};
      const entry=slideState.get(key)||{};
      ['in','out'].forEach(slot=>{
        const value=cfg[slot];
        if(DIRECTIONAL_SLIDES.has(String(value)))entry[slot]=String(value);
        else delete entry[slot];
      });
      if(entry.in||entry.out)slideState.set(key,entry);
      else slideState.delete(key);
    });
    while(slideState.size>512)slideState.delete(slideState.keys().next().value);
  }
  function stripDirectionalSlides(value,parentKey){
    if(Array.isArray(value))return value.map(item=>stripDirectionalSlides(item,''));
    if(!value||typeof value!=='object')return value;
    const out={};
    Object.keys(value).sort().forEach(key=>{
      if(parentKey==='clipAnimations'&&(key==='in'||key==='out')&&DIRECTIONAL_SLIDES.has(String(value[key])))return;
      const child=stripDirectionalSlides(value[key],key);
      if(child!==undefined)out[key]=child;
    });
    if(parentKey==='clipAnimations'){
      const hasActive=['in','out','overall'].some(key=>out[key]&&out[key]!=='none');
      if(!hasActive){
        // getAnimationConfig() materializes these defaults when the first preset is
        // selected. Remove that inert scaffolding too, so a slide-only choice has the
        // exact same history signature as the untouched clip and cannot cut Redo.
        if(out.in==='none')delete out.in;
        if(out.out==='none')delete out.out;
        if(out.overall==='none')delete out.overall;
        if(Number(out.inDuration)===0.45)delete out.inDuration;
        if(Number(out.outDuration)===0.45)delete out.outDuration;
        if(Number(out.overallDuration)===1.2)delete out.overallDuration;
      }
      if(!Object.keys(out).length)return undefined;
    }
    return out;
  }
  function normalizeHistorySignature(value,inCharacters,parentKey){
    if(Array.isArray(value))return value.map(item=>normalizeHistorySignature(item,inCharacters,''));
    if(!value||typeof value!=='object')return value;
    const out={};
    Object.keys(value).sort().forEach(key=>{
      if(key.startsWith('__')||key==='peaks'||key==='thumbs')return;
      if(['currentAbsoluteTimeSec','playheadTimeSec','playhead','currentTime','frame','currentFrame','runtimeFrame'].includes(key))return;
      if(inCharacters&&['bones','baseAngles','modelTransform','restTransform','hipRestY'].includes(key))return;
      if(parentKey==='clipAnimations'&&(key==='in'||key==='out')&&DIRECTIONAL_SLIDES.has(String(value[key])))return;
      out[key]=normalizeHistorySignature(value[key],inCharacters||key==='characters',key);
    });
    return out;
  }
  function stableHistoryString(value){
    if(Array.isArray(value))return '['+value.map(stableHistoryString).join(',')+']';
    if(value&&typeof value==='object')return '{'+Object.keys(value).sort().map(key=>JSON.stringify(key)+':'+stableHistoryString(value[key])).join(',')+'}';
    return JSON.stringify(value);
  }

  const priorSnapshotState=window.snapshotState;
  if(typeof priorSnapshotState==='function'&&!priorSnapshotState.__ethvelleV141SlideHistory){
    const wrappedSnapshotState=function(){
      captureCurrentSlidePresets();
      const baseSignature=priorSnapshotState.apply(this,arguments);
      if(!historyPayloads||typeof baseSignature!=='string')return baseSignature;
      try{
        const payloadText=historyPayloads.get(baseSignature);
        if(!payloadText)return baseSignature;
        const payload=stripDirectionalSlides(JSON.parse(payloadText),'' );
        const normalized=normalizeHistorySignature(payload,false,'');
        const signature=HISTORY_PREFIX+stableHistoryString(normalized);
        // Store a compact, direction-free payload under the returned signature.
        historyPayloads.set(signature,JSON.stringify(payload));
        return signature;
      }catch(err){
        console.warn('Ethvelle directional-slide history normalization:',err);
        return baseSignature;
      }
    };
    wrappedSnapshotState.__ethvelleV141SlideHistory=true;
    window.snapshotState=wrappedSnapshotState;
  }

  function mergeCurrentDirectionalSlides(restoredData){
    if(!restoredData||!Array.isArray(restoredData.overlays))return;
    const currentById=new Map((Array.isArray(overlays)?overlays:[]).filter(Boolean).map(ov=>[slideKey(ov),ov]));
    restoredData.overlays.forEach(target=>{
      if(!target||target.id==null)return;
      const key=slideKey(target),current=currentById.get(key)||null;
      const cached=slideState.get(key)||{};
      const targetCfg=target.clipAnimations&&typeof target.clipAnimations==='object'?target.clipAnimations:(target.clipAnimations={});
      const liveCfg=current&&current.clipAnimations&&typeof current.clipAnimations==='object'?current.clipAnimations:{};
      ['in','out'].forEach(slot=>{
        const liveValue=liveCfg[slot];
        const cachedValue=cached[slot];
        const savedValue=targetCfg[slot];
        let preserve;
        if(DIRECTIONAL_SLIDES.has(String(liveValue)))preserve=String(liveValue);
        else if(DIRECTIONAL_SLIDES.has(String(cachedValue)))preserve=String(cachedValue);
        else if(DIRECTIONAL_SLIDES.has(String(savedValue))){
          // A legacy/older snapshot may still contain a directional slide. Keep the
          // current non-slide choice (Fade/None/Zoom/etc), or remove the slide if none
          // is currently selected; do not let Undo/Redo introduce it.
          if(liveValue!==undefined&&liveValue!==null)preserve=liveValue;
        }
        if(preserve!==undefined)targetCfg[slot]=preserve;
        else if(DIRECTIONAL_SLIDES.has(String(savedValue)))delete targetCfg[slot];
      });
      if(!Object.keys(targetCfg).length)delete target.clipAnimations;
    });
  }

  const priorApplySnapshot=window.applySnapshot;
  if(typeof priorApplySnapshot==='function'&&!priorApplySnapshot.__ethvelleV141SlideHistory){
    const wrappedApplySnapshot=function(snapshot){
      captureCurrentSlidePresets();
      if(historyPayloads&&typeof snapshot==='string'){
        try{
          let payloadText=historyPayloads.get(snapshot)||null;
          if(!payloadText&&snapshot.startsWith(HISTORY_PREFIX))payloadText=snapshot.slice(HISTORY_PREFIX.length);
          let data=payloadText?JSON.parse(payloadText):JSON.parse(snapshot);
          if(data&&typeof data==='object'&&Array.isArray(data.overlays)){
            mergeCurrentDirectionalSlides(data);
            // The previous v140 restore wrapper still performs waveform-cache rehydration,
            // UI rebuild and its settled history-head refresh when given ordinary JSON.
            return priorApplySnapshot.call(this,JSON.stringify(data));
          }
        }catch(err){
          console.warn('Ethvelle directional-slide restore merge:',err);
        }
      }
      return priorApplySnapshot.apply(this,arguments);
    };
    wrappedApplySnapshot.__ethvelleV141SlideHistory=true;
    window.applySnapshot=wrappedApplySnapshot;
  }
