  const editor=document.getElementById('screen-editor');
  const activePointers=new Set();
  const originalPushHistory=window.pushHistory;
  let transactionOpen=false;
  let commitTimer=null;
  // Invalidates delayed commits when Undo/Redo or project reset is requested. Without this,
  // a pending pointer/input timer could run just after Undo and accidentally erase redo.
  let transactionEpoch=0;

  function isEditorEvent(e){
    if(!editor||!e||!e.target)return false;
    if(!editor.contains(e.target))return false;
    // Undo and redo must not record themselves as new edits.
    if(e.target.closest&&e.target.closest('#undoBtn,#redoBtn'))return false;
    return true;
  }
  function clearCommitTimer(){
    if(commitTimer!==null){clearTimeout(commitTimer);commitTimer=null;}
  }
  function beginTransaction(){
    clearCommitTimer();
    if(transactionOpen)return;
    try{
      // Project opening already seeds history with resetHistory(). Only seed here as a
      // safety net for a newly-created editor; avoid serializing large media projects on
      // every ordinary tap/drag start when the previous action is already committed.
      if(typeof pushHistory==='function'&&!restoringHistory&&typeof historyIndex!=='undefined'&&historyIndex<0)pushHistory();
    }catch(err){console.warn('Ethvelle history baseline:',err);}
    transactionOpen=true;
  }
  function commitTransaction(delay){
    if(!transactionOpen)return;
    clearCommitTimer();
    const epoch=transactionEpoch;
    commitTimer=setTimeout(()=>{
      commitTimer=null;
      if(epoch!==transactionEpoch)return;
      // Pointer capture is occasionally cancelled by Android WebView while a panel is
      // rebuilding. Retry briefly rather than leaving the transaction permanently open.
      if(activePointers.size){commitTransaction(90);return;}
      try{
        // Bypass the transaction-aware wrapper below for this one final snapshot.
        if(typeof originalPushHistory==='function'&&!restoringHistory)originalPushHistory();
      }catch(err){console.warn('Ethvelle history commit:',err);}
      transactionOpen=false;
    },Math.max(0,Number(delay)||0));
  }
  function flushPendingTransaction(){
    clearCommitTimer();
    const hadPending=transactionOpen;
    activePointers.clear();

    // Undo creates a redo entry even if the normal gesture debounce had not fired yet.
    if(hadPending&&!restoringHistory&&typeof originalPushHistory==='function'){
      try{originalPushHistory();}catch(err){console.warn('Ethvelle history flush:',err);}
    }

    // outside the original pointer gesture. In that case the current history head is the
