/* ================= PLAN PRICING (admin discount + first-purchase offer + promo popup) =====
   A configured promo-popup discount has priority over plan-level and first-purchase discounts
   for its target plan. A popup may target one plan or all paid plans (__all_paid__). If no promo
   applies to a plan, the best applicable regular discount is used as before.
   Returns { finalPrice, originalPrice, off, label } or null if no discount applies. */
function computePlanDiscount(plan){
  const price = Number(plan.price) || 0;
  if(price <= 0) return null;
  function offFor(d){
    if(!d || !d.on) return 0;
    const v = Number(d.value) || 0;
    return d.type === 'amount' ? v : Math.round(price * v / 100);
  }
  const pd = window.activePromoDiscount;
  const promoTargetsPlan = pd && (pd.applyToAll === true || pd.planId === '__all_paid__' || pd.planId === plan.id);
  if(promoTargetsPlan){
    const value = Number(pd.value) || 0;
    const promoOff = pd.type === 'amount' ? value : Math.round(price * value / 100);
    if(promoOff > 0){
      const off = Math.min(promoOff, price); // never discount below ৳0
      return { finalPrice: price - off, originalPrice: price, off, label: pd.label || 'Promo discount', isPromo:true };
    }
  }
  const candidates = [];
  if(plan.discount && plan.discount.on){
    const off = offFor(plan.discount);
    if(off > 0) candidates.push({ off, label: plan.discount.label || 'Discount' });
  }
  if(plan.firstPurchaseOffer && plan.firstPurchaseOffer.on && currentUser && !myHasEverOrdered){
    const off = offFor(plan.firstPurchaseOffer);
    if(off > 0) candidates.push({ off, label: plan.firstPurchaseOffer.label || 'First purchase offer' });
  }
  if(!candidates.length) return null;
  candidates.sort((a,b)=>b.off - a.off);
  const best = candidates[0];
  const off = Math.min(best.off, price); // never discount below ৳0
  return { finalPrice: price - off, originalPrice: price, off, label: best.label };
}
function renderPaymentPlans(){
  const scroll = document.querySelector('#screen-payment .demo-scroll');
  if(!scroll) return;
  const plans = getPlans();
  // Whether the signed-in user currently holds ANY paid (non-free) plan — once true, the
  // Free card is no longer shown at all, the same way most subscription apps stop offering
  // you the free tier once you're already a paying customer. Signed-out visitors and
  // Free-tier users still see every card, Free included.
  const hasActivePaidPlan = !!(currentUser && myActivePlanId && myActivePlanId !== 'free');
  const orderSig = myLatestOrderInfo ? [myLatestOrderInfo.id,myLatestOrderInfo.planId,myLatestOrderInfo.status].join(':') : '';
  const planOrderSig = JSON.stringify(Object.keys(myPlanOrderStates).sort().map(k=>[k,myPlanOrderStates[k].id,myPlanOrderStates[k].status]));
  const sig = JSON.stringify(plans) + '|' + myActivePlanId + '|' + (currentUser?'in':'out') + '|' + myHasEverOrdered + '|' + orderSig + '|' + planOrderSig + '|' + JSON.stringify(window.activePromoDiscount||null);
  if(sig === _plansSig) return;
  _plansSig = sig;
  // The user's own plan always sorts to the very top of the list, ahead of every other
  // card, so they see "this is what I have" first rather than having to scroll to find it.
  const ordered = plans.slice().sort((a,b)=>{
    const aMine = currentUser && a.id === myActivePlanId;
    const bMine = currentUser && b.id === myActivePlanId;
    if(aMine && !bMine) return -1;
    if(bMine && !aMine) return 1;

    // discount applies; otherwise just the plain price, same as before.

        : `৳${p.price}<span>/month</span>`;
    const featuresHtml = (p.features||[]).map(f=>`<div class="plan-feature"><svg viewBox="0 0 24 24" fill="none" stroke-width="2" stroke-linecap="round"><path d="M20 6L9 17l-5-5"/></svg>${escapeHtml(f)}</div>`).join('');

    // A card is the user's current plan if they're signed in and its id matches their
    // admin-approved plan (myActivePlanId), OR it's the free card and they have no active
    // paid plan at all. Everything else always reads "Upgrade to X", even the ৳0 card, so a
    // Pro user browsing back to Free correctly sees "Upgrade" isn't offered but Free isn't
    // mislabeled as their current plan either.
    const isMine = currentUser ? (p.id === myActivePlanId) : isFree;
    // A submitted pending order replaces the Upgrade button on that exact plan card.
    // Normalize ids/status because admin-published plan ids may not use the built-in names.
    const orderForPlan = currentUser ? (myPlanOrderStates[String(p.id)] || (myLatestOrderInfo && String(myLatestOrderInfo.planId) === String(p.id) ? myLatestOrderInfo : null)) : null;
    const orderStatusForPlan = String(orderForPlan && orderForPlan.status || '').toLowerCase();

    // Badge: "Current" always wins (shows what the user actually owns). Otherwise, a discount
    // badge takes priority over the plain custom badge, since "33% off" is more useful to see
    // at a glance than "Popular" — but a plan can still carry both a badgeText AND a discount;
    // only one badge chip is shown at a time to avoid crowding the card.
    let badgeHtml = '';

      badgeHtml = `<div class="plan-badge">${escapeHtml(p.badgeText)}</div>`;
    }

      <div class="plan-card${p.popular?' featured':''}${isMine?' current-plan':''}" data-plan="${p.id}" data-name="${escapeHtml(p.name)}" onclick="selectPlan(this, true)">

        ${ctaHtml}
      </div>`;
  }).join('');
  scroll.innerHTML = cardsHtml;
  // No plan is highlighted merely because it is the current/default plan.
  // A border and checked radio appear only after the visitor explicitly taps a card.
  scroll.querySelectorAll('.plan-card.selected').forEach(card=>card.classList.remove('selected'));

function ethvelleEstimateFace(srcCanvas,joints){
  if(!srcCanvas||!joints||!joints.headTop||!joints.neck) return null;
  const top=joints.headTop, neck=joints.neck;
  const headH=Math.max(0.025,neck.y-top.y);

    mouth:{x:cx,y:Math.min(.995,top.y+headH*.88),w:headW*.24,h:headH*.095},
    source:'estimated'

  try{

    const px=Math.round(Math.max(0,Math.min(srcCanvas.width-1,cx*srcCanvas.width)));

    const required=['headTop','neck','shoulderL','shoulderR','elbowL','elbowR','wristL','wristR','hip','kneeL','kneeR','ankleL','ankleR'];
    const merged={...autoRigState.joints};
    required.forEach(k=>{if(detected[k])merged[k]=detected[k];});
    // Keep Gemini's lateral shoulder positions; buildRigOverrideFromJoints converts them to
    // torso-local attachment offsets so the shoulder pivots stay on the drawn silhouette.
    const validOrder=merged.headTop.y<merged.neck.y&&merged.neck.y<merged.hip.y&&merged.kneeL.y>merged.hip.y&&merged.kneeR.y>merged.hip.y&&merged.ankleL.y>merged.kneeL.y&&merged.ankleR.y>merged.kneeR.y;
    if(!validOrder)throw new Error('Gemini returned inconsistent body-joint order. Use the local rig and adjust the dots manually.');
    Object.keys(detected).forEach(k=>{if(!required.includes(k))merged[k]=detected[k];});
    autoRigState.joints=merged;
    autoRigState.originalJoints=JSON.parse(JSON.stringify(merged));
    autoRigState.attachAt={armL:Math.max(0.2,Math.min(0.98,(merged.hip.y-merged.shoulderL.y)/Math.max(.01,merged.hip.y-merged.neck.y))),armR:Math.max(0.2,Math.min(.98,(merged.hip.y-merged.shoulderR.y)/Math.max(.01,merged.hip.y-merged.neck.y)))};
    autoRigState.face=ethvelleNormalizeFace(parsed.face,autoRigState.srcCanvas,merged);
    autoRigState.aiProvider='Gemini';
    autoRigState.aiConfidence=Number(parsed.confidence)||null;
    autoRigState.handRig=!!(merged.palmL&&merged.palmR&&['L','R'].some(side=>['thumb','index','middle','ring','pinky'].some(f=>merged[f+'Base'+side]&&merged[f+'Mid'+side]&&merged[f+'Tip'+side])));
    autoRigState.warnings=Array.isArray(parsed.warnings)?parsed.warnings.filter(x=>typeof x==='string').slice(0,5):[];
    renderAutoRigWarnings();renderAutoRigOverlay();
    if(status)status.textContent='Gemini analysis complete. Check the points before creating the rig; missing fingertips may still need manual correction.';
    if(!silent)showToast('Gemini detected the body and face landmarks',{icon:toastIcons.check,duration:1800});
  }catch(err){
    console.warn('Ethvelle Gemini rig analysis failed:',err);
    if(status)status.textContent='Gemini analysis failed: '+(err.message||'unknown error')+' Local detection is still available.';
    if(!silent)showToast('Gemini analysis failed — local rig is unchanged.',{icon:toastIcons.info,duration:2600});
  }finally{
    if(btn){btn.disabled=false;btn.textContent='Analyze with Gemini · hands + face';}
  }
}

function ethvelleFaceTime(){
  if(Number.isFinite(faceAnimationTimeOverride))return faceAnimationTimeOverride;
  return Number.isFinite(window.currentAbsoluteTimeSec)?window.currentAbsoluteTimeSec:(typeof currentAbsoluteTimeSec==='number'?currentAbsoluteTimeSec:0);
}
function ethvelleMapFacePoint(char,point,bStart,bEnd){
  const mesh=char.skinMesh,face=char.autoRig?.face;
  if(!mesh||!mesh.restBones?.head||!point)return null;
  const rest=mesh.restBones.head,s=bStart('head'),e=bEnd('head');
  const angle=Math.atan2(e.y-s.y,e.x-s.x),len=Math.max(1e-5,Math.hypot(e.x-s.x,e.y-s.y));
  const x=point.x*mesh.srcW,y=point.y*mesh.srcH,dx=x-rest.ax,dy=y-rest.ay;
  const ca=Math.cos(-rest.angle),sa=Math.sin(-rest.angle),lx=dx*ca-dy*sa,ly=dx*sa+dy*ca;
  const scale=Math.max(.92,Math.min(1.08,len/rest.len)),rot=Math.cos(angle),sin=Math.sin(angle),delta=angle-rest.angle;
  return {x:s.x+(lx*scale*Math.cos(angle)-ly*scale*Math.sin(angle)),y:s.y+(lx*scale*Math.sin(angle)+ly*scale*Math.cos(angle)),angle:delta,scale};
}
function drawEthvelleFaceMotion(targetCtx,char,bStart,bEnd){
  const face=char.autoRig?.face,mesh=char.skinMesh;
  if(!face||!mesh||!mesh.restBones?.head)return;
  const t=ethvelleFaceTime();
  const hash=String(char.id||'c').split('').reduce((a,c)=>a+c.charCodeAt(0),0)%37/37;
  const phase=((t+hash*1.7+0.8)%3.7+3.7)%3.7;
  const blink=phase<0.105 || (phase>3.48&&phase<3.58);
  const clipKey=(char.id===activeCharacter()?.id&&activeClip)?activeClip.animKey:(char.clip?.animKey||null);
  const talking=(clipKey==='talk'||clipKey==='talkBlink') || char.faceMotion?.talking===true;
  const skin= /^#[0-9a-f]{6}$/i.test(face.skinColor||'')?face.skinColor:'#dba982';
  function drawFeature(name,mode){
    const f=face[name];if(!f)return;
    if(mode==='eye'&&!blink)return;
    if(mode==='mouth'&&!talking)return;
    const center=ethvelleMapFacePoint(char,f,bStart,bEnd);if(!center)return;
    const w=Math.max(1,Number(f.w||.02)*mesh.srcW*center.scale),h=Math.max(1,Number(f.h||.01)*mesh.srcH*center.scale);
    targetCtx.save();targetCtx.translate(center.x,center.y);targetCtx.rotate(center.angle);
    targetCtx.fillStyle=skin;targetCtx.beginPath();targetCtx.ellipse(0,0,w*.72,h*.9,0,0,Math.PI*2);targetCtx.fill();
    targetCtx.lineCap='round';targetCtx.lineJoin='round';
    if(mode==='eye'){
      if(blink){targetCtx.strokeStyle='#392922';targetCtx.lineWidth=Math.max(1,w*.10);targetCtx.beginPath();targetCtx.moveTo(-w*.48,h*.05);targetCtx.quadraticCurveTo(0,h*.34,w*.48,h*.05);targetCtx.stroke();}
    }else if(mode==='mouth'&&(talking)){
      const open=.10+.5*(.5+.5*Math.sin(t*16.5+hash*3));
      targetCtx.fillStyle='#3e1720';targetCtx.strokeStyle='#3e1720';targetCtx.lineWidth=Math.max(1,w*.055);
      targetCtx.beginPath();targetCtx.ellipse(0,0,w*.47,Math.max(h*.12,h*open),0,0,Math.PI*2);targetCtx.fill();targetCtx.stroke();
      if(open>.34){targetCtx.fillStyle='#e9788a';targetCtx.beginPath();targetCtx.ellipse(0,h*open*.43,w*.22,h*.10,0,0,Math.PI*2);targetCtx.fill();}
    }
    targetCtx.restore();
  }
  drawFeature('leftEye','eye');drawFeature('rightEye','eye');drawFeature('mouth','mouth');
}

// (pants). Earlier builds forced one arm behind the body, which hid that hand under the hoodie/pants
// while the other arm stayed on top. Arm pieces that are already in front are left exactly where they

// Bird tail artwork is always layered over body art.
function sheetDepthOrderForView(order, modelType, view){
  let out=Array.isArray(order)?order.slice():[];
  // v352: Mermaid and Naga / Lamia both need the head artwork behind the torso.
  // The body overlaps the lower jaw/neck join so the head does not look pasted onto
  // the front of the base body. Keep every full-expression-head swap in the same slot;
  // independently nested eyes and brows are still painted in the face pass.
  if(modelType==='lamia'||modelType==='mermaid'){
    const backHeadKeys=['head','head_blink','head_open','head_sad','head_angry','head_shock','head_laugh'];
    const headParts=out.filter(k=>backHeadKeys.includes(k));
    out=out.filter(k=>!backHeadKeys.includes(k));
    const torsoAt=out.indexOf('torso');
    out.splice(torsoAt<0?0:torsoAt,0,...headParts);
  }
  if(modelType==='bird'){
    const ti=out.indexOf('tail'),bi=out.indexOf('torso');
    if(ti>=0&&bi>=0&&ti<bi){out.splice(ti,1);out.splice(out.indexOf('torso')+1,0,'tail');}
    // v325: the closed-eye lid must sit UNDER the open eye, never on top of it.
    const bci=out.indexOf('bird_eye_closed'),boi=out.indexOf('bird_eye');
    if(bci>=0&&boi>=0&&bci>boi){out.splice(bci,1);out.splice(out.indexOf('bird_eye'),0,'bird_eye_closed');}
  }
  if(modelType==='humanoid'||isWingedHumanoidType(modelType)){
    const chainL=['handL','handL_fist','handL_point','handL_thumb','handL_palm','handL_hold','armL_lower','armL_upper'];
    const chainR=['handR','handR_fist','handR_point','handR_thumb','handR_palm','handR_hold','armR_lower','armR_upper'];
    const armKeys=[...chainL,...chainR];
    const bodyKeys=['torso','legL_upper','legL_lower','shoeL','legR_upper','legR_lower','shoeR'];
    // v324 FRONT VIEW: the viewer-left arm (L keys) is the character's RIGHT arm and stays IN FRONT of the
    // body, pants and shoes. The viewer-right arm (R keys) is the character's LEFT arm and is tucked UNDER
    // the body and pants (drawn first). Distal-first order inside each limb is preserved.
    if(!view || view==='front'){
      const under=out.filter(k=>chainR.includes(k));
      out=out.filter(k=>!chainR.includes(k));
      out.unshift(...under);
      let lastBodyF=-1;
      out.forEach((k,i)=>{ if(bodyKeys.includes(k)) lastBodyF=i; });
      if(lastBodyF>=0){
        const behindL=out.filter((k,i)=>chainL.includes(k)&&i<lastBodyF);
        if(behindL.length){
          out=out.filter((k,i)=>!(chainL.includes(k)&&i<lastBodyF));
          out.push(...behindL);
        }
      }
      return out;
    }
    let lastBody=-1;
    out.forEach((k,i)=>{ if(bodyKeys.includes(k)) lastBody=i; });
    if(lastBody>=0){
      const behind=out.filter((k,i)=>armKeys.includes(k)&&i<lastBody);
      if(behind.length){
        out=out.filter((k,i)=>!(armKeys.includes(k)&&i<lastBody));
        // keep distal-first order inside each limb, and keep the arms after body/legs/head
        out.push(...behind);
      }
    }
  }
  return out;
}

// Given a base key and a character's expression state, returns which key should actually be drawn.
// `parts` is whatever object holds the sheet's known part keys (sh.parts at runtime, ED.arts while
// editing) -- if the alt texture isn't actually part of this particular model, always falls back to
// the base so an expression flag left set on an old/basic model can never blank out a part.
function sheetActiveKey(baseKey, expr, parts){
  if(!expr || !parts) return baseKey;
  if(baseKey === 'head'){
    const emo = expr.emo ? 'head_'+expr.emo : null;
    // Keep expression heads intact; blinking is composited as an eye-only overlay for moods.
    if(expr.eyeState==='closed' && !(emo && parts[emo])) return parts.head_blink ? 'head_blink' : baseKey;
    if(expr.eyeState==='open' && !(emo && parts[emo]) && parts.head_open) return 'head_open';
    if((expr.blink || expr.ambientBlink) && !(emo && parts[emo])) return parts.head_blink ? 'head_blink' : baseKey;
    return emo && parts[emo] ? emo : baseKey;
  }
  if(baseKey === 'beak_closed' && expr.mouth === 'open' && parts.beak_open) return 'beak_open';
  if((baseKey === 'armL_upper' || baseKey === 'armR_upper') && expr.flying){ const fly=baseKey==='armL_upper'?'wingL_fly':'wingR_fly'; if(parts[fly]) return fly; }
  if(baseKey === 'mouth_closed') {
    if(expr.mouth === 'open' && parts.mouth_open) return 'mouth_open';
    if(expr.mouth === 'hold' && parts.mouth_hold) return 'mouth_hold';
    return baseKey;
  }
  const side = baseKey === 'handL' ? 'L' : baseKey === 'handR' ? 'R' : '';
  if(side){
    if((side === 'L' ? expr.fistL : expr.fistR) && parts['hand'+side+'_fist']) return 'hand'+side+'_fist';
    const g = side === 'L' ? expr.gestL : expr.gestR;
    if(g && parts['hand'+side+'_'+g]) return 'hand'+side+'_'+g;
  }
  return baseKey;
}
