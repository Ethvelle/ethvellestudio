  function fft(re,im,inverse){
    const n=re.length;
    for(let i=1,j=0;i<n;i++){
      let bit=n>>1;for(;j&bit;bit>>=1)j^=bit;j^=bit;
      if(i<j){let t=re[i];re[i]=re[j];re[j]=t;t=im[i];im[i]=im[j];im[j]=t;}
    }
    for(let len=2;len<=n;len<<=1){
      const angle=(inverse?2:-2)*Math.PI/len,wr0=Math.cos(angle),wi0=Math.sin(angle);
      for(let i=0;i<n;i+=len){let wr=1,wi=0;const half=len>>1;
        for(let j=0;j<half;j++){
          const u=i+j,v=u+half,vr=re[v]*wr-im[v]*wi,vi=re[v]*wi+im[v]*wr;
          re[v]=re[u]-vr;im[v]=im[u]-vi;re[u]+=vr;im[u]+=vi;
          const nwr=wr*wr0-wi*wi0;wi=wr*wi0+wi*wr0;wr=nwr;
        }
      }
    }
    if(inverse)for(let i=0;i<n;i++){re[i]/=n;im[i]/=n;}
  }
  async function spectralClean(buffer,mode,strength,onProgress){
    const fftSize=1024,hop=256,half=fftSize>>1,windowFn=new Float32Array(fftSize);
    for(let i=0;i<fftSize;i++)windowFn[i]=Math.sqrt(.5-.5*Math.cos(2*Math.PI*i/(fftSize-1)));
    const channels=[],totalFrames=Math.ceil(buffer.length/hop),channelCount=buffer.numberOfChannels;
    for(let ch=0;ch<channelCount;ch++){
      const source=buffer.getChannelData(ch),out=new Float32Array(buffer.length),weight=new Float32Array(buffer.length);
      const starts=[];for(let s=-fftSize+hop;s<buffer.length;s+=hop)starts.push(s);
      const training=[];for(const s of starts){if(s>buffer.sampleRate*1.8)break;let sum=0,n=0;for(let i=0;i<fftSize;i+=4){const ix=s+i;if(ix>=0&&ix<source.length){const v=source[ix];sum+=v*v;n++;}}if(n)training.push({s,rms:Math.sqrt(sum/n)});}
      training.sort((a,b)=>a.rms-b.rms);const quiet=training.slice(0,Math.max(3,Math.ceil(training.length*.24)));
      const profile=new Float32Array(half+1),re=new Float64Array(fftSize),im=new Float64Array(fftSize);
      for(const item of quiet){re.fill(0);im.fill(0);for(let i=0;i<fftSize;i++){const ix=item.s+i;re[i]=(ix>=0&&ix<source.length?source[ix]:0)*windowFn[i];}fft(re,im,false);for(let k=0;k<=half;k++)profile[k]+=Math.hypot(re[k],im[k]);}
      const denom=Math.max(1,quiet.length);for(let k=0;k<=half;k++)profile[k]/=denom;
      // A short moving average keeps the learned noise floor smooth between FFT bins.
      const smooth=profile.slice();for(let k=1;k<half;k++)profile[k]=(smooth[k-1]+2*smooth[k]+smooth[k+1])/4;
      const modeFactor=mode==='noise'?(0.95+strength*1.8):(0.25+strength*.62);
      const floorGain=mode==='noise'?(0.035+(1-strength)*.20):.42;
      let done=0;
      for(const s of starts){
        re.fill(0);im.fill(0);
        for(let i=0;i<fftSize;i++){const ix=s+i;re[i]=(ix>=0&&ix<source.length?source[ix]:0)*windowFn[i];}
        fft(re,im,false);
        const gains=new Float32Array(half+1);
        for(let k=0;k<=half;k++){
          const mag=Math.hypot(re[k],im[k]),noise=profile[k]*modeFactor;
          const raw=mag>1e-9?Math.max(floorGain,Math.min(1,(mag-noise)/mag)):1;
          gains[k]=raw;
        }
        // Frequency smoothing reduces the watery artifacts of a hard spectral gate.
        const copy=gains.slice();for(let k=1;k<half;k++)gains[k]=(copy[k-1]+2*copy[k]+copy[k+1])/4;
        for(let k=0;k<=half;k++){const g=gains[k];re[k]*=g;im[k]*=g;if(k>0&&k<half){re[fftSize-k]*=g;im[fftSize-k]*=g;}}
        fft(re,im,true);
        for(let i=0;i<fftSize;i++){const ix=s+i;if(ix>=0&&ix<out.length){const w=windowFn[i];out[ix]+=re[i]*w;weight[ix]+=w*w;}}
        done++;if(done%32===0){onProgress(((ch+done/starts.length)/channelCount)*82,mode==='noise'?'Removing background noise…':'Enhancing audio…');await yieldUI();}
      }
      const clean=new Float32Array(buffer.length);
      for(let i=0;i<clean.length;i++)clean[i]=weight[i]>1e-6?out[i]/weight[i]:source[i];
      if(mode==='enhance'){
        const amount=strength;let peak=0;
        for(let i=0;i<clean.length;i++){let x=clean[i],a=Math.abs(x);if(a>.22){const compressed=.22+(a-.22)/(1+amount*1.8);x=Math.sign(x)*compressed;}clean[i]=x;peak=Math.max(peak,Math.abs(x));}
        const gain=Math.min(2.8,peak>1e-6?.90/peak:1);for(let i=0;i<clean.length;i++)clean[i]=Math.max(-1,Math.min(1,clean[i]*gain));
      }
      channels.push(clean);onProgress(((ch+1)/channelCount)*86,mode==='noise'?'Finalizing cleaned audio…':'Finalizing enhanced audio…');await yieldUI();
    }
    return {numberOfChannels:channelCount,length:buffer.length,sampleRate:buffer.sampleRate,duration:buffer.duration,getChannelData:(ch)=>channels[ch]};
  }
