// BUGFIX ("it show a use smaller background for save notification all time" — the toast
// kept reappearing on every single edit once a project was over quota): persistProjectState
// calls this on almost every interaction (every drag, every trim, every keyframe, many with
// skipHistory:true for live-updates-while-dragging), so once localStorage's ~5-10MB quota
// was exceeded, EVERY one of those calls failed and re-showed the exact same toast — it
// wasn't stuck on screen, it was being re-triggered dozens of times a minute, which reads
// identically to "won't go away". lastQuotaToastAt throttles it to once per 8 seconds no
// matter how many saves fail in that window, so the person can keep editing without a wall
// of repeat toasts, while still being told the truth the moment they pause.
let lastQuotaToastAt = 0;
function saveProjectsToStorage(){
  try{
    localStorage.setItem(STORAGE_KEY, JSON.stringify(projects));
    return true;
  }catch(err){
    // A background image, or any of several background/overlay images/videos now that both
    // rows support multiple clips, is stored as base64 (see the various upload handlers'
    // BUGFIX notes) rather than a tiny blob: URL reference, which makes hitting
    // localStorage's quota (~5-10MB depending on browser) a real possibility once a project
    // has accumulated enough media. Surfacing this with a toast instead of only a
    // console.warn matters: silently failing to save would mean edits look fine in the live
    // editor but quietly never persist, only to vanish on next reopen.
    console.warn('Ethvelle: could not save projects', err);
    const now = Date.now();
    if(now - lastQuotaToastAt > 8000){
      lastQuotaToastAt = now;
      // BUGFIX: no longer blames "background image" specifically — with multiple
      // backgrounds AND multiple overlay clips now possible, the actual cause just as often
      // is several images/videos added over time rather than one large background, so the
      // message names the real, fixable options instead of pointing at only one of them.
      showToast("Storage is full — try removing an unused clip/background or using a smaller image", {icon: toastIcons.info, duration:3200});
    }
    return false;
  }
}

// Builds a brand-new character (own rig + own default animation clip) from a saved model
// asset and adds it to the CURRENT project — this is the "use this model in any project"
// action. Mirrors finalizeAutoRiggedCharacter's own character-construction path exactly, just
// sourced from a saved asset's data instead of a freshly-uploaded image.
function addCharacterFromModelAsset(assetOrId){
  // Resolve the live library entry again at click time. The card can outlive an edit, and a
  // published copy can be stale; the locally edited asset is always the authoritative version.
  const id=String(assetOrId && typeof assetOrId==='object' ? assetOrId.id : assetOrId || '');
  const local=modelAssets.find(a=>String(a.id)===id);
  const published=Array.isArray(publishedModelAssets)?publishedModelAssets.find(a=>String(a.id)===id):null;
  const source=local || published || (assetOrId && typeof assetOrId==='object' ? assetOrId : null);
  if(!source) return null;
  const asset=JSON.parse(JSON.stringify(source));
  // The Lab's sheet override lives inside `sheet`; never feed it into the main site's legacy rig.
  const c = makeCharacter(asset.name, asset.sheet ? undefined : asset.rigOverride);
  c.type = 'custom';
  c.modelType = (asset.sheet && asset.sheet.modelType) || asset.modelType || 'humanoid';
  c.thumb = asset.thumb || (asset.sheet && asset.sheet.thumb) || undefined;
  c.assetId = asset.id;
  c.sheet = asset.sheet ? JSON.parse(JSON.stringify(asset.sheet)) : undefined;
  ensureSpeciesTailBone(c);
  c.expr = c.sheet ? (typeof sheetExprFresh === 'function' ? sheetExprFresh() : {}) : undefined;
  c.parts = c.sheet ? {} : (asset.parts ? JSON.parse(JSON.stringify(asset.parts)) : undefined);
  c.autoRig = c.sheet ? undefined : (asset.autoRig ? JSON.parse(JSON.stringify(asset.autoRig)) : undefined);
  if(c.sheet){ c.rigOverride = undefined; sheetPlace(c); }
  // UPDATE (default placeholder clip on add): makeCharacter() now gives every new character
  // a 'default' (static, no-motion) clip automatically — see makeCharacter's own comment —
  // so a model pulled in from the library shows up on the timeline immediately too, same as
  // plain addCharacter()/finalizeAutoRiggedCharacter. No forced 'walk'/remembered-default
  // animation here, just the same static placeholder every other creation path gets.
  characters.push(c);
  activeCharIdx = characters.length-1;
  syncActiveCharacterRefs();
  selectedJoint = null;
  if(navigator.vibrate) navigator.vibrate(12);
  trackEvent('model_reused');
  showToast('Model added', { duration:1400 });
  preloadCustomCharacterImages(ctx, [c]).then(()=>c.sheet ? undefined : rebuildSkinMeshesForCharacters([c])).then(()=>{

    drawStick();
    renderCharacterLayers(); syncTransformSliders(); persistProjectState();
    renderTimelineClip(); syncTimelineInnerWidth(); renderRuler();
  });
  return c;
}

// A project cover represents only the content visible at absolute time zero. Compare that
// slice of the saved state before scheduling a refresh, so routine timeline scrubs, selection
// changes, keyframe edits away from zero, etc. never queue a cover capture.
function ethvelleFrameZeroCoverContentView(state){
  if(!state || typeof state!=='object') return null;
  const atZero=item=>!!item && (Number(item.trimStart)||0)<=0.0001 && Number(item.trimEnd)>0.0001;
  const pick=(item,keys)=>{
    const out={};
    keys.forEach(key=>{if(item && Object.prototype.hasOwnProperty.call(item,key))out[key]=item[key];});
    return out;
  };
  const backgroundRows=Array.isArray(state.backgrounds)?state.backgrounds:[];
  const overlaysRows=Array.isArray(state.overlays)?state.overlays:[];
  const charactersRows=Array.isArray(state.characters)?state.characters:[];
  return {
    backgrounds:backgroundRows.filter(atZero).map(b=>pick(b,[
      'id','src','trimStart','trimEnd','transform','keyframes','ethvelleMediaKeyframes','mediaPerspectiveX','mediaPerspectiveY','mediaWide','mediaLength'
    ])),
    legacyBackground:backgroundRows.length?null:pick(state,['currentBg','bgTransform','bgDuration']),
    overlays:overlaysRows.filter(atZero).map(o=>pick(o,[
      'id','type','src','trimStart','trimEnd','sourceDuration','sourceOffset','speed',
      'reversePlayback','__ethvelleBehindBackground','transform','opacity','adjustments','filterPreset','chromaKey','mediaPerspectiveX','mediaPerspectiveY','mediaWide','mediaLength',
      'keyframes','ethvelleMediaKeyframes','textContent','fontFamily','fontSize',
      'fontWeight','fontStyle','textAlign','textColor','stroke'
    ])),
    characters:charactersRows.map(c=>({
      id:c.id,type:c.type,modelType:c.modelType,parts:c.parts,rigOverride:c.rigOverride,
      autoRig:c.autoRig,expr:c.expr,handProps:c.handProps,
      restTransform:c.restTransform||c.modelTransform,
      clip:atZero(c.clip)?c.clip:null,
      segments:(Array.isArray(c.segments)?c.segments:[]).filter(atZero)
    }))
  };
}
function ethvelleFrameZeroCoverContentEqual(a,b){
  if(a===b)return true;
  if(a==null||b==null||typeof a!==typeof b)return false;
  if(typeof a!=='object')return a===b;
  const aa=Array.isArray(a),bb=Array.isArray(b);
  if(aa!==bb)return false;
  const ak=Object.keys(a),bk=Object.keys(b);
  if(ak.length!==bk.length)return false;
  for(let i=0;i<ak.length;i++){
    const key=ak[i];
    if(!Object.prototype.hasOwnProperty.call(b,key)||!ethvelleFrameZeroCoverContentEqual(a[key],b[key]))return false;
  }
  return true;
}
// Build isolated character objects for first-frame cover rendering. Bone records and model
// transforms are cloned, so reset/pose evaluation never mutates the live editor's rig.
function ethvelleCloneCharactersForFirstFrameCover(){
  return characters.map(char=>{
    const rest={...(char.restTransform||char.modelTransform||{})};
    const copy={
      ...char,
      bones:Object.fromEntries(Object.entries(char.bones||{}).map(([key,bone])=>[key,{...bone}])),
      baseAngles:{...(char.baseAngles||{})},
      modelTransform:{...rest},
      restTransform:{...rest}
    };
    resetCharacterForTimelineFrame(copy);
    if(copy.clip){
      const pose=poseAtClipTime(0,copy.clip,copy.segments||[]);
      if(pose)applyPoseToBones(pose,copy.bones,copy.baseAngles,copy.hipRestY);
      const clipTransform=transformAtClipTime(0,copy.clip,copy.segments||[]);
      if(clipTransform)applyTransformToModel(clipTransform,copy.modelTransform);
    }
    return copy;
  });
}

// Persist the live editor state (pose, clip, transform, bg, overlays) into the open project
function persistProjectState(opts){
  if(!currentProject) return;
  const previousSavedStateForCover=currentProject.savedState;
  currentProject.savedState = {
    currentAnimKey,
    timelineDuration,
    // Preserve the timeline position so time-based text overlays are still visible after reload.
    playheadTimeSec: Number.isFinite(Number(currentAbsoluteTimeSec)) ? Math.max(0, Number(currentAbsoluteTimeSec)) : 0,
    // Every character's own Transform (size/position/rotation) AND its own animation
    // clip/segments — bone pose itself is never persisted (same as the original
    // single-character app), only rebuilt fresh at rest pose on reopen. modelTransform
    // (singular, top-level) is intentionally no longer written — openEditor's migration path
    // reads it only from OLD saves that predate this array. clip/segments are saved per
    // character now (not as one shared top-level activeClip/splitStickmanSegments) so every
    // character keeps its own animation independently across save/reload, not just whichever
    // one happened to be active when you saved.
    characters: characters.map(c=>({
      id:c.id, name:c.name, modelTransform:{...c.modelTransform},
      restTransform:{...(c.restTransform || c.modelTransform)},
      clip: c.clip ? JSON.parse(JSON.stringify(c.clip)) : null,
      segments: JSON.parse(JSON.stringify(c.segments||[])),
      // Custom (image-based) character data — see PART_SLOTS/finishCustomCharacterBuild.
      // `parts` holds each uploaded part's base64 image data URL directly (same pattern as
      // overlays' `src` and currentBg above), so a custom character's artwork survives a
      // reload exactly like an uploaded background/clip image already does. Undefined for
      // an ordinary stickman character, so old saves and new plain characters both stay the
      // same tiny size.
      type: c.type,
      parts: c.parts ? JSON.parse(JSON.stringify(c.parts)) : undefined,
      // Auto-rig data (see buildRigOverrideFromJoints/applyRigOverride and autorig-detect's
      // ported detectCharacter): `rigOverride` is the bone length/angle/attachAt fractions
      // openEditor's restore path re-applies on top of a fresh buildFreshRig() (bone pose is
      // never persisted, per the note above, but these PROPORTIONS must be, or an auto-rigged
      // character would revert to the generic stickman proportions on reload). `autoRig` keeps
      // the original uploaded image + detected/adjusted joints so "Rig my art" can re-open this
      // character straight into the Auto Rig Preview for further correction later (section 16/17
      // of the feature — re-detect must start from the ORIGINAL image, not a re-compressed part
      // crop). Both undefined for a stickman or an old manual-parts custom character.
      rigOverride: c.rigOverride ? JSON.parse(JSON.stringify(c.rigOverride)) : undefined,
      autoRig: c.autoRig ? JSON.parse(JSON.stringify(c.autoRig)) : undefined,
      modelType: c.modelType || undefined,
      thumb: c.thumb || undefined,
      sheet: c.sheet ? JSON.parse(JSON.stringify(c.sheet)) : undefined,
      expr: c.expr ? JSON.parse(JSON.stringify(c.expr)) : undefined,
      handProps: c.handProps ? JSON.parse(JSON.stringify(c.handProps)) : undefined,
      assetId: c.assetId || undefined
    })),
    activeCharIdx,
    // FEATURE (multiple sequential backgrounds): `backgrounds` is now the real source of
    // truth — an array of clips, same shape as `overlays`, each with its own src/transform/
    // trimStart/trimEnd. bgTransform/currentBg/bgDuration below are kept ADDITIONALLY,
    // written from the currently-active clip, purely so a project saved by an older version
    // of this file (or opened by code that hasn't been updated) still has something sane to
    // read — they are never the authoritative data once `backgrounds` is present (see
    // openEditor's migration path below, which only falls back to them when `backgrounds`
    // is missing entirely from an old save).
    backgrounds: backgrounds.map(b=>({...b, transform:{...b.transform}})),
    bgTransform: {...bgTransform},
    currentBg,
    bgDuration,
    // BUGFIX (this was the "clips aren't saving after refreshing" bug — see the fileInput
    // 'change' handler's matching BUGFIX note for the fix itself): `src` now always holds a
    // `url(data:...)`-free base64 data: URL string too (image/video/audio alike), exactly
    // like currentBg above — never a `url(blob:...)` reference. Blob URLs die the moment the
    // page session ends, so a saved clip's actual picture/video/audio used to go blank on
    // reopen even though its trim/position data (and thumbs/peaks, already plain data)
    // survived fine — this spread now carries a `src` that survives right along with them.
    overlays: overlays.map(o=>({...o, thumbs:o.thumbs?[...o.thumbs]:undefined, peaks:o.peaks?[...o.peaks]:undefined}))
  };
  currentProject.duration = activeClip ? +(combinedStickmanDuration() - activeClip.trimStart).toFixed(1) : currentProject.duration;
  const idx = projects.findIndex(p=>p.id===currentProject.id);
  if(idx>-1) projects[idx] = currentProject;
  saveProjectsToStorage();
  const previousFrameZeroCoverContent=ethvelleFrameZeroCoverContentView(previousSavedStateForCover);
  const nextFrameZeroCoverContent=ethvelleFrameZeroCoverContentView(currentProject.savedState);
  if(!ethvelleFrameZeroCoverContentEqual(previousFrameZeroCoverContent,nextFrameZeroCoverContent))queueProjectFirstFrameCover(1100);
  if(!opts || !opts.skipHistory) pushHistory();
}

/* ---------------- Undo / redo history ---------------- */
let historyStack = [];
let historyIndex = -1;
let restoringHistory = false;
const HISTORY_LIMIT = 40;

function snapshotState(){
  return JSON.stringify({
    currentAnimKey,
    activeClip,
    splitStickmanSegments,
    timelineDuration,
    bgDuration,
    characters,
    activeCharIdx,
    bgTransform,
    currentBg,
    backgrounds,
    overlays
  });
}
function pushHistory(){
  if(restoringHistory) return;
  const snap = snapshotState();
  if(historyIndex>=0 && historyStack[historyIndex]===snap) return; // no real change
  historyStack = historyStack.slice(0, historyIndex+1);
  historyStack.push(snap);
  if(historyStack.length>HISTORY_LIMIT) historyStack.shift();
  historyIndex = historyStack.length-1;
  updateHistoryButtons();
}
function resetHistory(){
  historyStack = [snapshotState()];
  historyIndex = 0;
  updateHistoryButtons();
}
function updateHistoryButtons(){
  const undoBtn = document.getElementById('undoBtn');
  const redoBtn = document.getElementById('redoBtn');
  if(!undoBtn || !redoBtn) return;
  const canUndo = historyIndex > 0;
  const canRedo = historyIndex < historyStack.length-1;
  undoBtn.classList.toggle('history-available', canUndo);
  redoBtn.classList.toggle('history-available', canRedo);
  undoBtn.style.opacity = canUndo ? '1' : '0.4';
  redoBtn.style.opacity = canRedo ? '1' : '0.4';
  undoBtn.setAttribute('aria-disabled', canUndo ? 'false' : 'true');
  redoBtn.setAttribute('aria-disabled', canRedo ? 'false' : 'true');
}
function applySnapshot(snap){
  const s = JSON.parse(snap);
  restoringHistory = true;
  // Undo/redo mutates activeClip/segments/keyframes directly — never do that while the
  // playback loop might be mid-tick reading them (PART 36: history must apply cleanly,
  // not race a running animation).
  pausePlayback();
  currentAnimKey = s.currentAnimKey;
  timelineDuration = s.timelineDuration;
  // Full rig objects were captured verbatim in this snapshot (unlike the lighter
  // Transform-only shape used for disk saves) — undo/redo restores bone pose exactly as it
  // was at that point in history, since it's an in-memory-only structure with no migration
  // concerns the way older disk saves have. Each character's own clip/segments live nested
  // on the character object itself (see makeCharacter) and are restored as part of this.
  characters = s.characters && s.characters.length ? s.characters : characters;
  // Old history entries saved before per-character clips existed kept activeClip/
  // splitStickmanSegments only at the top level, not nested on any character — patch those
  // onto the (single, pre-multi-character) character here so undo/redo across that boundary
  // never silently drops the clip that was on the timeline at the time.
  characters.forEach(c=>{ if(c.clip===undefined) c.clip = null; if(c.segments===undefined) c.segments = []; });
  activeCharIdx = Math.min(s.activeCharIdx ?? 0, characters.length-1);
  if(s.activeClip !== undefined && characters[activeCharIdx] && !characters[activeCharIdx].clip){
    characters[activeCharIdx].clip = s.activeClip;
    characters[activeCharIdx].segments = s.splitStickmanSegments || [];
  }
  syncActiveCharacterRefs();
  // FEATURE (multiple sequential backgrounds): `backgrounds` is the real data now. Old
  // history entries captured before this array existed only have the single legacy
  // currentBg/bgTransform/bgDuration fields — migrate those into a single one-clip
  // `backgrounds` array on the fly so undo/redo across that boundary never silently loses
  // whatever background was set at that point in history.
  if(s.backgrounds){
    backgrounds = s.backgrounds;
  } else if(s.currentBg){
    backgrounds = [{ id:'bg'+Date.now(), src:s.currentBg, trimStart:0, trimEnd: s.bgDuration ?? s.timelineDuration, transform: s.bgTransform ? {rotation:0, ...s.bgTransform} : {scale:1, offsetX:0, offsetY:0, rotation:0} }];
  } else {
    backgrounds = [];
  }
  refreshCurrentBgFromPlayhead();
  overlays = s.overlays || [];
  applyBackground();
  deselectAllClips();
  frame = 0;
  currentAbsoluteTimeSec = 0;
  renderOverlayLayers();
  renderBackgroundTrack();
  renderCharacterLayers(); // rebuilds every character row AND the active one's clip markup
  renderFrame();
  syncTransformSliders();
  persistProjectState({skipHistory:true});
  restoringHistory = false;
}
function undoHistory(){
  if(historyIndex<=0) return;
  historyIndex--;
  applySnapshot(historyStack[historyIndex]);
  updateHistoryButtons();
  showToast('Undo', {icon: toastIcons.info, duration:900});
}
function redoHistory(){
  if(historyIndex>=historyStack.length-1) return;
  historyIndex++;
  applySnapshot(historyStack[historyIndex]);
  updateHistoryButtons();
  showToast('Redo', {icon: toastIcons.info, duration:900});
}

// Renders the Projects → Models grid. The grid always opens with an "Add model" tile (so
// uploading a new model never requires starting a project first — see the "add model's
// batton" request), followed by every saved model asset. Tapping a saved model's thumbnail/
// name opens it straight in the project-free model editor (openModelAssetEditor) so it can be
// re-posed/re-rigged in place; the small pencil/trash icons on the corner rename or delete the
// asset itself, same as the in-project "My models" library sheet already does.
// Model-grid preview helpers. The sheet compiler's legacy `thumb` is intentionally just a
// small head portrait. For sheet-based models we render the saved neutral/rest rig once into an
// offscreen canvas and crop to the visible alpha bounds, producing a full-body preview without
// changing any rig data or the thumbnail used elsewhere in the editor.
const modelGridPreviewPending = new Map();
// v299: model cards now show ONLY the model's face (its neutral head part) rendered from the
// original full-resolution head artwork, not a full-body crop. The result is cached in memory
// only (never written to localStorage) so large high-quality previews cannot bloat saved data.
const modelFacePreviewCache = new Map();
const modelFacePreviewFailed = new Set();
const modelFacePreviewPending = new Map();
function modelFaceSourceFor(asset){
  try{
    if(asset && asset.sheet && asset.sheet.parts && asset.sheet.parts.head && asset.sheet.parts.head.src) return String(asset.sheet.parts.head.src);
    if(asset && asset.parts && asset.parts.head && asset.parts.head.src) return String(asset.parts.head.src);
  }catch(_e){}
  return '';
}
function modelGridThumbSrc(asset){
  const faceSrc = modelFaceSourceFor(asset);
  if(faceSrc){
    if(modelFacePreviewCache.has(faceSrc)) return modelFacePreviewCache.get(faceSrc);
    if(modelFacePreviewFailed.has(faceSrc)) return (asset && asset.thumb) || modelAssetThumbSrc(asset);
    return ''; // wait for the high-quality face render instead of flashing a full-body image
  }
  if(asset && asset.autoRig && asset.autoRig.sourceImage) return asset.autoRig.sourceImage;
  if(asset && asset.thumb) return asset.thumb;
  return asset ? modelAssetThumbSrc(asset) : '';
}
// Face tiles are rendered as a 2048px (2K) square so they stay sharp on high-density phone screens.
// The head artwork is enlarged in gentle <=2x steps (one big jump looks soft or jagged) and then given a
// light unsharp-mask pass. The result is held in memory as a blob URL only - never written to localStorage.
const MODEL_FACE_PREVIEW_PX = 2048;
let modelFaceQueue = Promise.resolve();
function modelFaceResizeSteps(src, tw, th){
  let cur = src, guard = 0;
  while((cur.width !== tw || cur.height !== th) && guard++ < 14){
    const nw = tw > cur.width ? Math.min(tw, cur.width*2) : Math.max(tw, Math.ceil(cur.width/2));
    const nh = th > cur.height ? Math.min(th, cur.height*2) : Math.max(th, Math.ceil(cur.height/2));
    const c = document.createElement('canvas'); c.width = Math.max(1,nw); c.height = Math.max(1,nh);
    const x = c.getContext('2d'); x.imageSmoothingEnabled = true; x.imageSmoothingQuality = 'high';
    x.drawImage(cur,0,0,cur.width,cur.height,0,0,c.width,c.height);
    cur = c;
  }
  return cur;
}
function modelFaceUnsharp(canvas, radius, amount){
  try{
    const ctx = canvas.getContext('2d');
    if(!('filter' in ctx)) return;
    const w = canvas.width, h = canvas.height;
    const blur = document.createElement('canvas'); blur.width = w; blur.height = h;
    const bc = blur.getContext('2d'); bc.filter = 'blur('+radius+'px)'; bc.drawImage(canvas,0,0);
    const a = ctx.getImageData(0,0,w,h), b = bc.getImageData(0,0,w,h), ad = a.data, bd = b.data;
    for(let i=0;i<ad.length;i+=4){
      if(ad[i+3] < 40 || bd[i+3] < 90) continue;   // leave the transparent edge alone: no halos
      for(let c=0;c<3;c++){
        const o = ad[i+c], diff = o-bd[i+c];
        if(diff > 2 || diff < -2){ const v = o+diff*amount; ad[i+c] = v < 0 ? 0 : v > 255 ? 255 : v; }
      }
    }
    ctx.putImageData(a,0,0);
  }catch(err){ console.warn('Ethvelle: face sharpen skipped',err); }
}
// Name kept for existing callers; it now builds the face-only preview.
function generateModelGridFullBodyPreview(asset){
  const faceSrc = modelFaceSourceFor(asset);
  if(!faceSrc) return Promise.resolve('');
  if(modelFacePreviewCache.has(faceSrc)) return Promise.resolve(modelFacePreviewCache.get(faceSrc));
  if(modelFacePreviewPending.has(faceSrc)) return modelFacePreviewPending.get(faceSrc);
  const task = (modelFaceQueue = modelFaceQueue.then(()=>new Promise(r=>setTimeout(r,0))).then(()=>new Promise(resolve=>{
    const img = new Image();
    img.onload = ()=>{
      try{
        const iw = img.naturalWidth, ih = img.naturalHeight;
        if(!iw || !ih){ resolve(''); return; }
        // Crop to the visible (non-transparent) pixels so the face fills the card.
        let sx=0, sy=0, sw=iw, sh=ih;
        if(iw*ih <= 9000000){
          const probe = document.createElement('canvas'); probe.width=iw; probe.height=ih;
          const pc = probe.getContext('2d',{willReadFrequently:true});
          pc.drawImage(img,0,0);
          const px = pc.getImageData(0,0,iw,ih).data;
          let minX=iw, minY=ih, maxX=-1, maxY=-1;
          for(let y=0;y<ih;y++){
            const row=y*iw*4;
            for(let x=0;x<iw;x++){
              if(px[row+x*4+3]>12){ if(x<minX)minX=x; if(x>maxX)maxX=x; if(y<minY)minY=y; if(y>maxY)maxY=y; }
            }
          }
          if(maxX>=minX && maxY>=minY){
            const pad = Math.max(2,Math.round(Math.max(maxX-minX+1,maxY-minY+1)*.03));
            minX=Math.max(0,minX-pad); minY=Math.max(0,minY-pad);
            maxX=Math.min(iw-1,maxX+pad); maxY=Math.min(ih-1,maxY+pad);
            sx=minX; sy=minY; sw=maxX-minX+1; sh=maxY-minY+1;
          }
        }
        // Square 2K output. The face is fitted to ~96% of the tile and ENLARGED when the artwork is smaller,
        // so every card gets a full-resolution tile instead of a small bitmap the browser has to stretch.
        const SIZE = MODEL_FACE_PREVIEW_PX, longest = Math.max(sw,sh);
        const scale = (SIZE*.96)/longest;
        const tw = Math.max(1,Math.round(sw*scale)), th = Math.max(1,Math.round(sh*scale));
        const crop = document.createElement('canvas'); crop.width = sw; crop.height = sh;
        crop.getContext('2d').drawImage(img,sx,sy,sw,sh,0,0,sw,sh);
        const big = modelFaceResizeSteps(crop,tw,th);
        const out = document.createElement('canvas'); out.width = SIZE; out.height = SIZE;
        const oc = out.getContext('2d'); oc.imageSmoothingEnabled = true; oc.imageSmoothingQuality = 'high';
        oc.drawImage(big,Math.round((SIZE-tw)/2),Math.round((SIZE-th)/2));
        if(scale > 1.25) modelFaceUnsharp(out, Math.min(4,Math.max(1.2,scale*.4)), Math.min(1.1,.45+scale*.08));
        else modelFaceUnsharp(out, 1, .35);
        if(out.toBlob) out.toBlob(blob=>resolve(blob ? URL.createObjectURL(blob) : out.toDataURL('image/png')),'image/png');
        else resolve(out.toDataURL('image/png'));
      }catch(err){
        console.warn('Ethvelle: could not build face model preview',err);
        resolve('');
      }
    };
    img.onerror = ()=>resolve('');
    img.src = faceSrc;
  }))).then(url=>{
    modelFacePreviewPending.delete(faceSrc);
    if(url) modelFacePreviewCache.set(faceSrc,url); else modelFacePreviewFailed.add(faceSrc);
    return url;
  });
  modelFacePreviewPending.set(faceSrc,task);
  return task;
}

// Tapping a saved model from the Projects list (as opposed to from inside an already-open

function useModelAssetFromProjects(id){
  const asset = getAvailableModelAssets().find(a=>String(a.id)===String(id));
  if(!asset) return;
  const plan = getMyPlanData();
  if(!plan.unlimitedProjects && plan.maxProjects != null && projects.length >= plan.maxProjects){
    showToast('Your plan allows up to ' + plan.maxProjects + ' projects — upgrade for more, or delete one first', {icon: toastIcons.info, duration:3000});
    return;
  }
  const proj = {
    id: Date.now(),
    name: asset.name || ('Untitled ' + (projects.length+1)),
    ratio: '9:16',
    created: new Date().toISOString(),
    duration: 2,
    savedState: null
  };
  projects.unshift(proj);
  saveProjectsToStorage();
  trackEvent('project');
  renderProjects();
  openEditor(proj);
  setTimeout(()=>addCharacterFromModelAsset(asset), 0);
}

function openEditorById(id){
  const proj = projects.find(p=>p.id===id);
  if(proj){ trackEvent('project_opened'); openEditor(proj); }
}

/* ---- Keyframe info screen (home tile → goTo('keyframeInfo')) ----
   The "Jump in" card's copy depends on whether the person has an existing project to drop
   back into — see refreshKeyframeEntryCard, called by goTo every time this screen is shown
   (projects can change between visits via delete/create elsewhere in the app). */
function refreshKeyframeEntryCard(){
  const title = document.getElementById('keyframeEntryTitle');
  const desc = document.getElementById('keyframeEntryDesc');
  if(!title || !desc) return;
  if(projects.length){
    title.textContent = 'Open your timeline';
    desc.textContent = 'Continue "' + (projects[0].name || 'your last project') + '" and start keying poses.';
  }else{
    title.textContent = 'Start your first animation';
    desc.textContent = 'No projects yet — create one to get a timeline with keyframe tools.';
  }
}
// Projects are unshifted onto `projects` (newest first — see e.g. useModelAssetFromProjects),
// so projects[0] is the most recently touched one. With none yet, falls through to the same
// new-animation flow as "Start a new animation" below it instead of opening nothing.
function openKeyframeEntry(){
  if(projects.length){ openEditorById(projects[0].id); }
  else{ openRatioModal(); }
}
function deleteProject(id){
  selectedProjectIds.delete(id);
  const removedProject=projects.find(p=>String(p.id)===String(id));
  if(removedProject)delete removedProject.textCustomColors;
  try{if(currentProject&&String(currentProject.id)===String(id))currentProject=null;}catch(_e){}
  projects = projects.filter(p=>p.id!==id);
  saveProjectsToStorage();
  trackEvent('project_deleted');
  renderProjects();
  showToast('Project deleted', {icon: toastIcons.check, duration:1200});
}
