function usageBarClass(used, max){
  if(max == null) return '';
  const ratio = max > 0 ? used / max : 1;
  if(ratio >= 1) return 'at-limit';
  if(ratio >= 0.75) return 'near-limit';
  return '';
}
function getMyOwnedPlansForBilling(){
  const plans = Array.isArray(getPlans()) ? getPlans() : [];
  const byId = new Map(plans.map(p=>[String(p.id),p]));
  const ids = [];
  if(currentUser){
    Object.keys(myPlanOrderStates || {}).forEach(id=>{
      const state=myPlanOrderStates[id];
      if(state && String(state.status||'').toLowerCase()==='approved') ids.push(String(id));
    });
    if(myActivePlanId && String(myActivePlanId)!=='free' && !ids.includes(String(myActivePlanId))) ids.unshift(String(myActivePlanId));
  }
  if(!ids.length){
    const fallback = currentUser ? (byId.get(String(myActivePlanId)) || plans.find(p=>Number(p.price)===0)) : plans.find(p=>Number(p.price)===0);
    if(fallback) ids.push(String(fallback.id));
  }
  const unique = [...new Set(ids)];
  const found = unique.map(id=>byId.get(id)).filter(Boolean);
  return found.length ? found : [{id:'free',name:'Free',price:0,features:['3 projects','720p export'],maxProjects:3,unlimitedProjects:false,unlimitedExports:false,maxExports:null,maxQuality:720,creditsPerCycle:0}];
}
function getMyCombinedCreditAllowance(ownedPlans){
  const list=Array.isArray(ownedPlans)?ownedPlans:getMyOwnedPlansForBilling();
  return list.reduce((sum,p)=>sum+Math.max(0,Number(getPlanCreditGrant(p))||0),0);
}
function renderHomeCreditSummary(){

  }

  const names=[...new Set((owned||[]).map(plan=>String(plan&&plan.name||'').trim()).filter(Boolean))];

  const balance=Math.max(0,Number(myCreditsBalance)||0);
  const allowance=typeof getMyCombinedCreditAllowance==='function'?getMyCombinedCreditAllowance(owned):0;
  planEl.textContent=planLabel;
  balanceEl.textContent=(allowance>0?balance.toLocaleString()+' / '+allowance.toLocaleString():balance.toLocaleString())+' credits';
  button.setAttribute('aria-label',planLabel+', '+balance.toLocaleString()+' credits. Open billing and plan');
}

async function hideMyCreditExpense(transactionId){
  const id=String(transactionId||'');
  if(!id||!(currentUser&&currentUser.uid)){showToast('Sign in to manage credit history.',{icon:toastIcons.info,duration:2400});return;}
  const entry=(Array.isArray(myCreditTransactions)?myCreditTransactions:[]).find(t=>String(t&&t.id)===id);
  if(!entry||String(entry.uid||'')!==String(currentUser.uid)||String(entry.type||'').toLowerCase()!=='usage'||!(Number(entry.amount)<0)){
    showToast('Only your credit-usage expenses can be removed.',{icon:toastIcons.info,duration:2600});return;
  }
  if(myHiddenCreditTransactionIds.includes(id))return;
  const confirmed=window.confirm('Delete this expense from your Credits History? Your credit balance and original accounting record will not change.');
  if(!confirmed)return;
  const uid=String(currentUser.uid);
  const next=[...new Set([...myHiddenCreditTransactionIds,id])];
  let synced=false;
  try{
    if(!(firebaseReady&&fbDb&&firebase&&firebase.firestore&&firebase.firestore.FieldValue))throw new Error('CREDIT_HISTORY_SYNC_UNAVAILABLE');
    await fbDb.collection('users').doc(uid).set({hiddenCreditTransactionIds:firebase.firestore.FieldValue.arrayUnion(id)},{merge:true});
    synced=true;
  }catch(err){console.warn('Ethvelle: could not sync hidden credit expense; keeping this removal on this device.',err);}
  const savedLocally=writeLocalHiddenCreditIds(uid,next);
  myHiddenCreditTransactionIds=next;
  if(typeof renderMyPlanScreen==='function')renderMyPlanScreen();
  if(synced){showToast('Expense removed from your visible history. Your credit balance was not changed.',{icon:toastIcons.check,duration:3000});}
  else if(savedLocally){showToast('Expense hidden on this device. Account sync is unavailable.',{icon:toastIcons.info,duration:3200});}
  else{showToast('Could not save this history change. Please check device storage and try again.',{icon:toastIcons.info,duration:3200});}
}

window.hideMyCreditExpense=hideMyCreditExpense;

// Legal acceptance is collected in a separate confirmation panel only after a user
// starts an email or Google sign-in/sign-up action. No auth request runs before acceptance.
let pendingLoginConsentAction = null; // 'email' | 'google' | null
let loginLegalConsentAcceptedOnce = false;
function updateLoginConsentContinueButton(){
  const privacy=document.getElementById('loginPrivacyConsent');
  const terms=document.getElementById('loginTermsConsent');
  const btn=document.getElementById('loginConsentContinueBtn');
  if(btn)btn.disabled=!(privacy&&terms&&privacy.checked&&terms.checked);
  const error=document.getElementById('loginConsentError');
  if(error&&privacy&&terms&&privacy.checked&&terms.checked)error.textContent='';
}
function openLoginConsentPanel(action,preserveChoices){
  pendingLoginConsentAction=action;
  loginLegalConsentAcceptedOnce=false;
  const privacy=document.getElementById('loginPrivacyConsent');
  const terms=document.getElementById('loginTermsConsent');
  if(!preserveChoices){if(privacy)privacy.checked=false;if(terms)terms.checked=false;}
  const title=document.getElementById('loginConsentTitle');
  const copy=document.getElementById('loginConsentCopy');
  if(title)title.innerHTML='Before you <span>continue</span>';
  if(copy)copy.textContent=action==='email'
    ?'Please accept both documents to create your Ethvelle account.'
    :'Please accept both documents before continuing with Google.';
  const error=document.getElementById('loginConsentError');if(error)error.textContent='';
  updateLoginConsentContinueButton();
  const backdrop=document.getElementById('loginConsentBackdrop');
  if(backdrop)backdrop.classList.add('active');
}
function cancelLoginConsentPanel(){
  const backdrop=document.getElementById('loginConsentBackdrop');
  if(backdrop)backdrop.classList.remove('active');
  pendingLoginConsentAction=null;
  loginLegalConsentAcceptedOnce=false;
  const privacy=document.getElementById('loginPrivacyConsent');
  const terms=document.getElementById('loginTermsConsent');
  if(privacy)privacy.checked=false;if(terms)terms.checked=false;
  const error=document.getElementById('loginConsentError');if(error)error.textContent='';
  updateLoginConsentContinueButton();
}
function requireLoginLegalConsent(){
  const privacy=document.getElementById('loginPrivacyConsent');
  const terms=document.getElementById('loginTermsConsent');
  if(!privacy||!terms||!privacy.checked||!terms.checked){
    const error=document.getElementById('loginConsentError');
    if(error)error.textContent='Please accept both the Privacy Policy and Terms & Conditions to continue.';
    updateLoginConsentContinueButton();
    return false;
  }
  return true;
}
function acceptLoginConsentAndContinue(){
  if(!requireLoginLegalConsent())return;
  const action=pendingLoginConsentAction;
  if(!action)return;
  const backdrop=document.getElementById('loginConsentBackdrop');
  if(backdrop)backdrop.classList.remove('active');
  pendingLoginConsentAction=null;
  loginLegalConsentAcceptedOnce=true;
  if(action==='email')performEmailAuth();
  else if(action==='google')performGoogleSignIn();
}
function openLoginLegalPage(which){
  window.__ethvelleLegalReturnScreen='login';
  window.__ethvelleReturnToLoginConsent=true;
  const backdrop=document.getElementById('loginConsentBackdrop');
  if(backdrop)backdrop.classList.remove('active');
  if(typeof window.openEthvelleLegalPage==='function')window.openEthvelleLegalPage(which);
  else goTo(which==='terms'?'terms':'privacy');
}
function returnFromEthvelleLegalPage(){
  const target=window.__ethvelleLegalReturnScreen||'profile';
  goTo(target);
  window.__ethvelleLegalReturnScreen='profile';
  if(target==='login'&&window.__ethvelleReturnToLoginConsent){
    window.__ethvelleReturnToLoginConsent=false;
    openLoginConsentPanel(pendingLoginConsentAction,true);
  }
}
// Opens the login screen. `after` is an optional zero-arg function run automatically once
// sign-in succeeds (used to resume Export right after the user logs in).
function openLoginScreen(after){
  pendingAfterLogin = (typeof after === 'function') ? after : null;
  const active = document.querySelector('.screen.active');
  const activeId = active && active.id && active.id.indexOf('screen-') === 0 ? active.id.slice(7) : null;
  pendingBeforeLoginScreen = (activeId && activeId !== 'login') ? activeId : 'editor';
  setLoginMode('signin');
  document.getElementById('loginError').classList.remove('show');
  showLoginStatus('');
  document.getElementById('loginEmailInput').value = '';
  document.getElementById('loginPasswordInput').value = '';
  cancelLoginConsentPanel();
  window.__ethvelleLegalReturnScreen='profile';
  window.__ethvelleReturnToLoginConsent=false;
  goTo('login');
}
function cancelLoginFlow(){
  cancelLoginConsentPanel();
  pendingAfterLogin = null;
  goTo(currentUser ? 'profile' : 'home');
}
function setLoginMode(mode){
  loginMode = mode;
  const isSignup = mode === 'signup';
  document.getElementById('loginTitle').textContent = isSignup ? 'Create your account' : 'Sign in for Export';
  document.getElementById('loginSub').textContent = isSignup ? 'Sign up to export your animations' : 'Sign in to export your animations';
  document.getElementById('loginSubmitBtn').textContent = isSignup ? 'Sign up' : 'Sign in';
  document.getElementById('loginSwitchText').textContent = isSignup ? 'Already have an account?' : "Don't have an account?";
  document.getElementById('loginSwitchLink').textContent = isSignup ? 'Sign in' : 'Sign up';
  const forgotLink = document.getElementById('forgotPasswordLink');
  if(forgotLink) forgotLink.style.display = isSignup ? 'none' : 'inline';
  document.getElementById('loginError').classList.remove('show');
  showLoginStatus('');
}

}
function showLoginError(msg){
  showLoginStatus('');
  const el = document.getElementById('loginError');
  el.textContent = msg;
  el.classList.add('show');
}
function showLoginStatus(msg){
  const el = document.getElementById('loginAuthStatus');
  if(!el) return;
  el.textContent = msg || '';
  el.classList.toggle('show', !!msg);
}
async function handleForgotPassword(){
  if(loginMode !== 'signin') return;
  const email = document.getElementById('loginEmailInput').value.trim();
  if(!email){ showLoginError('Enter your email address first to reset your password'); return; }
  if(!firebaseReady || !fbAuth){ showLoginError('Password reset is unavailable right now — try again later'); return; }
  const link = document.getElementById('forgotPasswordLink');
  if(link && link.dataset.busy === '1') return;
  if(link){ link.dataset.busy = '1'; link.textContent = 'Sending…'; link.style.pointerEvents = 'none'; }
  document.getElementById('loginError').classList.remove('show');
  showLoginStatus('');
  try{
    await fbAuth.sendPasswordResetEmail(email);
    showLoginStatus('Password reset email sent. Check your inbox and spam folder.');
  }catch(err){
    showLoginError(friendlyAuthError(err));
  }finally{
    if(link){ link.dataset.busy = '0'; link.textContent = 'Forgot password?'; link.style.pointerEvents = ''; }
  }
}
function friendlyAuthError(err){
  const code = err && err.code || '';
  if(code.indexOf('wrong-password') > -1 || code.indexOf('invalid-credential') > -1) return 'Wrong email or password';
  if(code.indexOf('user-not-found') > -1) return 'No account found with that email';
  if(code.indexOf('email-already-in-use') > -1) return 'That email is already registered — try signing in instead';
  if(code.indexOf('weak-password') > -1) return 'Password should be at least 6 characters';
  if(code.indexOf('invalid-email') > -1) return 'Enter a valid email address';
  if(code.indexOf('network-request-failed') > -1) return 'Network error — check your connection';
  if(code.indexOf('popup-closed-by-user') > -1) return 'Google sign-in was closed before finishing';
  return (err && err.message) || 'Something went wrong — try again';
}
function handleEmailAuthSubmit(){
  const email=document.getElementById('loginEmailInput').value.trim();
  const password=document.getElementById('loginPasswordInput').value;
  document.getElementById('loginError').classList.remove('show');
  if(!email||!password){showLoginError('Enter both email and password');return;}
  if(!firebaseReady||!fbAuth){showLoginError('Sign-in is unavailable right now — try again later');return;}
  // Legal acceptance is a required step only when the user is creating an account.
  // The consent checkboxes live in their own modal and are never embedded in the login form.
  if(loginMode==='signup'){
    openLoginConsentPanel('email',false);
    return;
  }
  performEmailAuth();
}
async function performEmailAuth(){
  const wasSignup=loginMode==='signup';
  if(wasSignup&&(!loginLegalConsentAcceptedOnce||!requireLoginLegalConsent()))return;
  loginLegalConsentAcceptedOnce=false;
  const email=document.getElementById('loginEmailInput').value.trim();

/* ================= USERS DIRECTORY (per-user Firestore profile doc) =================
   Mirrors the signed-in user into users/{uid} so the admin panel can list, filter and
   message real accounts. Previously plan/activity only ever lived in this device's
   localStorage — nothing about a user existed in the cloud beyond Firebase Auth's own
   record (which the admin panel cannot list without the Admin SDK). This runs once per
   sign-in and again whenever the active plan changes, as a best-effort merge write —
   never blocks the UI and never throws if Firestore is unreachable. */
function syncUserProfileDoc(user){
  if(!(firebaseReady && fbDb) || !user) return;
  try{
    const ref = fbDb.collection('users').doc(user.uid);
    ref.set({
      uid: user.uid,
      email: user.email || null,
      displayName: user.displayName || null,
      photoURL: user.photoURL || null,
      planId: myActivePlanId,
      planName: myActivePlanName,
      lastSeen: firebase.firestore.FieldValue.serverTimestamp(),
      createdAt: firebase.firestore.FieldValue.serverTimestamp() // ignored by merge if the doc already has one — see below
    }, { merge:true }).catch(()=>{});
    // createdAt should only ever be set once (on first sync) — a plain merge:true above
    // would otherwise overwrite it with "now" on every later login. Firestore has no
    // "set only if absent" in a single merge call, so read-then-maybe-write once.
    ref.get().then(snap=>{
      const data = snap.data();
      if(data && data._createdOnce) return;
      try{window.evQueueAutoEvent&&window.evQueueAutoEvent('account_created',{},user);}catch(_e){}
      ref.set({ _createdOnce:true, createdAt: firebase.firestore.FieldValue.serverTimestamp() }, { merge:true }).catch(()=>{});
    }).catch(()=>{});
  }catch(e){ console.warn('Ethvelle: could not sync user profile doc', e); }
}

/* ================= SITE-CONTENT TRACKING (feeds the admin dashboard) =================
   Declared here (before goTo/trackEvent('visitor') below) since function declarations are
   hoisted but the const/let bindings they close over are not — calling trackEvent() from
   the bottom of the script before TRACKING_KEY had executed used to throw a
   "Cannot access before initialization" error that silently broke visitor counting.
   Every event bumps a localStorage counter immediately (so the dashboard always has
   something to show) and mirrors to Firestore in the background when configured, keyed by
   today's date so "today" counts reset naturally each day while totals keep accumulating.

   EXTENDED DATA MODEL (backward compatible):
   The original three fields (visitors/exports/projects) keep working exactly as before —
   every old call site (`trackEvent('visitor')`, `trackEvent('project')`, `trackEvent('export')`)
   still increments the same three counters, so historical totals are never broken or reset.
   New event kinds below just add MORE fields alongside those three, in the same dailyStats
   day-doc and the same stats/totals doc, using the same increment-based writes. A new
   monthlyStats/{YYYY-MM} doc is written alongside dailyStats so the admin dashboard can show
   month/quarter/year views without scanning every daily doc ever created.

   ANONYMOUS IDENTITY (privacy-conscious, no personal data):
   - visitorId: a random id persisted in localStorage — identifies "this browser" across visits,
     so the dashboard can distinguish page-load COUNT from actual unique visitors. Only a random
     string; nothing that identifies a person.
   - sessionId: a random id kept in sessionStorage — identifies "this tab session"; a fresh
     session starts whenever the app is opened in a new tab (sessionStorage is cleared then),
     so multiple tabs from the same visitor count as multiple sessions but the SAME visitorId.
   Both are anonymous identifiers with no link to a real identity unless the visitor separately
   signs in, in which case authenticatedUid (the existing Firebase UID) is used for user-level
   stats — never these anonymous ids. */
function todayKey(d){
  d = d || new Date();
  return d.getFullYear()+'-'+String(d.getMonth()+1).padStart(2,'0')+'-'+String(d.getDate()).padStart(2,'0');
}
function monthKey(d){
  d = d || new Date();
  return d.getFullYear()+'-'+String(d.getMonth()+1).padStart(2,'0');
}
function currentHour(){ return new Date().getHours(); } // 0-23, local time

/* ---- anonymous visitor/session identity ---- */
const VISITOR_ID_KEY = 'riggy.visitorId.v1';
const VISITOR_FIRSTSEEN_KEY = 'riggy.visitorFirstSeen.v1';
const SESSION_ID_KEY = 'riggy.sessionId.v1'; // sessionStorage — one per tab, cleared when the tab closes
function makeAnonId(){ return 'v_'+Date.now().toString(36)+Math.random().toString(36).slice(2,10); }
function getOrCreateVisitorId(){
  try{
    let id = localStorage.getItem(VISITOR_ID_KEY);
    if(!id){
      id = makeAnonId();
      localStorage.setItem(VISITOR_ID_KEY, id);
      localStorage.setItem(VISITOR_FIRSTSEEN_KEY, new Date().toISOString());
      return { id, isNew:true };
    }
    return { id, isNew:false };
  }catch(e){
    // localStorage unavailable (private browsing lockdown, etc) — fall back to a per-load id;
    // this visit will count as "unique" every time, which is the best available approximation.
    return { id: makeAnonId(), isNew:true };
  }
}
function getOrCreateSessionId(){
  try{
    let id = sessionStorage.getItem(SESSION_ID_KEY);
    if(!id){
      id = makeAnonId();
      sessionStorage.setItem(SESSION_ID_KEY, id);
      return { id, isNew:true };
    }
    return { id, isNew:false };
  }catch(e){
    return { id: makeAnonId(), isNew:true };
  }
}
// Tracks, once per calendar day per browser, whether THIS visitorId has already been counted as
// a "unique visitor" today — so refreshing the page 20 times in one day still shows as 1 unique
// visitor (though 20 plain "visitors"/page-loads), without needing any server-side lookup.
const VISITOR_LASTDAY_KEY = 'riggy.visitorLastCountedDay.v1';
function isFirstVisitToday(){
  try{
    const last = localStorage.getItem(VISITOR_LASTDAY_KEY);
    const today = todayKey();
    if(last === today) return false;
    localStorage.setItem(VISITOR_LASTDAY_KEY, today);
    return true;
  }catch(e){
    return true; // can't dedupe without storage — treat every load as unique rather than undercount
  }
}

// kind: any key of EVENT_FIELDS above (includes the original 'visitor' | 'export' | 'project')
let _lastPaymentPageViewAt = 0;
function trackEvent(kind, details){
  // Payment entry could be requested twice in quick succession by overlapping navigation
  // callbacks (notably promo-driven navigation). Count one page-view per actual entry, while
  // still allowing a genuine later return to Payment to count again. The in-memory guard also
  // covers restricted WebViews where sessionStorage can throw.
  if(kind === 'payment_page_view'){
    const stampKey = 'riggy.paymentPageView.lastAt.v1';
    const now = Date.now();
    if(_lastPaymentPageViewAt && now - _lastPaymentPageViewAt < 1800) return;
    try{
      const last = Number(sessionStorage.getItem(stampKey) || 0);
      if(last && now - last < 1800){ _lastPaymentPageViewAt = last; return; }
      sessionStorage.setItem(stampKey, String(now));
    }catch(e){ /* in-memory debounce remains active if storage is unavailable */ }
    _lastPaymentPageViewAt = now;
  }
  const fields = EVENT_FIELDS[kind];
  if(!fields || !fields.length) return; // unknown kind — no-op rather than throw, keeps callers safe
  const t = loadTracking();
  const key = todayKey();
  const hour = currentHour();
  if(!t.days[key]) t.days[key] = blankDay();
  // Older saved localStorage days (from before this update) may be missing newer fields —
  // backfill them in place so incrementing never throws on `undefined + 1`.
  const day = t.days[key];
  Object.keys(blankDay()).forEach(f=>{ if(f!=='hours' && day[f]===undefined) day[f]=0; });
  if(!day.hours) day.hours = {};
  if(!day.hours[hour]) day.hours[hour] = {};
  fields.forEach(field=>{
    day[field] = (day[field]||0) + 1;
    day.hours[hour][field] = (day.hours[hour][field]||0) + 1;
    t.totals[field] = (t.totals[field]||0) + 1;
  });
  saveTracking(t);
  // Detailed visitor timeline is independent of the roll-up counters. Failures are swallowed
  // inside the recorder so normal editing, payment and navigation continue as usual.
  recordVisitorSessionEvent(kind, details || {});
  // Mirrors this event to Firestore (dailyStats/{today}, monthlyStats/{YYYY-MM} + stats/totals)
  // so the separate admin panel's dashboard can show live, global counts covering every visitor
  // — not just this device. Uses FieldValue.increment (atomic on Firestore's servers) rather than
  // a read-modify-write, so two people using the app at the same moment both land correctly
  // instead of one overwriting the other's count. No-ops safely if Firestore is unreachable.
  if(firebaseReady && fbDb){
    try{
      const inc = firebase.firestore.FieldValue.increment(1);
      const dayPatch = {};
      fields.forEach(field=>{ dayPatch[field] = inc; });
      dayPatch.hours = {};
      fields.forEach(field=>{ dayPatch.hours[hour] = Object.assign(dayPatch.hours[hour]||{}, { [field]: inc }); });
      const dayRef = fbDb.collection('dailyStats').doc(key);
      dayRef.set(dayPatch, { merge:true }).catch(()=>{});

      const totalsPatch = {};
      fields.forEach(field=>{ totalsPatch[field] = inc; });
      const totalsRef = fbDb.collection('stats').doc('totals');
      totalsRef.set(totalsPatch, { merge:true }).catch(()=>{});

      // Monthly rollup — same fields, no hourly breakdown (the hourly chart only ever needs
      // "today", so monthlyStats skips that cost entirely). This lets the dashboard show a
      // month/quarter/year view by reading ~12-36 small docs instead of scanning every day.
      const mPatch = {};
      fields.forEach(field=>{ mPatch[field] = inc; });
      const monthRef = fbDb.collection('monthlyStats').doc(monthKey());
      monthRef.set(mPatch, { merge:true }).catch(()=>{});
    }catch(e){}
  }
}

/* ---- session/visitor lifecycle: called once per page load, right before the legacy
   trackEvent('visitor') call further down, so a single page view always produces at least
   the original 'visitor' count, PLUS a 'session_start' and (once per day per browser) a
   'unique_visitor' count. Kept as a separate small function so the legacy call site below
   stays a clean one-liner. ---- */
function detectTrafficSource(){
  // Prefer explicit campaign tags; otherwise classify the referring hostname. Store only the
  // category, never a full referrer URL or search query, so no unnecessary user data is logged.
  let params;
  try{ params = new URLSearchParams(window.location.search || ''); }catch(e){ params = null; }
  const utmSource = params ? String(params.get('utm_source') || params.get('source') || '').toLowerCase() : '';
  const utmMedium = params ? String(params.get('utm_medium') || '').toLowerCase() : '';
  let refHost = '';
  try{ refHost = document.referrer ? new URL(document.referrer).hostname.toLowerCase().replace(/^www\./,'') : ''; }catch(e){}
  const tagged = (utmSource + ' ' + utmMedium).trim();
  const hostAndTag = (tagged + ' ' + refHost).toLowerCase();
  if(/facebook|(^|[\s._-])fb($|[\s._-])|fb\.com|meta\.com/.test(hostAndTag)) return 'facebook';
  if(/youtube|youtu\.be|youtube-nocookie/.test(hostAndTag)) return 'youtube';
  if(/google|bing|duckduckgo|yahoo|yandex|ecosia|search/.test(hostAndTag)) return 'search';
  if(/instagram|tiktok|twitter|x\.com|linkedin|pinterest|reddit|threads\.net/.test(hostAndTag)) return 'social';
  if(!refHost || refHost === String(location.hostname || '').toLowerCase().replace(/^www\./,'')) return 'direct';
  return 'other';
}
function trackVisitAndSession(){
  getOrCreateVisitorId();       // ensures a persistent id exists; return value unused here
  const session = getOrCreateSessionId();
  if(session.isNew) trackEvent('session_start');
  if(isFirstVisitToday()) trackEvent('unique_visitor');
  const trafficSource = detectTrafficSource();
  // sessionStorage survives same-tab refreshes, preventing repeated attribution on reload.
  trackOncePerTab('traffic-source', 'traffic_source_' + trafficSource, { source:trafficSource });
}

/* ================= NAVIGATION ================= */
function goTo(screenName){
  if(screenName === 'settings'){ openSettingsScreen(); return; }
  const settingsBackdrop = document.getElementById('settingsSheetBackdrop');
  if(settingsBackdrop) settingsBackdrop.classList.remove('active');
  // 'models' isn't a real screen — it's the Projects screen's Models tab. Routed here (rather
  // than only from the home tile's onclick) so ANY destination picker that stores plain
  // screen names — including the admin panel's "When tapped → Go to a screen" dropdown for
  // banners/tiles (see runAdminAction) — can target the Models tab just by name, with no
  // separate actionType of its own.
  if(screenName==='models'){
    goTo('projects');
    setProjectsTab('models');
    return;
  }
  const previousScreenEl = document.getElementById('screen-settings')?.classList.contains('active')
    ? document.getElementById('screen-settings') : document.querySelector('.screen.active');
  const previousScreenName = previousScreenEl && previousScreenEl.id ? previousScreenEl.id.replace(/^screen-/, '') : '';
  document.querySelectorAll('.screen').forEach(s=>s.classList.remove('active'));
  const nextScreenEl = document.getElementById('screen-'+screenName);
  if(nextScreenEl) nextScreenEl.classList.add('active');
  if(previousScreenName !== screenName){
    trackEvent('screen_view', { screen:screenName });
    if(screenName === 'myplan') trackEvent('billing_plan_view', { screen:'myplan' });
    if(screenName === 'payment'){
      trackEvent('payment_page_view', { screen:'payment' });
      trackOncePerTab('payment-page', 'payment_page_session', { screen:'payment' });
    }
  }
  if(screenName==='profile'){
    const statEl = document.getElementById('profileStatCount');
    if(statEl) statEl.textContent = projects.length;
  }
  if(screenName==='tutorials' && typeof renderTutorialsGrid==='function') renderTutorialsGrid();
  if(screenName==='home' && typeof renderPromptLibrary==='function') renderPromptLibrary();
  if(screenName==='keyframeInfo' && typeof renderPromptLibrary==='function') renderPromptLibrary();
  if(screenName==='home' && typeof renderHomeCreditSummary==='function') renderHomeCreditSummary();
  if(screenName==='keyframeInfo' && typeof refreshKeyframeEntryCard==='function') refreshKeyframeEntryCard();
  if(screenName==='help' && typeof renderHelpFaq==='function') renderHelpFaq();
  if(screenName==='help' && typeof renderHelpContactCards==='function') renderHelpContactCards();
  if(screenName==='settings' && typeof prefillSettingsScreen==='function') prefillSettingsScreen();
  if(screenName==='settings' && typeof renderNotifToggles==='function') renderNotifToggles();
  if(screenName==='notifications' && typeof openNotificationsScreen==='function') openNotificationsScreen();
  if(typeof wirePromoPopups==='function') setTimeout(()=>wirePromoPopups(true,screenName),0);
  if(typeof maybeShowLocationNotification==='function'){ setTimeout(()=>maybeShowLocationNotification(screenName),30); if(screenName==='home') setTimeout(()=>maybeShowLocationNotification(screenName),450); }
}
