<script id="ethvelle-v142-smooth-transition-character-ui-fix">
(function(){
  'use strict';
  if(window.__ethvelleV142SmoothTransitionCharacterUI)return;
  window.__ethvelleV142SmoothTransitionCharacterUI=true;

  /* ----------------------------------------------------------------------
     1) Smooth video cuts.
     v141 rewound each gesture-primed incoming decoder in its pause listener.
     That discarded the decoded first frame and forced Android WebView to seek
     and decode it again at the clip boundary. Keep the already-decoded parked
     frame instead. While the element is paused and still carries this marker,
     treat its parked currentTime as its prepared start frame. This skips only
     the redundant boundary seek; once the incoming video actually starts,
     normal timeline-time sync/rate correction resumes immediately.
     ---------------------------------------------------------------------- */
  const V142_WARM_ROLE='v142-gesture-prime';

  function markGesturePrimedVideo(el){
    if(!el||el.tagName!=='VIDEO'||el.dataset.evWarmElement!=='1')return;
    el.dataset.evWarmRole=V142_WARM_ROLE; // v141's pause handler only rewinds the old role.
    el.dataset.evV142GestureFrame='1';
  }
  const priorSetupVideo=window.setupOverlayVideoEl;
  if(typeof priorSetupVideo==='function'&&!priorSetupVideo.__ethvelleV142KeepDecodedFrame){
    const wrappedSetupVideo=function(el,ov){
      const result=priorSetupVideo.apply(this,arguments);
      if(el&&el.tagName==='VIDEO'&&el.dataset.evWarmRole==='v140-gesture-prime')markGesturePrimedVideo(el);
      return result;
    };
    wrappedSetupVideo.__ethvelleV142KeepDecodedFrame=true;
    window.setupOverlayVideoEl=wrappedSetupVideo;
  }

  const priorExpected=window.evOverlayVideoExpectedLocalTime;
  if(typeof priorExpected==='function'&&!priorExpected.__ethvelleV142ParkedFrame){
    const wrappedExpected=function(ov,timeSec,el){
      const real=priorExpected.apply(this,arguments);
      // BUGFIX v150: this used to trust el.currentTime unconditionally for ANY paused
      // gesture-primed element, so a clip whose priming play() ran a little long before
      // being paused again reported itself "on time" at the WRONG source frame — it then
      // opened on that wrong frame and had to be dragged back into sync by the rate servo
      // after it went live (visible as a jump/offset start on every clip). Only skip the
      // redundant seek when the parked frame is already within one frame of the real
      // expected position; otherwise defer to the real value so the normal alignment/seek
      // path corrects it before the clip is ever shown.
      if(el&&el.tagName==='VIDEO'&&el.dataset.evV142GestureFrame==='1'&&
         el.paused&&!el.seeking&&Number.isFinite(Number(el.currentTime))&&Number.isFinite(real)&&
         Math.abs((Number(el.currentTime)||0)-real)<=(1/Math.max(1,Number(FPS)||30))){
        // The priming play has already decoded this frame, and it's genuinely where the
        // timeline expects it. Do not ask the boundary gate or v141's adoption wrapper to
        // re-seek back to the same source start.
        if(window.__evVideoDebug)try{console.log('[evVideo] keep-parked-frame',el.dataset.evV72OverlayId,'parked',el.currentTime,'expected',real);}catch(_e){}
        return Math.max(0,Number(el.currentTime)||0);
      }
      if(window.__evVideoDebug&&el&&el.dataset&&el.dataset.evV142GestureFrame==='1'&&el.paused&&!el.seeking&&Number.isFinite(real))try{console.log('[evVideo] discard-parked-frame (drifted)',el.dataset.evV72OverlayId,'parked',el.currentTime,'expected',real);}catch(_e){}
      return real;
    };
    wrappedExpected.__ethvelleV142ParkedFrame=true;
    window.evOverlayVideoExpectedLocalTime=wrappedExpected;
  }

  const priorTakeWarm=window.takeWarmOverlayVideo;
  if(typeof priorTakeWarm==='function'&&!priorTakeWarm.__ethvelleV142AdoptFrame){
    const wrappedTakeWarm=function(ov){
      const id=String(ov&&ov.id!=null?ov.id:'');
      const candidate=typeof overlayWarmEls!=='undefined'&&overlayWarmEls?overlayWarmEls[id]:null;
      const wasGesturePrimed=!!(candidate&&candidate.dataset.evV142GestureFrame==='1');
      const el=priorTakeWarm.apply(this,arguments);
      if(el&&wasGesturePrimed){
        // Keep the marker through adoption. The actual 'playing' event clears it;
        // that lets the parked frame pass readiness checks without masking later sync.
        el.dataset.evV142GestureFrame='1';
        if(!el.__evV142ClearPrimeOnLivePlay){
          el.addEventListener('playing',function(){
            if(el.dataset.evWarmElement!=='1')delete el.dataset.evV142GestureFrame;
          });
          el.__evV142ClearPrimeOnLivePlay=true;
        }
      }
      return el;
    };
    wrappedTakeWarm.__ethvelleV142AdoptFrame=true;
    window.takeWarmOverlayVideo=wrappedTakeWarm;
  }

  // The v140 wrapper primes every video synchronously inside the trusted Play gesture.
  // Existing warm elements (created before Play) also need the v142 role before their
  // asynchronous play() promise pauses them, so v141 cannot rewind them on that pause.
  const priorStartPlayback=window.startPlayback;
  if(typeof priorStartPlayback==='function'&&!priorStartPlayback.__ethvelleV142KeepPrimed){
    const wrappedStartPlayback=function(){
      const markExistingWarmVideos=()=>{
        if(typeof overlayWarmEls!=='undefined'&&overlayWarmEls){
          Object.keys(overlayWarmEls).forEach(id=>markGesturePrimedVideo(overlayWarmEls[id]));
        }
      };
      // Mark already-created warm elements before v140 calls play(). This also covers
      // legacy WebViews whose play() returns no Promise and whose fallback pauses inline.
      markExistingWarmVideos();
      const result=priorStartPlayback.apply(this,arguments);
      if(typeof isPlaying!=='undefined'&&isPlaying)markExistingWarmVideos();
      return result;
    };
    wrappedStartPlayback.__ethvelleV142KeepPrimed=true;
    window.startPlayback=wrappedStartPlayback;
  }

  /* ----------------------------------------------------------------------
     2) Character held-object preview: retain the full-body crop and 1080px-wide
     backing bitmap across panel re-renders, and prevent touch drags on the preview
     or its sliders from scrolling the sheet. The actual hand/object drag logic is
     unchanged and continues to use the original pointer handlers.
     ---------------------------------------------------------------------- */
  const propsBackdrop=document.getElementById('ethvelleCharacterPropsBackdrop');
  let lastPreviewCanvas=null;
  function rangeProgress(input){
    if(!input||input.type!=='range')return;
    const min=Number(input.min||0),max=Number(input.max||100),value=Number(input.value||0);
    const percent=max>min?Math.max(0,Math.min(100,(value-min)/(max-min)*100)):0;
    input.style.setProperty('--ev-range-progress',percent+'%');
  }
  function tuneHeldPropsPanel(){
    const back=document.getElementById('ethvelleCharacterPropsBackdrop');
    if(!back||!back.classList.contains('active'))return;
    const canvas=back.querySelector('#evpaPropStagePreview');
    if(canvas&&canvas!==lastPreviewCanvas){
      lastPreviewCanvas=canvas;
      if(canvas.width!==1080)canvas.width=1080;
      if(canvas.height!==1512)canvas.height=1512;
      canvas.style.touchAction='pan-y';
      canvas.style.userSelect='none';
      canvas.style.webkitUserSelect='none';
      // Do not cancel touch on the whole preview. The sheet must remain vertically
      // scrollable when a finger starts on the character; the existing object-hit
      // pointer handler below only captures gestures that begin on the held object.
      canvas.__evV142NoSheetScroll=true;
    }
    back.querySelectorAll('.evpa-props-sheet .evpa-prop-field input[type="range"]').forEach(rangeProgress);
  }
  if(typeof MutationObserver==='function'&&document.body){
    const observer=new MutationObserver(()=>{
      const canvas=document.querySelector('#ethvelleCharacterPropsBackdrop #evpaPropStagePreview');
      if(canvas!==lastPreviewCanvas)tuneHeldPropsPanel();
    });
    observer.observe(document.body,{subtree:true,childList:true});
  }
  document.addEventListener('input',function(e){
    if(e.target&&e.target.matches&&e.target.matches('#ethvelleCharacterPropsBackdrop .evpa-props-sheet input[type="range"]'))rangeProgress(e.target);
  },true);
  ['openCharacterPropsPanel','setCharacterPropSide','toggleCharacterHeldProp','removeCharacterHeldProp'].forEach(name=>{
    const prior=window[name];
    if(typeof prior!=='function'||prior.__ethvelleV142PanelTune)return;
    const wrapped=function(){const result=prior.apply(this,arguments);tuneHeldPropsPanel();return result;};
    wrapped.__ethvelleV142PanelTune=true;window[name]=wrapped;
  });
  // FileReader completion rebuilds the panel asynchronously, so MutationObserver above
  // also catches that canvas replacement and restores its high-resolution backing store.

  /* ----------------------------------------------------------------------
     3) Put each custom-part '+' action inside the choice grid as its final tile.
     Existing model editor navigation and upload handling are reused unchanged.
     ---------------------------------------------------------------------- */
  function placePartAddTiles(){
    const back=document.getElementById('ethvelleClipFaceActionsBackdrop');
    if(!back||!back.classList.contains('active'))return;
    back.querySelectorAll('.evpa-section').forEach(section=>{
      const headingRow=section.querySelector(':scope > .evpa-section-heading');
      let heading=section.querySelector(':scope > .evpa-section-title');
      let oldAdd=null;
      if(headingRow){
        if(!heading)heading=headingRow.querySelector('.evpa-section-title');
        oldAdd=headingRow.querySelector('.evpa-section-add');
        if(oldAdd)oldAdd.remove();
        if(heading)section.insertBefore(heading,headingRow);
        headingRow.remove();
      }
      if(!heading)return;
      const label=String(heading.textContent||'').trim();
      if(/^blink\s+mode$/i.test(label))return;
      const grid=section.querySelector(':scope > .evpa-choice-grid');
      if(!grid||grid.querySelector(':scope > .evpa-choice-add'))return;
      const add=document.createElement('button');
      add.type='button';add.className='evpa-choice evpa-choice-add';
      add.title='Add custom '+label.toLowerCase()+' artwork';
      add.setAttribute('aria-label','Add custom '+label.toLowerCase()+' artwork');
      add.innerHTML='<span class="evpa-choice-preview"><span class="evpa-choice-add-icon" aria-hidden="true"><svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.6" stroke-linecap="round"><path d="M12 5v14M5 12h14"/></svg></span></span><span class="evpa-choice-label">Add part</span>';
      add.addEventListener('click',function(e){
        e.preventDefault();e.stopPropagation();
        if(typeof window.openCustomPartEditorFromAction==='function')window.openCustomPartEditorFromAction();
        else if(typeof window.openSelectedCharacterModelEditor==='function'){window.closeClipCharacterActions();window.openSelectedCharacterModelEditor();}
      });
      grid.appendChild(add);
    });
  }
  const priorOpenCharacterActions=window.openClipCharacterActions;
  if(typeof priorOpenCharacterActions==='function'&&!priorOpenCharacterActions.__ethvelleV142GridAddTiles){
    const wrappedOpenCharacterActions=function(){
      const result=priorOpenCharacterActions.apply(this,arguments);
      placePartAddTiles();
      requestAnimationFrame(placePartAddTiles);
      return result;
    };
    wrappedOpenCharacterActions.__ethvelleV142GridAddTiles=true;
    window.openClipCharacterActions=wrappedOpenCharacterActions;
  }
  if(document.readyState==='loading')document.addEventListener('DOMContentLoaded',tuneHeldPropsPanel,{once:true});
  else tuneHeldPropsPanel();
})();

(function(){
  'use strict';
  if(window.__ethvelleV146HeldPreviewFullScrollInstalled)return;
  window.__ethvelleV146HeldPreviewFullScrollInstalled=true;

  let gesture=null;
  const ROOT='#ethvelleCharacterPropsBackdrop';
  const STAGE='.evpa-prop-stage-wrap';

  function getPreviewSheet(e){
    const back=document.querySelector(ROOT+'.active');
    if(!back)return null;
    const stage=back.querySelector(STAGE);
    const sheet=back.querySelector('.evpa-props-sheet');
    if(!stage||!sheet||!(e.target===stage||stage.contains(e.target)))return null;

  // A vertical gesture scrolls the complete sheet; a horizontal gesture is left
  // to the existing object-positioning handler.
  document.addEventListener('pointerdown',function(e){
    if(e.pointerType==='mouse'&&e.button!==0)return;
    const target=getPreviewSheet(e);
    if(!target)return;
    gesture={pointerId:e.pointerId,startX:e.clientX,startY:e.clientY,lastX:e.clientX,
      lastY:e.clientY,mode:'pending',sheet:target.sheet};
  },{capture:true,passive:true});

  document.addEventListener('pointermove',function(e){
    const g=gesture;
    if(!g||g.pointerId!==e.pointerId)return;
    const totalX=e.clientX-g.startX;
    const totalY=e.clientY-(g.startY===undefined?e.clientY:g.startY);
    if(g.mode==='pending'&&Math.max(Math.abs(totalX),Math.abs(totalY))>6){
      if(Math.abs(totalY)>=Math.abs(totalX)*0.9)g.mode='scroll';
      else if(Math.abs(totalX)>Math.abs(totalY)*0.9)g.mode='object';
    }
    if(g.mode==='scroll'){
      const max=Math.max(0,g.sheet.scrollHeight-g.sheet.clientHeight);
      if(max>0){
        // Incremental scrolling avoids the short travel/clamping seen with the old
        // fixed startScroll calculation and continues correctly across long swipes.
        const delta=g.lastY-e.clientY;
        g.sheet.scrollTop=Math.max(0,Math.min(max,g.sheet.scrollTop+delta));
        if(e.cancelable)e.preventDefault();
        e.stopPropagation();
        if(e.stopImmediatePropagation)e.stopImmediatePropagation();
      }
    }
    g.lastX=e.clientX;
    g.lastY=e.clientY;
  },{capture:true,passive:false});

  function finish(e){
    if(gesture&&(!e||gesture.pointerId===e.pointerId))gesture=null;
  }
  document.addEventListener('pointerup',finish,true);
  document.addEventListener('pointercancel',finish,true);
  document.addEventListener('lostpointercapture',finish,true);

    return null;
  }
  function ensurePartInput(){
    if(partInput&&partInput.isConnected)return partInput;
    partInput=document.createElement('input');partInput.type='file';partInput.accept='image/png,image/webp,image/jpeg,image/*';
    partInput.id='ethvelleV149SectionPartFile';partInput.hidden=true;
    partInput.addEventListener('change',handlePartFile);
    ( $('screen-editor')||document.body ).appendChild(partInput);
    return partInput;
  }
  function openSectionPicker(route){
    const char=activeChar();
    if(!char||!char.sheet){if(typeof showToast==='function')showToast('This character needs a Model Lab sheet before adding custom parts',{icon:toastIcons.info,duration:2200});return;}
    pendingPart={route,charId:char.id};
    const input=ensurePartInput();input.value='';input.click();
  }
  function cleanPartName(name,route){
    const cleaned=String(name||'').replace(/\.[^.]+$/,'').replace(/[_-]+/g,' ').replace(/\s+/g,' ').trim().slice(0,36);
    if(cleaned)return cleaned;
    return route.side==='L'?'Left hand gesture':route.side==='R'?'Right hand gesture':route.display;
  }
  function handlePartFile(){
    const file=partInput&&partInput.files&&partInput.files[0],pending=pendingPart;
    if(partInput)partInput.value='';pendingPart=null;
    if(!file||!pending||!pending.route)return;
    const char=activeChar();if(!char||char.id!==pending.charId||!char.sheet){if(typeof showToast==='function')showToast('The selected character changed. Please try again.',{icon:toastIcons.info});return;}
    if(file.type&&!/^image\/(png|jpeg|webp|gif|avif)/i.test(file.type)){if(typeof showToast==='function')showToast('Choose a PNG, JPG or WebP image',{icon:toastIcons.info});return;}
    const reader=new FileReader();
    reader.onerror=()=>{if(typeof showToast==='function')showToast('Could not read this image',{icon:toastIcons.info});};
    reader.onload=()=>{
      const img=new Image();
      img.onerror=()=>{if(typeof showToast==='function')showToast('Could not open this image',{icon:toastIcons.info});};
      img.onload=()=>{
        try{
          const max=1024,ratio=Math.min(1,max/Math.max(img.naturalWidth,img.naturalHeight));
          const w=Math.max(1,Math.round(img.naturalWidth*ratio)),h=Math.max(1,Math.round(img.naturalHeight*ratio));
          const canvas=document.createElement('canvas');canvas.width=w;canvas.height=h;
          const ctx=canvas.getContext('2d',{willReadFrequently:true});if(!ctx)throw new Error('Canvas unavailable');
          ctx.clearRect(0,0,w,h);ctx.drawImage(img,0,0,w,h);
          if(typeof edVariantRemoveFlatBackground==='function')edVariantRemoveFlatBackground(canvas);
          const variants=ensureVariants(char);if(!variants)throw new Error('Model sheet unavailable');
          const route=pending.route,id='extra_'+Date.now().toString(36)+'_'+Math.random().toString(36).slice(2,7);
          const uploadedName=String(file.name||'Custom part').trim()||'Custom part';
          const item={id,name:uploadedName,fileName:uploadedName,originalFileName:uploadedName,kind:route.kind,src:canvas.toDataURL('image/png'),w,h,createdAt:Date.now()};
          if(route.side)item.side=route.side;
          variants[route.category].push(item);
          if(typeof updateModelAssetFromChar==='function')updateModelAssetFromChar(char);
          if(typeof persistProjectState==='function')persistProjectState();
          if(typeof window.closeClipCharacterActions==='function')window.closeClipCharacterActions();
          const idx=Array.isArray(characters)?characters.indexOf(char):-1;
          if(idx>=0&&typeof window.edOpenFromChar==='function'){
            const opened=window.edOpenFromChar(idx);
            if(opened&&typeof opened.catch==='function')opened.catch(err=>console.error('Could not open Model Lab after custom part upload',err));
          }else if(typeof showToast==='function')showToast('Custom part saved, but the Model Lab could not be opened',{icon:toastIcons.info,duration:2200});
        }catch(err){console.error('Ethvelle custom section upload failed',err);if(typeof showToast==='function')showToast('Could not save this custom part',{icon:toastIcons.info,duration:2200});}
      };
      img.src=String(reader.result||'');
    };
    reader.readAsDataURL(file);
  }

  function wirePartAddButtons(){
    const back=$('ethvelleClipFaceActionsBackdrop');if(!back||!back.classList.contains('active'))return;
    back.querySelectorAll('.evpa-section').forEach(section=>{
      const route=routeForSection(section),grid=section.querySelector(':scope > .evpa-choice-grid');
      if(!route||!grid)return;
      const old=grid.querySelector(':scope > .evpa-choice-add');if(!old)return;
      if(old.dataset.evV149Route===route.field)return;
      const replacement=old.cloneNode(true);replacement.dataset.evV149Route=route.field;
      replacement.title='Upload '+route.display.toLowerCase();replacement.setAttribute('aria-label','Upload '+route.display.toLowerCase());
      replacement.addEventListener('click',e=>{e.preventDefault();e.stopPropagation();openSectionPicker(route);});
      old.replaceWith(replacement);
    });
  }
