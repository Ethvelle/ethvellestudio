// Animal face kit: counts transparent pixels fully ENCLOSED inside a head piece (the hollow mouth) - transparent
// pixels that cannot be reached from the piece border through other transparent pixels.
function animalHeadHollowPixels(art){
  if(!art || !art.alpha || !(art.w>0) || !(art.h>0)) return 0;
  const w=art.w, h=art.h, a=art.alpha, seen=new Uint8Array(w*h), stack=[];
  const push=(x,y)=>{ const i=y*w+x; if(!a[i] && !seen[i]){ seen[i]=1; stack.push(i); } };
  for(let x=0;x<w;x++){ push(x,0); push(x,h-1); }
  for(let y=0;y<h;y++){ push(0,y); push(w-1,y); }
  while(stack.length){
    const i=stack.pop(), x=i%w, y=(i-x)/w;
    if(x>0)push(x-1,y); if(x<w-1)push(x+1,y); if(y>0)push(x,y-1); if(y<h-1)push(x,y+1);
  }
  let n=0; for(let i=0;i<w*h;i++) if(!a[i] && !seen[i]) n++;
  return n;
}
// Names of the emotion heads that do NOT have a hollow mouth (empty list = all four heads are hollow).
function animalFaceKitHollowReport(arts){
  const names={head_sad:'Sad',head_angry:'Angry',head_shock:'Surprised',head_laugh:'Happy'}, missing=[];
  Object.keys(names).forEach(k=>{
    const art=arts&&arts[k]; if(!art) return;
    const need=Math.max(60,Math.round(art.w*art.h*0.0006));
    if(animalHeadHollowPixels(art)<need) missing.push(names[k]);
  });
  return missing;
}

// Automatically detects dark outline strokes on hidden species attachment edges and blends
// them into the local coat/feather colour before a cut-out part is stored or sent to the rig editor.
// Only known mating zones are inspected; all exposed outer contours and internal markings remain intact.
function edCleanSpeciesJointEdge(canvas, key, modelType){
  if(modelType==='animal' && /^(?:head(?:_(?:sad|angry|shock|laugh))?|mouth_(?:closed|open|hold|inner)|eye_[LR]|eye_closed_[LR]|brow_[LR])$/.test(key)) return {__evJointCleanup:true, cleanedPixels:0};
  if(/^(?:mouth_inner|eye_[LR]|eye_closed_[LR]|brow_[LR])$/.test(key)) return {__evJointCleanup:true, cleanedPixels:0};
  if(!canvas || !canvas.width || !canvas.height) return {__evJointCleanup:true, cleanedPixels:0};
  const w=canvas.width,h=canvas.height,ctx=canvas.getContext('2d',{willReadFrequently:true});
  let image; try{image=ctx.getImageData(0,0,w,h);}catch(_e){return {__evJointCleanup:true, cleanedPixels:0};}
  const d=image.data,src=new Uint8ClampedArray(d); let minX=w,minY=h,maxX=-1,maxY=-1;
  for(let y=0;y<h;y++)for(let x=0;x<w;x++)if(src[(y*w+x)*4+3]>35){minX=Math.min(minX,x);maxX=Math.max(maxX,x);minY=Math.min(minY,y);maxY=Math.max(maxY,y);}
  if(maxX<minX||maxY<minY)return {__evJointCleanup:true,cleanedPixels:0};
  const bw=maxX-minX+1,bh=maxY-minY+1,short=Math.max(1,Math.min(bw,bh));
  const depth=Math.max(3,Math.min(13,Math.round(short*.075)));
  const zones=[];
  const add=(side,a,b,th=depth)=>zones.push({side,a,b,th});
  const animal=modelType==='animal', bird=modelType==='bird';
  const isHead=/^head(?:_|$)/.test(key), isMouth=/^mouth_(?:closed|open|hold)$/.test(key);
  if(isMouth){
    // The three talking overlays carry the nose/beak base too: erase only their outer patch
    // contour, never the dark mouth opening/lip detail in the middle.
    add('top',.02,.98); add('bottom',.02,.98); add('left',.02,.98); add('right',.02,.98);
  }else if(isHead){
    // Full head/face swaps attach along the lower jaw/neck cut.
    add('bottom',animal?.28:.25,animal?.82:.78);
  }else if(key==='torso'){
    if(animal){
      // Rump/tail root on the left, head/body connection at the right, and the two
      // underside leg-root windows. The visible belly contour between roots is preserved.
      add('left',.25,.76); add('right',.24,.76);
      add('bottom',.12,.39); add('bottom',.60,.88);
    }else{
      // Bird body: tail attachment at rear, head/neck at front, wing roots on the flanks,
      // and leg roots underneath. Restrict each zone to its expected anatomical window.
      add('left',.28,.72); add('right',.22,.55);
      add('top',.36,.68); add('bottom',.30,.70);
    }
  }else if(key==='tail'){
    add('right',.20,.80);
  }else if(/^arm[LR]_upper$/.test(key) || /^leg[LR]_upper$/.test(key)){
    // Proximal limb pieces connect to the body at the top and to the next segment at bottom.
    add('top',.02,.98); add('bottom',.02,.98);
  }else if(/^arm[LR]_lower$/.test(key)){
    add('top',.02,.98); add('bottom',.02,.98);
  }else if(/^leg[LR]_lower$/.test(key)){
    // Mammal lower legs have a separate paw at the ankle; bird lower legs include the foot,
    // so their bottom contour is exposed and must stay outlined.
    add('top',.02,.98);
    if(animal)add('bottom',.02,.98);
  }else if(/^(?:hand[LR]|shoe[LR])(?:_|$)/.test(key)){
    // The wrist/ankle opening is the top edge of a separate paw or foot piece.
    add('top',.10,.90);
  }
  if(!zones.length)return {__evJointCleanup:true,cleanedPixels:0};
  const lum=i=>.2126*src[i]+.7152*src[i+1]+.0722*src[i+2];
  const inside=(x,y,z)=>{
    let dd=0,t=0;
    if(z.side==='top'){dd=y-minY;t=(x-minX)/Math.max(1,bw-1);}
    else if(z.side==='bottom'){dd=maxY-y;t=(x-minX)/Math.max(1,bw-1);}
    else if(z.side==='left'){dd=x-minX;t=(y-minY)/Math.max(1,bh-1);}
    else {dd=maxX-x;t=(y-minY)/Math.max(1,bh-1);}
    return dd>=0&&dd<z.th&&t>=z.a&&t<=z.b;
  };
  function inwardSample(x,y,side,fromDepth){
    const vals=[];
    for(let along=-2;along<=2;along++){
      let xx=x,yy=y;
      if(side==='top'){xx+=along;yy=minY+fromDepth;}
      else if(side==='bottom'){xx+=along;yy=maxY-fromDepth;}
      else if(side==='left'){xx=minX+fromDepth;yy+=along;}
      else {xx=maxX-fromDepth;yy+=along;}
      if(xx<minX||xx>maxX||yy<minY||yy>maxY)continue;
      const i=(yy*w+xx)*4;if(src[i+3]<100)continue;
      const l=lum(i);if(l>45)vals.push([src[i],src[i+1],src[i+2],l]);
    }
    if(vals.length<2)return null;
    vals.sort((a,b)=>a[3]-b[3]);
    const bright=vals.filter(v=>v[3]>vals[0][3]+16);
    const chosen=bright.length>=2?bright:vals.slice(Math.floor(vals.length/2));
    if(!chosen.length)return null;
    const med=k=>{const a=chosen.map(v=>v[k]).sort((a,b)=>a-b);return a[Math.floor(a.length/2)];};
    return [med(0),med(1),med(2)];
  }
  let changed=0;
  for(let y=minY;y<=maxY;y++)for(let x=minX;x<=maxX;x++){
    const i=(y*w+x)*4,a=src[i+3];if(a<75)continue;
    let zone=null;
    for(let z=0;z<zones.length;z++)if(inside(x,y,zones[z])){zone=zones[z];break;}
    if(!zone)continue;
    const l=lum(i);if(l>188)continue;
    const inward=inwardSample(x,y,zone.side,Math.min(Math.max(3,zone.th-1),Math.max(3,Math.round(zone.th*.62))));
    if(!inward)continue;
    const il=.2126*inward[0]+.7152*inward[1]+.0722*inward[2];
    // Require a clear local outline-to-fill contrast. Dark fur patches with matching dark
    // colour immediately inward are deliberately left untouched.
    if(il-l<24 || l>il*.86)continue;
    for(let c=0;c<3;c++)d[i+c]=Math.round(src[i+c]*.06+inward[c]*.94);
    changed++;
  }
  ctx.putImageData(image,0,0);
  return {__evJointCleanup:true,cleanedPixels:changed};
}

// Remove dark contour/shadow pixels only on the small end zones that are meant to be hidden at a joint.
// The outer silhouette elsewhere (faces, hair, clothes and shoe soles) is untouched. We recolour contour
// pixels instead of erasing alpha, so antialiased edges stay closed when parts overlap.
function edCleanJointEdge(canvas, key, modelType){
  // v318: automatic joint-border removal is switched OFF for now. Parts keep their original artwork edges.
  // Set window.ETHVELLE_JOINT_BORDER_CLEANUP = true to bring the old behaviour back.
  if(!window.ETHVELLE_JOINT_BORDER_CLEANUP) return;
  if(!canvas || !canvas.width || !canvas.height) return;
  // Species art has its own anatomy-aware attachment zones. Run the automatic local
  // outline detector now, while each extracted piece is still a separate canvas and before
  // the model can be assembled/created in the rig editor.
  if(modelType === 'animal' || modelType === 'bird') return edCleanSpeciesJointEdge(canvas,key,modelType);
  const w=canvas.width, h=canvas.height, ctx=canvas.getContext('2d',{willReadFrequently:true});
  let im; try{ im=ctx.getImageData(0,0,w,h); }catch(e){ return; }
  const d=im.data, original=new Uint8ClampedArray(d), isHead=/^head(?:_|$)/.test(key), torso=key==='torso';
  let cleanedPixels=0;
  const isMouth=key==='mouth_closed'||key==='mouth_open'||key==='mouth_hold';
  // Mouth pieces are overlays on top of the full head. Remove dark pixels only within a few pixels
  // of the outer alpha silhouette, recolouring them to the same skin palette; internal lip/smile
  // details remain untouched because they are surrounded by opaque mouth/skin pixels.
  if(isMouth){
    const skinSamples=[[],[],[]];
    for(let p=0;p<w*h;p++){
      const i=p*4, a=original[i+3]; if(a<150) continue;
      const r=original[i],g=original[i+1],b=original[i+2];
      if(r>165 && g>75 && b>30 && r>g*1.12 && g>b*1.12 && r-g>28 && g-b>12){
        skinSamples[0].push(r); skinSamples[1].push(g); skinSamples[2].push(b);
      }
    }
    const med=a=>{ if(!a.length) return null; a.sort((x,y)=>x-y); return a[Math.floor(a.length/2)]; };
    const skin=skinSamples.map(med);
    if(skin.every(v=>v!==null)){
      for(let y=0;y<h;y++) for(let x=0;x<w;x++){
        const i=(y*w+x)*4; if(original[i+3]<35) continue;
        const lum=.2126*original[i]+.7152*original[i+1]+.0722*original[i+2];
        if(lum>220) continue;
        let nearOutside=false;
        for(let oy=-5;oy<=5&&!nearOutside;oy++) for(let ox=-5;ox<=5;ox++){
          if(ox*ox+oy*oy>64) continue;
          const nx=x+ox, ny=y+oy;
          if(nx<0||ny<0||nx>=w||ny>=h||original[(ny*w+nx)*4+3]<28){ nearOutside=true; break; }
        }
        if(!nearOutside) continue;
        // Preserve edge alpha for anti-aliasing while removing its dark RGB fringe.
        const blend=1;
        for(let c=0;c<3;c++) d[i+c]=Math.round(original[i+c]*(1-blend)+skin[c]*blend);
        cleanedPixels++;
      }
    }
    ctx.putImageData(im,0,0);
    return null;
  }
  const point=/^hand[LR]_point$/.test(key), thumbOrPalm=/^hand[LR]_(?:thumb|palm)$/.test(key);
  const hand=/^hand[LR](?:_fist)?$/.test(key), upperArm=/^arm[LR]_upper$/.test(key), lowerArm=/^arm[LR]_lower$/.test(key);
  const upperLeg=/^leg[LR]_upper$/.test(key), lowerLeg=/^leg[LR]_lower$/.test(key);
  const inJointBand=(x,y)=>{
    // Keep the jaw/chin silhouette intact. A recovered head ends at the face/jaw boundary,
    // not around the neck, so only the final few pixels are a hidden overlap zone.
    if(isHead) return y>=h*.97 && x>=w*.30 && x<=w*.70;
    if(torso){
      // The bottom of this Body image may contain the REAL dark-blue trouser/waist piece from the
      // uploaded sheet. It is artwork, not a hidden joint seam: never recolour/flatten the bottom
      // centre band, or the denim triangle can turn shirt-blue and look like an added blue patch.
      // Only clean the tiny shoulder/arm overlap corners; preserve the complete lower hem and
      // every pants-start pixel with its original RGB and alpha values.
      if(y>=h*.05 && y<=h*.27 && ((x<=w*.14)||(x>=w*.86))) return true;
      return false;
    }
    if(point){ return (key==='handL_point' ? x<=w*.10 : x>=w*.90) && y>=h*.22 && y<=h*.78; }
    if(thumbOrPalm) return y>=h*.89 && x>=w*.18 && x<=w*.82;
    if(hand) return y<=h*.20 && x>=w*.14 && x<=w*.86;
    if(upperArm || lowerArm || upperLeg) return (y<=h*.14 || y>=h*.86) && x>=w*.08 && x<=w*.92;
    // Shin/foot artwork keeps its shoe outline; only the knee end is a joining edge.
    if(lowerLeg) return y<=h*.15 && x>=w*.08 && x<=w*.92;
    return false;
  };
  const lum=(i)=>.2126*original[i]+.7152*original[i+1]+.0722*original[i+2];
  for(let y=1;y<h-1;y++) for(let x=1;x<w-1;x++){
    const i=(y*w+x)*4; if(original[i+3]<80 || !inJointBand(x,y)) continue;
    const l=lum(i); if(l>185) continue;
    let rr=0,gg=0,bb=0,n=0,maxL=0;
    for(let oy=-1;oy<=1;oy++) for(let ox=-1;ox<=1;ox++){
      if(!ox&&!oy) continue; const j=((y+oy)*w+x+ox)*4; if(original[j+3]<90) continue;
      const lj=lum(j); if(lj>maxL) maxL=lj;
      if(lj>l+4){ rr+=original[j]; gg+=original[j+1]; bb+=original[j+2]; n++; }
    }
    // Only soften a dark pixel that is clearly a contour against a lighter interior fill.
    if(n && maxL>l+6 && l < maxL*.97){ d[i]=Math.round(rr/n); d[i+1]=Math.round(gg/n); d[i+2]=Math.round(bb/n); cleanedPixels++; }
  }
  // Flatten mild skin/fabric shading in the joint overlap itself. Several generated sheets draw a
  // brown/black wrist crease or dark cuff-shadow a few pixels INSIDE the crop, not just on its edge,
  // so edge-only decontouring cannot remove it. Sample each material from the interior and reuse its
  // own median colour just in those tiny hidden zones; silhouette alpha and all non-joint pixels stay.
  const samples={skin:[[],[],[]],blue:[[],[],[]],gray:[[],[],[]],other:[[],[],[]]};
  for(let y=2;y<h-2;y++) for(let x=2;x<w-2;x++){
    const i=(y*w+x)*4; if(original[i+3]<150 || inJointBand(x,y)) continue;
    const r=original[i],gg=original[i+1],b=original[i+2],mx=Math.max(r,gg,b),mn=Math.min(r,gg,b);
    let cls=null;
    if(r>135 && gg>70 && r>gg*1.12 && gg>b*1.12 && r-b>28) cls='skin';
    else if(b>r*1.08 && b>gg*1.02 && b-gg>12) cls='blue';
    else if(mx<135 && mx-mn<55) cls='gray';
    else cls='other';
    if(cls) for(let c=0;c<3;c++) samples[cls][c].push(original[i+c]);
  }
  const median=arr=>{ if(arr.length<12) return null; arr.sort((a,b)=>a-b); return arr[Math.floor(arr.length/2)]; };
  const palette={}; Object.keys(samples).forEach(k=>{ palette[k]=samples[k].map(median); if(palette[k].some(v=>v===null)) palette[k]=null; });
  for(let y=0;y<h;y++) for(let x=0;x<w;x++){
    if(!inJointBand(x,y)) continue; const i=(y*w+x)*4; if(original[i+3]<45) continue;
    const r=original[i],gg=original[i+1],b=original[i+2],mx=Math.max(r,gg,b),mn=Math.min(r,gg,b); let cls=null;
    if(r>125 && gg>65 && r>gg*1.10 && gg>b*1.10 && r-b>28) cls='skin';
    else if(b>r*1.06 && b>gg*.99 && b-gg>8) cls='blue';
    else if(mx<140 && mx-mn<62) cls='gray';
    const col=cls&&palette[cls]; if(!col) continue;
    for(let c=0;c<3;c++) d[i+c]=Math.round(original[i+c]*.08+col[c]*.92);
    cleanedPixels++;
  }
  ctx.putImageData(im,0,0);
  palette.cleanedPixels=cleanedPixels;
  return palette;
}
// One cropped part -> "art": pixels + hit mask + default pivot (the joint it rotates around) + a few measurements.
function edMakeArt(canvas, key, modelType){
  const jointColors = edCleanJointEdge(canvas,key,modelType);
  const w = canvas.width, h = canvas.height, d = canvas.getContext('2d').getImageData(0,0,w,h).data, alpha = new Uint8Array(w*h);
  for(let i=0;i<w*h;i++) alpha[i] = d[i*4+3] > 30 ? 1 : 0;
  const cx = (a,b)=>{ let s = 0, n = 0; for(let y = Math.floor(a); y < Math.min(h, Math.ceil(b)); y++) for(let x = 0; x < w; x++) if(alpha[y*w+x]){ s += x; n++; } return n ? s/n : w/2; };
  let skinTop = -1;                         // first row that is mostly skin-coloured (where a sleeve ends and the bare forearm begins)
  for(let y = 0; y < h && skinTop < 0; y++){ let sk = 0, op = 0; for(let x = 0; x < w; x++){ const i = (y*w+x)*4; if(d[i+3] > 30){ op++; const r = d[i], g = d[i+1], b = d[i+2]; if(r > 120 && r > g && g > b && r-b > 35) sk++; } } if(op > 3 && sk/op > 0.5) skinTop = y; }
  let dpx, dpy;
  if(/^head/.test(key)){ dpx = cx(h*0.88, h); dpy = h*0.97; }
  else if(key === 'torso'){ dpx = w/2; dpy = h*0.9; }
  else if(isMythicModelType(modelType) && /^tail(?:_mid|_tip)?$/.test(key)){ dpx = cx(0,Math.max(2,h*0.06)); dpy = h*0.035; }   // v347: pivot at the real top-edge centre (works for curved tails)
  else if(key === 'tail' && (modelType === 'animal' || modelType === 'bird')){ dpx = w*0.94; dpy = h*0.5; }
  else if(key === 'tail'){ dpx = cx(0,h*0.1); dpy = h*0.04; }   // v344 beast-man tail: root at the TOP of the upright piece
  else if(key === 'mouth_closed' || key === 'mouth_open' || key === 'mouth_hold' || /^(?:mouth_inner|eye_[LR]|eye_closed_[LR]|brow_[LR])$/.test(key)){ dpx = w/2; dpy = h/2; }
  // Animal paw and bird wingtip alternatives all share a top-edge attachment point. The humanoid
  // point/thumb/palm pivots below are intentionally kept for human hand silhouettes only.
  else if((modelType === 'animal' || modelType === 'bird') && /^hand[LR](?:_(?:fist|point|thumb|palm))?$/.test(key)){ dpx = cx(0,h*0.1); dpy = h*0.06; }
  // Gesture art is not always upright: pointing hands have a wrist on the side, while thumbs-up
  // and palm pieces have the wrist at the bottom. Put the pivot on the actual wrist edge instead
  // of treating every variant like the plain open hand (this prevents detached gesture swaps).
  else if(/^handL_point$/.test(key)){ dpx = w*0.035; dpy = h*0.5; }
  else if(/^handR_point$/.test(key)){ dpx = w*0.965; dpy = h*0.5; }
  else if(/^hand[LR]_(?:thumb|palm)$/.test(key)){ dpx = cx(h*0.88,h); dpy = h*0.96; }
  else if(/^hand[LR](?:_fist)?$/.test(key)){ dpx = cx(0,h*0.1); dpy = h*0.06; }
  else { dpx = cx(0, h*0.1); dpy = h*(/lower|fist|point|thumb|palm/.test(key) ? 0.06 : 0.05); }
  return { canvas, w, h, alpha, dpx, dpy, jointColors:jointColors||{}, jointCleanedPixels:Number(jointColors&&jointColors.cleanedPixels)||0, botCx:cx(h*0.92, h), topCx:cx(0,Math.max(2,h*0.06)), topEdgeX:cx(0,2), botEdgeX:cx(Math.max(0,h-2),h), skinTop: skinTop < 0 ? -1 : skinTop/h };
}
// The new humanoid format places cut-outs around an upright anatomical silhouette (head above
// torso, arms at the sides, legs below) instead of putting every piece into a rectangular grid.
// This detector keeps legacy two-row 12-piece sheets usable by checking that the topmost piece is
// near the horizontal centre and that a separate central torso-like component sits below it.
function sheetLooksLikeRegularPartGrid(parts, w, h){
  if(!parts || parts.length < 10 || !w || !h) return false;
  const bucket = (values, tol) => {
    const out=[];
    values.slice().sort((a,b)=>a-b).forEach(v=>{
      const last=out[out.length-1];
      if(!last || Math.abs(v-last.mean)>tol) out.push({values:[v],mean:v,count:1});
      else { last.values.push(v); last.mean=last.values.reduce((a,b)=>a+b,0)/last.values.length; last.count=last.values.length; }
    });
    return out;
  };
  const xs=parts.map(b=>(b.x0+b.x1)/2), ys=parts.map(b=>(b.y0+b.y1)/2);
  const cols=bucket(xs,Math.max(3,w*.055)), rows=bucket(ys,Math.max(3,h*.035));
  if(cols.length<4 || rows.length<3) return false;
  const gapStats=clusters=>{
    const gaps=[]; for(let i=1;i<clusters.length;i++) gaps.push(clusters[i].mean-clusters[i-1].mean);
    const mean=gaps.reduce((a,b)=>a+b,0)/Math.max(1,gaps.length);
    return {mean,spread:mean?Math.max(...gaps.map(v=>Math.abs(v-mean)))/mean:Infinity,min:Math.min(...gaps)};
  };
  const gx=gapStats(cols), gy=gapStats(rows);
  // A true model-sheet grid repeats cell-centre spacing horizontally AND vertically.
  // Anatomical cut-outs may form rows, but their irregular shoulder/hand/leg x positions
  // must not be mistaken for a grid simply because some centres happen to line up.
  if(gx.spread>.38 || gy.spread>.48 || gx.min<w*.075 || gy.min<h*.035) return false;
  const repeatedCols=cols.filter(c=>c.count>=2).length;
  const repeatedRows=rows.filter(r=>r.count>=3).length;
  const strongRows=rows.filter(r=>r.count>=Math.max(2,Math.ceil(cols.length*.55))).length;
  const strongCols=cols.filter(c=>c.count>=2).length;
  return repeatedCols>=Math.ceil(cols.length*.65) && repeatedRows>=2 && strongRows>=Math.ceil(rows.length*.6) && strongCols>=Math.ceil(cols.length*.7) && parts.length/cols.length>=1.8;
}
function sheetIsAnatomy12(parts, w, h){
  if(!parts || parts.length !== 12) return false;
  const cx = b => (b.x0 + b.x1) / 2;
  const cy = b => (b.y0 + b.y1) / 2;
  const head = parts.reduce((best,b)=>cy(b)<cy(best)?b:best, parts[0]);
  const headX = cx(head) / Math.max(1,w);
  if(headX < 0.30 || headX > 0.70) return false;
  const possible = parts.filter(b=>b!==head && cx(b)/Math.max(1,w)>=0.31 && cx(b)/Math.max(1,w)<=0.69 && cy(b)>cy(head)+h*0.06 && cy(b)<h*0.67);
  if(!possible.length) return false;
  const torso = possible.sort((a,b)=>b.n-a.n)[0];
  const torsoX = cx(torso) / Math.max(1,w), torsoY = cy(torso) / Math.max(1,h), headY = cy(head) / Math.max(1,h);
  return torsoX >= 0.32 && torsoX <= 0.68 && torsoY > headY + 0.06 && torsoY < 0.68;
}
function sheetOrderAnatomy12(parts, w, h){
  const cx = b => (b.x0 + b.x1) / 2;
  const cy = b => (b.y0 + b.y1) / 2;
  const centreX = w / 2;
  const list = parts.slice();
  const head = list.reduce((best,b)=>cy(b)<cy(best)?b:best, list[0]);
  let rest = list.filter(b=>b!==head);
  const torsoPool = rest.filter(b=>cx(b) >= w*0.31 && cx(b) <= w*0.69 && cy(b) > cy(head)+h*0.05 && cy(b) < h*0.68);
  if(!torsoPool.length) return null;
  const torso = torsoPool.sort((a,b)=>b.n-a.n)[0];
  rest = rest.filter(b=>b!==torso);
  // Legs sit lower AND closer to the canvas centre than the side-positioned arms/hands.
  // Combining both signals is safer than relying on vertical position alone, because a relaxed
  // hand can hang down to hip height in an anatomy-position sheet.
  const legParts = rest.slice().sort((a,b)=>{
    const score = item => Math.abs(cx(item)-centreX)/Math.max(1,w)*1.2 - (cy(item)/Math.max(1,h))*0.4;
    return score(a)-score(b);
  }).slice(0,4);
  const armParts = rest.filter(b=>!legParts.includes(b));
  if(armParts.length !== 6 || legParts.length !== 4) return null;
  const byY = items => items.slice().sort((a,b)=>cy(a)-cy(b));
  const bySide = pair => pair.slice().sort((a,b)=>cx(a)-cx(b)); // viewer-left, then viewer-right
  const armSorted = byY(armParts);
  const upperArms = bySide(armSorted.slice(0,2));
  const forearms = bySide(armSorted.slice(2,4));
  const hands = bySide(armSorted.slice(4,6));
  const legSorted = byY(legParts);
  const thighs = bySide(legSorted.slice(0,2));
  const shins = bySide(legSorted.slice(2,4));
  if([upperArms,forearms,hands,thighs,shins].some(pair=>pair.length!==2)) return null;
  return [head, torso, upperArms[0], forearms[0], hands[0], upperArms[1], forearms[1], hands[1], thighs[0], shins[0], thighs[1], shins[1]];
}

// Detect a humanoid sheet arranged around one standing figure, rather than in a fixed grid.
// Several image generators let the head/neck/t-shirt edges touch, or let a wrist connect to a hand.
// These helpers use each component's position relative to the whole figure and can split those
// fused regions at the visible anatomical boundary before assigning editor part names.
function sheetFigureMetrics(parts){
  const x0=Math.min(...parts.map(b=>b.x0)), y0=Math.min(...parts.map(b=>b.y0));
  const x1=Math.max(...parts.map(b=>b.x1)), y1=Math.max(...parts.map(b=>b.y1));
  const w=Math.max(1,x1-x0+1), h=Math.max(1,y1-y0+1), mid=x0+w/2;
  const cx=b=>(b.x0+b.x1)/2, cy=b=>(b.y0+b.y1)/2;
  return {x0,y0,x1,y1,w,h,mid,cx,cy,yr:b=>(cy(b)-y0)/h,dx:b=>Math.abs(cx(b)-mid)/w,bw:b=>b.x1-b.x0+1,bh:b=>b.y1-b.y0+1};
}
