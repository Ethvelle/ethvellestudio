function hexToRgb(hex){
  let h=String(hex||'#00ff00').replace('#','').trim();
  if(h.length===3) h=h.split('').map(ch=>ch+ch).join('');
  if(!/^[0-9a-f]{6}$/i.test(h)) h='00ff00';
  return {r:parseInt(h.slice(0,2),16),g:parseInt(h.slice(2,4),16),b:parseInt(h.slice(4,6),16)};
}
function paintChromaKeyFrame(source,canvas,cfg,maxSide){
  const sw=source?.videoWidth||source?.naturalWidth||source?.width||0;
  const sh=source?.videoHeight||source?.naturalHeight||source?.height||0;
  if(!sw||!sh) return false;
  const ratio=Math.min(1,(Number(maxSide)>0?Number(maxSide):Math.max(sw,sh))/Math.max(sw,sh));
  const w=Math.max(1,Math.round(sw*ratio)),h=Math.max(1,Math.round(sh*ratio));
  if(canvas.width!==w) canvas.width=w; if(canvas.height!==h) canvas.height=h;
  const ctx=canvas.getContext('2d',{willReadFrequently:true});
  if(!ctx) return false;
  try{
    ctx.clearRect(0,0,w,h); ctx.drawImage(source,0,0,w,h);
    const frame=ctx.getImageData(0,0,w,h),d=frame.data,key=hexToRgb(cfg?.color);
    const tolerance=Math.max(0,Number(cfg?.tolerance)||75),softness=Math.max(0,Number(cfg?.softness)||0),edge=Math.max(1,softness);
    for(let i=0;i<d.length;i+=4){
      const dr=d[i]-key.r,dg=d[i+1]-key.g,db=d[i+2]-key.b;
      const distance=Math.sqrt(dr*dr+dg*dg+db*db);
      if(distance<=tolerance) d[i+3]=0;
      else if(softness>0 && distance<tolerance+softness) d[i+3]=Math.round(d[i+3]*Math.max(0,Math.min(1,(distance-tolerance)/edge)));
    }
    ctx.putImageData(frame,0,0); return true;
  }catch(err){
    console.warn('Ethvelle: chroma-key pixel processing failed (the video may be cross-origin)',err);
    return false;
  }
}

function openPanel(key){
  // BUGFIX (Animate panel could be opened with zero characters on stage): the Animate panel
  // lets you pick/replace an animation preset for "the active character" (see renderAnimList/
  // appendAnimationClip), which is meaningless with no character to apply it to —
  // activeCharacter() is null and bones/baseAngles are never set up yet. Previously the panel
  // still opened and let you pick a preset, which either silently did nothing useful or could
  // leave inconsistent state. Now it's blocked at the door with a clear nudge toward the real
  // first step — the same "Add character" affordance already on the timeline/Bones panel —
  // instead of opening into a panel that has nothing to actually animate yet.
  if(key === 'anim' && !characters.length){
    showToast('Add a character first', {icon: toastIcons.info, duration:1600});
    return;
  }
  document.querySelectorAll('.tool-item').forEach(t=>t.classList.remove('active'));
  const btn = document.querySelector(`.tool-item[data-tool="${key}"]`);
  if(btn) btn.classList.add('active');

  const content = panelContent[key];
  // Panel title reflects what Transform will actually edit (see transformTarget()) — so it's
  // clear at a glance whether Size/Position/Rotation below are about to move the Background,
  // a Clip, or the stickman, instead of the title always just saying "Transform" regardless.
  if(key === 'transform'){
    const { kind } = transformTarget();
    document.getElementById('panelTitle').textContent =
      kind==='background' ? 'Transform — Background' :
      kind==='overlay' ? 'Transform Clip' : 'Transform';
  } else {
    document.getElementById('panelTitle').textContent = content.title;
  }
  document.getElementById('panelBody').innerHTML = content.body;
  document.getElementById('panelBackdrop').classList.add('active');
  document.getElementById('toolPanel').classList.add('open');

  if(key === 'transform') syncTransformSliders();
  if(key === 'anim') renderAnimList();
  if(key === 'overlay') renderOverlayListPanel();
  if(key === 'assets') renderAssetsPanel();
  if(key === 'trim') renderTrimPanel();
  if(key === 'speed') renderSpeedPanel();
  if(key === 'volume') renderVolumePanel();
  if(key === 'text') renderTextPanel();
  if(key === 'chroma') renderChromaKeyPanel();
}

// Tapping the ROW ITSELF (not the toggle) now only PREVIEWS a preset — plays it briefly on
// the canvas so you can see the motion before committing to it — without touching the
// selected clip's saved animKey at all. This replaces the old behavior where a single tap
// both selected AND immediately applied the preset to whatever clip was selected.
let previewAnimKey = null;
let previewRAF = null;
let previewSnapshot = null;
let previewBonesRef = null;
let previewCharacterId = null;
let previewExprSnapshot = null;
function cancelAnimationPreview(redraw){
  if(previewRAF){ cancelAnimationFrame(previewRAF); previewRAF = null; }
  if(previewSnapshot && previewBonesRef) restoreBonesSnapshot(previewSnapshot, previewBonesRef);
  if(previewCharacterId && previewExprSnapshot){
    const targetChar=characters.find(c=>c&&c.id===previewCharacterId);
    if(targetChar)targetChar.expr=previewExprSnapshot.had?JSON.parse(JSON.stringify(previewExprSnapshot.value)):null;
  }
  previewExprSnapshot = null; previewSnapshot = null; previewBonesRef = null; previewCharacterId = null; previewAnimKey = null;
  if(redraw !== false) drawStick();
}
// v344: a beast-man animation fits beast-man models only; plain humanoid animations still fit beast men too (same rig).
function animPresetFitsCharacter(preset,character){
  const pt=(preset&&preset.modelType)||'humanoid';
  const ct=String(character&&(character.modelType||(character.sheet&&character.sheet.modelType))||'humanoid');
  if(pt==='beastman')return ct==='humanoid'&&!!(character&&((character.sheet&&character.sheet.beast)||character.beast));
  if(pt==='mythic')return isMythicModelType(ct);
  if(pt==='humanoid'&&isWingedHumanoidType(ct))return true;
  return pt===ct;
}
function previewAnim(key){
  const character = activeCharacter();
  const preset = animPresets[key];
  if(!character || !bones || !preset) return;
  const activeType=String(character.modelType||(character.sheet&&character.sheet.modelType)||'humanoid');
  if(!animPresetFitsCharacter(preset,character)){if(typeof showToast==='function')showToast('Select a matching character type for this animation.',{icon:toastIcons.info,duration:1800});return;}
  pausePlayback();
  // A preview is temporary only. Keep its exact target rig so a character switch cannot
  // restore the preview pose into a different character by accident.
  cancelAnimationPreview(false);
  previewBonesRef = character.bones;
  previewCharacterId = character.id;
  previewSnapshot = JSON.parse(JSON.stringify(previewBonesRef));
  previewExprSnapshot = {had:!!character.expr,value:character.expr?JSON.parse(JSON.stringify(character.expr)):null};
  previewAnimKey = key;
  renderAnimList();
  const LOOPS = 2;
  const startTime = performance.now();
  const totalMs = preset.duration * 1000 * LOOPS;
  const step = (now)=>{
    // If the preview belongs to a character which is no longer the active canvas layer,
    // stop cleanly; never keep animating an invisible/stale rig.
    if(!previewBonesRef || !characters.some(c=>c.id===previewCharacterId) || !activeCharacter() || activeCharacter().id!==previewCharacterId){
      cancelAnimationPreview(false); drawStick(); renderAnimList(); return;
    }
    const elapsed = now - startTime;
    if(elapsed >= totalMs){
      cancelAnimationPreview(false);
      drawStick();
      renderAnimList();
      return;
    }
    const tFrac = (elapsed % (preset.duration*1000)) / (preset.duration*1000);
    applyPoseToBones(evalPreset(preset, tFrac), previewBonesRef, character.baseAngles, character.hipRestY);
    if(preset.faceActions){
      if(!character.expr)character.expr=typeof sheetExprFresh==='function'?sheetExprFresh():{};
      const fa=preset.faceActions,period=Math.max(.45,Math.min(6,Number(fa.blinkRate)||3.15)),phase=(elapsed/1000)%period,blink=fa.blink==='auto'&&(phase<.13||phase>period-.13);
      const talkPhase=(elapsed/1000)%0.82;
      const talkOpen=fa.mouth==='talk'&&(talkPhase<.22||(talkPhase>.38&&talkPhase<.58));
      const p=character.sheet&&character.sheet.parts||{},mt=String(character.modelType||(character.sheet&&character.sheet.modelType)||'');
      const animalFaceKit=mt==='animal'&&!!(p.mouth_inner&&p.head_laugh);
      const humanoidFaceKit=mt==='humanoid'&&!!(p.mouth_inner&&p.head_laugh&&p.eye_L&&p.eye_R);
      const birdFaceKit=mt==='bird'&&!!(p.bird_eye&&p.bird_brow&&p.beak_open&&p.beak_closed);
      character.expr.emo=fa.mouth==='talk'&&(animalFaceKit||humanoidFaceKit||birdFaceKit)
        ? (talkOpen?(fa.expression&&fa.expression!=='none'?fa.expression:'laugh'):null)
        : (fa.expression&&fa.expression!=='none'?fa.expression:null);
      character.expr.eyeState=['open','closed'].includes(fa.eyes)?fa.eyes:'auto';
      character.expr.blink=character.expr.eyeState==='closed'?false:!!blink;
      character.expr.ambientBlink=fa.blink==='ambient'&&phase<.13;
      character.expr.disableAmbientBlink=fa.blink==='off'||character.expr.eyeState!=='auto';
      character.expr.flying=birdFaceKit&&fa.wings==='fly';
      if(fa.mouth==='talk')character.expr.mouth=talkOpen?'open':'closed';
      else if(fa.mouth==='closed'||fa.mouth==='hold')character.expr.mouth=fa.mouth;
      else character.expr.mouth=birdFaceKit&&['angry','shock','laugh'].includes(fa.expression)?'open':'closed';
      character.expr.faceKitHideMouth=(animalFaceKit||humanoidFaceKit)&&(fa.mouth==='closed'||(fa.mouth==='talk'&&!talkOpen));
    }
    drawStick();
    previewRAF = requestAnimationFrame(step);
  };
  previewRAF = requestAnimationFrame(step);
}

// The toggle switch is the ONLY thing that actually commits a preset to the selected clip
// now (see the .anim-toggle CSS note). Tapping an already-on toggle turns the animation back
// off (the clip reverts to no preset / the character's rest pose) rather than being a one-way
// switch, since a real on/off control should go both ways.
function toggleApplyAnim(key){
  const preset = animPresets[key];
  const character=activeCharacter();
  if(!preset || !character) return;
  const activeType=String(character.modelType||(character.sheet&&character.sheet.modelType)||'humanoid');
  if(!animPresetFitsCharacter(preset,character)){if(typeof showToast==='function')showToast('Select a matching character type for this animation.',{icon:toastIcons.info,duration:1800});return;}
  cancelAnimationPreview(false);
  pausePlayback();

  // Only an explicitly selected main/split animation clip is a toggle target. If the user
  // has selected a background, overlay, or character row, tapping Add must create a new
  // animation clip instead of silently toggling currentAnimKey without changing the clip.
  let target = null;
  if(selectedClip === 'main' && activeClip){
    target = {ref:'main', clip:activeClip};
  } else if(typeof selectedClip === 'string' && selectedClip.startsWith('seg')){
    const idx = parseInt(selectedClip.slice(3),10);
    if(Number.isInteger(idx) && splitStickmanSegments[idx]) target = {ref:'seg'+idx, clip:splitStickmanSegments[idx]};
  } else if(selectedClip == null && activeClip){
    // No explicit selection means the main animation clip is the default target. This avoids
    // an apparent flaky toggle where the list UI showed the main animation as selected but
    // the click instead created a separate clip because selectedClip was null.
    target = {ref:'main', clip:activeClip};
  }
  const currentlyApplied = !!(target && target.clip.animKey === key);
  if(!currentlyApplied && isAnimationLocked(key)){
    openAnimationUpgrade(getAnimationRequiredPlan(key)?.id);
    drawStick();
    renderAnimList();
    return;
  }

  if(currentlyApplied){
    if(Object.prototype.hasOwnProperty.call(target.clip,'__evFaceActionSnapshot')){
      const before=target.clip.__evFaceActionSnapshot;
      if(before===null)delete target.clip.faceActions;else target.clip.faceActions=JSON.parse(JSON.stringify(before));
      delete target.clip.__evFaceActionSnapshot;
    }
    target.clip.animKey = null;
    if(target.ref === 'main') currentAnimKey = null;
    renderTimelineClip();
    renderAnimList();
    // renderFrame restores the authored rest pose when a clip has no active preset, so
    // repeatedly toggling an animation on/off cannot leave the character stuck mid-motion.
    renderFrame();
    showToast(preset.name + ' turned off', {duration:1200});
    persistProjectState();
    return;
  }

  if(target){
    target.clip.animKey = key;
    if(target.ref === 'main') currentAnimKey = key;
    renderTimelineClip();
    renderAnimList();
    seekAbsoluteTimelineSeconds(target.clip.trimStart);
  } else if(activeClip){
    appendAnimationClip(key);
    renderAnimList();
  } else {
    setActiveAnimation(key);
    renderAnimList();
  }
  const committedFaceClip=target?target.clip:activeClip;
  if(preset.faceActions&&committedFaceClip){
    if(!Object.prototype.hasOwnProperty.call(committedFaceClip,'__evFaceActionSnapshot'))committedFaceClip.__evFaceActionSnapshot=committedFaceClip.faceActions?JSON.parse(JSON.stringify(committedFaceClip.faceActions)):null;
    committedFaceClip.faceActions={expression:'none',mouth:'preset',blink:'auto',eyes:'auto',leftHand:'none',rightHand:'none',...preset.faceActions};
  }
  showToast(preset.name + (target ? ' applied' : ' added'), {icon:toastIcons.check,duration:1400});
  // Start at the chosen clip's first frame to make the newly committed animation visible
  // immediately. The row itself only previews; this real button is the commit action.
  renderFrame();
  startPlayback();
  persistProjectState();
}

// Dispatches the contextual toolbar's "Replace" button (see bottomToolsClip markup) based on
// what's currently selected — this replaces the old numeric Trim panel entirely, per request.
//  - Background selected: reopens the same image picker "Add background" already used —
//    picking a new image just overwrites currentBg in place, same as setting it the first time.
//  - A media/overlay clip selected: opens the picker and swaps that exact clip's own
//    picture/video/audio content in place (see the 'replace-overlay' upload branch), keeping
//    its position and trim untouched.
//  - The stickman animation clip (main/segN) selected: "replacing" it means swapping its
//    animation preset — opens the Animate panel, where the toggle already handles applying a
//    different preset onto the currently-selected clip in place.
function replaceSelectedClip(){
  if(!selectedClip) return;
  const bgClip = backgroundClipForSelection();
  if(bgClip){
    replaceTargetId = bgClip.id;
    triggerUpload('replace-background');
    return;
  }
  const isStickman = selectedClip==='main' || String(selectedClip).startsWith('seg');
  if(isStickman){
    openPanel('anim');
    return;
  }
  const ov = overlays.find(o=>o.id===selectedClip);
  if(ov){
    replaceTargetId = ov.id;
    triggerUpload('replace-overlay');
  }
}

/* ---------------- Real media generation: video thumbnails + audio waveforms ----------------
   CapCut shows the actual content of a clip in its track (filmstrip of real frames for
   video, a rendered waveform for audio) rather than a flat color block. We generate both
   client-side from the uploaded file so what's shown is genuine, not a placeholder. */

// Captures `count` evenly-spaced frames from a <video> as JPEG data URLs using an offscreen
// canvas. Resolves once all frames are captured (or on failure, resolves with []).
function generateVideoThumbnails(url, count){
  // Kept as a compatibility name for older upload handlers. Video frames are no longer
  // batch-extracted here; the timeline's single lazy decoder owns all video thumbnails.
  return new Promise(resolve=>{
    const video=document.createElement('video');
    let settled=false,timer=0;
    video.muted=true;video.playsInline=true;video.preload='metadata';
    video.setAttribute('playsinline','');
    if(/^https?:/i.test(String(url||'')))video.crossOrigin='anonymous';
    const finish=()=>{
      if(settled)return;settled=true;clearTimeout(timer);
      const result={frames:[],duration:Number(video.duration)||1,width:Number(video.videoWidth)||0,height:Number(video.videoHeight)||0};
      try{video.pause();video.removeAttribute('src');video.load();}catch(_e){}
      resolve(result);
    };
    video.addEventListener('loadedmetadata',finish,{once:true});
    video.addEventListener('error',finish,{once:true});
    timer=setTimeout(finish,15000);
    try{video.src=String(url||'');video.load();}catch(_e){finish();}
  });
}

// Decodes an audio file with Web Audio API and returns { peaks: Float32Array-like downsampled
// min/max pairs, duration } so we can draw a real waveform without keeping the whole PCM buffer.
function generateAudioWaveform(url){
  return fetch(url)
    .then(r=>r.arrayBuffer())
    .then(buf=>{
      const AC = window.AudioContext || window.webkitAudioContext;
      const actx = new AC();
      return actx.decodeAudioData(buf).then(audioBuffer=>{
        const raw = audioBuffer.getChannelData(0);
        const samples = 200; // resolution of the drawn waveform, independent of clip length
        const blockSize = Math.max(1, Math.floor(raw.length/samples));
        const peaks = [];
        for(let i=0;i<samples;i++){
          const start = i*blockSize;
          let min=1, max=-1;
          for(let j=0;j<blockSize && (start+j)<raw.length;j++){
            const v = raw[start+j];
            if(v<min) min=v;
            if(v>max) max=v;
          }
          peaks.push([min,max]);
        }
        const duration = audioBuffer.duration;
        actx.close();
        return { peaks, duration };
      });
    })
    .catch(()=>({ peaks: [], duration: 1 }));
}

        ov.thumbs = newType==='video'?[]:(thumbs||[]);
        ov.peaks = peaks||[];
        // A shape/audio clip has no on-stage transform; an image/video does. Give it one if
        // it's newly gaining a visual and didn't have one before, same shape addMediaOverlay uses.
        if((newType==='video'||newType==='image') && !ov.transform) ov.transform = { scale:1, offsetX:0, offsetY:0, rotation:0 };
        renderOverlayLayers();
        renderOverlayListPanel();
        renderFrame();
        showToast('Replaced', {icon: toastIcons.check, duration:1200});
        persistProjectState();
      };
      if(isVideo){
        generateVideoThumbnails(dataUrl, 24, 0, null).then(({duration,width,height})=> finish({newType:'video', thumbs:[], duration, sourceWidth:width, sourceHeight:height}));
      } else if(isAudio){
        generateAudioWaveform(dataUrl).then(({peaks, duration})=> finish({newType:'audio', peaks, duration: duration||3}));
      } else {
        finish({newType:'image', thumbs:[dataUrl], duration: ov.sourceDuration||3});
      }
    };
    reader.onerror = ()=>{
      showToast("Couldn't read that file", {icon: toastIcons.info, duration:1800});
    };
    reader.readAsDataURL(file);
    closePanel();
    replaceTargetId = null;
  } else {
    showToast('Upload received — layer support coming next version');
    closePanel();
  }
  uploadAppendGroup = null;
  e.target.value='';
});

// Unified entry point for adding uploaded media (image/video/audio) into the shared
// Clips or Audio group track. The current playhead decides the insertion point: before/left
// of a clip in its first half, after/right in its second half, or before the next clip in a gap.
// Every item in that track is then packed into one sequential lane.
function addMediaOverlay({type, src, name, duration, thumbs, peaks, appendAfterGroup, deviceMediaKey, deviceMediaInfo, sourceDuration, sourceOffset, sourceWidth, sourceHeight, adjustments, chromaKey, enhanceAmount, filterPreset}){
  const id = 'ov'+Date.now();
  const color = OVERLAY_COLORS[overlays.length % OVERLAY_COLORS.length];
  const clipLen = type==='video' ? Math.max(MIN_CLIP_SEC, Number(duration)||MIN_CLIP_SEC) : Math.max(MIN_CLIP_SEC,Math.min(Number(duration)||3,12)); // video clips keep their full source duration; only stills use the short default
  let startT=0;
  const groupKey = appendAfterGroup || (type==='audio' ? 'audio' : 'clips');
  const groupClips = overlays.filter(o=> (o.type==='audio'?'audio':(o.type==='text'?'text':'clips')) === groupKey);
  // Insert relative to the playhead, like a single-track editor: first half of the
  // clip => before/left of it; second half => after/right of it; a playhead sitting in
  // the gap before a clip => before that clip. Then repack the row sequentially.
  const orderedGroupClips=[...groupClips].sort((a,b)=>(Number(a.trimStart)||0)-(Number(b.trimStart)||0));
  const groupClipLengths=new Map(orderedGroupClips.map(o=>[String(o.id),Math.max(MIN_CLIP_SEC,(Number(o.trimEnd)||0)-(Number(o.trimStart)||0))]));
  let playheadTime=0;
  try{playheadTime=Math.max(0,Number(typeof getCurrentAbsoluteTime==='function'?getCurrentAbsoluteTime():(typeof currentAbsoluteTimeSec!=='undefined'?currentAbsoluteTimeSec:0))||0);}catch(_e){}
  let insertIndex=orderedGroupClips.length;
  for(let i=0;i<orderedGroupClips.length;i++){
    const existing=orderedGroupClips[i];
    const start=Number(existing.trimStart)||0;
    const len=groupClipLengths.get(String(existing.id))||MIN_CLIP_SEC;
    const end=start+len;
    // A playhead exactly at / just before a clip's leading edge is treated as "before it".
    if(playheadTime <= start+0.001){insertIndex=i;break;}
    if(playheadTime < end-0.001){
      insertIndex=playheadTime < start+len/2 ? i : i+1;
      break;
    }
  }
  startT=orderedGroupClips.slice(0,insertIndex).reduce((sum,o)=>sum+(groupClipLengths.get(String(o.id))||MIN_CLIP_SEC),0);
  const newClip = {
    id, name: name || (type==='video'?'Video':type==='audio'?'Audio':'Image'), color,
    type, src, deviceMediaKey: deviceMediaKey || undefined,
    deviceMediaName: deviceMediaInfo&&deviceMediaInfo.name || undefined,
    deviceMediaSize: deviceMediaInfo&&Number(deviceMediaInfo.size)||undefined,
    deviceMediaLastModified: deviceMediaInfo&&Number(deviceMediaInfo.lastModified)||undefined,
    sourceDuration: Number(sourceDuration)>0?Number(sourceDuration):(Number(duration)||clipLen), sourceOffset: Math.max(0,Number(sourceOffset)||0),
    sourceWidth: Number(sourceWidth)>0?Number(sourceWidth):0, sourceHeight: Number(sourceHeight)>0?Number(sourceHeight):0,
    trimStart: startT, trimEnd: startT + clipLen,
    adjustments: adjustments&&typeof adjustments==='object'?{...adjustments}:undefined,
    chromaKey: chromaKey&&typeof chromaKey==='object'?{enabled:!!chromaKey.enabled,color:chromaKey.color||'#00ff00',tolerance:Number(chromaKey.tolerance)||75,softness:Number(chromaKey.softness)||24}:undefined,
    enhanceAmount: Number(enhanceAmount)||0, filterPreset:filterPreset||'original',
    thumbs: thumbs||[], peaks: peaks||[], speed: 1,
    // FEATURE (volume control + audio extraction): every clip that can carry sound (video
    // or audio) now has its own volume multiplier, 0 (silent) to 2 (2x boost), defaulting
    // to 1 (unchanged) — see the Volume panel (renderVolumePanel/volumePanelSetVolume,
    // mirrors the existing Speed panel) and its live application in renderOverlayMedia/
    // renderAudioOverlayMedia below. An image/shape clip has nothing to carry volume on, so
    // stays undefined there (same pattern transform already uses for audio).
    volume: (type==='video'||type==='audio') ? 1 : undefined,
    // Own independent size/position/rotation (image/video overlays only — audio/shape have
    // no on-stage visual of their own to move). Lets the Transform panel adjust THIS clip's
    // picture on the canvas the same way it already adjusts the stickman — see
    // updateTransform's selectedClip-aware BUGFIX note.
    transform: (type==='video'||type==='image') ? { scale:1, offsetX:0, offsetY:0, rotation:0 } : null
  };
  const orderedWithNew=[...orderedGroupClips.slice(0,insertIndex),newClip,...orderedGroupClips.slice(insertIndex)];
  let groupCursor=0;
  orderedWithNew.forEach(item=>{
    const oldStart=Number(item.trimStart)||0;
    const oldKeyframeTimes=Array.isArray(item.ethvelleMediaKeyframes)?item.ethvelleMediaKeyframes.map(k=>Number(k&&k.t)||0):[];
    const len=item===newClip?clipLen:(groupClipLengths.get(String(item.id))||Math.max(MIN_CLIP_SEC,(Number(item.trimEnd)||0)-(Number(item.trimStart)||0)));
    item.trimStart=groupCursor;
    item.trimEnd=groupCursor+len;
    const movedBy=item.trimStart-oldStart;
    if(item!==newClip&&Math.abs(movedBy)>0.000001)shiftTimelineKeyframesOnce(item.ethvelleMediaKeyframes,oldKeyframeTimes,movedBy);
    groupCursor=item.trimEnd;
  });
  startT=Number(newClip.trimStart)||0;
  overlays.push(newClip);
  timelineDuration = Math.max(Number(timelineDuration)||0, groupCursor, startT+clipLen);
  renderOverlayLayers();
  syncTimelineInnerWidth();
  renderRuler();
  onClipSelected(newClip,'add');
  renderOverlayListPanel();
  // Immediately redraw the current playhead frame after the timeline is rearranged, so
  // the preview reflects the selected clip order without requiring a later play/scrub action.
  renderFrame();
  showToast((type==='video'?'Video':type==='audio'?'Audio':'Image')+(appendAfterGroup?' added to layer':' added'), {icon: toastIcons.check, duration:1200});
  persistProjectState();
}

// Background clips use the same midpoint insertion rule as every other timeline row:
// first half inserts before/left; second half inserts after/right; gaps insert before next.
function addBackgroundClip(src){
  const id='bg'+Date.now(),len=Math.max(MIN_CLIP_SEC,Number(BG_DEFAULT_DURATION)||3);
  const ordered=[...backgrounds].sort((a,b)=>(Number(a.trimStart)||0)-(Number(b.trimStart)||0));
  let playhead=0;try{playhead=Math.max(0,Number(typeof getCurrentAbsoluteTime==='function'?getCurrentAbsoluteTime():currentAbsoluteTimeSec)||0);}catch(_e){}
  let at=ordered.length;
  for(let i=0;i<ordered.length;i++){
    const c=ordered[i],st=Number(c.trimStart)||0,l=Math.max(MIN_CLIP_SEC,(Number(c.trimEnd)||st+MIN_CLIP_SEC)-st),en=st+l;
    if(playhead<=st+.001){at=i;break;}
    if(playhead<en-.001){at=playhead<(st+en)/2?i:i+1;break;}
  }
  const clip={id,src,trimStart:0,trimEnd:len,transform:{scale:1,offsetX:0,offsetY:0,rotation:0}};
  const arranged=[...ordered.slice(0,at),clip,...ordered.slice(at)];let cursor=0;
  arranged.forEach(c=>{
    const oldStart=Number(c.trimStart)||0;
    const oldKeyframeTimes=Array.isArray(c.ethvelleMediaKeyframes)?c.ethvelleMediaKeyframes.map(k=>Number(k&&k.t)||0):[];
    const l=c===clip?len:Math.max(MIN_CLIP_SEC,(Number(c.trimEnd)||0)-(Number(c.trimStart)||0));
    c.trimStart=cursor;c.trimEnd=cursor+l;
    const movedBy=c.trimStart-oldStart;
    if(c!==clip&&Math.abs(movedBy)>0.000001)shiftTimelineKeyframesOnce(c.ethvelleMediaKeyframes,oldKeyframeTimes,movedBy);
    cursor=c.trimEnd;
  });
  backgrounds.splice(0,backgrounds.length,...arranged);timelineDuration=Math.max(Number(timelineDuration)||0,cursor);
  renderBackgroundTrack();syncTimelineInnerWidth();renderRuler();
  onClipSelected(clip,'add');
  refreshCurrentBgFromPlayhead();applyBackground();updateSelBox();persistProjectState();return clip;
}
