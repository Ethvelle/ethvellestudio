  function resolveTimelineClipAtTime(row,x,y){
    if(!row)return null;
    const now=typeof getCurrentAbsoluteTime==='function'?Number(getCurrentAbsoluteTime()):Number(currentAbsoluteTimeSec)||0;
    const entries=timelineEntriesForRow(row).filter(entry=>{
      const start=Number(entry.model&&entry.model.trimStart)||0;
      const end=Number(entry.model&&entry.model.trimEnd)||start;
      return end>start&&now>=start-0.000001&&now<end-0.000001;
    });
    if(!entries.length)return null;
    if(entries.length===1)return {id:entries[0].id,characterIndex:entries[0].characterIndex};
    // v258: stacked clips (x on top, y underneath, same time coverage). If the clip that is
    // already selected still covers the playhead, it stays selected for the whole slide —
    // the border never jumps to the other clip. Only when the selected clip no longer covers
    // the playhead does the pointer's lane pick a new one.
    const keepRef=currentSelectedRef();
    if(keepRef){
      const keepOwner=window.__ethvelleTimelineSelectionCharacterIndex;
      const keepEntry=entries.find(entry=>String(entry.id)===keepRef&&(entry.characterIndex==null||keepOwner==null||String(entry.characterIndex)===String(keepOwner)));
      if(keepEntry)return {id:keepEntry.id,characterIndex:keepEntry.characterIndex};
    }
    // Multiple clips can occupy separate vertical sub-lanes at the same time. Use the
    // pointer's lane only to choose between those data-matched candidates; time membership
    // always comes from trimStart/trimEnd, never horizontal DOM rectangles.

    entries.forEach(entry=>{
      const el=timelineEntryElement(row,entry);

      const score=Number.isFinite(hitTop)&&Number.isFinite(top)?Math.abs(top-hitTop):0;
      if(score<bestScore){best=entry;bestScore=score;}

    return {id:best.id,characterIndex:best.characterIndex};

<script id="ethvelle-v82-viewport-fit-js">
(function(){
  'use strict';
  const screen=document.getElementById('screen-editor');
  const wrap=screen&&screen.querySelector('.canvas-wrap');
  const stage=document.getElementById('canvasStage');
  if(!screen||!wrap||!stage)return;

  let fitRaf=0;
  function activeRatio(){
    try{
      const value=(typeof currentProject!=='undefined'&&currentProject&&currentProject.ratio)||'16:9';
      const match=String(value).match(/^\s*(\d+(?:\.\d+)?)\s*:\s*(\d+(?:\.\d+)?)\s*$/);
      return match&&Number(match[1])>0&&Number(match[2])>0 ? `${Number(match[1])}:${Number(match[2])}` : '16:9';
    }catch(_e){return '16:9';}
  }
  function refit(){
    fitRaf=0;
    if(!screen.classList.contains('active'))return;
    const ratio=activeRatio();
    const parts=ratio.split(':').map(Number);
    const rw=parts[0],rh=parts[1];
    if(!(rw>0&&rh>0))return;

    // Set the non-shrinking flex behaviour before measuring/re-sizing. setupCanvasStage
    // computes the largest frame that fits both available dimensions while preserving ratio.
    stage.style.flex='0 0 auto';
    stage.style.aspectRatio=rw+' / '+rh;
    if(typeof setupCanvasStage==='function')setupCanvasStage(ratio);

    // Wait for the WebView to commit the final post-rotation layout before repainting.
    requestAnimationFrame(()=>{
      if(!screen.classList.contains('active'))return;
      // renderFrame already invokes drawStick, so avoid drawing the stage twice on rotation.
      try{if(typeof renderFrame==='function')renderFrame();}catch(_e){}
      try{if(typeof updateSelBox==='function')updateSelBox();}catch(_e){}
    });
  }
  function scheduleRefit(){
    if(fitRaf)return;
    // Two frames let Android/WebView finish applying the new visual viewport dimensions.
    fitRaf=requestAnimationFrame(()=>{
      fitRaf=requestAnimationFrame(refit);
    });
  }

  // The observer catches the actual editor preview-area resize, including address-bar and
  // orientation changes that do not always arrive as a reliable window.resize event.
  if(typeof ResizeObserver==='function'){
    const observer=new ResizeObserver(scheduleRefit);
    observer.observe(wrap);
  }
  window.addEventListener('resize',scheduleRefit,{passive:true});
  window.addEventListener('orientationchange',scheduleRefit,{passive:true});
  if(window.visualViewport)window.visualViewport.addEventListener('resize',scheduleRefit,{passive:true});

  // Re-fit after the editor becomes visible; the initial observer notification can occur
  // while its parent screen is display:none.
  const editorObserver=new MutationObserver(()=>{
    if(screen.classList.contains('active'))scheduleRefit();
  });
  editorObserver.observe(screen,{attributes:true,attributeFilter:['class']});
  scheduleRefit();
})();
</script>
