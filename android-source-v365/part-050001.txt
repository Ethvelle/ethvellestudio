  function beginCropDrag(e){

    const layerRotation=((Number(tr.rotation)||0)+(Number(motion?.rotation)||0))*Math.PI/180;
    const stage=$('canvasStage'),stageRect=stage&&stage.getBoundingClientRect();
    const sx=stageRect&&stageRect.width?stage.clientWidth/stageRect.width:1;
    const sy=stageRect&&stageRect.height?stage.clientHeight/stageRect.height:1;
    // Capture on the stable frame, not on a handle whose position changes during resizing.
    // Invert the actual computed crop-root matrix so zoom, layer scale and layer rotation all
    // convert the finger delta into the same local coordinate space as the crop fractions.
    let inverseMatrix=null;
    try{
      const Matrix=window.DOMMatrixReadOnly||window.DOMMatrix;
      const cssTransform=cropRoot&&getComputedStyle(cropRoot).transform;
      if(Matrix&&cssTransform&&cssTransform!=='none'){
        const inv=new Matrix(cssTransform).inverse();

      }
    }catch(_e){}
    const capture=cropFrame;
    cropDrag={pointerId:e.pointerId,handle,startX:e.clientX,startY:e.clientY,m,c:getCrop(target.clip),
      invScale:1/layerScale,cos:Math.cos(layerRotation),sin:Math.sin(layerRotation),inverseMatrix,stageScaleX:sx,stageScaleY:sy,capture};
    try{cropDrag.capture.setPointerCapture(e.pointerId);}catch(_e){}

  function moveCropDrag(e){
    if(!cropDrag||e.pointerId!==cropDrag.pointerId)return;
    const target=cropTarget();if(!target)return;
    const d=cropDrag;
    // getBoundingClientRect() includes the canvas zoom scale; convert finger movement back
    // into stage CSS pixels before applying the selected layer's inverse scale/rotation.
    const screenX=(e.clientX-d.startX)*d.stageScaleX;
    const screenY=(e.clientY-d.startY)*d.stageScaleY;
    const localX=d.inverseMatrix?(d.inverseMatrix.a*screenX+d.inverseMatrix.c*screenY):((screenX*d.cos+screenY*d.sin)*d.invScale);
    const localY=d.inverseMatrix?(d.inverseMatrix.b*screenX+d.inverseMatrix.d*screenY):((-screenX*d.sin+screenY*d.cos)*d.invScale);
    const dx=localX/Math.max(1,d.m.w),dy=localY/Math.max(1,d.m.h),c={...d.c},h=d.handle;
    const w0=1-d.c.left-d.c.right,h0=1-d.c.top-d.c.bottom;
    if(h==='move'){
      const maxL=1-d.c.right-w0; // remains the original left unless clamped at the image edge
      c.left=clamp(d.c.left+dx,0,1-w0);c.right=1-w0-c.left;
      c.top=clamp(d.c.top+dy,0,1-h0);c.bottom=1-h0-c.top;
    }else if(cropRatio==='free'){
      if(h.includes('w'))c.left=clamp(d.c.left+dx,0,1-d.c.right-.08);
      if(h.includes('e'))c.right=clamp(d.c.right-dx,0,1-d.c.left-.08);
      if(h.includes('n'))c.top=clamp(d.c.top+dy,0,1-d.c.bottom-.08);
      if(h.includes('s'))c.bottom=clamp(d.c.bottom-dy,0,1-d.c.top-.08);
    }else{
      const p=cropRatio.split(':').map(Number),ratio=(p[0]/p[1])/d.m.aspect;
      if(h==='e'||h==='w'){
        let w=clamp(w0+(h==='e'?dx:-dx),.08,1);
        let hh=clamp(w/ratio,.08,1);w=hh*ratio;
        if(h==='e')c.right=1-d.c.left-w;else c.left=1-d.c.right-w;
        c.top=(1-hh)/2;c.bottom=1-hh-c.top;
      }else if(h==='n'||h==='s'){
        let hh=clamp(h0+(h==='s'?dy:-dy),.08,1),w=clamp(hh*ratio,.08,1);hh=w/ratio;
        if(h==='s')c.bottom=1-d.c.top-hh;else c.top=1-d.c.bottom-hh;
        c.left=(1-w)/2;c.right=1-w-c.left;
      }else{
        const dirX=h.includes('w')?-1:1,dirY=h.includes('n')?-1:1;
        const candidateW=clamp(w0+dirX*dx,.08,1),candidateH=clamp(h0+dirY*dy,.08,1);
        let w=Math.abs(candidateW-w0)/Math.max(.08,w0)>=Math.abs(candidateH-h0)/Math.max(.08,h0)?candidateW:candidateH*ratio;
        let hh=w/ratio;if(hh>.92){hh=.92;w=hh*ratio;}if(w>.92){w=.92;hh=w/ratio;}
        if(dirX<0)c.left=1-d.c.right-w;else c.right=1-d.c.left-w;
        if(dirY<0)c.top=1-d.c.bottom-hh;else c.bottom=1-d.c.top-hh;
      }
    }
    setCropFractions(target,c);paintCropFrame();e.preventDefault();e.stopPropagation();
  }

})();
