// A video frame is never drawn into an export frame until the requested source time is
// decoded. Avoid seeking when the frame is already at the requested source time.
async function waitForExportVideoFrame(video,targetSec,clipLabel){
  if(!video)throw new Error('Missing video element for '+clipLabel);
  video.preload='auto';
  const target=Math.max(0,Number(targetSec)||0);
  const needsSeek=Math.abs((Number(video.currentTime)||0)-target)>0.008;
  if(needsSeek){
    await new Promise((resolve,reject)=>{
      let settled=false;
      const timeout=setTimeout(()=>finish(new Error('Timed out decoding '+clipLabel)),12000);
      const cleanup=()=>{clearTimeout(timeout);['seeked','loadeddata','canplay','error'].forEach(name=>video.removeEventListener(name,onEvent));};
      const finish=err=>{if(settled)return;settled=true;cleanup();err?reject(err):resolve();};
      const onEvent=e=>{
        if(e.type==='error'){finish(new Error('Video decode failed for '+clipLabel));return;}
        if(video.readyState>=2&&(e.type==='seeked'||e.type==='loadeddata'||e.type==='canplay'))finish();
      };
      ['seeked','loadeddata','canplay','error'].forEach(name=>video.addEventListener(name,onEvent));
      try{video.currentTime=target;}catch(err){finish(err);}
    });
  }else if(video.readyState<2){
    await new Promise((resolve,reject)=>{
      let settled=false;
      const timeout=setTimeout(()=>finish(new Error('Timed out loading '+clipLabel)),12000);
      const cleanup=()=>{clearTimeout(timeout);['loadeddata','canplay','error'].forEach(name=>video.removeEventListener(name,onEvent));};
      const finish=err=>{if(settled)return;settled=true;cleanup();err?reject(err):resolve();};
      const onEvent=e=>{if(e.type==='error')finish(new Error('Video decode failed for '+clipLabel));else if(video.readyState>=2)finish();};
      ['loadeddata','canplay','error'].forEach(name=>video.addEventListener(name,onEvent));
      if(video.readyState>=2)finish();
    });
  }
  if(video.readyState<2)throw new Error('No decoded video frame available for '+clipLabel);
}

            window.ethvelleDrawTextBackplate(exportCtx,bx,-totalH/2-bgPad,widest+bgPad*2,totalH+bgPad*2,bgRadius,bgShape);
            exportCtx.restore();
          }
          usable.forEach((line,i)=>{
            const y=-totalH/2+lineHeight*(i+0.5);
            const lineW=Math.min(maxWidth,exportCtx.measureText(line).width);
            let lineX=0;
            if((ov.textAlign||'center')==='left') lineX=-maxWidth/2;
            else if((ov.textAlign||'center')==='right') lineX=maxWidth/2;
            if(ov.textBackgroundEnabled&&bgShape==='rounded'){
              exportCtx.save(); exportCtx.shadowColor='transparent'; exportCtx.shadowBlur=0; exportCtx.shadowOffsetX=0; exportCtx.shadowOffsetY=0;
              exportCtx.fillStyle=bgFill;
              const rectX=(ov.textAlign||'center')==='left' ? -maxWidth/2-bgPad : (ov.textAlign==='right' ? maxWidth/2-lineW-bgPad : -lineW/2-bgPad);
              drawRoundedRect(rectX,y-lineHeight/2-bgPad,lineW+bgPad*2,lineHeight+bgPad*2,bgRadius);
              exportCtx.restore();
            }
            exportCtx.shadowColor=shadowEnabled?ethvelleHexToRgba(ov.shadowColor||'#000000',Math.max(0,Math.min(100,Number(ov.shadowOpacity==null?100:ov.shadowOpacity)))/100):'transparent';
            exportCtx.shadowBlur=shadowEnabled?shadowBlur:0;
            exportCtx.shadowOffsetX=shadowEnabled?shadowX:0;
            exportCtx.shadowOffsetY=shadowEnabled?shadowY:0;
            if(strokeWidth>0){ exportCtx.lineWidth=strokeWidth*2; exportCtx.lineJoin='round'; exportCtx.strokeStyle=strokeColor; if(window.ethvelleDrawStrokeDepth)window.ethvelleDrawStrokeDepth(exportCtx,line,lineX,y,maxWidth,strokeWidth,exportScale); exportCtx.strokeText(line,lineX,y,maxWidth); }
            exportCtx.fillText(line,lineX,y,maxWidth);
          });
          if(ov.visualMask&&window.ethvelleApplyCanvasMaskClip)exportCtx.restore();
          exportCtx.restore();
          continue;
        }
        const rec = exportOverlayEls[ov.id];
        if(!rec) continue;
        if(ov.type==='video'){
          const speed = ov.speed || 1;
          // Speed-scaled the same way as the live preview (see renderOverlayMedia) so an
          // exported video matches exactly what was seen in the editor.
          const maxT = Math.max(0.05, (ov.sourceDuration || rec.el.duration || (ov.trimEnd-ov.trimStart)) - 0.02);
          const elapsed=Math.max(0,(tSec-Number(ov.trimStart))*speed);
          const span=Math.max(0,(Number(ov.trimEnd)-Number(ov.trimStart))*speed);
          const rawLocal=ov.reversePlayback?(Number(ov.sourceOffset)||0)+span-elapsed:(Number(ov.sourceOffset)||0)+elapsed;
          const localT = Math.max(0, Math.min(maxT, rawLocal));
          await waitForExportVideoFrame(rec.el,localT,ov.name||('Video clip '+String(ov.id||'')));
        }
        const evAnim = window.getEthvelleClipAnimationState ? window.getEthvelleClipAnimationState(ov,tSec) : {opacity:1,scale:1,x:0,y:0,rotation:0};
        const coverScale = Math.min(w/rec.iw, h/rec.ih) * (ov.transform ? ov.transform.scale : 1) * (Number(evAnim.scale)||1);
        const drawW = rec.iw*coverScale, drawH = rec.ih*coverScale;
        // Mirrors the live preview's authored transform plus the selected clip's animation.
        const ovExportScale = stageW ? (w/stageW) : 1;
        const trackedExportTransform = window.ethvelleGetTrackedOverlayTransform ? window.ethvelleGetTrackedOverlayTransform(ov,tSec) : ov.transform;
        const keyedExportTransform = ethvelleObjectKeyframeProperty(ov)==='transform' ? ethvelleGetObjectKeyframeValue(ov,'transform',tSec) : null;
        const t = ov.clipTracking&&ov.clipTracking.enabled&&ov.clipTracking.targetId ? trackedExportTransform : (keyedExportTransform||trackedExportTransform);
        const cx = w/2 + (((t ? Number(t.offsetX)||0 : 0) + (Number(evAnim.x)||0))*ovExportScale);
        const cy = h/2 + (((t ? Number(t.offsetY)||0 : 0) + (Number(evAnim.y)||0))*ovExportScale);
        const exportRotation = (t ? Number(t.rotation)||0 : 0) + (Number(evAnim.rotation)||0);
        try{
          exportCtx.save();
          exportCtx.globalAlpha = Math.max(0,Math.min(1,Number(evAnim.opacity)||0));
          const exportBaseFilter=window.ethvelleAdjustmentFilter ? window.ethvelleAdjustmentFilter(ov.adjustments, ov.enhanceAmount) : 'none';
          const exportBlurFx=window.ethvelleGetActiveClipBlurEffect?window.ethvelleGetActiveClipBlurEffect(ov,tSec):(ov.visualEffects&&ov.visualEffects.type==='blur'?ov.visualEffects:null);
          const exportBlurPx=exportBlurFx ? Math.max(0,Math.min(14,(Number(exportBlurFx.intensity)||135)/30))*ovExportScale : 0;
          exportCtx.filter=exportBlurPx>0 ? ((exportBaseFilter&&exportBaseFilter!=='none'?exportBaseFilter+' ':'')+'blur('+exportBlurPx.toFixed(2)+'px)') : exportBaseFilter;
          let drawSource=rec.el;
          if(ov.type==='video' && ov.chromaKey && ov.chromaKey.enabled){
            let keyed=exportChromaCanvases[ov.id];
            if(!keyed){ keyed=document.createElement('canvas'); exportChromaCanvases[ov.id]=keyed; }
            if(paintChromaKeyFrame(rec.el,keyed,ov.chromaKey,1280)) drawSource=keyed;
          }
          const crop=window.ethvelleCropFractions?window.ethvelleCropFractions(ov.crop):{left:0,top:0,right:0,bottom:0};
          const sourceX=rec.iw*crop.left,sourceY=rec.ih*crop.top;
          const sourceW=Math.max(1,rec.iw*(1-crop.left-crop.right)),sourceH=Math.max(1,rec.ih*(1-crop.top-crop.bottom));
          // Preserve the full source's contain scale. The crop only removes pixels from the
          // original source rectangle; it does not enlarge or recenter the remaining pixels.
          const baseContainScale=Math.min(w/rec.iw,h/rec.ih);
          const fullDrawW=rec.iw*baseContainScale,fullDrawH=rec.ih*baseContainScale;
          const baseDrawW=sourceW*baseContainScale,baseDrawH=sourceH*baseContainScale;
          const cropDestX=(w-fullDrawW)/2+sourceX*baseContainScale;
          const cropDestY=(h-fullDrawH)/2+sourceY*baseContainScale;
          const objectScale=(t?(Number(t.scale)||1):1)*(Number(evAnim.scale)||1);
          exportCtx.save();
          exportCtx.translate(cx,cy);
          if(exportRotation)exportCtx.rotate(exportRotation*Math.PI/180);
          exportCtx.scale(objectScale*(ov.mirrorX?-1:1),objectScale*(ov.mirrorY?-1:1));
          { const evStretch=window.ethvelleMediaStretch?window.ethvelleMediaStretch(ov):null; if(evStretch&&(evStretch.w!==1||evStretch.l!==1))exportCtx.scale(evStretch.w,evStretch.l); }
          exportCtx.translate(-w/2,-h/2);
          // The mask is clipped in the transformed layer's own full-stage box, so it moves,
          // rotates and mirrors together with the image/video exactly like the live CSS mask.
          if(window.ethvelleApplyCanvasMaskClip)window.ethvelleApplyCanvasMaskClip(exportCtx,ov.visualMask,w,h);
          const evPerspX=Number(ov.mediaPerspectiveX)||0,evPerspY=Number(ov.mediaPerspectiveY)||0;
          if((evPerspX||evPerspY)&&window.ethvelleDrawPerspectiveImage)window.ethvelleDrawPerspectiveImage(exportCtx,drawSource,sourceX,sourceY,sourceW,sourceH,cropDestX,cropDestY,baseDrawW,baseDrawH,evPerspX,evPerspY,w/2,h/2,(window.ethvelleMediaPerspectiveDistance||900)*ovExportScale);
          else exportCtx.drawImage(drawSource,sourceX,sourceY,sourceW,sourceH,cropDestX,cropDestY,baseDrawW,baseDrawH);
          exportCtx.restore();
          exportCtx.restore();
        }
        catch(err){ try{exportCtx.restore();}catch(_e){} throw new Error('Could not draw '+String(ov.name||('clip '+ov.id))+': '+(err&&err.message||err)); }
      }
    }

    // Render each frame of the FULL combined timeline (every clip/segment in its own
    // absolute-time slot) into the export canvas at the target FPS — using the exact same
    // poseAtClipTime()/transformAtClipTime() functions the live preview uses, so exported
    // motion always matches what was seen in the editor (PART 35). animateAllCharacters()
    // additionally poses every OTHER character from its own clip each frame, so multiple
    // characters with their own animations all move together in the exported video instead
    // of only the active one animating while the rest export frozen in their rest pose.
    if(exportMode==='mediarecorder'){
      exportRecorder.start();
      if(liveAudioCtx&&liveAudioCtx._exportSourceNode){try{liveAudioCtx._exportSourceNode.start();}catch(err){throw new Error('Could not start fallback export audio: '+err.message);}}
    }else if(exportMode==='webcodecs'&&mixedAudioBuffer&&exportAudioEncoder){
      // Encode offline-mixed PCM with sample-derived timestamps, not wall-clock playback.
      const channels=mixedAudioBuffer.numberOfChannels,sampleRate=mixedAudioBuffer.sampleRate,totalSamples=mixedAudioBuffer.length;
      const channelData=Array.from({length:channels},(_,i)=>mixedAudioBuffer.getChannelData(i));
      const chunkFrames=1024;
      for(let offset=0;offset<totalSamples;offset+=chunkFrames){
        const frameCount=Math.min(chunkFrames,totalSamples-offset);
        const planar=new Float32Array(frameCount*channels);
        for(let ch=0;ch<channels;ch++)planar.set(channelData[ch].subarray(offset,offset+frameCount),ch*frameCount);
        const audioData=new AudioData({format:'f32-planar',sampleRate,numberOfFrames:frameCount,numberOfChannels:channels,timestamp:Math.round(offset*1000000/sampleRate),data:planar});
        try{exportAudioEncoder.encode(audioData);}finally{audioData.close();}
        if(exportAudioEncoder.encodeQueueSize>24)await exportAudioEncoder.flush();
        const audioError=window.__ethvelleExportEncoderErrors&&window.__ethvelleExportEncoderErrors().audio;
        if(audioError)throw audioError;
      }
      await exportAudioEncoder.flush();
      const audioError=window.__ethvelleExportEncoderErrors&&window.__ethvelleExportEncoderErrors().audio;
      if(audioError)throw audioError;
    }

    const fallbackStartMs=exportMode==='mediarecorder'?performance.now():0;
    const frameIntervalMs=1000/renderFps;
    let fallbackFrameIndex=0;
    for(let f=0;exportMode==='webcodecs'?f<totalFrames:true;f++){
      let clipAbsoluteT;
      if(exportMode==='webcodecs')clipAbsoluteT=f/renderFps;
      else{
        const elapsed=Math.max(0,(performance.now()-fallbackStartMs)/1000);
        clipAbsoluteT=elapsed>=durationSec?Math.max(0,durationSec-1/renderFps):Math.min(elapsed,Math.max(0,durationSec-1/renderFps));
      }
      faceAnimationTimeOverride=clipAbsoluteT;
      if(activeCharacter()) resetCharacterForTimelineFrame(activeCharacter());
      if(activeCharacter() && activeClip){
        const pose = poseAtClipTime(clipAbsoluteT);
        if(pose) applyPoseToBones(pose);
        const clipTransform = transformAtClipTime(clipAbsoluteT);
        if(clipTransform) applyTransformToModel(clipTransform);
      }
      if(window.__capExportCancelled)throw new Error('export-cancelled');
      animateAllCharacters(clipAbsoluteT);

      exportCtx.clearRect(0,0,w,h);
      // Paint overlays explicitly sent behind the Background first. The background is then
      // drawn over them, matching the live canvas z-order instead of letting export silently
      // pull every overlay back above the background layer.
      await drawOverlayMediaToContext(clipAbsoluteT,true);
      // FEATURE (multiple sequential backgrounds): resolved freshly for THIS frame's own
      // time — a project that switches background partway through now actually shows that
      // switch in the exported video, instead of exporting whichever background was active
      // only at the moment export started.
      const bgNow = bgDrawFor(clipAbsoluteT);
      if(bgNow.image){
        exportCtx.save();
        // Keep the stage's neutral white visible outside a masked/moved background image.
        exportCtx.fillStyle = '#ffffff'; exportCtx.fillRect(0,0,w,h);
        exportCtx.filter = window.ethvelleAdjustmentFilter ? window.ethvelleAdjustmentFilter(bgNow.adjustments, bgNow.enhanceAmount) : 'none';
        exportCtx.translate(w/2+bgNow.offsetX,h/2+bgNow.offsetY);
        if(bgNow.rotation) exportCtx.rotate(bgNow.rotation * Math.PI/180);
        exportCtx.scale(bgNow.scale*(bgNow.mirrorX?-1:1),bgNow.scale*(bgNow.mirrorY?-1:1));
        if((bgNow.stretchW&&bgNow.stretchW!==1)||(bgNow.stretchL&&bgNow.stretchL!==1))exportCtx.scale(bgNow.stretchW||1,bgNow.stretchL||1);
        exportCtx.translate(-w/2,-h/2);
        if(window.ethvelleApplyCanvasMaskClip) window.ethvelleApplyCanvasMaskClip(exportCtx,bgNow.visualMask,w,h);
        if((bgNow.perspectiveX||bgNow.perspectiveY)&&window.ethvelleDrawPerspectiveImage)window.ethvelleDrawPerspectiveImage(exportCtx,bgNow.image,bgNow.sourceX||0,bgNow.sourceY||0,bgNow.sourceW||bgNow.image.width,bgNow.sourceH||bgNow.image.height,bgNow.destX??((w-bgNow.baseW)/2),bgNow.destY??((h-bgNow.baseH)/2),bgNow.baseW,bgNow.baseH,bgNow.perspectiveX,bgNow.perspectiveY,w/2,h/2,(window.ethvelleMediaPerspectiveDistance||900)*(bgNow.perspectiveScale||1));
        else exportCtx.drawImage(bgNow.image,bgNow.sourceX||0,bgNow.sourceY||0,bgNow.sourceW||bgNow.image.width,bgNow.sourceH||bgNow.image.height,bgNow.destX??((w-bgNow.baseW)/2),bgNow.destY??((h-bgNow.baseH)/2),bgNow.baseW,bgNow.baseH);
        exportCtx.restore();
      }
      else {
        exportCtx.save();
        exportCtx.filter = window.ethvelleAdjustmentFilter ? window.ethvelleAdjustmentFilter(bgNow.adjustments, bgNow.enhanceAmount) : 'none';
        exportCtx.fillStyle = bgNow.fill; exportCtx.fillRect(0,0,w,h);
        exportCtx.restore();
      }

      await drawOverlayMediaToContext(clipAbsoluteT,false);

      window.__evCharFxTime = clipAbsoluteT;
      try{ drawStickToContext(exportCtx, w, h); } finally { window.__evCharFxTime = undefined; }
      if(window.ethvelleDrawVisualEffectsToContext) window.ethvelleDrawVisualEffectsToContext(exportCtx,clipAbsoluteT,w,h);

      if(exportMode==='webcodecs'){
        const timestamp=Math.round(f*1000000/renderFps);
        const frame=new VideoFrame(exportCanvas,{timestamp,duration:Math.round(1000000/renderFps)});
        try{exportVideoEncoder.encode(frame,{keyFrame:f===0||f%(renderFps*2)===0});}finally{frame.close();}
        while(exportVideoEncoder.encodeQueueSize>8){await new Promise(r=>{let d=false;const fin=()=>{if(d)return;d=true;exportVideoEncoder.ondequeue=null;r();};exportVideoEncoder.ondequeue=fin;setTimeout(fin,40);});if(window.__ethvelleExportEncoderErrors&&(window.__ethvelleExportEncoderErrors().video))break;}
        const encoderErrors=window.__ethvelleExportEncoderErrors&&window.__ethvelleExportEncoderErrors();
        if(encoderErrors&&encoderErrors.video)throw encoderErrors.video;
        if(f%Math.max(1,Math.floor(renderFps/2))===0){showExportProgress(Math.round(((f+1)/totalFrames)*96));window.__capExportFrame&&window.__capExportFrame(exportCanvas);await new Promise(resolve=>setTimeout(resolve,0));}
      }else{
        // MediaRecorder timestamps from the real clock. The source time is also taken from
        // that clock, and the offline mix starts with recorder.start(), keeping this fallback
        // in sync even when the renderer cannot maintain the requested frame rate.
        exportCaptureTrack.requestFrame();
        fallbackFrameIndex++;
        const elapsed=(performance.now()-fallbackStartMs)/1000;
        showExportProgress(Math.min(96,Math.round(elapsed/durationSec*96)));window.__capExportFrame&&window.__capExportFrame(exportCanvas);
        if(elapsed>=durationSec)break;
        const nextDue=fallbackStartMs+fallbackFrameIndex*frameIntervalMs;
        const waitMs=nextDue-performance.now();
        if(waitMs>1)await new Promise(resolve=>setTimeout(resolve,waitMs));
      }
    }

    let blob;
    if(exportMode==='webcodecs'){
      await exportVideoEncoder.flush();
      const encoderErrors=window.__ethvelleExportEncoderErrors&&window.__ethvelleExportEncoderErrors();
      if(encoderErrors&&(encoderErrors.video||encoderErrors.audio))throw (encoderErrors.video||encoderErrors.audio);
      exportMuxer.finalize();
      const finalBuffer=muxerTarget&&muxerTarget.buffer;
      if(!finalBuffer||!finalBuffer.byteLength)throw new Error('The video encoder produced an empty export.');
      blob=new Blob([finalBuffer],{type:exportOutputMime});
    }else{
      exportRecorder.stop();
      await exportRecorderFinished;
      blob=new Blob(exportRecorderChunks,{type:exportOutputMime||'video/webm'});
    }
    showExportProgress(100);
    const url = URL.createObjectURL(blob);
    const safeName = (currentProject?.name || 'ethvelle-animation').replace(/[^a-z0-9\-_]+/gi,'_');
    const savedFileName = `${safeName}-${exportQuality}p.${exportFileExtension}`;
    // Direct save: the file is written to the device (Downloads) automatically the moment export ends.
    const a = document.createElement('a');
    a.href = url; a.download = savedFileName;
    document.body.appendChild(a); a.click(); a.remove();
    if(window.__capExportDone) window.__capExportDone({blob, url, filename:savedFileName}); else { hideExportProgress(); showToast('Saved to device', {icon: toastIcons.export, duration:1800}); }
    recordExportUsed();
    trackEvent('export');
  }catch(err){
    console.error('Ethvelle export failed', err);
    hideExportProgress();
    trackEvent('export_failed');
    if(chargedCreditsAmount>0){
      try{await changeMyCreditBalance(chargedCreditsAmount,{type:'refund',feature:'export',description:'Refund for failed video export',quality:exportQuality,durationSeconds:Math.ceil(plannedDurationSec),referenceId:creditChargeId});}
      catch(refundErr){console.error('Ethvelle: automatic credit refund failed',refundErr);}
    }
    showToast((err&&err.message==='export-cancelled')?"Export cancelled — credits refunded when possible":"Export failed — credits were refunded when possible. Try a lower quality", {icon: toastIcons.info, duration:3000});
  }finally{
    faceAnimationTimeOverride=null;
    isExporting = false;
    // Release the offscreen <video> elements created for overlay export compositing —
    // without this, each export would leak a decoder/video element per video overlay clip.
    Object.values(exportOverlayEls).forEach(rec=>{
      if(rec.el.tagName === 'VIDEO'){ rec.el.pause(); rec.el.removeAttribute('src'); rec.el.load(); }
    });
    // Tear down the real-time audio-mix context (if the project had any sound) — stopping
    // the source node is harmless even if it already finished playing on its own (a
    // buffer source that's already ended throws on a second stop() in some browsers, so
    // this is wrapped defensively), and closing the context releases its audio-processing
    // resources instead of leaving one AudioContext alive per export.
    if(exportRecorder&&exportRecorder.state!=='inactive'){try{exportRecorder.stop();}catch(_e){}}
    if(exportVideoEncoder){try{if(exportVideoEncoder.state!=='closed')exportVideoEncoder.close();}catch(_e){}}
    if(exportAudioEncoder){try{if(exportAudioEncoder.state!=='closed')exportAudioEncoder.close();}catch(_e){}}
    if(liveAudioCtx){
      try{ if(liveAudioCtx._exportSourceNode) liveAudioCtx._exportSourceNode.stop(); }catch(e){}
      try{ liveAudioCtx.close(); }catch(e){}
    }
    // Put every character's modelTransform AND bone pose back exactly as they were before
    // export touched them, then let renderFrame() re-apply whatever pose/transform-keyframe
    // (if any) actually belongs at the current playhead on top of that — for the active
    // character via the shared bones/modelTransform, and for every other character via
    // animateAllCharacters() inside that same renderFrame() call.
    characters.forEach((c, idx)=>{
      const saved = preExportState[idx];
      if(!saved) return;
      Object.assign(c.modelTransform, saved.modelTransform);
      Object.keys(saved.angles).forEach(k=>{ if(c.bones[k]) c.bones[k].angle = saved.angles[k]; });
      c.bones.hip.y = saved.hipY;
    });
    // restore live canvas to current playhead position
    renderFrame();
    if(wasPlaying) startPlayback();
  }
}

// Draw the current bone pose into an arbitrary canvas context, scaled to that context's size.
// Draws ONE character's rig (given the character object itself, independent of whichever
// character is currently "active") into an arbitrary target context, scaled to that
// context's size. Shared by drawStickToContext (loops this over every character) so export/
// thumbnails render the whole cast, not just whichever one happens to be selected. Takes the
// full character (not just bones/modelTransform) so a custom image-based character exports
// with its own artwork (see drawCustomCharacterOnContext) instead of always falling back to
// the green stroke stickman.
function drawCharacterToContext(targetCtx, char){
  const charBones = char.bones, charModelTransform = char.modelTransform;
  function boneEnd(key){
    const b = charBones[key];
    if(!b.parent) return {x:b.x, y:b.y};
    const pEnd = boneEnd(b.parent);
    let sx=pEnd.x, sy=pEnd.y;
    if(b.attachAt!==undefined){
      const pStart = boneStart(b.parent);
      sx = pStart.x + (pEnd.x-pStart.x)*b.attachAt;
      sy = pStart.y + (pEnd.y-pStart.y)*b.attachAt;
    }
    return { x: sx+Math.cos(b.angle)*b.length, y: sy+Math.sin(b.angle)*b.length };
  }
  function boneStart(key){
    const b = charBones[key];
    if(!b.parent) return {x:b.x, y:b.y};
    const pEnd = boneEnd(b.parent);
    if(b.attachAt!==undefined){
      const pStart = boneStart(b.parent);
      return { x: pStart.x+(pEnd.x-pStart.x)*b.attachAt, y: pStart.y+(pEnd.y-pStart.y)*b.attachAt };
    }
    return pEnd;
  }
  targetCtx.save();
  const anchor = charBones.hip;
  targetCtx.translate(anchor.x + charModelTransform.offsetX, anchor.y + charModelTransform.offsetY);
  targetCtx.rotate(charModelTransform.rotation * Math.PI/180);
  targetCtx.scale(charModelTransform.scale, charModelTransform.scale);
  targetCtx.scale(char.mirrorX?-1:1,char.mirrorY?-1:1);
  // Keep the exported mask aligned to the same local stage coordinates as the rig.
  targetCtx.translate(-anchor.x, -anchor.y);
  if(window.ethvelleApplyCanvasMaskClip) window.ethvelleApplyCanvasMaskClip(targetCtx,char.visualMask,stageW,stageH);
  if(char.type==='custom' && char.parts){
    drawCustomCharacterOnContext(targetCtx, char, boneStart, boneEnd);
  } else {
    targetCtx.strokeStyle = '#2F5CE8';
    targetCtx.lineWidth = 5;
    targetCtx.lineCap = 'round';
    ['spine','armL','armL_lo','armR','armR_lo','legL','legL_lo','legR','legR_lo'].forEach(key=>{
      const s = boneStart(key), e = boneEnd(key);
      targetCtx.beginPath();
      targetCtx.moveTo(s.x, s.y);
      targetCtx.lineTo(e.x, e.y);
      targetCtx.stroke();
    });
    const headStart = boneStart('head'), headEnd = boneEnd('head');
    const headCenter = { x:(headStart.x+headEnd.x)/2, y:(headStart.y+headEnd.y)/2 };
    targetCtx.beginPath();
    targetCtx.arc(headCenter.x, headCenter.y, charBones.head.length*0.6, 0, Math.PI*2);
    targetCtx.stroke();
  }
  if(typeof window.ethvelleDrawCharacterProps==='function'){
    try{window.ethvelleDrawCharacterProps(targetCtx,char,boneStart,boneEnd);}catch(err){console.debug('Held-item render skipped',err);}
  }
  targetCtx.restore();
}
// Draws EVERY character (not just the active one) into an arbitrary target context, scaled
// to that context's size — used for video export frames and project thumbnails, so both
// show the full cast exactly as it appears live on stage.
function drawStickToContext(targetCtx, w, h){
  const scaleX = w/stageW, scaleY = h/stageH;
  targetCtx.save();
  targetCtx.scale(scaleX, scaleY);
  characters.forEach(c=>{
    drawCharacterToContext(targetCtx, c);
  });
  targetCtx.restore();
}

// Renders a small real preview — background + the clip's first-frame pose — for this
// project's card in the Projects list, CapCut-style, instead of the generic placeholder
// character every project used to share. Leaves the live editor's own pose/canvas alone
// (restores it afterward) since this can run right as the user is navigating away.
async function generateProjectThumbnail(){
  if(!stageW || !stageH) return null;
  try{
    const THUMB_W = 220;
    const THUMB_H = Math.max(1, Math.round(THUMB_W * (stageH/stageW)));
    const cv = document.createElement('canvas');
    cv.width = THUMB_W; cv.height = THUMB_H;
    const tctx = cv.getContext('2d');
    const firstFrameBackground = currentBackgroundClip(0);
    const thumbnailBgSource = firstFrameBackground ? firstFrameBackground.src : currentBg;
    const thumbnailBgTransform = firstFrameBackground ? firstFrameBackground.transform : bgTransform;

    if(thumbnailBgSource && String(thumbnailBgSource).startsWith('url(')){
      const urlMatch = String(thumbnailBgSource).match(/url\((.*)\)/);
      const src = urlMatch ? urlMatch[1].replace(/^["']|["']$/g,'') : null;
      let drew = false;
      if(src){
        const img = new Image();
        img.src = src;
        await new Promise(res=>{ img.onload = res; img.onerror = res; });
        if(img.naturalWidth){
          const bgScale = THUMB_W/stageW;
          // UPDATED: "contain" (Math.min) instead of "cover" (Math.max) so the project-list
          // thumbnail shows the same uncropped background as the editor and the export.
          tctx.fillStyle = '#ffffff'; tctx.fillRect(0,0,THUMB_W,THUMB_H);
          const coverScale = Math.min(THUMB_W/img.naturalWidth, THUMB_H/img.naturalHeight) * thumbnailBgTransform.scale;
          const dw = img.naturalWidth*coverScale, dh = img.naturalHeight*coverScale;
          const cx = THUMB_W/2 + thumbnailBgTransform.offsetX*bgScale, cy = THUMB_H/2 + thumbnailBgTransform.offsetY*bgScale;
          // Mirrors the export routine's rotate-around-center handling (see startExport's own
          // bgDraw comment) so a rotated background still reads correctly in the small
          // project-list thumbnail too, not just in the live editor/export.
          if(thumbnailBgTransform.rotation){
            tctx.save();
            tctx.translate(cx, cy);
            tctx.rotate(thumbnailBgTransform.rotation * Math.PI/180);
            tctx.drawImage(img, -dw/2, -dh/2, dw, dh);
            tctx.restore();
          } else {
            tctx.drawImage(img, cx-dw/2, cy-dh/2, dw, dh);
          }
          drew = true;
        }
      }
      if(!drew){ tctx.fillStyle = '#ffffff'; tctx.fillRect(0,0,THUMB_W,THUMB_H); }
    } else {
      tctx.fillStyle = thumbnailBgSource || '#ffffff';
      tctx.fillRect(0,0,THUMB_W,THUMB_H);
    }

    // Render every character at absolute time zero from isolated rig clones. The live
    // character bones/model transforms and the playback clock are never temporarily changed.
    const coverCharacters=ethvelleCloneCharactersForFirstFrameCover();
    ethvelleDrawCoverCharacters(tctx,coverCharacters,THUMB_W,THUMB_H);

    return cv.toDataURL('image/jpeg', 0.72);
  }catch(err){
    console.warn('Ethvelle: thumbnail generation failed', err);
    return null;
  }
}


let projectCoverTimer=null;
let lastLifecycleCoverAt=0;
let lastLifecycleCoverProjectId=null;
function saveProjectThumbnailResult(project, thumb){
  if(!project || !thumb) return;
  project.thumb=thumb;
  const idx=projects.findIndex(p=>p.id===project.id);
  if(idx>-1) projects[idx]=project;
  saveProjectsToStorage();
  if(document.getElementById('screen-projects')?.classList.contains('active')) renderProjects();
}
function captureProjectFirstFrameCoverSync(project){
  // This capture is deliberately isolated from the live editor clock/compositor. Never pause,
  // seek, or call renderFrame() here: doing so removes/recreates the active overlay <video>
  // nodes while a user is scrubbing and can leave the live preview stuck on a stale frame.
  if(!project || !stageW || !stageH) return false;
  try{
    const w=220, h=Math.max(1,Math.round(w*(stageH/stageW)));
    const cover=document.createElement('canvas');cover.width=w;cover.height=h;
    const cx=cover.getContext('2d');
    if(!cx)return false;
    cx.fillStyle='#FFFFFF';cx.fillRect(0,0,w,h);
    const coverCharacters=ethvelleCloneCharactersForFirstFrameCover();
    ethvelleDrawCoverCharacters(cx,coverCharacters,w,h);
    const data=cover.toDataURL('image/jpeg',0.76);
    if(!data || data.length<100)return false;
    project.thumb=data;
    const idx=projects.findIndex(p=>p.id===project.id);
    if(idx>-1)projects[idx]=project;
    saveProjectsToStorage();
    if(document.getElementById('screen-projects')?.classList.contains('active'))renderProjects();
    return true;
  }catch(err){
    console.warn('Ethvelle: first-frame cover capture failed',err);
    return false;
  }
}
function queueProjectFirstFrameCover(delay){
  if(!currentProject || !stageW || !stageH) return;
  const pid=currentProject.id;
  clearTimeout(projectCoverTimer);
  projectCoverTimer=setTimeout(()=>{
    if(!currentProject || currentProject.id!==pid || !document.getElementById('screen-editor')?.classList.contains('active')) return;
    if(isPlaying){ queueProjectFirstFrameCover(1800); return; }
    const project=currentProject;
    captureProjectFirstFrameCoverSync(project);
    generateProjectThumbnail().then(thumb=>{
      // Don't let a slow background-image load from an old project overwrite another project's cover.
      if(!thumb || !currentProject || currentProject.id!==pid || !document.getElementById('screen-editor')?.classList.contains('active')) return;
      saveProjectThumbnailResult(project,thumb);
    });
  },Math.max(0,Number(delay)||0));
}
function saveEditorStateOnLeave(){
  if(!currentProject || !document.getElementById('screen-editor')?.classList.contains('active')) return;
  const now=Date.now();
  if(lastLifecycleCoverProjectId===currentProject.id && now-lastLifecycleCoverAt<700) return;
  lastLifecycleCoverAt=now; lastLifecycleCoverProjectId=currentProject.id;
  clearTimeout(projectCoverTimer);
  try{ persistProjectState({skipHistory:true}); }catch(e){ console.warn('Ethvelle: leave-save state error',e); }
  if(isPlaying) pausePlaybackAtCurrentTime();
  const project=currentProject, pid=currentProject.id;
  captureProjectFirstFrameCoverSync(project);
  generateProjectThumbnail().then(thumb=>{
    if(!thumb || !currentProject || currentProject.id!==pid) return;
    saveProjectThumbnailResult(project,thumb);
  });
}
document.addEventListener('visibilitychange',()=>{if(document.visibilityState==='hidden')saveEditorStateOnLeave();});
window.addEventListener('pagehide',saveEditorStateOnLeave);
