<script id="ethvelle-v42-focus-and-short-clip-fix">
(function(){
  'use strict';
  const SHORT_CLIP_WIDTH = 42;
  const CLIP_SELECTOR = '#timelineInner .anim-clip, #timelineInner .overlay-clip, #timelineInner .track-clip';
  const observedClips = new WeakSet();
  let timelineRoot = null;
  let timelineMutationObserver = null;
  let bodyWaitObserver = null;
  let clipResizeObserver = null;

  function updateShortClip(el){
    if(!el || !el.isConnected) return;
    const width = el.getBoundingClientRect().width || Number.parseFloat(el.style.width) || 0;
    el.classList.toggle('ev-short-clip', width > 0 && width < SHORT_CLIP_WIDTH);
  }
  function observeClip(el){
    if(!el || !el.matches || !el.matches('.anim-clip, .overlay-clip, .track-clip')) return;
    updateShortClip(el);
    if(clipResizeObserver && !observedClips.has(el)){
      observedClips.add(el);
      try{clipResizeObserver.observe(el);}catch(_e){}
    }
  }
  function observeAdded(node){
    if(!node || node.nodeType !== 1) return;
    if(node.matches && node.matches('.anim-clip, .overlay-clip, .track-clip')) observeClip(node);
    if(node.querySelectorAll) node.querySelectorAll('.anim-clip, .overlay-clip, .track-clip').forEach(observeClip);
  }
  function attachTimelineObservers(){
    const root = document.getElementById('timelineInner');
    if(!root || root === timelineRoot) return;
    if(timelineMutationObserver) timelineMutationObserver.disconnect();
    timelineRoot = root;
    if(typeof ResizeObserver === 'function'){
      clipResizeObserver = clipResizeObserver || new ResizeObserver(entries=>entries.forEach(entry=>updateShortClip(entry.target)));
    }
    root.querySelectorAll('.anim-clip, .overlay-clip, .track-clip').forEach(observeClip);
    if(typeof MutationObserver === 'function'){
      timelineMutationObserver = new MutationObserver(records=>records.forEach(record=>record.addedNodes.forEach(observeAdded)));
      timelineMutationObserver.observe(root,{childList:true,subtree:true});
    }
    if(bodyWaitObserver){bodyWaitObserver.disconnect();bodyWaitObserver=null;}
  }
  if(document.readyState === 'loading') document.addEventListener('DOMContentLoaded', attachTimelineObservers, {once:true});
  else attachTimelineObservers();
  if(document.body && !timelineRoot && typeof MutationObserver === 'function'){
    bodyWaitObserver = new MutationObserver(attachTimelineObservers);
    bodyWaitObserver.observe(document.body,{childList:true,subtree:true});
  }
  window.addEventListener('resize',()=>{if(timelineRoot)timelineRoot.querySelectorAll('.anim-clip, .overlay-clip, .track-clip').forEach(updateShortClip);},{passive:true});

  /* Preserve the composing text input focus while applying a swatch. This keeps the software
     keyboard available when a user picks a text colour rather than forcing them to tap the text
     field again after every colour change. */
  let focusToRestore = null;
  let palettePointerDown = false;
  document.addEventListener('pointerdown',e=>{
    const inPalette = !!(e.target && e.target.closest && e.target.closest(
      '#ethvellePicsartTextDock #evpaPalette button, #ethvelleTextComposer #evtcColorStrip button, #ethvelleTextComposer .evpa-palette button'
    ));
    palettePointerDown = inPalette;
    focusToRestore = null;
    if(!inPalette) return;
    const active = document.activeElement;
    if(active && active.matches && active.matches('textarea, input:not([type="range"]), [contenteditable="true"]') && active.closest('#ethvelleTextComposer')){
      focusToRestore = active;
    }
  },true);
  document.addEventListener('pointerup',()=>{
    const input = palettePointerDown ? focusToRestore : null;
    palettePointerDown = false;
    focusToRestore = null;
    if(!input) return;
    setTimeout(()=>{
      const composer = document.getElementById('ethvelleTextComposer');
      if(!input.isConnected || !composer || !composer.classList.contains('active')) return;
      if(document.activeElement !== input){
        try{input.focus({preventScroll:true});}catch(_e){try{input.focus();}catch(__e){}}
      }
    },90);
  },true);
  document.addEventListener('pointercancel',()=>{palettePointerDown=false;focusToRestore=null;},true);
})();
